   159→              <div className="p-5 space-y-3">

   247→            </AlertDialogDescription>

   249→          <AlertDialogFooter>
   250→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   251→            <AlertDialogAction

   255→                try {
   256→                  await apiPost(
   257→                    `/api/admin/universities/${deleteTarget.id}`,
   258→                    null,

   262→                    title: "University deleted",

   419→// ─── Form Dialog (Add / Edit) ───────────────────────────────────────────
   420→interface FormValues {
   421→  name: string;
   422→  email: string;
   423→  password: string;
   424→  description: string;
   425→  logoFile: File | null;
   426→  logoPreview: string | null;
   427→}
   428→
   429→function UniversityFormDialog({
   430→  open,
   431→  onOpenChange,
   432→  target,
   433→  onSaved,
   434→}: {
   435→  open: boolean;
   436→  onOpenChange: (v: boolean) => void;
   437→  target?: UniversityRow | null;
   438→  onSaved: () => void;
   439→}) {
   440→  const isEdit = !!target;
   441→  const [values, setValues] = useState<FormValues>({
   442→    name: "",
   443→    email: "",
   444→    password: "",
   445→    description: "",
   446→    logoFile: null,
   447→    logoPreview: null,
   448→  });
   449→  const [saving, setSaving] = useState(false);
   450→
   451→  useEffect(() => {
   452→    if (target) {
   453→      setValues({
   454→        name: target.name,
   455→        email: target.email,
   456→        password: "",
   457→        description: target.description || "",
   458→        logoFile: null,
   459→        logoPreview: target.logo,
   460→      });
   461→    } else if (open) {
   462→      setValues({
   463→        name: "",
   464→        email: "",
   465→        password: "",
   466→        description: "",
   467→        logoFile: null,
   468→        logoPreview: null,
   469→      });
   470→    }
   471→  }, [target, open]);
   472→
   473→  function onLogoSelected(file: File | null) {
   474→    if (!file) {
   475→      setValues((v) => ({ ...v, logoFile: null, logoPreview: target?.logo || null }));
   476→      return;
   477→    }
   478→    const previewUrl = URL.createObjectURL(file);
   479→    setValues((v) => ({ ...v, logoFile: file, logoPreview: previewUrl }));
   480→  }
   481→
   482→  async function submit() {
   483→    if (!values.name.trim()) {
   484→      toast({ title: "Name required", variant: "destructive" });
   485→      return;
   486→    }
   487→    if (!values.email.trim()) {
   488→      toast({ title: "Email required", variant: "destructive" });
   489→      return;
   490→    }
   491→    if (!isEdit && !values.password) {
   492→      toast({ title: "Password required", variant: "destructive" });
   493→      return;
   494→    }
   495→    if (values.password && values.password.length < 6) {
   496→      toast({
   497→        title: "Password too short",
   498→        description: "Must be at least 6 characters.",
   499→        variant: "destructive",
   500→      });
   501→      return;
   502→    }
   503→
   504→    setSaving(true);
   505→    try {
   506→      const fd = new FormData();
   507→      fd.append("name", values.name.trim());
   508→      fd.append("email", values.email.trim());
   509→      if (values.password) fd.append("password", values.password);
   510→      if (values.description.trim()) fd.append("description", values.description.trim());
   511→      if (values.logoFile) fd.append("logo", values.logoFile);
   512→
   513→      const url = isEdit
   514→        ? `/api/admin/universities/${target!.id}`
   515→        : "/api/admin/universities";
   516→
   517→      // apiUpload uses POST; for PUT we fetch directly with FormData.
   518→      if (isEdit) {
   519→        const res = await fetch(url, {
   520→          method: "PUT",
   521→          body: fd,
   522→          credentials: "include",
   523→        });
   524→        const json = await res.json();
   525→        if (!res.ok) {
   526→          const msg = json.error || `Request failed (${res.status})`;
   527→          toast({ title: "Error", description: msg, variant: "destructive" });
   528→          throw new Error(msg);
   529→        }
   530→      } else {
   531→        await apiUpload(url, fd);
   532→      }
   533→      onSaved();
   534→    } catch {
   535→      /* toast handled in helpers */
   536→    } finally {
   537→      setSaving(false);
   538→    }
   539→  }
   540→
   541→  const previewInitials = values.name
   542→    .split(/\s+/)
   543→    .map((w) => w[0])
   544→    .filter(Boolean)
   545→    .slice(0, 2)
   546→    .join("")
   547→    .toUpperCase();
   548→
   549→  return (
   550→    <Dialog open={open} onOpenChange={onOpenChange}>
   551→      <DialogContent className="sm:max-w-lg">
   552→        <DialogHeader>
   553→          <DialogTitle className="font-heading text-lg font-semibold">
   554→            {isEdit ? "Edit University" : "Add New University"}
   555→          </DialogTitle>
   556→          <DialogDescription>
   557→            {isEdit
   558→              ? "Update the university profile and its linked login account."
   559→              : "Create a university profile and a login account with role UNIVERSITY."}
   560→          </DialogDescription>
   561→        </DialogHeader>
   562→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1">
   563→          {/* Logo preview + upload */}
   564→          <div className="space-y-1.5">
   565→            <Label className="text-xs font-medium text-muted-foreground">Logo</Label>
   566→            <div className="flex items-center gap-3">
   567→              <div className="h-16 w-16 overflow-hidden rounded-lg bg-muted flex items-center justify-center shrink-0 ring-1 ring-border">
   568→                {values.logoPreview ? (
   569→                  <img
   570→                    src={values.logoPreview}
   571→                    alt=""
   572→                    className="h-full w-full object-contain p-1"
   573→                  />
   574→                ) : previewInitials ? (
   575→                  <span className="font-heading text-lg font-bold text-primary">
   576→                    {previewInitials}
   577→                  </span>
   578→                ) : (
   579→                  <Building2 className="h-5 w-5 text-muted-foreground" />
   580→                )}
   581→              </div>
   582→              <div className="flex-1 space-y-1.5">
   583→                <label className="inline-flex items-center gap-1.5 cursor-pointer text-xs text-accent hover:underline">
   584→                  <Upload className="h-3.5 w-3.5" />
   585→                  {values.logoFile ? "Change image" : "Upload logo"}
   586→                  <input
   587→                    type="file"
   588→                    accept="image/*"
   589→                    className="hidden"
   590→                    onChange={(e) => {
   591→                      const f = e.target.files?.[0] || null;
   592→                      onLogoSelected(f);
   593→                    }}
   594→                  />
   595→                </label>
   596→                <p className="text-xs text-muted-foreground">
   597→                  PNG, JPG, WebP — square format recommended.
   598→                </p>
   599→              </div>
   600→            </div>
   601→          </div>
   602→
   603→          <div className="space-y-1.5">

   672→function UniversityDetailDrawer({
   673→  target,
   674→  onOpenChange,
   675→}: {
   676→  target: UniversityRow | null;
   677→  onOpenChange: (v: boolean) => void;
   678→}) {
   679→  const open = !!target;
   680→  const { data, loading, error } = useApiFetch<UniversityDetailResponse>(
   681→    target ? `/api/admin/universities/${target.id}` : null,
   682→  );
   683→
   684→  // Reset fetch when drawer closes — pass null url above handles that
   685→  const university = data?.university || target;
   686→  const initials = useMemo(() => {
   687→    if (!university) return "";
   688→    return university.name
   689→      .split(/\s+/)
   690→      .map((w) => w[0])
   691→      .filter(Boolean)
   692→      .slice(0, 2)
   693→      .join("")
   694→      .toUpperCase();
   695→  }, [university]);
   696→
   697→  return (
   698→    <Drawer open={open} onOpenChange={onOpenChange}>
   699→      <DrawerContent className="max-h-[88vh]">
   700→        <div className="mx-auto w-full max-w-3xl">
   701→          <DrawerHeader className="border-b border-border">
   702→            <DrawerTitle className="font-heading text-xl font-semibold text-foreground">
   703→              {university?.name}
   704→            </DrawerTitle>
   705→            <DrawerDescription>
   706→              {university?.description || "No description provided."}
   707→            </DrawerDescription>
   708→          </DrawerHeader>
   709→
   710→          <div className="space-y-4 p-4 md:p-6 overflow-y-auto max-h-[calc(88vh-6rem)] scrollbar-thin">
   711→            {loading ? (
   712→              <div className="space-y-3">
   713→                <Skeleton className="h-24 w-full rounded-lg" />
   714→                <Skeleton className="h-6 w-40" />
   715→                <Skeleton className="h-32 w-full rounded-lg" />
   716→              </div>
   717→            ) : error ? (
   718→              <EmptyState
   719→                icon={<Building2 className="h-5 w-5" />}
   720→                title="Couldn't load details"
   721→                description={error}
   722→              />
   723→            ) : university ? (
   724→              <>
   725→                {/* Header card */}
   726→                <Card className="overflow-hidden border-border p-0">
   727→                  <div className="relative h-24 bg-gradient-to-br from-primary via-primary to-[#2A3F35]">
   728→                    <div
   729→                      className="absolute inset-0 opacity-30"
   730→                      style={{
   731→                        backgroundImage:
   732→                          "radial-gradient(circle at 20% 50%, rgba(120,151,134,0.55), transparent 55%), radial-gradient(circle at 80% 20%, rgba(252,252,250,0.12), transparent 45%)",
   733→                      }}
   734→                    />
   735→                  </div>
   736→                  <div className="relative -mt-10 px-5 pb-5">
   737→                    <div className="flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
   738→                      <div className="flex items-end gap-3">
   739→                        {university.logo ? (
   740→                          <img
   741→                            src={university.logo}
   742→                            alt={`${university.name} logo`}
   743→                            className="h-20 w-20 rounded-xl bg-background object-contain p-1.5 ring-2 ring-background shadow-sm"
   744→                          />
   745→                        ) : (
   746→                          <div className="flex h-20 w-20 items-center justify-center rounded-xl bg-background text-primary font-heading text-2xl font-bold ring-2 ring-background shadow-sm">
   747→                            {initials || <GraduationCap className="h-7 w-7" />}
   748→                          </div>
   749→                        )}
   750→                        <div className="pb-1">
   751→                          <Badge className="gap-1">
   752→                            <Building2 className="h-3 w-3" /> University Partner
   753→                          </Badge>
   754→                        </div>
   755→                      </div>
   756→                    </div>
   757→                  </div>
   758→                </Card>
   759→
   760→                {/* Stats row */}
   761→                <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">

   764→                    label="Courses"
   765→                    value={university.courseCount}

   769→                    label="Email"
   770→                    value={university.email}
   771→                    isText
   772→                  />
   773→                  <DetailStat
   774→                    icon={<Calendar className="h-4 w-4" />}
   775→                    label="Joined"
   776→                    value={formatDate(university.createdAt)}
   777→                    isText
   778→                  />
   779→                  <DetailStat
   780→                    icon={<GraduationCap className="h-4 w-4" />}
   781→                    label="Status"
   782→                    value="Active"
   783→                    isText
   784→                  />
   785→                </div>
   786→
   787→                {/* Description block */}
   788→                {university.description && (
   789→                  <Card className="border-border p-5">
   790→                    <h4 className="font-heading text-sm font-semibold text-foreground mb-2">

   829→    </Drawer>

   833→function DetailStat({
   834→  icon,
   835→  label,
   836→  value,
   837→  isText,
   838→}: {
   839→  icon: React.ReactNode;
   840→  label: string;
   841→  value: number | string;
   842→  isText?: boolean;
   843→}) {
   844→  return (
   845→    <Card className="border-border p-3">
   846→      <div className="flex items-center gap-2 text-accent">
   847→        {icon}
   848→        <span className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground">
   849→          {label}
   850→        </span>
   851→      </div>
   852→      <p
   853→        className={`mt-1.5 font-heading font-semibold text-foreground ${
   854→          isText ? "text-sm truncate" : "text-xl"
   855→        }`}
   856→        title={isText ? String(value) : undefined}
   857→      >
   858→        {isText && typeof value === "string"
   859→          ? value.length > 22
   860→            ? value.slice(0, 20) + "…"
   861→            : value
   862→          : typeof value === "number"
   863→            ? value.toLocaleString()
   864→            : value}
   865→      </p>
   866→    </Card>
   867→  );
