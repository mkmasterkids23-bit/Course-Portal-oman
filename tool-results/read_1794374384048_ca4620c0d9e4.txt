   425→// ─── Form Dialog (Add / Edit) ───────────────────────────────────────────
   426→interface FormValues {
   427→  name: string;
   428→  email: string;
   429→  password: string;
   430→  description: string;
   431→  logoFile: File | null;
   432→  logoPreview: string | null;
   433→}
   434→
   435→function UniversityFormDialog({
   436→  open,
   437→  onOpenChange,
   438→  target,
   439→  onSaved,
   440→}: {
   441→  open: boolean;
   442→  onOpenChange: (v: boolean) => void;
   443→  target?: UniversityRow | null;
   444→  onSaved: () => void;
   445→}) {
   446→  const isEdit = !!target;
   447→  const [values, setValues] = useState<FormValues>({
   448→    name: "",
   449→    email: "",
   450→    password: "",
   451→    description: "",
   452→    logoFile: null,
   453→    logoPreview: null,
   454→  });
   455→  const [saving, setSaving] = useState(false);
   456→
   457→  useEffect(() => {
   458→    if (target) {
   459→      setValues({
   460→        name: target.name,
   461→        email: target.email,
   462→        password: "",
   463→        description: target.description || "",
   464→        logoFile: null,
   465→        logoPreview: target.logo,
   466→      });
   467→    } else if (open) {
   468→      setValues({
   469→        name: "",
   470→        email: "",
   471→        password: "",
   472→        description: "",
   473→        logoFile: null,
   474→        logoPreview: null,
   475→      });
   476→    }
   477→  }, [target, open]);
   478→
   479→  function onLogoSelected(file: File | null) {
   480→    if (!file) {
   481→      setValues((v) => ({ ...v, logoFile: null, logoPreview: target?.logo || null }));
   482→      return;
   483→    }
   484→    const previewUrl = URL.createObjectURL(file);
   485→    setValues((v) => ({ ...v, logoFile: file, logoPreview: previewUrl }));
   486→  }
   487→
   488→  async function submit() {
   489→    if (!values.name.trim()) {
   490→      toast({ title: "Name required", variant: "destructive" });
   491→      return;
   492→    }
   493→    if (!values.email.trim()) {
   494→      toast({ title: "Email required", variant: "destructive" });
   495→      return;
   496→    }
   497→    if (!isEdit && !values.password) {
   498→      toast({ title: "Password required", variant: "destructive" });
   499→      return;
   500→    }
   501→    if (values.password && values.password.length < 6) {
   502→      toast({
   503→        title: "Password too short",
   504→        description: "Must be at least 6 characters.",
   505→        variant: "destructive",
   506→      });
   507→      return;
   508→    }
   509→
   510→    setSaving(true);
   511→    try {
   512→      const fd = new FormData();
   513→      fd.append("name", values.name.trim());
   514→      fd.append("email", values.email.trim());
   515→      if (values.password) fd.append("password", values.password);
   516→      if (values.description.trim()) fd.append("description", values.description.trim());
   517→      if (values.logoFile) fd.append("logo", values.logoFile);
   518→
   519→      const url = isEdit
   520→        ? `/api/admin/universities/${target!.id}`
   521→        : "/api/admin/universities";
   522→
   523→      // apiUpload uses POST; for PUT we fetch directly with FormData.
   524→      if (isEdit) {
   525→        const res = await fetch(url, {
   526→          method: "PUT",
   527→          body: fd,
   528→          credentials: "include",
   529→        });
   530→        const json = await res.json();
   531→        if (!res.ok) {
   532→          const msg = json.error || `Request failed (${res.status})`;
   533→          toast({ title: "Error", description: msg, variant: "destructive" });
   534→          throw new Error(msg);
   535→        }
   536→      } else {
   537→        await apiUpload(url, fd);
   538→      }
   539→      onSaved();
   540→    } catch {
   541→      /* toast handled in helpers */
   542→    } finally {
   543→      setSaving(false);
   544→    }
   545→  }
   546→
   547→  const previewInitials = values.name
   548→    .split(/\s+/)
   549→    .map((w) => w[0])
   550→    .filter(Boolean)
   551→    .slice(0, 2)
   552→    .join("")
   553→    .toUpperCase();
   554→
   555→  return (
   556→    <Dialog open={open} onOpenChange={onOpenChange}>
   557→      <DialogContent className="sm:max-w-lg">
   558→        <DialogHeader>
   559→          <DialogTitle className="font-heading text-lg font-semibold">
   560→            {isEdit ? "Edit University" : "Add New University"}
   561→          </DialogTitle>
   562→          <DialogDescription>
   563→            {isEdit
   564→              ? "Update the university profile and its linked login account."
   565→              : "Create a university profile and a login account with role UNIVERSITY."}
   566→          </DialogDescription>
   567→        </DialogHeader>
   568→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1">
   569→          {/* Logo preview + upload */}
   570→          <div className="space-y-1.5">
   571→            <Label className="text-xs font-medium text-muted-foreground">Logo</Label>
   572→            <div className="flex items-center gap-3">
   573→              <div className="h-16 w-16 overflow-hidden rounded-lg bg-muted flex items-center justify-center shrink-0 ring-1 ring-border">
   574→                {values.logoPreview ? (
   575→                  <img
   576→                    src={values.logoPreview}
   577→                    alt=""
   578→                    className="h-full w-full object-contain p-1"
   579→                  />
   580→                ) : previewInitials ? (
   581→                  <span className="font-heading text-lg font-bold text-primary">
   582→                    {previewInitials}
   583→                  </span>
   584→                ) : (
   585→                  <Building2 className="h-5 w-5 text-muted-foreground" />
   586→                )}
   587→              </div>
   588→              <div className="flex-1 space-y-1.5">
   589→                <label className="inline-flex items-center gap-1.5 cursor-pointer text-xs text-accent hover:underline">
   590→                  <Upload className="h-3.5 w-3.5" />
   591→                  {values.logoFile ? "Change image" : "Upload logo"}
   592→                  <input
   593→                    type="file"
   594→                    accept="image/*"
   595→                    className="hidden"
   596→                    onChange={(e) => {
   597→                      const f = e.target.files?.[0] || null;
   598→                      onLogoSelected(f);
   599→                    }}
   600→                  />
   601→                </label>
   602→                <p className="text-xs text-muted-foreground">
   603→                  PNG, JPG, WebP — square format recommended.
   604→                </p>
   605→              </div>
   606→            </div>
   607→          </div>
   608→
   609→          <div className="space-y-1.5">

   612→            </Label>

   617→              placeholder="e.g. Sultan Qaboos University"
   618→              className="h-9 rounded-lg"
   619→            />
   620→          </div>
   621→
   622→          <div className="space-y-1.5">
   623→            <Label htmlFor="uf-email" className="text-xs font-medium text-muted-foreground">
   624→              Login Email
   625→            </Label>
   626→            <Input
   627→              id="uf-email"
   628→              type="email"
   629→              value={values.email}
   630→              onChange={(e) => setValues({ ...values, email: e.target.value })}
   631→              placeholder="university@example.com"
   632→              className="h-9 rounded-lg"
   633→            />
   634→          </div>
   635→
   636→          <div className="space-y-1.5">

   654→            <Textarea
   655→              id="uf-desc"

   658→              placeholder="Short summary shown on the university card"
   659→              rows={3}
   660→              className="rounded-lg"
   661→            />
   662→          </div>
   663→        </div>
   664→        <DialogFooter>

   667→          </Button>

   669→            {saving ? "Saving…" : isEdit ? "Save Changes" : "Create University"}
   670→          </Button>
   671→        </DialogFooter>
   672→      </DialogContent>
   673→    </Dialog>
   674→  );

   678→function UniversityDetailDrawer({
   679→  target,
   680→  onOpenChange,
   681→}: {
   682→  target: UniversityRow | null;
   683→  onOpenChange: (v: boolean) => void;
   684→}) {
   685→  const open = !!target;
   686→  const { data, loading, error } = useApiFetch<UniversityDetailResponse>(
   687→    target ? `/api/admin/universities/${target.id}` : null,
   688→  );
   689→
   690→  // Reset fetch when drawer closes — pass null url above handles that
   691→  const university = data?.university || target;
   692→  const initials = useMemo(() => {
   693→    if (!university) return "";
   694→    return university.name
   695→      .split(/\s+/)
   696→      .map((w) => w[0])
   697→      .filter(Boolean)
   698→      .slice(0, 2)
   699→      .join("")
   700→      .toUpperCase();
   701→  }, [university]);
   702→
   703→  return (
   704→    <Drawer open={open} onOpenChange={onOpenChange}>
   705→      <DrawerContent className="max-h-[88vh]">
   706→        <div className="mx-auto w-full max-w-3xl">
   707→          <DrawerHeader className="border-b border-border">
   708→            <DrawerTitle className="font-heading text-xl font-semibold text-foreground">
   709→              {university?.name}
   710→            </DrawerTitle>
   711→            <DrawerDescription>
   712→              {university?.description || "No description provided."}
   713→            </DrawerDescription>
   714→          </DrawerHeader>
   715→
   716→          <div className="space-y-4 p-4 md:p-6 overflow-y-auto max-h-[calc(88vh-6rem)] scrollbar-thin">
   717→            {loading ? (
   718→              <div className="space-y-3">
   719→                <Skeleton className="h-24 w-full rounded-lg" />
   720→                <Skeleton className="h-6 w-40" />
   721→                <Skeleton className="h-32 w-full rounded-lg" />
   722→              </div>
   723→            ) : error ? (
   724→              <EmptyState
   725→                icon={<Building2 className="h-5 w-5" />}
   726→                title="Couldn't load details"
   727→                description={error}
   728→              />
   729→            ) : university ? (
   730→              <>
   731→                {/* Header card */}
   732→                <Card className="overflow-hidden border-border p-0">
   733→                  <div className="relative h-24 bg-gradient-to-br from-primary/85 via-primary to-primary/70">
   734→                    <div
   735→                      className="absolute inset-0 opacity-30"
   736→                      style={{
   737→                        backgroundImage:
   738→                          "radial-gradient(circle at 20% 50%, rgba(15,118,110,0.55), transparent 55%), radial-gradient(circle at 80% 20%, rgba(255,255,255,0.12), transparent 45%)",
   739→                      }}
   740→                    />
   741→                  </div>
   742→                  <div className="relative -mt-10 px-5 pb-5">
   743→                    <div className="flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
   744→                      <div className="flex items-end gap-3">
   745→                        {university.logo ? (
   746→                          <img
   747→                            src={university.logo}
   748→                            alt={`${university.name} logo`}
   749→                            className="h-20 w-20 rounded-xl bg-background object-contain p-1.5 ring-2 ring-background shadow-sm"
   750→                          />
   751→                        ) : (
   752→                          <div className="flex h-20 w-20 items-center justify-center rounded-xl bg-background text-primary font-heading text-2xl font-bold ring-2 ring-background shadow-sm">
   753→                            {initials || <GraduationCap className="h-7 w-7" />}
   754→                          </div>
   755→                        )}
   756→                        <div className="pb-1">
   757→                          <Badge className="gap-1">

   759→                          </Badge>
   760→                        </div>
   761→                      </div>
   762→                    </div>
   763→                  </div>

   765→

   778→                  />

   781→                    label="Joined"
   782→                    value={formatDate(university.createdAt)}
   783→                    isText
   784→                  />
   785→                  <DetailStat
   786→                    icon={<GraduationCap className="h-4 w-4" />}
   787→                    label="Status"
   788→                    value="Active"
   789→                    isText
   790→                  />
   791→                </div>
   792→
   793→                {/* Description block */}

   839→function DetailStat({
   840→  icon,
   841→  label,
   842→  value,
   843→  isText,
   844→}: {
   845→  icon: React.ReactNode;
   846→  label: string;
   847→  value: number | string;
   848→  isText?: boolean;
   849→}) {

   852→      <div className="flex items-center gap-2 text-accent">

   861→        }`}
   862→        title={isText ? String(value) : undefined}
   863→      >
   864→        {isText && typeof value === "string"
   865→          ? value.length > 22
   866→            ? value.slice(0, 20) + "…"
   867→            : value
   868→          : typeof value === "number"
   869→            ? value.toLocaleString()
   870→            : value}
   871→      </p>
   872→    </Card>
   873→  );
