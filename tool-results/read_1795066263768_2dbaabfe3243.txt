   311→function TopicsTab({ course, onRefresh }: { course: CourseDetail; onRefresh: () => void }) {
   312→  const { t } = useI18n();
   313→  const [addTopicOpen, setAddTopicOpen] = useState(false);
   314→  const [expandedId, setExpandedId] = useState<string | null>(null);
   315→  const [editTopic, setEditTopic] = useState<TopicRow | null>(null);
   316→  const [deleteTopic, setDeleteTopic] = useState<TopicRow | null>(null);
   317→  const [addVideoFor, setAddVideoFor] = useState<TopicRow | null>(null);
   318→  const [editVideo, setEditVideo] = useState<VideoRow | null>(null);
   319→  const [deleteVideo, setDeleteVideo] = useState<VideoRow | null>(null);
   320→  const [questionsOpenFor, setQuestionsOpenFor] = useState<string | null>(null);
   321→  const [questionCounts, setQuestionCounts] = useState<Record<string, number>>({});
   322→  const reportCount = useCallback((videoId: string, n: number) => {
   323→    setQuestionCounts((prev) => (prev[videoId] === n ? prev : { ...prev, [videoId]: n }));
   324→  }, []);
   325→
   326→  async function moveTopic(topic: TopicRow, dir: -1 | 1) {
   327→    const sorted = [...course.topics].sort((a, b) => a.displayOrder - b.displayOrder);
   328→    const idx = sorted.findIndex((t) => t.id === topic.id);
   329→    const target = idx + dir;
   330→    if (target < 0 || target >= sorted.length) return;
   331→    const reordered = [...sorted];
   332→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   333→    try {
   334→      await apiPost(`/api/university/courses/${course.id}/topics/reorder`, {
   335→        orderedIds: reordered.map((t) => t.id),
   336→      });
   337→      onRefresh();
   338→    } catch {
   339→      /* toast */
   340→    }
   341→  }
   342→
   343→  return (
   344→    <div className="space-y-4">
   345→      <div className="flex items-center justify-between">
   346→        <div>
   347→          <h3 className="font-heading text-base font-semibold text-foreground">Course Topics</h3>
   348→          <p className="text-xs text-muted-foreground mt-0.5">
   349→            {course.topics.length} topic{course.topics.length === 1 ? "" : "s"} ·{" "}
   350→            {course.topics.reduce((a, t) => a + t.videos.length, 0)} videos
   351→          </p>
   352→        </div>
   353→        <Button className="h-9 gap-2" onClick={() => setAddTopicOpen(true)}>
   354→          <Plus className="h-4 w-4" /> Add Topic
   355→        </Button>
   356→      </div>
   357→
   358→      {course.topics.length === 0 ? (
   359→        <EmptyState
   360→          icon={<VideoIcon className="h-5 w-5" />}
   361→          title="No topics yet"
   362→          description="Topics are containers for video lessons. Add your first topic to get started."
   363→          action={
   364→            <Button onClick={() => setAddTopicOpen(true)} className="gap-2">
   365→              <Plus className="h-4 w-4" /> Add Topic
   366→            </Button>
   367→          }
   368→        />
   369→      ) : (
   370→        <div className="space-y-2">
   371→          {course.topics.map((topic, idx) => (
   372→            <Card
   373→              key={topic.id}
   374→              className="overflow-hidden rounded-lg border-border bg-card p-0 shadow-sm transition-shadow hover:shadow-md"
   375→            >
   376→              <div className="flex items-center gap-2 p-3 bg-gradient-to-r from-muted/40 to-muted/10">
   377→                <div className="flex flex-col">
   378→                  <Button
   379→                    variant="ghost"
   380→                    size="icon"
   381→                    className="h-5 w-5"
   382→                    disabled={idx === 0}
   383→                    onClick={() => moveTopic(topic, -1)}
   384→                    aria-label="Move topic up"
   385→                  >
   386→                    <ChevronUp className="h-3 w-3" />
   387→                  </Button>
   388→                  <Button
   389→                    variant="ghost"
   390→                    size="icon"
   391→                    className="h-5 w-5"
   392→                    disabled={idx === course.topics.length - 1}
   393→                    onClick={() => moveTopic(topic, 1)}
   394→                    aria-label="Move topic down"
   395→                  >
   396→                    <ChevronDown className="h-3 w-3" />
   397→                  </Button>
   398→                </div>
   399→                <GripVertical className="h-4 w-4 text-muted-foreground/40 hidden sm:block" />
   400→                <button
   401→                  className="flex flex-1 items-center gap-2 text-left min-w-0"
   402→                  onClick={() => setExpandedId(expandedId === topic.id ? null : topic.id)}
   403→                >
   404→                  <span className="flex h-7 w-7 items-center justify-center rounded-full bg-primary/10 text-primary text-xs font-semibold shrink-0 ring-1 ring-border">
   405→                    {idx + 1}
   406→                  </span>
   407→                  <div className="min-w-0">
   408→                    <p className="font-medium text-foreground truncate">{topic.title}</p>
   409→                    <p className="text-xs text-muted-foreground truncate">
   410→                      {topic.videos.length} video{topic.videos.length === 1 ? "" : "s"}
   411→                      {topic.materials.length > 0 &&
   412→                        ` · ${topic.materials.length} material${topic.materials.length === 1 ? "" : "s"}`}
   413→                    </p>
   414→                  </div>
   415→                </button>
   416→                <div className="flex items-center gap-1">
   417→                  <Button
   418→                    variant="ghost"
   419→                    size="sm"
   420→                    className="h-8 gap-1 text-muted-foreground hover:text-accent"
   421→                    onClick={() => setAddVideoFor(topic)}
   422→                  >
   423→                    <Plus className="h-3.5 w-3.5" /> Add Video
   424→                  </Button>
   425→                  <Button
   426→                    variant="ghost"
   427→                    size="icon"
   428→                    aria-label="Edit topic"
   429→                    className="h-8 w-8 text-muted-foreground hover:text-foreground"
   430→                    onClick={() => setEditTopic(topic)}
   431→                  >
   432→                    <Pencil className="h-3.5 w-3.5" />
   433→                  </Button>
   434→                  <Button
   435→                    variant="ghost"
   436→                    size="icon"
   437→                    aria-label="Delete topic"
   438→                    className="h-8 w-8 text-muted-foreground hover:text-destructive"
   439→                    onClick={() => setDeleteTopic(topic)}
   440→                  >
   441→                    <Trash2 className="h-3.5 w-3.5" />
   442→                  </Button>
   443→                </div>
   444→              </div>
   445→
   446→              {expandedId === topic.id && (
   447→                <div className="border-t border-border p-3 bg-card">
   448→                  {topic.description && (
   449→                    <p className="text-sm text-muted-foreground mb-3">{topic.description}</p>
   450→                  )}
   451→                  {topic.videos.length === 0 ? (
   452→                    <div className="text-center py-6">
   453→                      <VideoIcon className="h-6 w-6 text-muted-foreground/40 mx-auto mb-2" />
   454→                      <p className="text-sm text-muted-foreground">No videos in this topic yet</p>
   455→                      <Button
   456→                        variant="outline"
   457→                        size="sm"
   458→                        className="mt-3 gap-1.5 h-8"
   459→                        onClick={() => setAddVideoFor(topic)}
   460→                      >
   461→                        <Plus className="h-3.5 w-3.5" /> Add First Video
   462→                      </Button>
   463→                    </div>
   464→                  ) : (
   465→                    <ul className="divide-y divide-border">
   466→                      {topic.videos.map((v, vi) => {
   467→                        const qOpen = questionsOpenFor === v.id;
   468→                        const qCount = questionCounts[v.id] ?? 0;
   469→                        return (
   470→                        <li key={v.id} className="-mx-2 px-2 rounded">
   471→                          <div className="flex items-center gap-3 py-2.5 transition-colors hover:bg-muted/40">
   472→                            <span className="flex h-7 w-7 items-center justify-center rounded-md bg-accent/10 text-accent text-[11px] font-semibold shrink-0">
   473→                              {vi + 1}
   474→                            </span>
   475→                            <div className="min-w-0 flex-1">
   476→                              <p className="text-sm font-medium text-foreground truncate">{v.title}</p>
   477→                              <p className="text-xs text-muted-foreground truncate">
   478→                                {v.source === "UPLOAD"
   479→                                  ? "Uploaded video"
   480→                                  : v.source === "YOUTUBE"
   481→                                  ? "YouTube"
   482→                                  : "External URL"}
   483→                              </p>
   484→                            </div>
   485→                            {qCount > 0 && (
   486→                              <Badge
   487→                                variant="outline"
   488→                                className="gap-1 border-accent/30 bg-accent/5 text-accent text-[10px]"
   489→                                title={`${qCount} question${qCount === 1 ? "" : "s"}`}
   490→                              >
   491→                                <ListChecks className="h-3 w-3" />
   492→                                {qCount}
   493→                              </Badge>
   494→                            )}
   495→                            {v.isPublished ? (
   496→                              <Badge className="bg-success/15 text-success text-[10px]">{t("common.published")}</Badge>
   497→                            ) : (
   498→                              <Badge variant="outline" className="text-muted-foreground text-[10px]">
   499→                                Draft
   500→                              </Badge>
   501→                            )}
   502→                            <Button
   503→                              variant={qOpen ? "secondary" : "ghost"}
   504→                              size="icon"
   505→                              aria-label="Manage questions"
   506→                              aria-pressed={qOpen}
   507→                              className={cn(
   508→                                "h-7 w-7 shrink-0",
   509→                                qOpen ? "text-accent" : "text-muted-foreground hover:text-accent"
   510→                              )}
   511→                              onClick={() => setQuestionsOpenFor(qOpen ? null : v.id)}
   512→                            >
   513→                              <ListChecks className="h-3.5 w-3.5" />
   514→                            </Button>
   515→                            <Button
   516→                              variant="ghost"
   517→                              size="icon"
   518→                              aria-label="Edit video"
   519→                              className="h-7 w-7 text-muted-foreground hover:text-foreground"
   520→                              onClick={() => setEditVideo(v)}
   521→                            >
   522→                              <Pencil className="h-3.5 w-3.5" />
   523→                            </Button>
   524→                            <Button
   525→                              variant="ghost"
   526→                              size="icon"
   527→                              aria-label="Delete video"
   528→                              className="h-7 w-7 text-muted-foreground hover:text-destructive"
   529→                              onClick={() => setDeleteVideo(v)}

   531→                              <Trash2 className="h-3.5 w-3.5" />
