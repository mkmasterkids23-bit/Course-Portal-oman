   512→            <div className="space-y-3">
   513→              {[1, 2, 3, 4].map((i) => (
   514→                <Skeleton key={i} className="h-14 w-full rounded-lg" />
   515→              ))}
   516→            </div>
   517→          ) : activityData?.activities && activityData.activities.length > 0 ? (
   518→            <ol className="relative space-y-1 max-h-[28rem] overflow-y-auto scrollbar-thin pr-1">
   519→              {/* vertical spine */}

   521→                aria-hidden

   525→                <ActivityTimelineRow key={a.id} item={a} isFirst={i === 0} />

   528→          ) : (
   529→            <EmptyState
   530→              icon={<History className="h-5 w-5" />}
   531→              title="No activity yet"

   537→              }
   538→            />
   539→          )}
   540→        </CardContent>
   541→      </Card>
   542→    </div>
   543→  );

   641→  item,

   644→  item: ActivityItem;
   645→  isFirst: boolean;

   649→    <li

   659→        <div className="flex items-start justify-between gap-2">
   660→          <p className="text-sm font-medium text-foreground line-clamp-1">
   661→            {item.title}
   662→          </p>
   663→          <span className="text-[11px] text-muted-foreground whitespace-nowrap shrink-0">
   664→            {formatDateTime(item.timestamp)}
   665→          </span>

   677→
   678→function activityMeta(item: ActivityItem): {
   679→  icon: React.ReactNode;
   680→  badge?: React.ReactNode;
   681→} {
   682→  switch (item.type) {
   683→    case "video_completed":
   684→      return {
   685→        icon: <Video className="h-4 w-4" />,
   686→        badge: (
   687→          <Badge className="bg-accent/10 text-accent border-transparent">
   688→            <CheckCircle2 className="h-3 w-3" /> Completed
   689→          </Badge>
   690→        ),
   691→      };
   692→    case "assessment":
   693→      return {
   694→        icon: item.passed ? (
   695→          <CheckCircle2 className="h-4 w-4 text-success" />
   696→        ) : (
   697→          <AlertTriangle className="h-4 w-4 text-warning" />
   698→        ),
   699→        badge: (
   700→          <Badge
   701→            className={
   702→              item.passed
   703→                ? "bg-success/15 text-success border-transparent"
   704→                : "bg-warning/15 text-warning border-transparent"
   705→            }
   706→          >
   707→            {item.passed ? "Passed" : "Attempted"}
   708→            {typeof item.score === "number" ? ` · ${item.score}%` : ""}
   709→          </Badge>
   710→        ),
   711→      };
   712→    case "submission":
   713→      return {
   714→        icon: <FileText className="h-4 w-4" />,
   715→        badge: (
   716→          <Badge className="bg-primary/10 text-primary border-transparent">
   717→            <CheckCircle2 className="h-3 w-3" /> Submitted
   718→          </Badge>
   719→        ),
   720→      };
   721→    case "certificate":
   722→      return {
   723→        icon: <Award className="h-4 w-4 text-success" />,
   724→        badge: item.certificateNumber ? (
   725→          <Badge className="bg-success/15 text-success border-transparent">
   726→            <Award className="h-3 w-3" /> {item.certificateNumber}
   727→          </Badge>
   728→        ) : undefined,
   729→      };
   730→    default:
   731→      return {
   732→        icon: <XCircle className="h-4 w-4" />,
   733→      };
   734→  }
   735→}

   737→function StatCard({
   738→  icon, label, value, tint,
   739→}: {
   740→  icon: React.ReactNode;
   741→  label: string;
   742→  value: number | string;
   743→  tint: "accent" | "primary" | "success" | "warning";
   744→}) {
   745→  const tintClasses: Record<typeof tint, string> = {
   746→    accent: "bg-accent/10 text-accent ring-accent/20",
   747→    primary: "bg-primary/10 text-primary ring-primary/20",
   748→    success: "bg-success/15 text-success ring-success/20",
   749→    warning: "bg-warning/15 text-warning ring-warning/20",
   750→  };
   751→  return (
   752→    <Card className="p-5 shadow-sm transition-all duration-200 hover:shadow-md hover:-translate-y-0.5">
   753→      <div className="flex items-center gap-3">
   754→        <div className={`flex h-11 w-11 items-center justify-center rounded-xl ring-1 ${tintClasses[tint]}`}>
   755→          {icon}
   756→        </div>
   757→        <div className="min-w-0">
   758→          <p className="font-heading text-2xl font-bold text-foreground leading-none">{value}</p>
   759→          <p className="text-xs text-muted-foreground mt-1.5 truncate">{label}</p>
   760→        </div>
   761→      </div>
   762→    </Card>
   763→  );
   764→}
