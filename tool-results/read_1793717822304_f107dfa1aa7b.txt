   112→   112→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   113→   113→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   114→   114→    `/api/student/courses/${courseId}`
   115→   115→  );
   116→   116→
   117→   117→  // Pick the initial video: last watched, or first available.
   118→   118→  const initialVideoId = (() => {
   119→   119→    if (!data?.course) return null;
   120→   120→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   121→   121→    //    most recently updated progress entry in the topics tree.
   122→   122→    let candidate: VideoItem | null = null;
   123→   123→    for (const t of data.course.topics) {
   124→   124→      for (const v of t.videos) {
   125→   125→        if (v.progress && !v.progress.isCompleted) {
   126→   126→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   127→   127→            candidate = v;
   128→   128→          }
   129→   129→        }
   130→   130→      }
   131→   131→    }
   132→   132→    if (candidate) return candidate.id;
   133→   133→    // 2) First video of the first topic
   134→   134→    return data.course.topics[0]?.videos[0]?.id ?? null;
   135→   135→  })();
   136→   136→
   137→   137→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   138→   138→  // Reset user selection when courseId changes — "adjust state during render"
   139→   139→  // pattern (avoids setState-in-effect warnings).
   140→   140→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   141→   141→  if (courseId !== prevCourseId) {
   142→   142→    setPrevCourseId(courseId);
   143→   143→    setUserSelectedVideoId(null);
   144→   144→  }
   145→   145→  // Effective active video: explicit user selection overrides the auto-pick
   146→   146→  // (which is the last-watched or first video). When data loads and we have an
   147→   147→  // initial pick, this falls back to it without needing an effect.
   148→   148→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   149→   149→
   150→   150→  if (loading) {
   151→   151→    return (
   152→   152→      <div className="space-y-4">
   153→   153→        <Skeleton className="h-12 w-32 rounded-md" />
   154→   154→        <div className="grid gap-4 lg:grid-cols-[1fr_360px]">
   155→   155→          <Skeleton className="h-[480px] rounded-xl" />
   156→   156→          <Skeleton className="h-[480px] rounded-xl" />

   158→   158→      </div>

   163→   163→    return (

   191→   191→  const activeVideo =
   192→   192→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   193→   193→
   194→   194→  // Find the topic containing the active video
   195→   195→  const activeTopic = course.topics.find((t) =>
   196→   196→    t.videos.some((v) => v.id === activeVideo?.id)
   197→   197→  );
   198→   198→
   199→   199→  function handleProgressUpdate(updated: {
   200→   200→    videoId: string;
   201→   201→    progress: VideoProgressInfo;
   202→   202→    justCompleted: boolean;
   203→   203→  }) {

   205→   205→    const newCourse = {
   206→   206→      ...data.course,
   207→   207→      topics: data.course.topics.map((t) => ({
   208→   208→        ...t,
   209→   209→        videos: t.videos.map((v) =>
   210→   210→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   211→   211→        ),
   212→   212→      })),
   213→   213→      courseCompletion: (() => {

   215→   215→        const videos = data.course.topics.flatMap((t) => t.videos);

   217→   217→          (v) =>

   225→   225→          totalVideos: total,
   226→   226→          completedVideos: completed,
   227→   227→          percent,
   228→   228→          allCompleted: total > 0 && completed === total,

   230→   230→      })(),

   232→   232→    setData({ course: newCourse });
   233→   233→
   234→   234→    if (updated.justCompleted) {

   236→   236→        title: "Lesson completed",

   250→   250→  return (

   267→   267→            <Progress value={course.courseCompletion.percent} className="h-2 w-24" />

   272→   272→
   273→   273→      {/* Mobile: stacked. Desktop: split. */}

   284→   284→            <EmptyState
   285→   285→              icon={<VideoIcon className="h-5 w-5" />}
   286→   286→              title="No videos in this course"
   287→   287→              description="This course doesn't have any published video lessons yet."
   288→   288→            />
   289→   289→          )}
   290→   290→
   291→   291→          {/* Materials */}
   292→   292→          {activeTopic && activeTopic.materials.length > 0 && (
   293→   293→            <Card>
   294→   294→              <CardHeader>
   295→   295→                <CardTitle className="text-base flex items-center gap-2">
   296→   296→                  <FileText className="h-4 w-4 text-accent" />
   297→   297→                  Materials for {activeTopic.title}
   298→   298→                </CardTitle>
   299→   299→                <CardDescription>Reference documents for this topic</CardDescription>
   300→   300→              </CardHeader>
   301→   301→              <CardContent>
   302→   302→                <ul className="space-y-2">
   303→   303→                  {activeTopic.materials.map((m) => (
   304→   304→                    <MaterialRow key={m.id} material={m} />
   305→   305→                  ))}
   306→   306→                </ul>
   307→   307→              </CardContent>
   308→   308→            </Card>
   309→   309→          )}
   310→   310→
   311→   311→          {/* Assignment panel — completely hidden until ALL videos are completed.
   312→   312→              Before completion, show a locked placeholder that does NOT reveal

   314→   314→          {course.assignment && course.courseCompletion.allCompleted ? (

   326→   326→
   327→   327→          {/* All materials (course-level) */}
   328→   328→          {course.materials.length > 0 && (
   329→   329→            <Card>
   330→   330→              <CardHeader>

   334→   334→                </CardTitle>

   336→   336→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   337→   337→                </CardDescription>
   338→   338→              </CardHeader>
   339→   339→              <CardContent>
   340→   340→                <ul className="space-y-2">
   341→   341→                  {course.materials.map((m) => (
   342→   342→                    <MaterialRow key={m.id} material={m} />
   343→   343→                  ))}
   344→   344→                </ul>
   345→   345→              </CardContent>
   346→   346→            </Card>
   347→   347→          )}
   348→   348→        </div>
   349→   349→
   350→   350→        {/* Right: topic sidebar */}
   351→   351→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col">
   352→   352→          <CardHeader className="border-b border-border pb-3">
   353→   353→            <CardTitle className="text-base">Course Lessons</CardTitle>
   354→   354→            <CardDescription>
   355→   355→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   356→   356→            </CardDescription>
   357→   357→          </CardHeader>
   358→   358→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-12rem)] scrollbar-thin">
   359→   359→            {course.topics.length === 0 ? (
   360→   360→              <p className="text-sm text-muted-foreground p-6 text-center">
   361→   361→                No topics published yet.
   362→   362→              </p>
   363→   363→            ) : (
   364→   364→              <Accordion
   365→   365→                type="multiple"
   366→   366→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   367→   367→                className="px-3"
   368→   368→              >
   369→   369→                {course.topics.map((topic) => {
   370→   370→                  const completedInTopic = topic.videos.filter(
   371→   371→                    (v) => v.progress?.isCompleted
   372→   372→                  ).length;
   373→   373→                  const isActiveTopic = topic.id === activeTopic?.id;

   376→   376→                      <AccordionTrigger className="hover:no-underline px-2">

   388→   388→                          {topic.videos.map((video) => {
   389→   389→                            const isActive = video.id === activeVideo?.id;
   390→   390→                            const completed = video.progress?.isCompleted;
   391→   391→                            return (
   392→   392→                              <li key={video.id}>
   393→   393→                                <button
   394→   394→                                  onClick={() => setUserSelectedVideoId(video.id)}
   395→   395→                                  className={`flex w-full items-start gap-2 rounded-md px-2 py-2 text-left text-sm transition-colors ${
   396→   396→                                    isActive
   397→   397→                                      ? "bg-accent/10 text-accent"
   398→   398→                                      : "text-foreground hover:bg-muted"
   399→   399→                                  }`}
   400→   400→                                >
   401→   401→                                  <span className="mt-0.5 shrink-0">
   402→   402→                                    {completed ? (
   403→   403→                                      <CheckCircle2 className="h-4 w-4 text-success" />

   405→   405→                                      <PlayCircle className="h-4 w-4 text-accent" />

   417→   417→                                  </span>
   418→   418→                                </button>
   419→   419→                              </li>
   420→   420→                            );
   421→   421→                          })}
   422→   422→                        </ul>
   423→   423→                      </AccordionContent>
   424→   424→                    </AccordionItem>
   425→   425→                  );
   426→   426→                })}
   427→   427→              </Accordion>
   428→   428→            )}
   429→   429→          </CardContent>
   430→   430→        </Card>
   431→   431→      </div>
   432→   432→    </div>
   433→   433→  );
   434→   434→}

   437→   437→
   438→   438→function VideoPlayer({
   439→   439→  video,
   440→   440→  onProgress,
   441→   441→}: {
   442→   442→  video: VideoItem;
   443→   443→  onProgress: (update: {
   444→   444→    videoId: string;
   445→   445→    progress: VideoProgressInfo;
   446→   446→    justCompleted: boolean;
   447→   447→  }) => void;
   448→   448→}) {
   449→   449→  const videoRef = useRef<HTMLVideoElement>(null);
   450→   450→  const lastTimeRef = useRef<number>(0);
   451→   451→  const watchTimeRef = useRef<number>(0);
   452→   452→  const durationRef = useRef<number>(0);
   453→   453→  const lastPostRef = useRef<number>(0);
   454→   454→  const resumeAppliedRef = useRef<boolean>(false);
   455→   455→  const [expanded, setExpanded] = useState(false);
   456→   456→
   457→   457→  // YouTube completion timer
   458→   458→  const ytStartRef = useRef<number | null>(null);
   459→   459→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   460→   460→  const ytWatchSecondsRef = useRef<number>(0);
   461→   461→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   462→   462→
   463→   463→  // Local UI state
   464→   464→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   465→   465→    video.progress?.isCompleted || false
   466→   466→  );
   467→   467→  const [posting, setPosting] = useState(false);
   468→   468→
   469→   469→  // Reset on video change
   470→   470→  useEffect(() => {
   471→   471→    lastTimeRef.current = 0;
   472→   472→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   473→   473→    durationRef.current = video.duration || video.progress?.duration || 0;
   474→   474→    lastPostRef.current = 0;
   475→   475→    resumeAppliedRef.current = false;
   476→   476→    setLocalCompleted(video.progress?.isCompleted || false);
   477→   477→    setYtCanMarkComplete(false);
   478→   478→    ytWatchSecondsRef.current = 0;
   479→   479→    if (ytTimerRef.current) {
   480→   480→      clearInterval(ytTimerRef.current);
   481→   481→      ytTimerRef.current = null;
   482→   482→    }
   483→   483→  }, [video.id]);
   484→   484→
   485→   485→  // For HTML5 video: apply resume position once metadata is loaded.
   486→   486→  const handleLoadedMetadata = () => {
   487→   487→    const el = videoRef.current;
   488→   488→    if (!el) return;
   489→   489→    durationRef.current = el.duration || durationRef.current;
   490→   490→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   491→   491→      // resume within (duration - 2) so we don't immediately fire ended
   492→   492→      const target = Math.min(
   493→   493→        video.progress.lastPosition,
   494→   494→        Math.max(0, (el.duration || 0) - 3)
   495→   495→      );
   496→   496→      try {
   497→   497→        el.currentTime = target;
   498→   498→      } catch {}
   499→   499→      lastTimeRef.current = target;
   500→   500→      resumeAppliedRef.current = true;
   501→   501→    }
   502→   502→  };
   503→   503→
   504→   504→  // Core: timeupdate handler — accumulate real watch time.
   505→   505→  const handleTimeUpdate = () => {
   506→   506→    const el = videoRef.current;
   507→   507→    if (!el) return;
   508→   508→    const cur = el.currentTime;
   509→   509→    const last = lastTimeRef.current;
   510→   510→    const delta = cur - last;
   511→   511→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   512→   512→    // (large positive deltas) and rewinds (negative deltas).
   513→   513→    if (delta > 0 && delta <= 1.5) {
   514→   514→      watchTimeRef.current += delta;
   515→   515→    }
   516→   516→    // Update lastTime to current — even on seek, so next delta is small.
   517→   517→    lastTimeRef.current = cur;
   518→   518→
   519→   519→    // Throttled server POST every 5 seconds
   520→   520→    const now = Date.now();
   521→   521→    if (now - lastPostRef.current > 5000) {
   522→   522→      lastPostRef.current = now;
   523→   523→      void postProgress(false);
   524→   524→    }
   525→   525→  };
   526→   526→
   527→   527→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   528→   528→  const handleSeeked = () => {
   529→   529→    const el = videoRef.current;
   530→   530→    if (!el) return;
   531→   531→    lastTimeRef.current = el.currentTime;
   532→   532→  };
   533→   533→
   534→   534→  // Ended event — submit final progress with completion hint.
   535→   535→  const handleEnded = () => {
   536→   536→    void postProgress(true);
   537→   537→  };
   538→   538→
   539→   539→  const postProgress = useCallback(
   540→   540→    async (isCompletionHint: boolean) => {
   541→   541→      setPosting(true);
   542→   542→      try {
   543→   543→        const res = await apiPost<{
   544→   544→          progress: VideoProgressInfo;
   545→   545→          justCompleted: boolean;
   546→   546→          rejectedCompletion: boolean;
   547→   547→          certificateIssued: boolean;
   548→   548→          certificateId: string | null;
   549→   549→        }>("/api/student/progress", {
   550→   550→          videoId: video.id,
   551→   551→          lastPosition: videoRef.current?.currentTime || 0,
   552→   552→          watchTimeSeconds: watchTimeRef.current,
   553→   553→          duration: durationRef.current,
   554→   554→          isCompleted: isCompletionHint,
   555→   555→        });
   556→   556→        setLocalCompleted(res.progress.isCompleted);
   557→   557→        onProgress({
   558→   558→          videoId: video.id,
   559→   559→          progress: res.progress,
   560→   560→          justCompleted: res.justCompleted,
   561→   561→        });
   562→   562→        if (isCompletionHint && res.rejectedCompletion) {
   563→   563→          toast({
   564→   564→            title: "Completion not recorded",
   565→   565→            description:
   566→   566→              "You need to watch at least 85% of the video before it's marked complete.",
   567→   567→            variant: "destructive",
   568→   568→          });
   569→   569→        }
   570→   570→        // ─── Certificate auto-issued upon course completion ───
   571→   571→        if (res.certificateIssued && res.certificateId) {
   572→   572→          toast({
   573→   573→            title: "🎉 Certificate Issued!",
   574→   574→            description:
   575→   575→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   576→   576→          });
   577→   577→        }
   578→   578→      } catch {
   579→   579→        // apiPost already surfaces errors
   580→   580→      } finally {
   581→   581→        setPosting(false);
   582→   582→      }
   583→   583→    },
   584→   584→    [video.id, onProgress]
   585→   585→  );
   586→   586→
   587→   587→  // Manual "Mark complete" for YouTube videos
   588→   588→  const markYtComplete = async () => {

   596→   596→        certificateId: string | null;

   910→   910→function AssignmentPanel({
   911→   911→  course,
   912→   912→  onSubmitted,
   913→   913→}: {
   914→   914→  course: CourseDetail;
   915→   915→  onSubmitted: () => void;
   916→   916→}) {
   917→   917→  const navigate = useAppStore((s) => s.navigate);
   918→   918→  const [uploading, setUploading] = useState(false);
   919→   919→  const [selectedFile, setSelectedFile] = useState<File | null>(null);
   920→   920→
   921→   921→  const assignment = course.assignment!;
   922→   922→  const submission = course.submission;
   923→   923→  const completed = course.courseCompletion.allCompleted;
   924→   924→
   925→   925→  async function handleSubmit() {
   926→   926→    if (!selectedFile) {

   931→   931→    try {
   932→   932→      const fd = new FormData();

   934→   934→      const res = await apiUpload<{
   935→   935→        certificateIssued: boolean;

   937→   937→      }>(`/api/student/assignments/${assignment.id}/submit`, fd);
   938→   938→      setSelectedFile(null);

   940→   940→      if (res.certificateIssued) {
   941→   941→        toast({

   946→   946→        toast({
