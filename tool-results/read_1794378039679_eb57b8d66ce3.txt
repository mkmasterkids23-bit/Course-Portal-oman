   154→   154→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   155→   155→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   156→   156→    `/api/student/courses/${courseId}`
   157→   157→  );
   158→   158→
   159→   159→  // Pick the initial video: last watched, or first available.
   160→   160→  const initialVideoId = (() => {
   161→   161→    if (!data?.course) return null;
   162→   162→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   163→   163→    //    most recently updated progress entry in the topics tree.
   164→   164→    let candidate: VideoItem | null = null;
   165→   165→    for (const t of data.course.topics) {
   166→   166→      for (const v of t.videos) {
   167→   167→        if (v.progress && !v.progress.isCompleted) {
   168→   168→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   169→   169→            candidate = v;
   170→   170→          }
   171→   171→        }
   172→   172→      }
   173→   173→    }
   174→   174→    if (candidate) return candidate.id;
   175→   175→    // 2) First video of the first topic
   176→   176→    return data.course.topics[0]?.videos[0]?.id ?? null;
   177→   177→  })();
   178→   178→
   179→   179→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   180→   180→  // Reset user selection when courseId changes — "adjust state during render"
   181→   181→  // pattern (avoids setState-in-effect warnings).
   182→   182→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   183→   183→  if (courseId !== prevCourseId) {
   184→   184→    setPrevCourseId(courseId);
   185→   185→    setUserSelectedVideoId(null);
   186→   186→  }
   187→   187→  // Effective active video: explicit user selection overrides the auto-pick
   188→   188→  // (which is the last-watched or first video). When data loads and we have an
   189→   189→  // initial pick, this falls back to it without needing an effect.
   190→   190→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   191→   191→
   192→   192→  if (loading) {
   193→   193→    return (
   194→   194→      <div className="space-y-5">
   195→   195→        <Skeleton className="h-10 w-32 rounded-lg" />
   196→   196→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   197→   197→          <div className="space-y-4">
   198→   198→            <Skeleton className="h-[420px] rounded-xl" />
   199→   199→            <Skeleton className="h-24 rounded-xl" />
   200→   200→          </div>
   201→   201→          <Skeleton className="h-[600px] rounded-xl" />
   202→   202→        </div>
   203→   203→      </div>
   204→   204→    );
   205→   205→  }
   206→   206→
   207→   207→  if (error) {
   208→   208→    return (
   209→   209→      <div className="space-y-4">
   210→   210→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   211→   211→          <ArrowLeft className="h-4 w-4" /> Back to courses
   212→   212→        </Button>
   213→   213→        <EmptyState
   214→   214→          icon={<AlertTriangle className="h-5 w-5" />}
   215→   215→          title="Couldn't open course"
   216→   216→          description={error}
   217→   217→          action={<Button onClick={onBack}>Back to courses</Button>}
   218→   218→        />
   219→   219→      </div>
   220→   220→    );
   221→   221→  }
   222→   222→
   223→   223→  if (!data?.course) {
   224→   224→    return (
   225→   225→      <EmptyState
   226→   226→        icon={<PlayCircle className="h-5 w-5" />}
   227→   227→        title="Course not found"
   228→   228→        description="This course may have been removed or is not accessible."
   229→   229→        action={<Button onClick={onBack}>Back to courses</Button>}
   230→   230→      />
   231→   231→    );
   232→   232→  }
   233→   233→
   234→   234→  const course = data.course;
   235→   235→  const allVideos = course.topics.flatMap((t) => t.videos);
   236→   236→  const activeVideo =
   237→   237→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   238→   238→
   239→   239→  // Find the topic containing the active video
   240→   240→  const activeTopic = course.topics.find((t) =>
   241→   241→    t.videos.some((v) => v.id === activeVideo?.id)
   242→   242→  );
   243→   243→
   244→   244→  // Build a flat ordered list for prev/next navigation
   245→   245→  const flatVideoIds = allVideos.map((v) => v.id);
   246→   246→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   247→   247→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   248→   248→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   249→   249→
   250→   250→  function handleProgressUpdate(updated: {
   251→   251→    videoId: string;
   252→   252→    progress: VideoProgressInfo;
   253→   253→    justCompleted: boolean;
   254→   254→  }) {
   255→   255→    if (!data) return;
   256→   256→    const newCourse = {
   257→   257→      ...data.course,
   258→   258→      topics: data.course.topics.map((t) => ({
   259→   259→        ...t,
   260→   260→        videos: t.videos.map((v) =>
   261→   261→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   262→   262→        ),
   263→   263→      })),
   264→   264→      courseCompletion: (() => {
   265→   265→        // Recompute local snapshot
   266→   266→        const videos = data.course.topics.flatMap((t) => t.videos);
   267→   267→        const completed = videos.filter(
   268→   268→          (v) =>
   269→   269→            v.id === updated.videoId
   270→   270→              ? updated.progress.isCompleted
   271→   271→              : v.progress?.isCompleted
   272→   272→        ).length;
   273→   273→        const total = videos.length;
   274→   274→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   275→   275→        return {
   276→   276→          totalVideos: total,
   277→   277→          completedVideos: completed,
   278→   278→          percent,
   279→   279→          allCompleted: total > 0 && completed === total,
   280→   280→        };
   281→   281→      })(),
   282→   282→    };
   283→   283→    setData({ course: newCourse });
   284→   284→
   285→   285→    if (updated.justCompleted) {
   286→   286→      toast({
   287→   287→        title: "Lesson completed",
   288→   288→        description: "Great work! Your progress has been saved.",
   289→   289→      });
   290→   290→    }
   291→   291→  }
   292→   292→
   293→   293→  async function handleSubmissionSuccess() {
   294→   294→    await refetch();
   295→   295→    toast({
   296→   296→      title: "Assignment submitted",
   297→   297→      description: "Your submission has been received.",
   298→   298→    });
   299→   299→  }
   300→   300→
   301→   301→  return (
   302→   302→    <div className="space-y-5">
   303→   303→      {/* ─── Premium header ─── */}
   304→   304→      <div className="rounded-xl border border-border bg-card p-4 md:p-5 shadow-sm">
   305→   305→        <div className="flex flex-wrap items-center justify-between gap-3">
   306→   306→          <div className="flex items-center gap-3 min-w-0 flex-1">
   307→   307→            <Button variant="outline" size="icon" onClick={onBack} className="shrink-0 h-9 w-9" aria-label="Back to courses">
   308→   308→              <ArrowLeft className="h-4 w-4" />
   309→   309→            </Button>
   310→   310→            <div className="min-w-0 flex-1">
   311→   311→              <div className="flex items-center gap-2 mb-0.5">
   312→   312→                {course.category && (
   313→   313→                  <Badge variant="outline" className="text-[11px] border-border text-muted-foreground">
   314→   314→                    {course.category}
   315→   315→                  </Badge>
   316→   316→                )}
   317→   317→                <Badge
   318→   318→                  className={
   319→   319→                    course.accessType === "FREE"
   320→   320→                      ? "bg-success/15 text-success border-transparent"
   321→   321→                      : "bg-primary text-primary-foreground border-transparent"
   322→   322→                  }
   323→   323→                >
   324→   324→                  {course.accessType === "FREE" ? "Free" : "Paid"}
   325→   325→                </Badge>
   326→   326→              </div>
   327→   327→              <h1 className="font-heading text-lg md:text-xl font-semibold text-foreground truncate">
   328→   328→                {course.title}
   329→   329→              </h1>
   330→   330→            </div>
   331→   331→          </div>
   332→   332→          <div className="flex items-center gap-4 shrink-0">
   333→   333→            <div className="hidden sm:flex flex-col items-end gap-0.5">
   334→   334→              <span className="text-xs font-medium text-foreground">
   335→   335→                {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} lessons
   336→   336→              </span>
   337→   337→              <span className="text-[11px] text-muted-foreground">completed</span>
   338→   338→            </div>
   339→   339→            <div className="flex items-center gap-2.5 min-w-[160px]">
   340→   340→              <Progress
   341→   341→                value={course.courseCompletion.percent}
   342→   342→                className={`h-2 w-32 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   343→   343→              />
   344→   344→              <span className="text-sm font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   345→   345→            </div>
   346→   346→          </div>
   347→   347→        </div>
   348→   348→      </div>
   349→   349→
   350→   350→      {/* ─── Mobile: stacked. Desktop: split. ─── */}
   351→   351→      <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   352→   352→        {/* Left: player + materials + assignment */}
   353→   353→        <div className="space-y-5 min-w-0">
   354→   354→          {activeVideo ? (
   355→   355→            <VideoPlayer
   356→   356→              key={activeVideo.id}
   357→   357→              video={activeVideo}
   358→   358→              topicTitle={activeTopic?.title || null}
   359→   359→              onProgress={handleProgressUpdate}
   360→   360→              prevVideo={prevVideo}
   361→   361→              nextVideo={nextVideo}
   362→   362→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   363→   363→              refetchCourse={refetch}
   364→   364→            />
   365→   365→          ) : (
   366→   366→            <EmptyState
   367→   367→              icon={<VideoIcon className="h-5 w-5" />}
   368→   368→              title="No videos in this course"
   369→   369→              description="This course doesn't have any published video lessons yet."
   370→   370→            />
   371→   371→          )}
   372→   372→
   373→   373→          {/* Topic materials */}
   374→   374→          {activeTopic && activeTopic.materials.length > 0 && (
   375→   375→            <Card className="shadow-sm">
   376→   376→              <CardHeader>
   377→   377→                <CardTitle className="text-base flex items-center gap-2">
   378→   378→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   379→   379→                    <FileText className="h-4 w-4" />
   380→   380→                  </span>
   381→   381→                  Materials · {activeTopic.title}
   382→   382→                </CardTitle>
   383→   383→                <CardDescription>Reference documents for this topic</CardDescription>
   384→   384→              </CardHeader>
   385→   385→              <CardContent>
   386→   386→                <ul className="space-y-2">
   387→   387→                  {activeTopic.materials.map((m) => (
   388→   388→                    <MaterialRow key={m.id} material={m} />
   389→   389→                  ))}
   390→   390→                </ul>
   391→   391→              </CardContent>
   392→   392→            </Card>
   393→   393→          )}
   394→   394→
   395→   395→          {/* Assignment panel — completely hidden until ALL videos are completed.
   396→   396→              Before completion, show a locked placeholder that does NOT reveal
   397→   397→              the assignment title, description, or instructions. */}
   398→   398→          {course.assignment && course.courseCompletion.allCompleted ? (
   399→   399→            <AssignmentPanel
   400→   400→              course={course}
   401→   401→              onSubmitted={handleSubmissionSuccess}
   402→   402→            />
   403→   403→          ) : course.assignment ? (
   404→   404→            <LockedAssignmentCard
   405→   405→              completedVideos={course.courseCompletion.completedVideos}
   406→   406→              totalVideos={course.courseCompletion.totalVideos}
   407→   407→              percent={course.courseCompletion.percent}
   408→   408→            />
   409→   409→          ) : null}
   410→   410→
   411→   411→          {/* All materials (course-level) */}
   412→   412→          {course.materials.length > 0 && (
   413→   413→            <Card className="shadow-sm">
   414→   414→              <CardHeader>
   415→   415→                <CardTitle className="text-base flex items-center gap-2">
   416→   416→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   417→   417→                    <FileText className="h-4 w-4" />
   418→   418→                  </span>
   419→   419→                  All Course Materials
   420→   420→                </CardTitle>
   421→   421→                <CardDescription>
   422→   422→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   423→   423→                </CardDescription>
   424→   424→              </CardHeader>
   425→   425→              <CardContent>
   426→   426→                <ul className="space-y-2">
   427→   427→                  {course.materials.map((m) => (
   428→   428→                    <MaterialRow key={m.id} material={m} />
   429→   429→                  ))}
   430→   430→                </ul>
   431→   431→              </CardContent>
   432→   432→            </Card>
   433→   433→          )}
   434→   434→        </div>
   435→   435→
   436→   436→        {/* Right: course sidebar */}
   437→   437→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col shadow-sm overflow-hidden">
   438→   438→          <CardHeader className="border-b border-border bg-gradient-to-br from-light-teal/60 to-card">
   439→   439→            <CardTitle className="text-base flex items-center gap-2">
   440→   440→              <BookOpen className="h-4 w-4 text-accent" />
   441→   441→              Course Lessons
   442→   442→            </CardTitle>
   443→   443→            <CardDescription className="mt-1">
   444→   444→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   445→   445→            </CardDescription>
   446→   446→            <div className="mt-2">
   447→   447→              <div className="flex justify-between text-xs mb-1">
   448→   448→                <span className="text-muted-foreground">Progress</span>
   449→   449→                <span className="font-semibold text-foreground">{course.courseCompletion.percent}%</span>
   450→   450→              </div>
   451→   451→              <Progress
   452→   452→                value={course.courseCompletion.percent}
   453→   453→                className={`h-1.5 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   454→   454→              />
   455→   455→            </div>
   456→   456→          </CardHeader>
   457→   457→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-15rem)] scrollbar-thin">
   458→   458→            {course.topics.length === 0 ? (
   459→   459→              <p className="text-sm text-muted-foreground p-6 text-center">
   460→   460→                No topics published yet.
   461→   461→              </p>
   462→   462→            ) : (
   463→   463→              <Accordion
   464→   464→                type="multiple"
   465→   465→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   466→   466→                className="px-3 py-2"
   467→   467→              >
   468→   468→                {course.topics.map((topic, topicIdx) => {
   469→   469→                  const completedInTopic = topic.videos.filter(
   470→   470→                    (v) => v.progress?.isCompleted
   471→   471→                  ).length;
   472→   472→                  const isActiveTopic = topic.id === activeTopic?.id;

   579→   579→
   580→   580→function VideoPlayer({

   582→   582→  topicTitle,
   583→   583→  onProgress,
   584→   584→  prevVideo,
   585→   585→  nextVideo,
   586→   586→  onNavigateLesson,
   587→   587→  refetchCourse,
   588→   588→}: {
   589→   589→  video: VideoItem;
   590→   590→  topicTitle: string | null;
   591→   591→  onProgress: (update: {
   592→   592→    videoId: string;
   593→   593→    progress: VideoProgressInfo;
   594→   594→    justCompleted: boolean;
   595→   595→  }) => void;
   596→   596→  prevVideo: VideoItem | null;
   597→   597→  nextVideo: VideoItem | null;
   598→   598→  onNavigateLesson: (id: string) => void;
   599→   599→  refetchCourse?: () => void;
   600→   600→}) {
   601→   601→  const videoRef = useRef<HTMLVideoElement>(null);
   602→   602→  const lastTimeRef = useRef<number>(0);
   603→   603→  const watchTimeRef = useRef<number>(0);
   604→   604→  const durationRef = useRef<number>(0);
   605→   605→  const lastPostRef = useRef<number>(0);
   606→   606→  const resumeAppliedRef = useRef<boolean>(false);
   607→   607→  const [expanded, setExpanded] = useState(false);
   608→   608→
   609→   609→  // ─── Video Assessment state ───
   610→   610→  // Fetches the questions for THIS video (without correctAnswer — security).
   611→   611→  // The hook is unconditional (rules of hooks); the URL just changes per video.
   612→   612→  const questionsUrl = `/api/student/videos/${video.id}/questions`;
   613→   613→  const {
   614→   614→    data: questionsData,
   615→   615→    loading: questionsLoading,
   616→   616→    refetch: refetchQuestions,
   617→   617→  } = useApiFetch<{
   618→   618→    questions: VideoQuestionData[];
   619→   619→    assessment: VideoAssessmentSummary | null;
   620→   620→  }>(questionsUrl);
   621→   621→
   622→   622→  // Currently-displayed question popup (null = no popup shown).
   623→   623→  const [activeQuestion, setActiveQuestion] = useState<VideoQuestionData | null>(null);
   624→   624→  // Map of questionId → selected option index (answers given this session).
   625→   625→  const [answeredMap, setAnsweredMap] = useState<Map<string, number>>(new Map());
   626→   626→  // Currently-selected option in the active popup (null = nothing selected yet).
   627→   627→  const [selectedAnswer, setSelectedAnswer] = useState<number | null>(null);
   628→   628→  // Brief "Answer saved" confirmation shown after submitting a popup answer.
   629→   629→  const [showFeedback, setShowFeedback] = useState(false);
   630→   630→  // Result returned from POST /assessment — when set, the result overlay shows.
   631→   631→  const [assessmentResult, setAssessmentResult] = useState<AssessmentResult | null>(null);
   632→   632→  // True while the assessment POST is in flight.
   633→   633→  const [submittingAssessment, setSubmittingAssessment] = useState(false);
   634→   634→  // Timer ref for the post-submit feedback delay (cleared on unmount/video change).
   635→   635→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   636→   636→
   637→   637→  // YouTube completion timer
   638→   638→  const ytStartRef = useRef<number | null>(null);
   639→   639→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   640→   640→  const ytWatchSecondsRef = useRef<number>(0);
   641→   641→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   642→   642→
   643→   643→  // Local UI state
   644→   644→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   645→   645→    video.progress?.isCompleted || false
   646→   646→  );
   647→   647→  const [posting, setPosting] = useState(false);
   648→   648→
   649→   649→  // Are question popups active for this viewing session?
   650→   650→  // They are NOT active if the student has already passed the assessment —
   651→   651→  // in that case the video plays freely and the "Passed" badge is shown.
   652→   652→  const questionsActive = Boolean(
   653→   653→    questionsData &&
   654→   654→      questionsData.questions.length > 0 &&
   655→   655→      questionsData.assessment?.passed !== true
   656→   656→  );
   657→   657→
   658→   658→  // Reset on video change
   659→   659→  useEffect(() => {
   660→   660→    lastTimeRef.current = 0;
   661→   661→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   662→   662→    durationRef.current = video.duration || video.progress?.duration || 0;
   663→   663→    lastPostRef.current = 0;
   664→   664→    resumeAppliedRef.current = false;
   665→   665→    setLocalCompleted(video.progress?.isCompleted || false);
   666→   666→    setYtCanMarkComplete(false);
   667→   667→    ytWatchSecondsRef.current = 0;
   668→   668→    if (ytTimerRef.current) {
   669→   669→      clearInterval(ytTimerRef.current);
   670→   670→      ytTimerRef.current = null;
   671→   671→    }
   672→   672→    // ─── Reset assessment state ───
   673→   673→    setActiveQuestion(null);
   674→   674→    setSelectedAnswer(null);
   675→   675→    setAnsweredMap(new Map());
   676→   676→    setShowFeedback(false);
   677→   677→    setAssessmentResult(null);
   678→   678→    setSubmittingAssessment(false);
   679→   679→    if (feedbackTimerRef.current) {
   680→   680→      clearTimeout(feedbackTimerRef.current);
   681→   681→      feedbackTimerRef.current = null;
   682→   682→    }
   683→   683→  }, [video.id]);
   684→   684→
   685→   685→  // ─── Missed-question catch-up ───
   686→   686→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   687→   687→  // the student answers a question. If the playhead has already moved past an
   688→   688→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   689→   689→  // the questions fetch landed late), pause and surface that question now.
   690→   690→  useEffect(() => {
   691→   691→    if (!questionsActive || !questionsData) return;
   692→   692→    if (activeQuestion || assessmentResult || showFeedback || submittingAssessment) return;
   693→   693→    const el = videoRef.current;
   694→   694→    if (!el) return;
   695→   695→    const cur = el.currentTime;
   696→   696→    const missed = questionsData.questions.find(
   697→   697→      (q) => !answeredMap.has(q.id) && cur >= q.timestamp - 0.5
   698→   698→    );
   699→   699→    if (missed && !el.paused) {
   700→   700→      try {
   701→   701→        el.pause();
   702→   702→        lastTimeRef.current = el.currentTime;
   703→   703→      } catch {}
   704→   704→      setActiveQuestion(missed);
   705→   705→      setSelectedAnswer(null);
   706→   706→    }
   707→   707→  }, [
   708→   708→    questionsActive,
   709→   709→    questionsData,
   710→   710→    activeQuestion,
   711→   711→    assessmentResult,
   712→   712→    showFeedback,
   713→   713→    submittingAssessment,
   714→   714→    answeredMap,
   715→   715→  ]);
   716→   716→
   717→   717→  // For HTML5 video: apply resume position once metadata is loaded.
   718→   718→  const handleLoadedMetadata = () => {
   719→   719→    const el = videoRef.current;
   720→   720→    if (!el) return;
   721→   721→    durationRef.current = el.duration || durationRef.current;
   722→   722→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   723→   723→      // resume within (duration - 2) so we don't immediately fire ended
   724→   724→      const target = Math.min(
   725→   725→        video.progress.lastPosition,
   726→   726→        Math.max(0, (el.duration || 0) - 3)
   727→   727→      );
   728→   728→      try {
   729→   729→        el.currentTime = target;
   730→   730→      } catch {}
   731→   731→      lastTimeRef.current = target;
   732→   732→      resumeAppliedRef.current = true;
   733→   733→    }
   734→   734→  };
   735→   735→
   736→   736→  // Core: timeupdate handler — accumulate real watch time.
   737→   737→  const handleTimeUpdate = () => {
   738→   738→    const el = videoRef.current;
   739→   739→    if (!el) return;
   740→   740→    const cur = el.currentTime;
   741→   741→    const last = lastTimeRef.current;
   742→   742→
   743→   743→    // ─── Question trigger check ───
   744→   744→    // Pause and surface a popup the moment the playhead crosses an unanswered
   745→   745→    // question's timestamp going forward. We require BOTH:
   746→   746→    //   - cur >= q.timestamp    (we're at or past it)
   747→   747→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   748→   748→    //                            this is a forward crossing, not a re-tick while
   749→   749→    //                            paused, and not a backward seek landing past it)
   750→   750→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   751→   751→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   752→   752→    if (
   753→   753→      questionsActive &&
   754→   754→      !activeQuestion &&
   755→   755→      !assessmentResult &&
   756→   756→      !showFeedback &&
   757→   757→      !submittingAssessment &&
   758→   758→      questionsData
   759→   759→    ) {
   760→   760→      for (const q of questionsData.questions) {
   761→   761→        if (answeredMap.has(q.id)) continue;
   762→   762→        if (cur >= q.timestamp && last < q.timestamp) {
   763→   763→          try {
   764→   764→            el.pause();
   765→   765→            lastTimeRef.current = cur;
   766→   766→          } catch {}
   767→   767→          setActiveQuestion(q);
   768→   768→          setSelectedAnswer(null);
   769→   769→          return; // do NOT accumulate watch time for this tick
   770→   770→        }
   771→   771→      }
   772→   772→    }
   773→   773→
   774→   774→    // Watch time tracking — only counts normal forward playback deltas.
   775→   775→    const delta = cur - last;
   776→   776→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   777→   777→    // (large positive deltas) and rewinds (negative deltas).
   778→   778→    if (delta > 0 && delta <= 1.5) {
   779→   779→      watchTimeRef.current += delta;
   780→   780→    }
   781→   781→    // Update lastTime to current — even on seek, so next delta is small.
   782→   782→    lastTimeRef.current = cur;
   783→   783→
   784→   784→    // Throttled server POST every 5 seconds
   785→   785→    const now = Date.now();
   786→   786→    if (now - lastPostRef.current > 5000) {
   787→   787→      lastPostRef.current = now;
   788→   788→      void postProgress(false);
   789→   789→    }
   790→   790→  };
   791→   791→
   792→   792→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   793→   793→  // Also catches forward-seek-past-question: if the student scrubs the
   794→   794→  // progress bar past an unanswered question, pause and surface that question
   795→   795→  // so they can't skip it.
   796→   796→  const handleSeeked = () => {
   797→   797→    const el = videoRef.current;
   798→   798→    if (!el) return;
   799→   799→    const newTime = el.currentTime;
   800→   800→    lastTimeRef.current = newTime;
   801→   801→
   802→   802→    if (
   803→   803→      questionsActive &&
   804→   804→      !activeQuestion &&
   805→   805→      !assessmentResult &&
   806→   806→      !showFeedback &&
   807→   807→      !submittingAssessment &&
   808→   808→      questionsData
   809→   809→    ) {
   810→   810→      // Find the earliest unanswered question we've now skipped past.
   811→   811→      const skipped = questionsData.questions.find(
   812→   812→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
   813→   813→      );
   814→   814→      if (skipped) {
   815→   815→        try {
   816→   816→          el.pause();
   817→   817→        } catch {}
   818→   818→        setActiveQuestion(skipped);
   819→   819→        setSelectedAnswer(null);
   820→   820→      }
   821→   821→    }
   822→   822→  };
   823→   823→
   824→   824→  // Ended event — submit final progress with completion hint.
   825→   825→  const handleEnded = () => {
   826→   826→    void postProgress(true);
   827→   827→  };
   828→   828→
   829→   829→  const postProgress = useCallback(
   830→   830→    async (isCompletionHint: boolean) => {
   831→   831→      setPosting(true);
   832→   832→      try {
   833→   833→        const res = await apiPost<{
   834→   834→          progress: VideoProgressInfo;
   835→   835→          justCompleted: boolean;
   836→   836→          rejectedCompletion: boolean;
   837→   837→          certificateIssued: boolean;
   838→   838→          certificateId: string | null;
   839→   839→        }>("/api/student/progress", {
   840→   840→          videoId: video.id,
   841→   841→          lastPosition: videoRef.current?.currentTime || 0,
   842→   842→          watchTimeSeconds: watchTimeRef.current,
   843→   843→          duration: durationRef.current,
   844→   844→          isCompleted: isCompletionHint,
   845→   845→        });
   846→   846→        setLocalCompleted(res.progress.isCompleted);
   847→   847→        onProgress({
   848→   848→          videoId: video.id,
   849→   849→          progress: res.progress,
   850→   850→          justCompleted: res.justCompleted,
   851→   851→        });
   852→   852→        if (isCompletionHint && res.rejectedCompletion) {
   853→   853→          toast({
   854→   854→            title: "Completion not recorded",
   855→   855→            description:
   856→   856→              "You need to watch at least 85% of the video before it's marked complete.",
   857→   857→            variant: "destructive",
   858→   858→          });
   859→   859→        }
   860→   860→        // ─── Certificate auto-issued upon course completion ───
   861→   861→        if (res.certificateIssued && res.certificateId) {
   862→   862→          toast({
   863→   863→            title: "🎉 Certificate Issued!",
   864→   864→            description:
   865→   865→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   866→   866→          });
   867→   867→        }
   868→   868→      } catch {
   869→   869→        // apiPost already surfaces errors
   870→   870→      } finally {
   871→   871→        setPosting(false);
   872→   872→      }
   873→   873→    },
   874→   874→    [video.id, onProgress]
   875→   875→  );
   876→   876→
   877→   877→  // Manual "Mark complete" for YouTube videos
   878→   878→  const markYtComplete = async () => {
   879→   879→    setPosting(true);
   880→   880→    try {
   881→   881→      const res = await apiPost<{
   882→   882→        progress: VideoProgressInfo;
   883→   883→        justCompleted: boolean;
   884→   884→        rejectedCompletion: boolean;
   885→   885→        certificateIssued: boolean;
   886→   886→        certificateId: string | null;
   887→   887→      }>("/api/student/progress", {
   888→   888→        videoId: video.id,
   889→   889→        lastPosition: durationRef.current,
   890→   890→        watchTimeSeconds: ytWatchSecondsRef.current,
   891→   891→        duration: durationRef.current,
   892→   892→        isCompleted: true,
   893→   893→      });
   894→   894→      setLocalCompleted(res.progress.isCompleted);
   895→   895→      onProgress({
   896→   896→        videoId: video.id,
   897→   897→        progress: res.progress,
   898→   898→        justCompleted: res.justCompleted,
   899→   899→      });
   900→   900→      if (res.justCompleted) {
   901→   901→        toast({
   902→   902→          title: "Lesson completed",
   903→   903→          description: "Great work! Your progress has been saved.",
   904→   904→        });
   905→   905→      }
   906→   906→      if (res.rejectedCompletion) {
   907→   907→        toast({
   908→   908→          title: "Completion not recorded",
   909→   909→          description:
   910→   910→            "You need to keep the video playing for at least 85% of its duration.",
   911→   911→          variant: "destructive",
   912→   912→        });
   913→   913→      }
   914→   914→      // Certificate auto-issued upon course completion
   915→   915→      if (res.certificateIssued && res.certificateId) {
   916→   916→        toast({
   917→   917→          title: "🎉 Certificate Issued!",
   918→   918→          description:
   919→   919→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   920→   920→        });
   921→   921→      }
   922→   922→    } catch {
   923→   923→    } finally {
   924→   924→      setPosting(false);
   925→   925→    }
   926→   926→  };

   928→   928→  /* ──────────────────────────────────────────────────────────────
   929→   929→     Video Assessment — submit, retake, dismiss handlers
   930→   930→     ────────────────────────────────────────────────────────────── */
