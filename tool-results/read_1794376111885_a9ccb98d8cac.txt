   112→   112→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   113→   113→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   114→   114→    `/api/student/courses/${courseId}`
   115→   115→  );
   116→   116→
   117→   117→  // Pick the initial video: last watched, or first available.
   118→   118→  const initialVideoId = (() => {
   119→   119→    if (!data?.course) return null;
   120→   120→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   121→   121→    //    most recently updated progress entry in the topics tree.
   122→   122→    let candidate: VideoItem | null = null;
   123→   123→    for (const t of data.course.topics) {
   124→   124→      for (const v of t.videos) {
   125→   125→        if (v.progress && !v.progress.isCompleted) {
   126→   126→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   127→   127→            candidate = v;
   128→   128→          }
   129→   129→        }
   130→   130→      }
   131→   131→    }
   132→   132→    if (candidate) return candidate.id;
   133→   133→    // 2) First video of the first topic
   134→   134→    return data.course.topics[0]?.videos[0]?.id ?? null;
   135→   135→  })();
   136→   136→
   137→   137→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   138→   138→  // Reset user selection when courseId changes — "adjust state during render"
   139→   139→  // pattern (avoids setState-in-effect warnings).
   140→   140→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   141→   141→  if (courseId !== prevCourseId) {
   142→   142→    setPrevCourseId(courseId);
   143→   143→    setUserSelectedVideoId(null);
   144→   144→  }
   145→   145→  // Effective active video: explicit user selection overrides the auto-pick
   146→   146→  // (which is the last-watched or first video). When data loads and we have an
   147→   147→  // initial pick, this falls back to it without needing an effect.
   148→   148→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   149→   149→
   150→   150→  if (loading) {
   151→   151→    return (
   152→   152→      <div className="space-y-5">
   153→   153→        <Skeleton className="h-10 w-32 rounded-lg" />
   154→   154→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   155→   155→          <div className="space-y-4">
   156→   156→            <Skeleton className="h-[420px] rounded-xl" />
   157→   157→            <Skeleton className="h-24 rounded-xl" />
   158→   158→          </div>
   159→   159→          <Skeleton className="h-[600px] rounded-xl" />
   160→   160→        </div>
   161→   161→      </div>
   162→   162→    );
   163→   163→  }
   164→   164→
   165→   165→  if (error) {
   166→   166→    return (
   167→   167→      <div className="space-y-4">
   168→   168→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   169→   169→          <ArrowLeft className="h-4 w-4" /> Back to courses
   170→   170→        </Button>
   171→   171→        <EmptyState
   172→   172→          icon={<AlertTriangle className="h-5 w-5" />}
   173→   173→          title="Couldn't open course"
   174→   174→          description={error}
   175→   175→          action={<Button onClick={onBack}>Back to courses</Button>}
   176→   176→        />
   177→   177→      </div>
   178→   178→    );
   179→   179→  }
   180→   180→
   181→   181→  if (!data?.course) {
   182→   182→    return (
   183→   183→      <EmptyState
   184→   184→        icon={<PlayCircle className="h-5 w-5" />}
   185→   185→        title="Course not found"
   186→   186→        description="This course may have been removed or is not accessible."
   187→   187→        action={<Button onClick={onBack}>Back to courses</Button>}
   188→   188→      />
   189→   189→    );
   190→   190→  }
   191→   191→
   192→   192→  const course = data.course;
   193→   193→  const allVideos = course.topics.flatMap((t) => t.videos);
   194→   194→  const activeVideo =
   195→   195→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   196→   196→
   197→   197→  // Find the topic containing the active video
   198→   198→  const activeTopic = course.topics.find((t) =>
   199→   199→    t.videos.some((v) => v.id === activeVideo?.id)
   200→   200→  );
   201→   201→
   202→   202→  // Build a flat ordered list for prev/next navigation
   203→   203→  const flatVideoIds = allVideos.map((v) => v.id);
   204→   204→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   205→   205→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   206→   206→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   207→   207→
   208→   208→  function handleProgressUpdate(updated: {
   209→   209→    videoId: string;
   210→   210→    progress: VideoProgressInfo;
   211→   211→    justCompleted: boolean;
   212→   212→  }) {
   213→   213→    if (!data) return;
   214→   214→    const newCourse = {
   215→   215→      ...data.course,
   216→   216→      topics: data.course.topics.map((t) => ({
   217→   217→        ...t,
   218→   218→        videos: t.videos.map((v) =>
   219→   219→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   220→   220→        ),
   221→   221→      })),
   222→   222→      courseCompletion: (() => {
   223→   223→        // Recompute local snapshot
   224→   224→        const videos = data.course.topics.flatMap((t) => t.videos);
   225→   225→        const completed = videos.filter(
   226→   226→          (v) =>
   227→   227→            v.id === updated.videoId
   228→   228→              ? updated.progress.isCompleted
   229→   229→              : v.progress?.isCompleted
   230→   230→        ).length;
   231→   231→        const total = videos.length;
   232→   232→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   233→   233→        return {
   234→   234→          totalVideos: total,
   235→   235→          completedVideos: completed,
   236→   236→          percent,
   237→   237→          allCompleted: total > 0 && completed === total,
   238→   238→        };
   239→   239→      })(),
   240→   240→    };
   241→   241→    setData({ course: newCourse });
   242→   242→
   243→   243→    if (updated.justCompleted) {
   244→   244→      toast({
   245→   245→        title: "Lesson completed",
   246→   246→        description: "Great work! Your progress has been saved.",
   247→   247→      });
   248→   248→    }
   249→   249→  }
   250→   250→
   251→   251→  async function handleSubmissionSuccess() {
   252→   252→    await refetch();
   253→   253→    toast({
   254→   254→      title: "Assignment submitted",
   255→   255→      description: "Your submission has been received.",
   256→   256→    });
   257→   257→  }
   258→   258→
   259→   259→  return (
   260→   260→    <div className="space-y-5">
   261→   261→      {/* ─── Premium header ─── */}
   262→   262→      <div className="rounded-xl border border-border bg-card p-4 md:p-5 shadow-sm">
   263→   263→        <div className="flex flex-wrap items-center justify-between gap-3">
   264→   264→          <div className="flex items-center gap-3 min-w-0 flex-1">
   265→   265→            <Button variant="outline" size="icon" onClick={onBack} className="shrink-0 h-9 w-9" aria-label="Back to courses">
   266→   266→              <ArrowLeft className="h-4 w-4" />
   267→   267→            </Button>
   268→   268→            <div className="min-w-0 flex-1">
   269→   269→              <div className="flex items-center gap-2 mb-0.5">
   270→   270→                {course.category && (
   271→   271→                  <Badge variant="outline" className="text-[11px] border-border text-muted-foreground">
   272→   272→                    {course.category}
   273→   273→                  </Badge>
   274→   274→                )}
   275→   275→                <Badge
   276→   276→                  className={
   277→   277→                    course.accessType === "FREE"
   278→   278→                      ? "bg-success/15 text-success border-transparent"
   279→   279→                      : "bg-primary text-primary-foreground border-transparent"
   280→   280→                  }
   281→   281→                >
   282→   282→                  {course.accessType === "FREE" ? "Free" : "Paid"}
   283→   283→                </Badge>
   284→   284→              </div>
   285→   285→              <h1 className="font-heading text-lg md:text-xl font-semibold text-foreground truncate">
   286→   286→                {course.title}
   287→   287→              </h1>
   288→   288→            </div>
   289→   289→          </div>
   290→   290→          <div className="flex items-center gap-4 shrink-0">
   291→   291→            <div className="hidden sm:flex flex-col items-end gap-0.5">
   292→   292→              <span className="text-xs font-medium text-foreground">
   293→   293→                {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} lessons
   294→   294→              </span>
   295→   295→              <span className="text-[11px] text-muted-foreground">completed</span>
   296→   296→            </div>
   297→   297→            <div className="flex items-center gap-2.5 min-w-[160px]">
   298→   298→              <Progress
   299→   299→                value={course.courseCompletion.percent}
   300→   300→                className={`h-2 w-32 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   301→   301→              />
   302→   302→              <span className="text-sm font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   303→   303→            </div>
   304→   304→          </div>
   305→   305→        </div>
   306→   306→      </div>
   307→   307→
   308→   308→      {/* ─── Mobile: stacked. Desktop: split. ─── */}
   309→   309→      <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   310→   310→        {/* Left: player + materials + assignment */}
   311→   311→        <div className="space-y-5 min-w-0">
   312→   312→          {activeVideo ? (
   313→   313→            <VideoPlayer
   314→   314→              key={activeVideo.id}
   315→   315→              video={activeVideo}
   316→   316→              topicTitle={activeTopic?.title || null}
   317→   317→              onProgress={handleProgressUpdate}
   318→   318→              prevVideo={prevVideo}
   319→   319→              nextVideo={nextVideo}
   320→   320→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   321→   321→            />
   322→   322→          ) : (
   323→   323→            <EmptyState
   324→   324→              icon={<VideoIcon className="h-5 w-5" />}
   325→   325→              title="No videos in this course"
   326→   326→              description="This course doesn't have any published video lessons yet."
   327→   327→            />
   328→   328→          )}
   329→   329→
   330→   330→          {/* Topic materials */}
   331→   331→          {activeTopic && activeTopic.materials.length > 0 && (
   332→   332→            <Card className="shadow-sm">
   333→   333→              <CardHeader>
   334→   334→                <CardTitle className="text-base flex items-center gap-2">
   335→   335→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   336→   336→                    <FileText className="h-4 w-4" />
   337→   337→                  </span>
   338→   338→                  Materials · {activeTopic.title}
   339→   339→                </CardTitle>
   340→   340→                <CardDescription>Reference documents for this topic</CardDescription>
   341→   341→              </CardHeader>
   342→   342→              <CardContent>
   343→   343→                <ul className="space-y-2">
   344→   344→                  {activeTopic.materials.map((m) => (
   345→   345→                    <MaterialRow key={m.id} material={m} />
   346→   346→                  ))}
   347→   347→                </ul>
   348→   348→              </CardContent>
   349→   349→            </Card>
   350→   350→          )}
   351→   351→
   352→   352→          {/* Assignment panel — completely hidden until ALL videos are completed.
   353→   353→              Before completion, show a locked placeholder that does NOT reveal
   354→   354→              the assignment title, description, or instructions. */}
   355→   355→          {course.assignment && course.courseCompletion.allCompleted ? (
   356→   356→            <AssignmentPanel
   357→   357→              course={course}
   358→   358→              onSubmitted={handleSubmissionSuccess}
   359→   359→            />
   360→   360→          ) : course.assignment ? (
   361→   361→            <LockedAssignmentCard
   362→   362→              completedVideos={course.courseCompletion.completedVideos}
   363→   363→              totalVideos={course.courseCompletion.totalVideos}
   364→   364→              percent={course.courseCompletion.percent}
   365→   365→            />
   366→   366→          ) : null}
   367→   367→
   368→   368→          {/* All materials (course-level) */}
   369→   369→          {course.materials.length > 0 && (
   370→   370→            <Card className="shadow-sm">
   371→   371→              <CardHeader>
   372→   372→                <CardTitle className="text-base flex items-center gap-2">
   373→   373→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   374→   374→                    <FileText className="h-4 w-4" />
   375→   375→                  </span>
   376→   376→                  All Course Materials
   377→   377→                </CardTitle>
   378→   378→                <CardDescription>
   379→   379→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   380→   380→                </CardDescription>
   381→   381→              </CardHeader>
   382→   382→              <CardContent>
   383→   383→                <ul className="space-y-2">
   384→   384→                  {course.materials.map((m) => (
   385→   385→                    <MaterialRow key={m.id} material={m} />
   386→   386→                  ))}
   387→   387→                </ul>
   388→   388→              </CardContent>
   389→   389→            </Card>
   390→   390→          )}
   391→   391→        </div>
   392→   392→
   393→   393→        {/* Right: course sidebar */}
   394→   394→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col shadow-sm overflow-hidden">
   395→   395→          <CardHeader className="border-b border-border bg-gradient-to-br from-light-teal/60 to-card">
   396→   396→            <CardTitle className="text-base flex items-center gap-2">
   397→   397→              <BookOpen className="h-4 w-4 text-accent" />
   398→   398→              Course Lessons
   399→   399→            </CardTitle>
   400→   400→            <CardDescription className="mt-1">
   401→   401→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   402→   402→            </CardDescription>
   403→   403→            <div className="mt-2">
   404→   404→              <div className="flex justify-between text-xs mb-1">
   405→   405→                <span className="text-muted-foreground">Progress</span>
   406→   406→                <span className="font-semibold text-foreground">{course.courseCompletion.percent}%</span>
   407→   407→              </div>
   408→   408→              <Progress
   409→   409→                value={course.courseCompletion.percent}
   410→   410→                className={`h-1.5 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   411→   411→              />
   412→   412→            </div>
   413→   413→          </CardHeader>
   414→   414→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-15rem)] scrollbar-thin">
   415→   415→            {course.topics.length === 0 ? (
   416→   416→              <p className="text-sm text-muted-foreground p-6 text-center">
   417→   417→                No topics published yet.
   418→   418→              </p>
   419→   419→            ) : (
   420→   420→              <Accordion
   421→   421→                type="multiple"
   422→   422→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   423→   423→                className="px-3 py-2"
   424→   424→              >
   425→   425→                {course.topics.map((topic, topicIdx) => {
   426→   426→                  const completedInTopic = topic.videos.filter(
   427→   427→                    (v) => v.progress?.isCompleted
   428→   428→                  ).length;
   429→   429→                  const isActiveTopic = topic.id === activeTopic?.id;
   430→   430→                  const topicComplete = completedInTopic === topic.videos.length && topic.videos.length > 0;
   431→   431→                  return (
   432→   432→                    <AccordionItem key={topic.id} value={topic.id} className="border-b border-border/60 last:border-b-0">
   433→   433→                      <AccordionTrigger className="hover:no-underline px-2 py-3 rounded-lg hover:bg-muted/40 transition-colors">
   434→   434→                        <div className="flex items-center gap-2.5 flex-1 min-w-0 text-left">
   435→   435→                          <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-[11px] font-semibold ${
   436→   436→                            topicComplete
   437→   437→                              ? "bg-success/15 text-success"
   438→   438→                              : isActiveTopic
   439→   439→                                ? "bg-accent/15 text-accent"
   440→   440→                                : "bg-muted text-muted-foreground"
   441→   441→                          }`}>
   442→   442→                            {topicComplete ? <CheckCircle2 className="h-3.5 w-3.5" /> : topicIdx + 1}
   443→   443→                          </span>
   444→   444→                          <div className="flex-1 min-w-0">
   445→   445→                            <p className={`text-sm font-semibold truncate ${isActiveTopic ? "text-accent" : "text-foreground"}`}>
   446→   446→                              {topic.title}
   447→   447→                            </p>
   448→   448→                            <p className="text-[11px] text-muted-foreground mt-0.5">
   449→   449→                              {completedInTopic}/{topic.videos.length} lessons
   450→   450→                            </p>
   451→   451→                          </div>
   452→   452→                        </div>
   453→   453→                      </AccordionTrigger>
   454→   454→                      <AccordionContent className="pb-2 pt-1">
   455→   455→                        <ul className="space-y-0.5 ml-9">
   456→   456→                          {topic.videos.map((video, videoIdx) => {
   457→   457→                            const isActive = video.id === activeVideo?.id;
   458→   458→                            const completed = video.progress?.isCompleted;

   508→   508→  );

   512→   512→
   513→   513→function VideoPlayer({
   514→   514→  video,
   515→   515→  topicTitle,
   516→   516→  onProgress,
   517→   517→  prevVideo,
   518→   518→  nextVideo,
   519→   519→  onNavigateLesson,
   520→   520→}: {
   521→   521→  video: VideoItem;
   522→   522→  topicTitle: string | null;
   523→   523→  onProgress: (update: {
   524→   524→    videoId: string;
   525→   525→    progress: VideoProgressInfo;
   526→   526→    justCompleted: boolean;
   527→   527→  }) => void;
   528→   528→  prevVideo: VideoItem | null;
   529→   529→  nextVideo: VideoItem | null;
   530→   530→  onNavigateLesson: (id: string) => void;
   531→   531→}) {
   532→   532→  const videoRef = useRef<HTMLVideoElement>(null);
   533→   533→  const lastTimeRef = useRef<number>(0);
   534→   534→  const watchTimeRef = useRef<number>(0);
   535→   535→  const durationRef = useRef<number>(0);
   536→   536→  const lastPostRef = useRef<number>(0);
   537→   537→  const resumeAppliedRef = useRef<boolean>(false);
   538→   538→  const [expanded, setExpanded] = useState(false);
   539→   539→
   540→   540→  // YouTube completion timer
   541→   541→  const ytStartRef = useRef<number | null>(null);
   542→   542→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   543→   543→  const ytWatchSecondsRef = useRef<number>(0);
   544→   544→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   545→   545→
   546→   546→  // Local UI state
   547→   547→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   548→   548→    video.progress?.isCompleted || false
   549→   549→  );
   550→   550→  const [posting, setPosting] = useState(false);
   551→   551→
   552→   552→  // Reset on video change
   553→   553→  useEffect(() => {
   554→   554→    lastTimeRef.current = 0;
   555→   555→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   556→   556→    durationRef.current = video.duration || video.progress?.duration || 0;
   557→   557→    lastPostRef.current = 0;
   558→   558→    resumeAppliedRef.current = false;
   559→   559→    setLocalCompleted(video.progress?.isCompleted || false);
   560→   560→    setYtCanMarkComplete(false);
   561→   561→    ytWatchSecondsRef.current = 0;
   562→   562→    if (ytTimerRef.current) {
   563→   563→      clearInterval(ytTimerRef.current);
   564→   564→      ytTimerRef.current = null;
   565→   565→    }
   566→   566→  }, [video.id]);
   567→   567→
   568→   568→  // For HTML5 video: apply resume position once metadata is loaded.
   569→   569→  const handleLoadedMetadata = () => {
   570→   570→    const el = videoRef.current;
   571→   571→    if (!el) return;
   572→   572→    durationRef.current = el.duration || durationRef.current;
   573→   573→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   574→   574→      // resume within (duration - 2) so we don't immediately fire ended
   575→   575→      const target = Math.min(
   576→   576→        video.progress.lastPosition,
   577→   577→        Math.max(0, (el.duration || 0) - 3)
   578→   578→      );
   579→   579→      try {
   580→   580→        el.currentTime = target;
   581→   581→      } catch {}
   582→   582→      lastTimeRef.current = target;
   583→   583→      resumeAppliedRef.current = true;
   584→   584→    }
   585→   585→  };
   586→   586→
   587→   587→  // Core: timeupdate handler — accumulate real watch time.
   588→   588→  const handleTimeUpdate = () => {
   589→   589→    const el = videoRef.current;
   590→   590→    if (!el) return;
   591→   591→    const cur = el.currentTime;
   592→   592→    const last = lastTimeRef.current;
   593→   593→    const delta = cur - last;
   594→   594→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   595→   595→    // (large positive deltas) and rewinds (negative deltas).
   596→   596→    if (delta > 0 && delta <= 1.5) {
   597→   597→      watchTimeRef.current += delta;
   598→   598→    }
   599→   599→    // Update lastTime to current — even on seek, so next delta is small.
   600→   600→    lastTimeRef.current = cur;
   601→   601→
   602→   602→    // Throttled server POST every 5 seconds
   603→   603→    const now = Date.now();
   604→   604→    if (now - lastPostRef.current > 5000) {
   605→   605→      lastPostRef.current = now;
   606→   606→      void postProgress(false);
   607→   607→    }
   608→   608→  };
   609→   609→
   610→   610→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   611→   611→  const handleSeeked = () => {
   612→   612→    const el = videoRef.current;
   613→   613→    if (!el) return;
   614→   614→    lastTimeRef.current = el.currentTime;
   615→   615→  };
   616→   616→
   617→   617→  // Ended event — submit final progress with completion hint.
   618→   618→  const handleEnded = () => {
   619→   619→    void postProgress(true);
   620→   620→  };
   621→   621→
   622→   622→  const postProgress = useCallback(
   623→   623→    async (isCompletionHint: boolean) => {
   624→   624→      setPosting(true);
   625→   625→      try {
   626→   626→        const res = await apiPost<{
   627→   627→          progress: VideoProgressInfo;
   628→   628→          justCompleted: boolean;
   629→   629→          rejectedCompletion: boolean;
   630→   630→          certificateIssued: boolean;
   631→   631→          certificateId: string | null;
   632→   632→        }>("/api/student/progress", {
   633→   633→          videoId: video.id,
   634→   634→          lastPosition: videoRef.current?.currentTime || 0,
   635→   635→          watchTimeSeconds: watchTimeRef.current,
   636→   636→          duration: durationRef.current,
   637→   637→          isCompleted: isCompletionHint,
   638→   638→        });
   639→   639→        setLocalCompleted(res.progress.isCompleted);
   640→   640→        onProgress({
   641→   641→          videoId: video.id,
   642→   642→          progress: res.progress,
   643→   643→          justCompleted: res.justCompleted,
   644→   644→        });
   645→   645→        if (isCompletionHint && res.rejectedCompletion) {
   646→   646→          toast({
   647→   647→            title: "Completion not recorded",
   648→   648→            description:
   649→   649→              "You need to watch at least 85% of the video before it's marked complete.",
   650→   650→            variant: "destructive",
   651→   651→          });
   652→   652→        }
   653→   653→        // ─── Certificate auto-issued upon course completion ───
   654→   654→        if (res.certificateIssued && res.certificateId) {
   655→   655→          toast({
   656→   656→            title: "🎉 Certificate Issued!",
   657→   657→            description:
   658→   658→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   659→   659→          });
   660→   660→        }
   661→   661→      } catch {
   662→   662→        // apiPost already surfaces errors
   663→   663→      } finally {
   664→   664→        setPosting(false);
   665→   665→      }
   666→   666→    },
   667→   667→    [video.id, onProgress]
   668→   668→  );
   669→   669→
   670→   670→  // Manual "Mark complete" for YouTube videos
   671→   671→  const markYtComplete = async () => {
   672→   672→    setPosting(true);
   673→   673→    try {
   674→   674→      const res = await apiPost<{
   675→   675→        progress: VideoProgressInfo;
   676→   676→        justCompleted: boolean;
   677→   677→        rejectedCompletion: boolean;
   678→   678→        certificateIssued: boolean;
   679→   679→        certificateId: string | null;
   680→   680→      }>("/api/student/progress", {
   681→   681→        videoId: video.id,
   682→   682→        lastPosition: durationRef.current,
   683→   683→        watchTimeSeconds: ytWatchSecondsRef.current,
   684→   684→        duration: durationRef.current,
   685→   685→        isCompleted: true,
   686→   686→      });
   687→   687→      setLocalCompleted(res.progress.isCompleted);
   688→   688→      onProgress({
   689→   689→        videoId: video.id,
   690→   690→        progress: res.progress,
   691→   691→        justCompleted: res.justCompleted,
   692→   692→      });
   693→   693→      if (res.justCompleted) {
   694→   694→        toast({
   695→   695→          title: "Lesson completed",
   696→   696→          description: "Great work! Your progress has been saved.",
   697→   697→        });
   698→   698→      }
   699→   699→      if (res.rejectedCompletion) {
   700→   700→        toast({
   701→   701→          title: "Completion not recorded",
   702→   702→          description:
   703→   703→            "You need to keep the video playing for at least 85% of its duration.",
   704→   704→          variant: "destructive",
   705→   705→        });
   706→   706→      }
   707→   707→      // Certificate auto-issued upon course completion
   708→   708→      if (res.certificateIssued && res.certificateId) {
   709→   709→        toast({
   710→   710→          title: "🎉 Certificate Issued!",
   711→   711→          description:
   712→   712→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   713→   713→        });
   714→   714→      }
   715→   715→    } catch {
   716→   716→    } finally {
   717→   717→      setPosting(false);
   718→   718→    }
   719→   719→  };
   720→   720→
   721→   721→  /* YouTube handling */
   722→   722→  const ytEmbedUrl = useMemo(() => toYouTubeEmbed(video.videoUrl), [video.videoUrl]);
   723→   723→  const isYouTube = video.source === "YOUTUBE" && ytEmbedUrl !== null;
   724→   724→
   725→   725→  // YouTube timer: accumulate watch time + enable mark-complete after 85%.
   726→   726→  useEffect(() => {
   727→   727→    if (!isYouTube) return;
   728→   728→    if (ytTimerRef.current) clearInterval(ytTimerRef.current);
   729→   729→    ytStartRef.current = Date.now();
   730→   730→
   731→   731→    ytTimerRef.current = setInterval(() => {
   732→   732→      ytWatchSecondsRef.current = (Date.now() - (ytStartRef.current || Date.now())) / 1000;
   733→   733→      const dur = durationRef.current;
   734→   734→      if (dur > 0 && ytWatchSecondsRef.current >= dur * 0.85) {
   735→   735→        setYtCanMarkComplete(true);
   736→   736→      }
   737→   737→    }, 1000);
   738→   738→
   739→   739→    return () => {
   740→   740→      if (ytTimerRef.current) {
   741→   741→        clearInterval(ytTimerRef.current);
   742→   742→        ytTimerRef.current = null;
   743→   743→      }
   744→   744→    };
   745→   745→  }, [isYouTube, video.id]);
   746→   746→
   747→   747→  // Render the right player
   748→   748→  return (
   749→   749→    <>
   750→   750→      {/* Expanded fullscreen overlay — rendered as a portal-like fixed layer */}
   751→   751→      {expanded && (
   752→   752→        <div className="fixed inset-0 z-50 bg-black flex flex-col">
   753→   753→          {/* Top bar with title + close */}
   754→   754→          <div className="flex items-center justify-between px-4 py-3 bg-gradient-to-b from-black/80 to-transparent backdrop-blur">
   755→   755→            <div className="min-w-0 flex-1">
   756→   756→              {topicTitle && (
   757→   757→                <p className="text-[11px] font-medium uppercase tracking-wider text-white/60 mb-0.5 truncate">
   758→   758→                  {topicTitle}
   759→   759→                </p>
   760→   760→              )}
   761→   761→              <p className="text-sm font-semibold text-white truncate">{video.title}</p>
   762→   762→            </div>
   763→   763→            <div className="flex items-center gap-2 shrink-0">
   764→   764→              {localCompleted && (
   765→   765→                <Badge className="bg-success/20 text-success border-transparent">
   766→   766→                  <CheckCircle2 className="h-3 w-3" /> Completed
   767→   767→                </Badge>
   768→   768→              )}
   769→   769→              <button
   770→   770→                onClick={() => setExpanded(false)}
   771→   771→                className="flex items-center gap-1.5 rounded-lg bg-white/10 px-3 py-1.5 text-sm font-medium text-white hover:bg-white/20 transition-colors"
   772→   772→                aria-label="Exit fullscreen"
   773→   773→              >
   774→   774→                <Minimize2 className="h-4 w-4" />
   775→   775→                <span className="hidden sm:inline">Exit fullscreen</span>
   776→   776→              </button>
   777→   777→            </div>
   778→   778→          </div>
   779→   779→          {/* Video fills the remaining space */}
   780→   780→          <div className="flex-1 flex items-center justify-center p-2 sm:p-4">
   781→   781→            <div className="relative w-full h-full flex items-center justify-center">
   782→   782→              {video.source === "YOUTUBE" && ytEmbedUrl ? (
   783→   783→                <iframe
   784→   784→                  key={video.id}
   785→   785→                  src={ytEmbedUrl}
   786→   786→                  title={video.title}
   787→   787→                  className="w-full h-full max-w-[1600px]"
   788→   788→                  style={{ aspectRatio: "16 / 9" }}
   789→   789→                  frameBorder={0}
   790→   790→                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share"
   791→   791→                  allowFullScreen
   792→   792→                />
   793→   793→              ) : (
   794→   794→                <video
   795→   795→                  key={video.id}
   796→   796→                  ref={videoRef}
   797→   797→                  src={video.videoUrl}
   798→   798→                  poster={video.thumbnail || undefined}
   799→   799→                  controls
   800→   800→                  autoPlay
   801→   801→                  className="w-full h-full max-h-full object-contain"
   802→   802→                  onLoadedMetadata={handleLoadedMetadata}
   803→   803→                  onTimeUpdate={handleTimeUpdate}
   804→   804→                  onSeeked={handleSeeked}
   805→   805→                  onEnded={handleEnded}
   806→   806→                />
   807→   807→              )}
   808→   808→            </div>
   809→   809→          </div>
   810→   810→        </div>
   811→   811→      )}
   812→   812→
   813→   813→      {/* ─── Default compact view ─── */}
   814→   814→      <Card className="overflow-hidden p-0 shadow-sm">
   815→   815→        {/* Compact video area — centered, max 640px, dark frame */}
   816→   816→        <div className="bg-gradient-to-b from-foreground/[0.03] to-card p-4 md:p-6">
   817→   817→          <div className="relative mx-auto overflow-hidden rounded-xl bg-black shadow-lg ring-1 ring-black/20" style={{ maxWidth: "640px" }}>
   818→   818→            <div className="aspect-video w-full flex items-center justify-center relative">
   819→   819→              {video.source === "YOUTUBE" && ytEmbedUrl ? (
   820→   820→                <iframe
   821→   821→                  key={video.id}
   822→   822→                  src={ytEmbedUrl}
   823→   823→                  title={video.title}
   824→   824→                  className="w-full h-full"
   825→   825→                  frameBorder={0}

   832→   832→                  ref={videoRef}
   833→   833→                  src={video.videoUrl}

   835→   835→                  controls
   836→   836→                  className="w-full h-full"
   837→   837→                  onLoadedMetadata={handleLoadedMetadata}
   838→   838→                  onTimeUpdate={handleTimeUpdate}
   839→   839→                  onSeeked={handleSeeked}

  1216→  1216→function toYouTubeEmbed(url: string): string | null {
  1217→  1217→  try {
  1218→  1218→    const u = new URL(url);
  1219→  1219→    if (u.hostname === "youtu.be") {
  1220→  1220→      const id = u.pathname.slice(1);
  1221→  1221→      return `https://www.youtube.com/embed/${id}?rel=0`;
  1222→  1222→    }
  1223→  1223→    if (u.hostname.includes("youtube.com")) {
  1224→  1224→      const id = u.searchParams.get("v");
  1225→  1225→      if (id) return `https://www.youtube.com/embed/${id}?rel=0`;
  1226→  1226→      // /embed/ID or /shorts/ID
  1227→  1227→      const m = u.pathname.match(/\/(embed|shorts)\/([\w-]+)/);
  1228→  1228→      if (m) return `https://www.youtube.com/embed/${m[2]}?rel=0`;
  1229→  1229→    }
  1230→  1230→    return null;
  1231→  1231→  } catch {
  1232→  1232→    return null;
  1233→  1233→  }

  1236→  1236→function formatVideoDuration(seconds: number): string {
  1237→  1237→  if (seconds <= 0) return "";
  1238→  1238→  const m = Math.floor(seconds / 60);
  1239→  1239→  const s = Math.floor(seconds % 60);
  1240→  1240→  if (m === 0) return `${s}s`;
  1241→  1241→  return `${m}:${s.toString().padStart(2, "0")}`;
  1242→  1242→}
