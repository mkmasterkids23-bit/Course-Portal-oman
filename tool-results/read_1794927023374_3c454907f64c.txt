   160→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   161→  const { t } = useI18n();
   162→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   163→    `/api/student/courses/${courseId}`
   164→  );
   165→
   166→  // Read the hash query for deep-link support. The global Discussions
   167→  // view navigates here with `?tab=discussions&discussion=<id>`; we honour
   168→  // that by defaulting the Materials/Discussions tab to "discussions" and
   169→  // asking the panel to highlight (and scroll to) the matching thread.
   170→  const routeQuery = useAppStore((s) => s.query);
   171→  const highlightDiscussionId = routeQuery.discussion || null;
   172→  const discussionsTabDefault = routeQuery.tab === "discussions";
   173→
   174→  // Pick the initial video: last watched, or first available.
   175→  const initialVideoId = (() => {
   176→    if (!data?.course) return null;
   177→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   178→    //    most recently updated progress entry in the topics tree.
   179→    let candidate: VideoItem | null = null;
   180→    for (const t of data.course.topics) {
   181→      for (const v of t.videos) {
   182→        if (v.progress && !v.progress.isCompleted) {
   183→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   184→            candidate = v;
   185→          }
   186→        }
   187→      }
   188→    }
   189→    if (candidate) return candidate.id;
   190→    // 2) First video of the first topic
   191→    return data.course.topics[0]?.videos[0]?.id ?? null;
   192→  })();
   193→
   194→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   195→  // Reset user selection when courseId changes — "adjust state during render"
   196→  // pattern (avoids setState-in-effect warnings).
   197→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   198→  if (courseId !== prevCourseId) {
   199→    setPrevCourseId(courseId);
   200→    setUserSelectedVideoId(null);
   201→  }
   202→  // Effective active video: explicit user selection overrides the auto-pick
   203→  // (which is the last-watched or first video). When data loads and we have an
   204→  // initial pick, this falls back to it without needing an effect.
   205→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   206→
   207→  if (loading) {
   208→    return (
   209→      <div className="space-y-5">
   210→        <Skeleton className="h-10 w-32 rounded-lg" />
   211→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   212→          <div className="space-y-4">
   213→            <Skeleton className="h-[420px] rounded-xl" />
   214→            <Skeleton className="h-24 rounded-xl" />
   215→          </div>
   216→          <Skeleton className="h-[600px] rounded-xl" />
   217→        </div>
   218→      </div>
   219→    );
   220→  }
   221→
   222→  if (error) {
   223→    return (
   224→      <div className="space-y-4">
   225→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   226→          <ArrowLeft className="h-4 w-4" /> {t("learning.backToCourses")}
   227→        </Button>
   228→        <EmptyState
   229→          icon={<AlertTriangle className="h-5 w-5" />}
   230→          title={t("learning.couldntOpenCourse")}
   231→          description={error}
   232→          action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   233→        />
   234→      </div>
   235→    );
   236→  }
   237→
   238→  if (!data?.course) {
   239→    return (
   240→      <EmptyState
   241→        icon={<PlayCircle className="h-5 w-5" />}
   242→        title={t("learning.courseNotFound")}
   243→        description={t("learning.courseMayRemoved")}
   244→        action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   245→      />
   246→    );
   247→  }
   248→
   249→  const course = data.course;
   250→  const allVideos = course.topics.flatMap((t) => t.videos);
   251→  const activeVideo =
   252→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   253→
   254→  // Find the topic containing the active video
   255→  const activeTopic = course.topics.find((t) =>
   256→    t.videos.some((v) => v.id === activeVideo?.id)
   257→  );
   258→
   259→  // Build a flat ordered list for prev/next navigation
   260→  const flatVideoIds = allVideos.map((v) => v.id);
   261→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   262→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   263→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   264→
   265→  function handleProgressUpdate(updated: {
   266→    videoId: string;
   267→    progress: VideoProgressInfo;
   268→    justCompleted: boolean;
   269→  }) {
   270→    if (!data) return;
   271→    const newCourse = {
   272→      ...data.course,
   273→      topics: data.course.topics.map((t) => ({
   274→        ...t,
   275→        videos: t.videos.map((v) =>
   276→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   277→        ),
   278→      })),
   279→      courseCompletion: (() => {
   280→        // Recompute local snapshot
   281→        const videos = data.course.topics.flatMap((t) => t.videos);
   282→        const completed = videos.filter(
   283→          (v) =>
   284→            v.id === updated.videoId
   285→              ? updated.progress.isCompleted
   286→              : v.progress?.isCompleted
   287→        ).length;
   288→        const total = videos.length;
   289→        // No videos → course is considered fully watched (assignment unlocked).
   290→        const percent = total > 0 ? Math.round((completed / total) * 100) : 100;
   291→        const totalTopics = data.course.topics.length;
   292→        const completedTopics = data.course.topics.filter((topic) =>
   293→          topic.videos.length > 0 &&
   294→          topic.videos.every(
   295→            (v) =>
   296→              (v.id === updated.videoId ? updated.progress.isCompleted : v.progress?.isCompleted)
   297→          )
   298→        ).length;
   299→        return {
   300→          totalVideos: total,
   301→          completedVideos: completed,
   302→          completedTopics,
   303→          totalTopics,
   304→          percent,
   305→          allCompleted: total === 0 || completed === total,
   306→        };
   307→      })(),
   308→    };
   309→    setData({ course: newCourse });
   310→
   311→    if (updated.justCompleted) {
   312→      toast({
   313→        title: t("learning.lessonCompletedToast"),
   314→        description: t("learning.greatWorkSaved"),
   315→      });
   316→    }
   317→  }
   318→
   319→  async function handleSubmissionSuccess() {
   320→    await refetch();
   321→    toast({
   322→      title: t("learning.assignmentSubmittedToast"),
   323→      description: t("learning.assignmentSubmittedDesc"),
   324→    });
   325→  }
   326→
   327→  return (
   328→    <div className="space-y-4 animate-page-in pb-20 lg:pb-4">
   329→      {/* ─── Compact breadcrumb header ─── */}
   330→      <div className="shadow-premium flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-4 py-3">
   331→        <div className="flex items-center gap-2 min-w-0 flex-1">
   332→          <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0 h-8 w-8" aria-label={t("learning.backToCourses")}>
   333→            <ArrowLeft className="h-4 w-4" />
   334→          </Button>
   335→          <div className="min-w-0 flex-1">
   336→            <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   337→              <button onClick={onBack} className="hover:text-foreground transition-colors">{t("learning.myCourses")}</button>
   338→              <span>/</span>
   339→              <span className="truncate">{course.title}</span>
   340→              {activeVideo && (
   341→                <>
   342→                  <span>/</span>
   343→                  <span className="truncate text-foreground font-medium">{activeVideo.title}</span>
   344→                </>
   345→              )}
   346→            </div>
   347→            <div className="flex items-center gap-2 mt-0.5">
   348→              {course.category && (
   349→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   350→                  {course.category}
   351→                </Badge>
   352→              )}
   353→              <Badge
   354→                className={
   355→                  course.accessType === "FREE"
   356→                    ? "bg-success/15 text-success border-transparent text-[10px] px-1.5 py-0 h-4"
   357→                    : "bg-primary text-primary-foreground border-transparent text-[10px] px-1.5 py-0 h-4"
   358→                }
   359→              >
   360→                {course.accessType === "FREE" ? t("common.free") : t("common.paid")}
   361→              </Badge>
   362→            </div>
   363→          </div>
   364→        </div>
   365→        <div className="flex items-center gap-3 shrink-0">
   366→          <div className="hidden sm:flex flex-col items-end">
   367→            <span className="text-xs font-medium text-foreground">
   368→              {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} {t("courses.lessons")}
   369→            </span>
   370→            <span className="text-[10px] text-muted-foreground">{t("learning.completed")}</span>
   371→          </div>
   372→          <div className="flex items-center gap-2 min-w-[120px]">
   373→            <Progress
   374→              value={course.courseCompletion.percent}
   375→              className={`h-1.5 w-24 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   376→            />
   377→            <span className="text-xs font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   378→          </div>
   379→        </div>
   380→      </div>
   381→
   382→      {/* ─── Main Layout: LEFT content + RIGHT curriculum ─── */}
   383→      <div className="grid grid-cols-1 lg:grid-cols-[1fr_340px] gap-4">
   384→        {/* ─── LEFT: Video + Tabs ─── */}
   385→        <div className="space-y-4 min-w-0">
   386→          {/* Video player */}
   387→          {activeVideo ? (
   388→            <VideoPlayer
   389→              key={activeVideo.id}
   390→              video={activeVideo}
   391→              topicTitle={activeTopic?.title || null}
   392→              courseId={course.id}
   393→              topicId={activeTopic?.id || null}
   394→              onProgress={handleProgressUpdate}
   395→              prevVideo={prevVideo}
   396→              nextVideo={nextVideo}
   397→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   398→              refetchCourse={refetch}
   399→            />
   400→          ) : (
   401→            <EmptyState
   402→              icon={<VideoIcon className="h-5 w-5" />}
   403→              title={t("courses.noVideosYet")}
   404→              description={t("courses.noVideosDesc")}
   405→            />
   406→          )}
   407→
   408→          {/* Assignment panel — hidden until all videos completed */}
   409→          {course.assignment && course.courseCompletion.allCompleted ? (
   410→            <AssignmentPanel course={course} onSubmitted={handleSubmissionSuccess} />
   411→          ) : course.assignment ? (
   412→            <LockedAssignmentCard
   413→              completedVideos={course.courseCompletion.completedVideos}
   414→              totalVideos={course.courseCompletion.totalVideos}
   415→              percent={course.courseCompletion.percent}
   416→            />
   417→          ) : null}
   418→
   419→          {/* ─── 4-Tab Interface below video ─── */}
   420→          <Tabs defaultValue="overview" className="gap-0">
   421→            <div className="border-b border-border overflow-x-auto scrollbar-thin">
   422→              <TabsList className="bg-transparent border-0 p-0 h-auto gap-0 flex-nowrap">
   423→                <TabsTrigger
   424→                  value="overview"
   425→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   426→                >
   427→                  <Info className="h-3.5 w-3.5" />
   428→                  {t("learning.tab.overview")}
   429→                </TabsTrigger>
   430→                <TabsTrigger
   431→                  value="notes"
   432→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   433→                >
   434→                  <StickyNote className="h-3.5 w-3.5" />
   435→                  {t("learning.tab.notes")}
   436→                </TabsTrigger>
   437→                <TabsTrigger
   438→                  value="discussions"
   439→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   440→                >
   441→                  <MessageSquare className="h-3.5 w-3.5" />
   442→                  {t("learning.tab.discussion")}
   443→                </TabsTrigger>
   444→                <TabsTrigger
   445→                  value="resources"
   446→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   447→                >
   448→                  <FileText className="h-3.5 w-3.5" />
   449→                  {t("learning.tab.resources")}
   450→                  {(course.materials.length > 0 || (activeTopic && activeTopic.materials.length > 0)) && (
   451→                    <Badge variant="outline" className="ml-0.5 text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   452→                      {course.materials.length + (activeTopic?.materials.length || 0)}
   453→                    </Badge>
   454→                  )}
   455→                </TabsTrigger>
   456→              </TabsList>
   457→            </div>
   458→
   459→            {/* Tab 1: Overview */}
   460→            <TabsContent value="overview" className="mt-0 p-4">
   461→              <div className="space-y-3">
   462→                {activeVideo && (
   463→                  <>
   464→                    <div>
   465→                      <h3 className="font-heading text-base font-semibold text-foreground">
   466→                        {activeVideo.title}
   467→                      </h3>
   468→                      {activeTopic && (
   469→                        <p className="text-xs text-muted-foreground mt-0.5">
   470→                          {t("learning.topic")} {activeTopic.title}
   471→                        </p>
   472→                      )}
   473→                    </div>
   474→                    {activeVideo.description && (
   475→                      <p className="text-sm text-muted-foreground leading-relaxed">
   476→                        {activeVideo.description}
   477→                      </p>
   478→                    )}
   479→                    <div className="flex items-center gap-3 pt-2 border-t border-border">
   480→                      {activeVideo.progress?.isCompleted ? (
   481→                        <Badge className="bg-success/15 text-success border-transparent gap-1">
   482→                          <CheckCircle2 className="h-3 w-3" /> {t("common.completed")}
   483→                        </Badge>
   484→                      ) : activeVideo.progress?.lastPosition ? (
   485→                        <Badge className="bg-warning/15 text-warning border-transparent gap-1">
   486→                          <Clock className="h-3 w-3" /> {t("common.inProgress")}
   487→                        </Badge>
   488→                      ) : (
   489→                        <Badge variant="outline" className="text-muted-foreground gap-1">
   490→                          <PlayCircle className="h-3 w-3" /> {t("common.notStarted")}
   491→                        </Badge>
   492→                      )}
   493→                      {activeVideo.duration > 0 && (
   494→                        <span className="text-xs text-muted-foreground">
   495→                          {t("common.duration")}: {formatVideoDuration(activeVideo.duration)}
   496→                        </span>
   497→                      )}
   498→                    </div>
   499→                  </>
   500→                )}
   501→              </div>
   502→            </TabsContent>
   503→
   504→            {/* Tab 2: My Notes */}
   505→            <TabsContent value="notes" className="mt-0 p-4">
   506→              {activeVideo && (
   507→                <VideoNotesPanel
   508→                  videoId={activeVideo.id}
   509→                  courseId={course.id}
   510→                  topicId={activeTopic?.id || null}
   511→                  videoTitle={activeVideo.title}
   512→                  canCaptureTimestamp={false}
   513→                  getCurrentTime={() => 0}
   514→                  seekTo={() => {}}
   515→                />
   516→              )}
   517→            </TabsContent>
   518→
   519→            {/* Tab 3: Discussion */}
   520→            <TabsContent value="discussions" className="mt-0">
   521→              <CourseDiscussionsPanel
   522→                courseId={course.id}
   523→                highlightDiscussionId={highlightDiscussionId}
   524→              />
   525→            </TabsContent>
   526→
   527→            {/* Tab 4: Resources */}
   528→            <TabsContent value="resources" className="mt-0 p-4">
   529→              <div className="space-y-3">
   530→                {activeTopic && activeTopic.materials.length > 0 && (
   531→                  <div>
   532→                    <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   533→                      {t("learning.topicMaterialsLabel")} · {activeTopic.title}
   534→                    </p>
   535→                    <ul className="space-y-2">
   536→                      {activeTopic.materials.map((m) => (
   537→                        <MaterialRow key={m.id} material={m} />
   538→                      ))}
   539→                    </ul>
   540→                  </div>
   541→                )}
   542→                {course.materials.length > 0 && (
   543→                  <div>

   718→/* ===================== Video player ===================== */
   719→
   720→function VideoPlayer({
   721→  video,
   722→  topicTitle,
   723→  courseId,
   724→  topicId,
   725→  onProgress,
   726→  prevVideo,
   727→  nextVideo,
   728→  onNavigateLesson,
   729→  refetchCourse,
   730→}: {
   731→  video: VideoItem;
   732→  topicTitle: string | null;
   733→  courseId: string;
   734→  topicId: string | null;
   735→  onProgress: (update: {
   736→    videoId: string;
   737→    progress: VideoProgressInfo;
   738→    justCompleted: boolean;
   739→  }) => void;
   740→  prevVideo: VideoItem | null;
   741→  nextVideo: VideoItem | null;
   742→  onNavigateLesson: (id: string) => void;
   743→  refetchCourse?: () => void;
   744→}) {
   745→  const videoRef = useRef<HTMLVideoElement>(null);
   746→  const lastTimeRef = useRef<number>(0);
   747→  const watchTimeRef = useRef<number>(0);
   748→  const durationRef = useRef<number>(0);
   749→  const lastPostRef = useRef<number>(0);
   750→  const resumeAppliedRef = useRef<boolean>(false);
   751→  const [expanded, setExpanded] = useState(false);
   752→
   753→  // ─── Video Assessment state ───
   754→  // Fetches the questions for THIS video (without correctAnswer — security).
   755→  // The hook is unconditional (rules of hooks); the URL just changes per video.
   756→  const questionsUrl = `/api/student/videos/${video.id}/questions`;
   757→  const {
   758→    data: questionsData,
   759→    loading: questionsLoading,
   760→    refetch: refetchQuestions,
   761→  } = useApiFetch<{
   762→    questions: VideoQuestionData[];
   763→    assessment: VideoAssessmentSummary | null;
   764→  }>(questionsUrl);
   765→
   766→  // Currently-displayed question popup (null = no popup shown).
   767→  const [activeQuestion, setActiveQuestion] = useState<VideoQuestionData | null>(null);
   768→  // Map of questionId → selected option index (answers given this session).
   769→  const [answeredMap, setAnsweredMap] = useState<Map<string, number>>(new Map());
   770→  // Currently-selected option in the active popup (null = nothing selected yet).
   771→  const [selectedAnswer, setSelectedAnswer] = useState<number | null>(null);
   772→  // Brief "Answer saved" confirmation shown after submitting a popup answer.
   773→  const [showFeedback, setShowFeedback] = useState(false);
   774→  // Result returned from POST /assessment — when set, the result overlay shows.
   775→  const [assessmentResult, setAssessmentResult] = useState<AssessmentResult | null>(null);
   776→  // True while the assessment POST is in flight.
   777→  const [submittingAssessment, setSubmittingAssessment] = useState(false);
   778→  // Timer ref for the post-submit feedback delay (cleared on unmount/video change).
   779→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   780→
   781→  // YouTube completion timer
   782→  const ytStartRef = useRef<number | null>(null);
   783→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   784→  const ytWatchSecondsRef = useRef<number>(0);
   785→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   786→
   787→  // Local UI state
   788→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   789→    video.progress?.isCompleted || false
   790→  );
   791→  const [posting, setPosting] = useState(false);
   792→
   793→  // Are question popups active for this viewing session?
   794→  // They are NOT active if the student has already passed the assessment —
   795→  // in that case the video plays freely and the "Passed" badge is shown.
   796→  const questionsActive = Boolean(
   797→    questionsData &&
   798→      questionsData.questions.length > 0 &&
   799→      questionsData.assessment?.passed !== true
   800→  );
   801→
   802→  // Reset on video change
   803→  useEffect(() => {
   804→    lastTimeRef.current = 0;
   805→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   806→    durationRef.current = video.duration || video.progress?.duration || 0;
   807→    lastPostRef.current = 0;
   808→    resumeAppliedRef.current = false;
   809→    setLocalCompleted(video.progress?.isCompleted || false);
   810→    setYtCanMarkComplete(false);
   811→    ytWatchSecondsRef.current = 0;
   812→    if (ytTimerRef.current) {
   813→      clearInterval(ytTimerRef.current);
   814→      ytTimerRef.current = null;
   815→    }
   816→    // ─── Reset assessment state ───
   817→    setActiveQuestion(null);
   818→    setSelectedAnswer(null);
   819→    setAnsweredMap(new Map());
   820→    setShowFeedback(false);
   821→    setAssessmentResult(null);
   822→    setSubmittingAssessment(false);
   823→    if (feedbackTimerRef.current) {
   824→      clearTimeout(feedbackTimerRef.current);
   825→      feedbackTimerRef.current = null;
   826→    }
   827→  }, [video.id]);
   828→
   829→  // ─── Missed-question catch-up ───
   830→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   831→  // the student answers a question. If the playhead has already moved past an
   832→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   833→  // the questions fetch landed late), pause and surface that question now.
   834→  useEffect(() => {
   835→    if (!questionsActive || !questionsData) return;
   836→    if (activeQuestion || assessmentResult || showFeedback || submittingAssessment) return;
   837→    const el = videoRef.current;
   838→    if (!el) return;
   839→    const cur = el.currentTime;
   840→    const missed = questionsData.questions.find(
   841→      (q) => !answeredMap.has(q.id) && cur >= q.timestamp - 0.5
   842→    );
   843→    if (missed && !el.paused) {
   844→      try {
   845→        el.pause();
   846→        lastTimeRef.current = el.currentTime;
   847→      } catch {}
   848→      setActiveQuestion(missed);
   849→      setSelectedAnswer(null);
   850→    }
   851→  }, [
   852→    questionsActive,
   853→    questionsData,
   854→    activeQuestion,
   855→    assessmentResult,
   856→    showFeedback,
   857→    submittingAssessment,
   858→    answeredMap,
   859→  ]);
   860→
   861→  // For HTML5 video: apply resume position once metadata is loaded.
   862→  const handleLoadedMetadata = () => {
   863→    const el = videoRef.current;
   864→    if (!el) return;
   865→    durationRef.current = el.duration || durationRef.current;
   866→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   867→      // resume within (duration - 2) so we don't immediately fire ended
   868→      const target = Math.min(
   869→        video.progress.lastPosition,
   870→        Math.max(0, (el.duration || 0) - 3)
   871→      );
   872→      try {
   873→        el.currentTime = target;
   874→      } catch {}
   875→      lastTimeRef.current = target;
   876→      resumeAppliedRef.current = true;
   877→    }
   878→  };
   879→
   880→  // Core: timeupdate handler — accumulate real watch time.
   881→  const handleTimeUpdate = () => {
   882→    const el = videoRef.current;
   883→    if (!el) return;
   884→    const cur = el.currentTime;
   885→    const last = lastTimeRef.current;
   886→
   887→    // ─── Question trigger check ───
   888→    // Pause and surface a popup the moment the playhead crosses an unanswered
   889→    // question's timestamp going forward. We require BOTH:
   890→    //   - cur >= q.timestamp    (we're at or past it)
   891→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   892→    //                            this is a forward crossing, not a re-tick while
   893→    //                            paused, and not a backward seek landing past it)
   894→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   895→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   896→    if (
   897→      questionsActive &&
   898→      !activeQuestion &&
   899→      !assessmentResult &&
   900→      !showFeedback &&
   901→      !submittingAssessment &&
   902→      questionsData
   903→    ) {
   904→      for (const q of questionsData.questions) {
   905→        if (answeredMap.has(q.id)) continue;
   906→        if (cur >= q.timestamp && last < q.timestamp) {
   907→          try {
   908→            el.pause();
   909→            lastTimeRef.current = cur;
   910→          } catch {}
   911→          setActiveQuestion(q);
   912→          setSelectedAnswer(null);
   913→          return; // do NOT accumulate watch time for this tick
   914→        }
   915→      }
   916→    }
   917→
   918→    // Watch time tracking — only counts normal forward playback deltas.
   919→    const delta = cur - last;
   920→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   921→    // (large positive deltas) and rewinds (negative deltas).
   922→    if (delta > 0 && delta <= 1.5) {
   923→      watchTimeRef.current += delta;
   924→    }
   925→    // Update lastTime to current — even on seek, so next delta is small.
   926→    lastTimeRef.current = cur;
   927→
   928→    // Throttled server POST every 5 seconds
   929→    const now = Date.now();
   930→    if (now - lastPostRef.current > 5000) {
   931→      lastPostRef.current = now;
   932→      void postProgress(false);
   933→    }
   934→  };
   935→
   936→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   937→  // Also catches forward-seek-past-question: if the student scrubs the
   938→  // progress bar past an unanswered question, pause and surface that question
   939→  // so they can't skip it.
   940→  const handleSeeked = () => {
   941→    const el = videoRef.current;
   942→    if (!el) return;
   943→    const newTime = el.currentTime;
   944→    lastTimeRef.current = newTime;
   945→
   946→    if (
   947→      questionsActive &&
   948→      !activeQuestion &&
   949→      !assessmentResult &&
   950→      !showFeedback &&
   951→      !submittingAssessment &&
   952→      questionsData
   953→    ) {
   954→      // Find the earliest unanswered question we've now skipped past.
   955→      const skipped = questionsData.questions.find(
   956→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
   957→      );
   958→      if (skipped) {
   959→        try {
   960→          el.pause();
   961→        } catch {}
   962→        setActiveQuestion(skipped);
   963→        setSelectedAnswer(null);
   964→      }
   965→    }
   966→  };
   967→
   968→  // Ended event — submit final progress with completion hint.
   969→  const handleEnded = () => {
   970→    void postProgress(true);
   971→  };
   972→
   973→  const postProgress = useCallback(
   974→    async (isCompletionHint: boolean) => {
   975→      setPosting(true);
   976→      try {
   977→        const res = await apiPost<{
   978→          progress: VideoProgressInfo;
   979→          justCompleted: boolean;
   980→          rejectedCompletion: boolean;
   981→          certificateIssued: boolean;
   982→          certificateId: string | null;
   983→        }>("/api/student/progress", {
   984→          videoId: video.id,
   985→          lastPosition: videoRef.current?.currentTime || 0,
   986→          watchTimeSeconds: watchTimeRef.current,
   987→          duration: durationRef.current,
   988→          isCompleted: isCompletionHint,
   989→        });
   990→        setLocalCompleted(res.progress.isCompleted);
   991→        onProgress({
   992→          videoId: video.id,
   993→          progress: res.progress,
   994→          justCompleted: res.justCompleted,
   995→        });
   996→        if (isCompletionHint && res.rejectedCompletion) {
   997→          toast({
   998→            title: "Completion not recorded",
   999→            description:
  1000→              "You need to watch at least 85% of the video before it's marked complete.",
  1001→            variant: "destructive",
  1002→          });
  1003→        }
  1004→        // ─── Certificate auto-issued upon course completion ───
  1005→        if (res.certificateIssued && res.certificateId) {
  1006→          toast({
  1007→            title: "🎉 Certificate Issued!",
  1008→            description:
  1009→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
  1010→          });
  1011→        }
  1012→      } catch {
  1013→        // apiPost already surfaces errors
  1014→      } finally {
  1015→        setPosting(false);
  1016→      }
  1017→    },
  1018→    [video.id, onProgress]
  1019→  );
  1020→
  1021→  // Manual "Mark complete" for YouTube videos
  1022→  const markYtComplete = async () => {
  1023→    setPosting(true);
  1024→    try {
  1025→      const res = await apiPost<{
  1026→        progress: VideoProgressInfo;
  1027→        justCompleted: boolean;
  1028→        rejectedCompletion: boolean;
  1029→        certificateIssued: boolean;
  1030→        certificateId: string | null;
  1031→      }>("/api/student/progress", {
  1032→        videoId: video.id,
  1033→        lastPosition: durationRef.current,
  1034→        watchTimeSeconds: ytWatchSecondsRef.current,
  1035→        duration: durationRef.current,
  1036→        isCompleted: true,
  1037→      });
  1038→      setLocalCompleted(res.progress.isCompleted);
  1039→      onProgress({
  1040→        videoId: video.id,
  1041→        progress: res.progress,
  1042→        justCompleted: res.justCompleted,
  1043→      });
  1044→      if (res.justCompleted) {
  1045→        toast({
  1046→          title: "Lesson completed",
  1047→          description: "Great work! Your progress has been saved.",
  1048→        });
  1049→      }
  1050→      if (res.rejectedCompletion) {
  1051→        toast({
  1052→          title: "Completion not recorded",
  1053→          description:
  1054→            "You need to keep the video playing for at least 85% of its duration.",
  1055→          variant: "destructive",
  1056→        });
  1057→      }
  1058→      // Certificate auto-issued upon course completion
  1059→      if (res.certificateIssued && res.certificateId) {
  1060→        toast({
  1061→          title: "🎉 Certificate Issued!",
  1062→          description:
  1063→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
  1064→        });
  1065→      }
  1066→    } catch {
  1067→    } finally {
  1068→      setPosting(false);
  1069→    }
  1070→  };
  1071→

  1543→/* ===================== Helpers ===================== */

  1570→  return (
