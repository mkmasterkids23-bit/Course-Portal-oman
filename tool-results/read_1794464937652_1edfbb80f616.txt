  1168→function AssignmentFormDialog({
  1169→  open, onOpenChange, courseId, target, onSaved,
  1170→}: {
  1171→  open: boolean;
  1172→  onOpenChange: (v: boolean) => void;
  1173→  courseId: string;
  1174→  target?: AssignmentRow | null;
  1175→  onSaved: () => void;
  1176→}) {
  1177→  const [title, setTitle] = useState("");
  1178→  const [description, setDescription] = useState("");
  1179→  const [instructions, setInstructions] = useState("");
  1180→  const [dueDate, setDueDate] = useState("");
  1181→  const [isPublished, setIsPublished] = useState(true);
  1182→  const [allowResubmission, setAllowResubmission] = useState(false);
  1183→  const [file, setFile] = useState<File | null>(null);
  1184→  const [clearFile, setClearFile] = useState(false);
  1185→  const [saving, setSaving] = useState(false);
  1186→
  1187→  useEffect(() => {
  1188→    if (target) {
  1189→      setTitle(target.title);
  1190→      setDescription(target.description || "");
  1191→      setInstructions(target.instructions || "");
  1192→      setDueDate(target.dueDate ? target.dueDate.slice(0, 10) : "");
  1193→      setIsPublished(target.isPublished);
  1194→      setAllowResubmission(target.allowResubmission);
  1195→    } else if (open) {
  1196→      setTitle("");
  1197→      setDescription("");
  1198→      setInstructions("");
  1199→      setDueDate("");
  1200→      setIsPublished(true);
  1201→      setAllowResubmission(false);
  1202→    }
  1203→    setFile(null);
  1204→    setClearFile(false);
  1205→  }, [target, open]);
  1206→
  1207→  async function submit() {
  1208→    if (!title.trim()) {
  1209→      toast({ title: "Title required", variant: "destructive" });
  1210→      return;
  1211→    }
  1212→    const fd = new FormData();
  1213→    fd.append("title", title.trim());
  1214→    fd.append("description", description.trim());
  1215→    fd.append("instructions", instructions.trim());

  1222→    setSaving(true);

  1224→      if (target) {
  1225→        const res = await fetch(`/api/admin/assignments/${target.id}`, {
  1226→          method: "PUT",
  1227→          body: fd,
  1228→          credentials: "include",
  1229→        });

  1231→        if (!res.ok) throw new Error(json.error || "Failed");
  1232→        toast({ title: "Assignment updated" });
  1233→      } else {
  1234→        await apiUpload(`/api/admin/courses/${courseId}/assignments`, fd);
  1235→        toast({ title: "Assignment added" });
  1236→      }
  1237→      onSaved();
  1238→    } catch (e) {
  1239→      if (e instanceof Error) {
  1240→        toast({ title: "Error", description: e.message, variant: "destructive" });
  1241→      }
  1242→    } finally {
  1243→      setSaving(false);
  1244→    }
  1245→  }
  1246→
  1247→  return (
  1248→    <Dialog open={open} onOpenChange={onOpenChange}>
  1249→      <DialogContent className="sm:max-w-lg">
