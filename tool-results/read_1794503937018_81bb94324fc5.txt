   494→            </div>

   496→        </Card>

   499→      {/* ─── Recent Activity timeline ─── */}
   500→      <Card className="shadow-sm transition-all duration-200 hover:shadow-md">
   501→        <CardHeader className="flex flex-row items-center justify-between space-y-0">

   503→            <CardTitle className="text-base flex items-center gap-2">

   507→            <CardDescription className="mt-1">
   508→              {t("student.recentActivityDesc")}
   509→            </CardDescription>
   510→          </div>
   511→        </CardHeader>
   512→        <CardContent>
   513→          {activityLoading ? (
   514→            <div className="space-y-3">
   515→              {[1, 2, 3, 4].map((i) => (
   516→                <Skeleton key={i} className="h-14 w-full rounded-lg" />

   521→              {/* vertical spine */}
   522→              <span
   523→                aria-hidden

   527→                <ActivityTimelineRow key={a.id} item={a} isFirst={i === 0} />

   529→            </ol>
   530→          ) : (
   531→            <EmptyState
   532→              icon={<History className="h-5 w-5" />}
   533→              title={t("student.noActivityYet")}
   534→              description={t("student.startCourseActivity")}

   538→                </Button>
   539→              }
   540→            />
   541→          )}
   542→        </CardContent>
   543→      </Card>
   544→    </div>
   545→  );

   682→function activityMeta(item: ActivityItem, t: (key: any) => string): {
   683→  icon: React.ReactNode;
   684→  badge?: React.ReactNode;
   685→} {
   686→  switch (item.type) {
   687→    case "video_completed":
   688→      return {
   689→        icon: <Video className="h-4 w-4" />,
   690→        badge: (
   691→          <Badge className="bg-accent/10 text-accent border-transparent">
   692→            <CheckCircle2 className="h-3 w-3" /> {t("common.completed")}
   693→          </Badge>
   694→        ),
   695→      };
   696→    case "assessment":
   697→      return {
   698→        icon: item.passed ? (
   699→          <CheckCircle2 className="h-4 w-4 text-success" />
   700→        ) : (
   701→          <AlertTriangle className="h-4 w-4 text-warning" />
   702→        ),
   703→        badge: (
   704→          <Badge
   705→            className={
   706→              item.passed
   707→                ? "bg-success/15 text-success border-transparent"
   708→                : "bg-warning/15 text-warning border-transparent"
   709→            }
   710→          >
   711→            {item.passed ? t("student.passedShort") : t("student.attempted")}
   712→            {typeof item.score === "number" ? ` · ${item.score}%` : ""}
   713→          </Badge>
   714→        ),
   715→      };
   716→    case "submission":
   717→      return {
   718→        icon: <FileText className="h-4 w-4" />,
   719→        badge: (
   720→          <Badge className="bg-primary/10 text-primary border-transparent">
   721→            <CheckCircle2 className="h-3 w-3" /> {t("student.submitted")}
   722→          </Badge>
   723→        ),
   724→      };
   725→    case "certificate":
   726→      return {
   727→        icon: <Award className="h-4 w-4 text-success" />,
   728→        badge: item.certificateNumber ? (
   729→          <Badge className="bg-success/15 text-success border-transparent">
   730→            <Award className="h-3 w-3" /> {item.certificateNumber}
   731→          </Badge>
   732→        ) : undefined,
   733→      };
   734→    default:
   735→      return {
   736→        icon: <XCircle className="h-4 w-4" />,
   737→      };
   738→  }
   739→}

   756→    <Card className="p-4 sm:p-5 shadow-sm transition-all duration-200 hover:shadow-md hover:-translate-y-0.5 overflow-hidden">

   758→        <div className={`flex h-10 w-10 sm:h-11 sm:w-11 items-center justify-center rounded-xl ring-1 ${tintClasses[tint]} shrink-0`}>

   765→      </div>

   767→  );
   768→}
