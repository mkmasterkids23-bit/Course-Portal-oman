   163→    if (search.trim()) p.set("q", search.trim());

   165→    if (planFilter !== "ALL") p.set("planId", planFilter);

   456→                }

   460→            </AlertDialogAction>
   461→          </AlertDialogFooter>
   462→        </AlertDialogContent>
   463→      </AlertDialog>
   464→    </div>

   466→}

   468→// ---------- Add Student Dialog ----------
   469→function AddStudentDialog({
   470→  open,
   471→  onOpenChange,
   472→  plans,
   473→  onCreated,
   474→}: {
   475→  open: boolean;
   476→  onOpenChange: (v: boolean) => void;
   477→  plans: SubscriptionPlanPublic[];
   478→  onCreated: () => void;
   479→}) {
   480→  const [name, setName] = useState("");
   481→  const [email, setEmail] = useState("");
   482→  const [password, setPassword] = useState("");
   483→  const [planId, setPlanId] = useState<string>("NONE");
   484→  const [saving, setSaving] = useState(false);
   485→
   486→  useEffect(() => {
   487→    if (open) {
   488→      setName("");
   489→      setEmail("");
   490→      setPassword("");
   491→      setPlanId("NONE");
   492→    }
   493→  }, [open]);
   494→
   495→  async function submit() {
   496→    if (!name.trim() || !email.trim() || !password) {
   497→      toast({ title: "Missing fields", description: "Name, email and password are required.", variant: "destructive" });
   498→      return;
   499→    }
   500→    setSaving(true);
   501→    try {
   502→      await apiPost("/api/admin/students", {
   503→        name: name.trim(),
   504→        email: email.trim(),
   505→        password,
   506→        planId: planId === "NONE" ? undefined : planId,
   507→      });
   508→      onCreated();
   509→      onOpenChange(false);
   510→    } catch {
   511→      /* toast shown by apiPost */
   512→    } finally {
   513→      setSaving(false);
   514→    }
   515→  }
   516→
   517→  return (
   518→    <Dialog open={open} onOpenChange={onOpenChange}>
   519→      <DialogContent className="sm:max-w-md">
   520→        <DialogHeader>
   521→          <DialogTitle>Add New Student</DialogTitle>
   522→          <DialogDescription>Create a student account. You can optionally assign a plan now.</DialogDescription>
   523→        </DialogHeader>
   524→        <div className="space-y-3">
   525→          <div className="space-y-1.5">
   526→            <Label htmlFor="as-name">Full Name</Label>
   527→            <Input id="as-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Aisha Khalid" />
   528→          </div>
   529→          <div className="space-y-1.5">
   530→            <Label htmlFor="as-email">Email</Label>
   531→            <Input id="as-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="aisha@example.com" />
   532→          </div>
   533→          <div className="space-y-1.5">
   534→            <Label htmlFor="as-pass">Password</Label>
   535→            <Input id="as-pass" type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="Minimum 6 characters" />
   536→          </div>
   537→          <div className="space-y-1.5">
   538→            <Label htmlFor="as-plan">Subscription Plan (optional)</Label>
   539→            <Select value={planId} onValueChange={setPlanId}>
   540→              <SelectTrigger id="as-plan" className="w-full">
   541→                <SelectValue placeholder="No plan — student only" />
   542→              </SelectTrigger>
   543→              <SelectContent>
   544→                <SelectItem value="NONE">No plan — student only</SelectItem>
   545→                {plans.map((p) => (
   546→                  <SelectItem key={p.id} value={p.id}>
   547→                    {p.name}
   548→                  </SelectItem>
   549→                ))}
   550→              </SelectContent>
   551→            </Select>
   552→          </div>
   553→        </div>
   554→        <DialogFooter>
   555→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   556→          <Button disabled={saving} onClick={submit}>
   557→            {saving ? "Creating…" : "Create Student"}
   558→          </Button>
   559→        </DialogFooter>
   560→      </DialogContent>
   561→    </Dialog>
   562→  );
   563→}

   565→// ---------- Edit Student Dialog ----------
   566→function EditStudentDialog({
   567→  open,
   568→  onOpenChange,
   569→  target,
   570→  onSaved,
   571→}: {
   572→  open: boolean;
   573→  onOpenChange: (v: boolean) => void;
   574→  target: StudentRow | null;
   575→  plans: SubscriptionPlanPublic[];
   576→  onSaved: () => void;
   577→}) {
   578→  const [name, setName] = useState("");
   579→  const [phone, setPhone] = useState("");
   580→  const [bio, setBio] = useState("");
   581→  const [saving, setSaving] = useState(false);
   582→
   583→  useEffect(() => {
   584→    if (target) {
   585→      setName(target.name);
   586→      setPhone(target.phone || "");
   587→      setBio("");
   588→    }
   589→  }, [target]);
   590→
   591→  async function submit() {
   592→    if (!target) return;
   593→    if (!name.trim()) {
   594→      toast({ title: "Name required", variant: "destructive" });
   595→      return;
   596→    }
   597→    setSaving(true);
   598→    try {
   599→      await apiPost(`/api/admin/students/${target.id}`, {
   600→        name: name.trim(),
   601→        phone: phone.trim() || null,
   602→        bio: bio.trim() || null,
   603→      }, { method: "PUT" });
   604→      onSaved();
   605→      onOpenChange(false);
   606→    } catch {
   607→      /* toast */
   608→    } finally {
   609→      setSaving(false);
   610→    }
   611→  }
   612→
   613→  return (
   614→    <Dialog open={open} onOpenChange={onOpenChange}>
   615→      <DialogContent className="sm:max-w-md">
   616→        <DialogHeader>
   617→          <DialogTitle>Edit Student</DialogTitle>
   618→          <DialogDescription>Update profile details for {target?.email}</DialogDescription>
   619→        </DialogHeader>
   620→        <div className="space-y-3">
   621→          <div className="space-y-1.5">
   622→            <Label htmlFor="es-name">Full Name</Label>
   623→            <Input id="es-name" value={name} onChange={(e) => setName(e.target.value)} />
   624→          </div>
   625→          <div className="space-y-1.5">
   626→            <Label htmlFor="es-phone">Phone</Label>
   627→            <Input id="es-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+968 9XXX XXXX" />
   628→          </div>
   629→          <div className="space-y-1.5">
   630→            <Label htmlFor="es-bio">Bio</Label>
   631→            <Textarea id="es-bio" value={bio} onChange={(e) => setBio(e.target.value)} placeholder="Short bio (optional)" rows={3} />
   632→          </div>
   633→        </div>
   634→        <DialogFooter>
   635→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   636→          <Button disabled={saving} onClick={submit}>
   637→            {saving ? "Saving…" : "Save Changes"}
   638→          </Button>
   639→        </DialogFooter>
   640→      </DialogContent>
   641→    </Dialog>
   642→  );
