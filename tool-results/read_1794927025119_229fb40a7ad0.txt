   160→   160→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   161→   161→  const { t } = useI18n();
   162→   162→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   163→   163→    `/api/student/courses/${courseId}`
   164→   164→  );
   165→   165→
   166→   166→  // Read the hash query for deep-link support. The global Discussions
   167→   167→  // view navigates here with `?tab=discussions&discussion=<id>`; we honour
   168→   168→  // that by defaulting the Materials/Discussions tab to "discussions" and
   169→   169→  // asking the panel to highlight (and scroll to) the matching thread.
   170→   170→  const routeQuery = useAppStore((s) => s.query);
   171→   171→  const highlightDiscussionId = routeQuery.discussion || null;
   172→   172→  const discussionsTabDefault = routeQuery.tab === "discussions";
   173→   173→
   174→   174→  // Pick the initial video: last watched, or first available.
   175→   175→  const initialVideoId = (() => {
   176→   176→    if (!data?.course) return null;
   177→   177→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   178→   178→    //    most recently updated progress entry in the topics tree.
   179→   179→    let candidate: VideoItem | null = null;
   180→   180→    for (const t of data.course.topics) {
   181→   181→      for (const v of t.videos) {
   182→   182→        if (v.progress && !v.progress.isCompleted) {
   183→   183→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   184→   184→            candidate = v;
   185→   185→          }
   186→   186→        }
   187→   187→      }
   188→   188→    }
   189→   189→    if (candidate) return candidate.id;
   190→   190→    // 2) First video of the first topic
   191→   191→    return data.course.topics[0]?.videos[0]?.id ?? null;
   192→   192→  })();
   193→   193→
   194→   194→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   195→   195→  // Reset user selection when courseId changes — "adjust state during render"
   196→   196→  // pattern (avoids setState-in-effect warnings).
   197→   197→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   198→   198→  if (courseId !== prevCourseId) {
   199→   199→    setPrevCourseId(courseId);
   200→   200→    setUserSelectedVideoId(null);
   201→   201→  }
   202→   202→  // Effective active video: explicit user selection overrides the auto-pick
   203→   203→  // (which is the last-watched or first video). When data loads and we have an
   204→   204→  // initial pick, this falls back to it without needing an effect.
   205→   205→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   206→   206→
   207→   207→  if (loading) {
   208→   208→    return (
   209→   209→      <div className="space-y-5">
   210→   210→        <Skeleton className="h-10 w-32 rounded-lg" />
   211→   211→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   212→   212→          <div className="space-y-4">
   213→   213→            <Skeleton className="h-[420px] rounded-xl" />
   214→   214→            <Skeleton className="h-24 rounded-xl" />
   215→   215→          </div>
   216→   216→          <Skeleton className="h-[600px] rounded-xl" />
   217→   217→        </div>
   218→   218→      </div>
   219→   219→    );
   220→   220→  }
   221→   221→
   222→   222→  if (error) {
   223→   223→    return (
   224→   224→      <div className="space-y-4">
   225→   225→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   226→   226→          <ArrowLeft className="h-4 w-4" /> {t("learning.backToCourses")}
   227→   227→        </Button>
   228→   228→        <EmptyState
   229→   229→          icon={<AlertTriangle className="h-5 w-5" />}
   230→   230→          title={t("learning.couldntOpenCourse")}
   231→   231→          description={error}
   232→   232→          action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   233→   233→        />
   234→   234→      </div>
   235→   235→    );
   236→   236→  }
   237→   237→
   238→   238→  if (!data?.course) {
   239→   239→    return (
   240→   240→      <EmptyState
   241→   241→        icon={<PlayCircle className="h-5 w-5" />}
   242→   242→        title={t("learning.courseNotFound")}
   243→   243→        description={t("learning.courseMayRemoved")}
   244→   244→        action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   245→   245→      />
   246→   246→    );
   247→   247→  }
   248→   248→
   249→   249→  const course = data.course;
   250→   250→  const allVideos = course.topics.flatMap((t) => t.videos);
   251→   251→  const activeVideo =
   252→   252→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   253→   253→
   254→   254→  // Find the topic containing the active video
   255→   255→  const activeTopic = course.topics.find((t) =>
   256→   256→    t.videos.some((v) => v.id === activeVideo?.id)
   257→   257→  );
   258→   258→
   259→   259→  // Build a flat ordered list for prev/next navigation
   260→   260→  const flatVideoIds = allVideos.map((v) => v.id);
   261→   261→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   262→   262→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   263→   263→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   264→   264→
   265→   265→  function handleProgressUpdate(updated: {
   266→   266→    videoId: string;
   267→   267→    progress: VideoProgressInfo;
   268→   268→    justCompleted: boolean;
   269→   269→  }) {
   270→   270→    if (!data) return;
   271→   271→    const newCourse = {
   272→   272→      ...data.course,
   273→   273→      topics: data.course.topics.map((t) => ({
   274→   274→        ...t,
   275→   275→        videos: t.videos.map((v) =>
   276→   276→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   277→   277→        ),
   278→   278→      })),
   279→   279→      courseCompletion: (() => {
   280→   280→        // Recompute local snapshot
   281→   281→        const videos = data.course.topics.flatMap((t) => t.videos);
   282→   282→        const completed = videos.filter(
   283→   283→          (v) =>
   284→   284→            v.id === updated.videoId
   285→   285→              ? updated.progress.isCompleted
   286→   286→              : v.progress?.isCompleted
   287→   287→        ).length;
   288→   288→        const total = videos.length;
   289→   289→        // No videos → course is considered fully watched (assignment unlocked).
   290→   290→        const percent = total > 0 ? Math.round((completed / total) * 100) : 100;
   291→   291→        const totalTopics = data.course.topics.length;
   292→   292→        const completedTopics = data.course.topics.filter((topic) =>
   293→   293→          topic.videos.length > 0 &&
   294→   294→          topic.videos.every(
   295→   295→            (v) =>
   296→   296→              (v.id === updated.videoId ? updated.progress.isCompleted : v.progress?.isCompleted)
   297→   297→          )
   298→   298→        ).length;
   299→   299→        return {
   300→   300→          totalVideos: total,
   301→   301→          completedVideos: completed,
   302→   302→          completedTopics,
   303→   303→          totalTopics,
   304→   304→          percent,
   305→   305→          allCompleted: total === 0 || completed === total,
   306→   306→        };
   307→   307→      })(),
   308→   308→    };
   309→   309→    setData({ course: newCourse });
   310→   310→
   311→   311→    if (updated.justCompleted) {
   312→   312→      toast({
   313→   313→        title: t("learning.lessonCompletedToast"),
   314→   314→        description: t("learning.greatWorkSaved"),
   315→   315→      });
   316→   316→    }
   317→   317→  }
   318→   318→
   319→   319→  async function handleSubmissionSuccess() {
   320→   320→    await refetch();
   321→   321→    toast({
   322→   322→      title: t("learning.assignmentSubmittedToast"),
   323→   323→      description: t("learning.assignmentSubmittedDesc"),
   324→   324→    });
   325→   325→  }
   326→   326→
   327→   327→  return (
   328→   328→    <div className="space-y-4 animate-page-in pb-20 lg:pb-4">
   329→   329→      {/* ─── Compact breadcrumb header ─── */}
   330→   330→      <div className="shadow-premium flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-4 py-3">
   331→   331→        <div className="flex items-center gap-2 min-w-0 flex-1">
   332→   332→          <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0 h-8 w-8" aria-label={t("learning.backToCourses")}>
   333→   333→            <ArrowLeft className="h-4 w-4" />
   334→   334→          </Button>
   335→   335→          <div className="min-w-0 flex-1">
   336→   336→            <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   337→   337→              <button onClick={onBack} className="hover:text-foreground transition-colors">{t("learning.myCourses")}</button>
   338→   338→              <span>/</span>
   339→   339→              <span className="truncate">{course.title}</span>
   340→   340→              {activeVideo && (
   341→   341→                <>
   342→   342→                  <span>/</span>
   343→   343→                  <span className="truncate text-foreground font-medium">{activeVideo.title}</span>
   344→   344→                </>
   345→   345→              )}
   346→   346→            </div>
   347→   347→            <div className="flex items-center gap-2 mt-0.5">
   348→   348→              {course.category && (
   349→   349→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   350→   350→                  {course.category}
   351→   351→                </Badge>
   352→   352→              )}
   353→   353→              <Badge
   354→   354→                className={
   355→   355→                  course.accessType === "FREE"
   356→   356→                    ? "bg-success/15 text-success border-transparent text-[10px] px-1.5 py-0 h-4"
   357→   357→                    : "bg-primary text-primary-foreground border-transparent text-[10px] px-1.5 py-0 h-4"
   358→   358→                }
   359→   359→              >
   360→   360→                {course.accessType === "FREE" ? t("common.free") : t("common.paid")}
   361→   361→              </Badge>
   362→   362→            </div>
   363→   363→          </div>
   364→   364→        </div>
   365→   365→        <div className="flex items-center gap-3 shrink-0">
   366→   366→          <div className="hidden sm:flex flex-col items-end">
   367→   367→            <span className="text-xs font-medium text-foreground">
   368→   368→              {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} {t("courses.lessons")}
   369→   369→            </span>
   370→   370→            <span className="text-[10px] text-muted-foreground">{t("learning.completed")}</span>
   371→   371→          </div>
   372→   372→          <div className="flex items-center gap-2 min-w-[120px]">
   373→   373→            <Progress
   374→   374→              value={course.courseCompletion.percent}
   375→   375→              className={`h-1.5 w-24 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   376→   376→            />
   377→   377→            <span className="text-xs font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   378→   378→          </div>
   379→   379→        </div>
   380→   380→      </div>
   381→   381→
   382→   382→      {/* ─── Main Layout: LEFT content + RIGHT curriculum ─── */}
   383→   383→      <div className="grid grid-cols-1 lg:grid-cols-[1fr_340px] gap-4">
   384→   384→        {/* ─── LEFT: Video + Tabs ─── */}
   385→   385→        <div className="space-y-4 min-w-0">
   386→   386→          {/* Video player */}
   387→   387→          {activeVideo ? (
   388→   388→            <VideoPlayer
   389→   389→              key={activeVideo.id}
   390→   390→              video={activeVideo}
   391→   391→              topicTitle={activeTopic?.title || null}
   392→   392→              courseId={course.id}
   393→   393→              topicId={activeTopic?.id || null}
   394→   394→              onProgress={handleProgressUpdate}
   395→   395→              prevVideo={prevVideo}
   396→   396→              nextVideo={nextVideo}
   397→   397→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   398→   398→              refetchCourse={refetch}
   399→   399→            />
   400→   400→          ) : (
   401→   401→            <EmptyState
   402→   402→              icon={<VideoIcon className="h-5 w-5" />}
   403→   403→              title={t("courses.noVideosYet")}
   404→   404→              description={t("courses.noVideosDesc")}
   405→   405→            />
   406→   406→          )}
   407→   407→
   408→   408→          {/* Assignment panel — hidden until all videos completed */}
   409→   409→          {course.assignment && course.courseCompletion.allCompleted ? (
   410→   410→            <AssignmentPanel course={course} onSubmitted={handleSubmissionSuccess} />
   411→   411→          ) : course.assignment ? (
   412→   412→            <LockedAssignmentCard
   413→   413→              completedVideos={course.courseCompletion.completedVideos}
   414→   414→              totalVideos={course.courseCompletion.totalVideos}
   415→   415→              percent={course.courseCompletion.percent}
   416→   416→            />
   417→   417→          ) : null}
   418→   418→
   419→   419→          {/* ─── 4-Tab Interface below video ─── */}
   420→   420→          <Tabs defaultValue="overview" className="gap-0">
   421→   421→            <div className="border-b border-border overflow-x-auto scrollbar-thin">
   422→   422→              <TabsList className="bg-transparent border-0 p-0 h-auto gap-0 flex-nowrap">
   423→   423→                <TabsTrigger
   424→   424→                  value="overview"
   425→   425→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   426→   426→                >
   427→   427→                  <Info className="h-3.5 w-3.5" />
   428→   428→                  {t("learning.tab.overview")}
   429→   429→                </TabsTrigger>
   430→   430→                <TabsTrigger
   431→   431→                  value="notes"
   432→   432→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   433→   433→                >
   434→   434→                  <StickyNote className="h-3.5 w-3.5" />
   435→   435→                  {t("learning.tab.notes")}
   436→   436→                </TabsTrigger>
   437→   437→                <TabsTrigger
   438→   438→                  value="discussions"
   439→   439→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   440→   440→                >
   441→   441→                  <MessageSquare className="h-3.5 w-3.5" />
   442→   442→                  {t("learning.tab.discussion")}
   443→   443→                </TabsTrigger>
   444→   444→                <TabsTrigger
   445→   445→                  value="resources"
   446→   446→                  className="tab-underline rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   447→   447→                >
   448→   448→                  <FileText className="h-3.5 w-3.5" />
   449→   449→                  {t("learning.tab.resources")}

   720→   720→function VideoPlayer({
   721→   721→  video,
   722→   722→  topicTitle,

   724→   724→  topicId,
   725→   725→  onProgress,

   728→   728→  onNavigateLesson,

   779→   779→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   780→   780→

   782→   782→  const ytStartRef = useRef<number | null>(null);
   783→   783→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);

   797→   797→    questionsData &&

   799→   799→      questionsData.assessment?.passed !== true

   809→   809→    setLocalCompleted(video.progress?.isCompleted || false);
   810→   810→    setYtCanMarkComplete(false);
   811→   811→    ytWatchSecondsRef.current = 0;

   830→   830→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   831→   831→  // the student answers a question. If the playhead has already moved past an
   832→   832→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   833→   833→  // the questions fetch landed late), pause and surface that question now.
   834→   834→  useEffect(() => {

   849→   849→      setSelectedAnswer(null);
   850→   850→    }
   851→   851→  }, [
   852→   852→    questionsActive,
   853→   853→    questionsData,
   854→   854→    activeQuestion,
   855→   855→    assessmentResult,
   856→   856→    showFeedback,
   857→   857→    submittingAssessment,
   858→   858→    answeredMap,
   859→   859→  ]);

   862→   862→  const handleLoadedMetadata = () => {

   866→   866→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   867→   867→      // resume within (duration - 2) so we don't immediately fire ended
   868→   868→      const target = Math.min(
   869→   869→        video.progress.lastPosition,
   870→   870→        Math.max(0, (el.duration || 0) - 3)
   871→   871→      );
   872→   872→      try {
   873→   873→        el.currentTime = target;
   874→   874→      } catch {}
   875→   875→      lastTimeRef.current = target;
   876→   876→      resumeAppliedRef.current = true;
   877→   877→    }
   878→   878→  };
   879→   879→
   880→   880→  // Core: timeupdate handler — accumulate real watch time.
   881→   881→  const handleTimeUpdate = () => {
   882→   882→    const el = videoRef.current;
   883→   883→    if (!el) return;
   884→   884→    const cur = el.currentTime;
   885→   885→    const last = lastTimeRef.current;
   886→   886→
   887→   887→    // ─── Question trigger check ───
   888→   888→    // Pause and surface a popup the moment the playhead crosses an unanswered
   889→   889→    // question's timestamp going forward. We require BOTH:
   890→   890→    //   - cur >= q.timestamp    (we're at or past it)
   891→   891→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   892→   892→    //                            this is a forward crossing, not a re-tick while
   893→   893→    //                            paused, and not a backward seek landing past it)
   894→   894→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   895→   895→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   896→   896→    if (
   897→   897→      questionsActive &&
   898→   898→      !activeQuestion &&
   899→   899→      !assessmentResult &&
   900→   900→      !showFeedback &&
   901→   901→      !submittingAssessment &&
   902→   902→      questionsData
   903→   903→    ) {
   904→   904→      for (const q of questionsData.questions) {
   905→   905→        if (answeredMap.has(q.id)) continue;
   906→   906→        if (cur >= q.timestamp && last < q.timestamp) {
   907→   907→          try {
   908→   908→            el.pause();
   909→   909→            lastTimeRef.current = cur;
   910→   910→          } catch {}
   911→   911→          setActiveQuestion(q);
   912→   912→          setSelectedAnswer(null);
   913→   913→          return; // do NOT accumulate watch time for this tick
   914→   914→        }
   915→   915→      }
   916→   916→    }
   917→   917→
   918→   918→    // Watch time tracking — only counts normal forward playback deltas.
   919→   919→    const delta = cur - last;
   920→   920→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   921→   921→    // (large positive deltas) and rewinds (negative deltas).
   922→   922→    if (delta > 0 && delta <= 1.5) {
   923→   923→      watchTimeRef.current += delta;
   924→   924→    }
   925→   925→    // Update lastTime to current — even on seek, so next delta is small.
   926→   926→    lastTimeRef.current = cur;
   927→   927→
   928→   928→    // Throttled server POST every 5 seconds
   929→   929→    const now = Date.now();
   930→   930→    if (now - lastPostRef.current > 5000) {
   931→   931→      lastPostRef.current = now;
   932→   932→      void postProgress(false);
   933→   933→    }
   934→   934→  };
   935→   935→
   936→   936→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   937→   937→  // Also catches forward-seek-past-question: if the student scrubs the
   938→   938→  // progress bar past an unanswered question, pause and surface that question
   939→   939→  // so they can't skip it.
   940→   940→  const handleSeeked = () => {
   941→   941→    const el = videoRef.current;
   942→   942→    if (!el) return;
   943→   943→    const newTime = el.currentTime;
   944→   944→    lastTimeRef.current = newTime;
   945→   945→
   946→   946→    if (
   947→   947→      questionsActive &&
   948→   948→      !activeQuestion &&
   949→   949→      !assessmentResult &&
   950→   950→      !showFeedback &&
   951→   951→      !submittingAssessment &&
   952→   952→      questionsData
   953→   953→    ) {
   954→   954→      // Find the earliest unanswered question we've now skipped past.
   955→   955→      const skipped = questionsData.questions.find(
   956→   956→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
   957→   957→      );
   958→   958→      if (skipped) {
   959→   959→        try {
   960→   960→          el.pause();
   961→   961→        } catch {}
   962→   962→        setActiveQuestion(skipped);
   963→   963→        setSelectedAnswer(null);
   964→   964→      }
   965→   965→    }
   966→   966→  };
   967→   967→
   968→   968→  // Ended event — submit final progress with completion hint.
   969→   969→  const handleEnded = () => {
   970→   970→    void postProgress(true);
   971→   971→  };

  1019→  1019→  );
  1020→  1020→

  1128→  1128→

  1130→  1130→      // after a forward seek that skipped several), surface the next one

  1157→  1157→  // afterAnswerSubmitted.
