   336→   336→
   337→   337→function CourseFormDialog({
   338→   338→  open,
   339→   339→  onOpenChange,
   340→   340→  target,
   341→   341→  plans,
   342→   342→  onSaved,
   343→   343→}: {
   344→   344→  open: boolean;
   345→   345→  onOpenChange: (v: boolean) => void;
   346→   346→  target?: CourseRow | null;
   347→   347→  plans: SubscriptionPlanPublic[];
   348→   348→  onSaved: (createdId?: string) => void;
   349→   349→}) {
   350→   350→  const isEdit = !!target;
   351→   351→  const [values, setValues] = useState<CourseFormValues>({
   352→   352→    title: "",
   353→   353→    description: "",
   354→   354→    category: "",
   355→   355→    accessType: "FREE",
   356→   356→    isPublished: false,
   357→   357→    thumbnail: "",
   358→   358→    planIds: [],
   359→   359→  });
   360→   360→  const [saving, setSaving] = useState(false);
   361→   361→  const [uploadingThumb, setUploadingThumb] = useState(false);
   362→   362→
   363→   363→  useEffect(() => {
   364→   364→    if (target) {
   365→   365→      setValues({
   366→   366→        title: target.title,
   367→   367→        description: target.description || "",
   368→   368→        category: target.category || "",
   369→   369→        accessType: target.accessType,
   370→   370→        isPublished: target.isPublished,
   371→   371→        thumbnail: target.thumbnail || "",
   372→   372→        planIds: [],
   373→   373→      });
   374→   374→    } else if (open) {
   375→   375→      setValues({
   376→   376→        title: "",
   377→   377→        description: "",
   378→   378→        category: "",
   379→   379→        accessType: "FREE",
   380→   380→        isPublished: false,
   381→   381→        thumbnail: "",
   382→   382→        planIds: [],
   383→   383→      });
   384→   384→    }
   385→   385→  }, [target, open]);
   386→   386→
   387→   387→  function togglePlan(id: string) {
   388→   388→    setValues((v) => ({
   389→   389→      ...v,
   390→   390→      planIds: v.planIds.includes(id)
   391→   391→        ? v.planIds.filter((p) => p !== id)
   392→   392→        : [...v.planIds, id],
   393→   393→    }));
   394→   394→  }
   395→   395→
   396→   396→  async function uploadThumbnail(file: File) {
   397→   397→    if (!target) {
   398→   398→      toast({
   399→   399→        title: "Save the course first",
   400→   400→        description: "Thumbnail upload is available after the course is created.",
   401→   401→        variant: "default",
   402→   402→      });
   403→   403→      return;
   404→   404→    }
   405→   405→    setUploadingThumb(true);
   406→   406→    try {
   407→   407→      const fd = new FormData();
   408→   408→      fd.append("file", file);
   409→   409→      const res = await apiUpload<{ thumbnail: string }>(
   410→   410→        `/api/university/courses/${target.id}/thumbnail`,
   411→   411→        fd
   412→   412→      );
   413→   413→      setValues((v) => ({ ...v, thumbnail: res.thumbnail }));
   414→   414→      toast({ title: "Thumbnail uploaded" });
   415→   415→    } catch {
   416→   416→      /* toast */
   417→   417→    } finally {
   418→   418→      setUploadingThumb(false);
   419→   419→    }
   420→   420→  }
   421→   421→
   422→   422→  async function submit() {
   423→   423→    if (!values.title.trim()) {
   424→   424→      toast({ title: "Title required", variant: "destructive" });
   425→   425→      return;
   426→   426→    }
   427→   427→    if (values.accessType === "PAID" && values.planIds.length === 0) {
   428→   428→      toast({
   429→   429→        title: "Select at least one plan",
   430→   430→        description: "Paid courses must be assigned to one or more subscription plans.",
   431→   431→        variant: "destructive",
   432→   432→      });
   433→   433→      return;
   434→   434→    }
   435→   435→
   436→   436→    setSaving(true);
   437→   437→    try {
   438→   438→      if (isEdit && target) {
   439→   439→        await apiPost(`/api/university/courses/${target.id}`, {
   440→   440→          title: values.title.trim(),
   441→   441→          description: values.description.trim() || null,
   442→   442→          category: values.category.trim() || null,
   443→   443→          accessType: values.accessType,
   444→   444→          isPublished: values.isPublished,
   445→   445→          thumbnail: values.thumbnail || null,
   446→   446→          ...(values.accessType === "PAID" ? { planIds: values.planIds } : {}),
   447→   447→        }, { method: "PUT" });
   448→   448→        onSaved();
   449→   449→      } else {
   450→   450→        const res = await apiPost<{ course: { id: string } }>(
   451→   451→          "/api/university/courses",
   452→   452→          {
   453→   453→            title: values.title.trim(),
   454→   454→            description: values.description.trim() || null,
   455→   455→            category: values.category.trim() || null,
   456→   456→            accessType: values.accessType,
   457→   457→            isPublished: values.isPublished,
   458→   458→            thumbnail: values.thumbnail || null,
   459→   459→            planIds: values.planIds,
   460→   460→          }
   461→   461→        );
   462→   462→        onSaved(res.course.id);
   463→   463→      }
   464→   464→    } catch {
   465→   465→      /* toast */
   466→   466→    } finally {
   467→   467→      setSaving(false);
   468→   468→    }
   469→   469→  }
   470→   470→
   471→   471→  return (
   472→   472→    <Dialog open={open} onOpenChange={onOpenChange}>
   473→   473→      <DialogContent className="sm:max-w-lg">
   474→   474→        <DialogHeader>
   475→   475→          <DialogTitle className="font-heading text-lg font-semibold">
   476→   476→            {isEdit ? "Edit Course" : "Create New Course"}
   477→   477→          </DialogTitle>
   478→   478→          <DialogDescription>
   479→   479→            {isEdit
   480→   480→              ? "Update course details. Plan assignment and content can be edited in the course manager."
   481→   481→              : "Set up the basics. The course will be automatically scoped to your university."}
   482→   482→          </DialogDescription>
   483→   483→        </DialogHeader>
   484→   484→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1 scrollbar-thin">
   485→   485→          <div className="space-y-1.5">
   486→   486→            <Label htmlFor="cf-title" className="text-xs font-medium text-muted-foreground">
   487→   487→              Title
   488→   488→            </Label>
   489→   489→            <Input
   490→   490→              id="cf-title"
   491→   491→              value={values.title}
   492→   492→              onChange={(e) => setValues({ ...values, title: e.target.value })}
   493→   493→              placeholder="e.g. Digital Marketing Fundamentals"
   494→   494→              className="h-9 rounded-lg"
   495→   495→            />
   496→   496→          </div>
   497→   497→          <div className="space-y-1.5">
   498→   498→            <Label htmlFor="cf-desc" className="text-xs font-medium text-muted-foreground">
   499→   499→              Description
   500→   500→            </Label>
   501→   501→            <Textarea
