   291→   291→interface CourseFormValues {

   300→   300→
   301→   301→function CourseFormDialog({
   302→   302→  open,
   303→   303→  onOpenChange,
   304→   304→  target,
   305→   305→  plans,
   306→   306→  onSaved,
   307→   307→}: {
   308→   308→  open: boolean;
   309→   309→  onOpenChange: (v: boolean) => void;
   310→   310→  target?: CourseRow | null;
   311→   311→  plans: SubscriptionPlanPublic[];
   312→   312→  onSaved: (createdId?: string) => void;
   313→   313→}) {
   314→   314→  const isEdit = !!target;
   315→   315→  const [values, setValues] = useState<CourseFormValues>({
   316→   316→    title: "",
   317→   317→    description: "",
   318→   318→    category: "",
   319→   319→    accessType: "FREE",
   320→   320→    isPublished: false,
   321→   321→    thumbnail: "",
   322→   322→    planIds: [],
   323→   323→  });
   324→   324→  const [saving, setSaving] = useState(false);
   325→   325→  const [uploadingThumb, setUploadingThumb] = useState(false);
   326→   326→
   327→   327→  useEffect(() => {
   328→   328→    if (target) {
   329→   329→      setValues({
   330→   330→        title: target.title,
   331→   331→        description: target.description || "",
   332→   332→        category: target.category || "",
   333→   333→        accessType: target.accessType,
   334→   334→        isPublished: target.isPublished,
   335→   335→        thumbnail: target.thumbnail || "",
   336→   336→        planIds: [],
   337→   337→      });
   338→   338→    } else if (open) {
   339→   339→      setValues({
   340→   340→        title: "",
   341→   341→        description: "",
   342→   342→        category: "",
   343→   343→        accessType: "FREE",
   344→   344→        isPublished: false,
   345→   345→        thumbnail: "",
   346→   346→        planIds: [],
   347→   347→      });
   348→   348→    }
   349→   349→  }, [target, open]);
   350→   350→
   351→   351→  function togglePlan(id: string) {
   352→   352→    setValues((v) => ({
   353→   353→      ...v,
   354→   354→      planIds: v.planIds.includes(id)
   355→   355→        ? v.planIds.filter((p) => p !== id)
   356→   356→        : [...v.planIds, id],
   357→   357→    }));
   358→   358→  }
   359→   359→
   360→   360→  async function uploadThumbnail(file: File) {
   361→   361→    if (!target) {
   362→   362→      // No course yet — defer; just keep file URL empty and warn user
   363→   363→      toast({
   364→   364→        title: "Save the course first",
   365→   365→        description: "Thumbnail upload is available after the course is created.",
   366→   366→        variant: "default",
   367→   367→      });
   368→   368→      return;
   369→   369→    }
   370→   370→    setUploadingThumb(true);
   371→   371→    try {
   372→   372→      const fd = new FormData();
   373→   373→      fd.append("file", file);
   374→   374→      const res = await apiUpload<{ thumbnail: string }>(`/api/admin/courses/${target.id}/thumbnail`, fd);
   375→   375→      setValues((v) => ({ ...v, thumbnail: res.thumbnail }));
   376→   376→      toast({ title: "Thumbnail uploaded" });
   377→   377→    } catch {
   378→   378→      /* toast */
   379→   379→    } finally {
   380→   380→      setUploadingThumb(false);
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  async function submit() {
   385→   385→    if (!values.title.trim()) {
   386→   386→      toast({ title: "Title required", variant: "destructive" });
   387→   387→      return;
   388→   388→    }
   389→   389→    if (values.accessType === "PAID" && values.planIds.length === 0) {
   390→   390→      toast({
   391→   391→        title: "Select at least one plan",
   392→   392→        description: "Paid courses must be assigned to one or more subscription plans.",
   393→   393→        variant: "destructive",
   394→   394→      });
   395→   395→      return;
   396→   396→    }
   397→   397→
   398→   398→    setSaving(true);
   399→   399→    try {
   400→   400→      if (isEdit && target) {
   401→   401→        await apiPost(`/api/admin/courses/${target.id}`, {
   402→   402→          title: values.title.trim(),
   403→   403→          description: values.description.trim() || null,
   404→   404→          category: values.category.trim() || null,
   405→   405→          accessType: values.accessType,
   406→   406→          isPublished: values.isPublished,
   407→   407→          thumbnail: values.thumbnail || null,
   408→   408→          ...(values.accessType === "PAID" ? { planIds: values.planIds } : {}),
   409→   409→        }, { method: "PUT" });
   410→   410→        onSaved();
   411→   411→      } else {
   412→   412→        const res = await apiPost<{ course: { id: string } }>("/api/admin/courses", {
   413→   413→          title: values.title.trim(),
   414→   414→          description: values.description.trim() || null,
   415→   415→          category: values.category.trim() || null,
   416→   416→          accessType: values.accessType,
   417→   417→          isPublished: values.isPublished,
   418→   418→          thumbnail: values.thumbnail || null,
   419→   419→          planIds: values.planIds,
   420→   420→        });
   421→   421→        onSaved(res.course.id);
   422→   422→      }
   423→   423→    } catch {
   424→   424→      /* toast */
   425→   425→    } finally {
   426→   426→      setSaving(false);
   427→   427→    }
   428→   428→  }
   429→   429→
   430→   430→  return (
   431→   431→    <Dialog open={open} onOpenChange={onOpenChange}>
   432→   432→      <DialogContent className="sm:max-w-lg">
   433→   433→        <DialogHeader>
   434→   434→          <DialogTitle className="font-heading text-lg font-semibold">{isEdit ? "Edit Course" : "Create New Course"}</DialogTitle>
   435→   435→          <DialogDescription>
   436→   436→            {isEdit
   437→   437→              ? "Update course details. Plan assignment and content can be edited in the course manager."
   438→   438→              : "Set up the basics. You'll be able to add topics, videos and assignments next."}
   439→   439→          </DialogDescription>
   440→   440→        </DialogHeader>
   441→   441→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1">
   442→   442→          <div className="space-y-1.5">
   443→   443→            <Label htmlFor="cf-title" className="text-xs font-medium text-muted-foreground">Title</Label>
   444→   444→            <Input id="cf-title" value={values.title} onChange={(e) => setValues({ ...values, title: e.target.value })} placeholder="e.g. Digital Marketing Fundamentals" className="h-9 rounded-lg" />
   445→   445→          </div>
   446→   446→          <div className="space-y-1.5">
   447→   447→            <Label htmlFor="cf-desc" className="text-xs font-medium text-muted-foreground">Description</Label>
   448→   448→            <Textarea id="cf-desc" value={values.description} onChange={(e) => setValues({ ...values, description: e.target.value })} placeholder="Short summary shown on course cards" rows={3} className="rounded-lg" />
   449→   449→          </div>
   450→   450→          <div className="grid grid-cols-2 gap-3">
   451→   451→            <div className="space-y-1.5">
   452→   452→              <Label htmlFor="cf-cat" className="text-xs font-medium text-muted-foreground">Category</Label>
   453→   453→              <Input id="cf-cat" value={values.category} onChange={(e) => setValues({ ...values, category: e.target.value })} placeholder="e.g. Marketing" className="h-9 rounded-lg" />
   454→   454→            </div>
   455→   455→            <div className="space-y-1.5">
   456→   456→              <Label className="text-xs font-medium text-muted-foreground">Access Type</Label>
   457→   457→              <Select value={values.accessType} onValueChange={(v) => setValues({ ...values, accessType: v as "FREE" | "PAID" })}>
   458→   458→                <SelectTrigger className="w-full h-9 rounded-lg">
   459→   459→                  <SelectValue />
   460→   460→                </SelectTrigger>
   461→   461→                <SelectContent>
   462→   462→                  <SelectItem value="FREE">Free — Open to all</SelectItem>
   463→   463→                  <SelectItem value="PAID">Paid — Plan-gated</SelectItem>
   464→   464→                </SelectContent>
   465→   465→              </Select>
   466→   466→            </div>
   467→   467→          </div>

   469→   469→          {values.accessType === "PAID" && (
