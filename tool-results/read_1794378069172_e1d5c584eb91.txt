   146→export function AdminStudents() {
   147→  const query = useAppStore((s) => s.query);
   148→  const [search, setSearch] = useState("");
   149→  const [statusFilter, setStatusFilter] = useState<"ALL" | "ACTIVE" | "EXPIRED">("ALL");
   150→  const [planFilter, setPlanFilter] = useState<string>("ALL");
   151→  const [page, setPage] = useState(1);
   152→  const [addOpen, setAddOpen] = useState(false);
   153→  const [editOpen, setEditOpen] = useState(false);
   154→  const [profileId, setProfileId] = useState<string | null>(null);
   155→  const [deleteTarget, setDeleteTarget] = useState<StudentRow | null>(null);
   156→  const [editTarget, setEditTarget] = useState<StudentRow | null>(null);
   157→
   158→  const { data: plansData } = useApiFetch<PlansResponse>("/api/public/plans");
   159→
   160→  // Build query string
   161→  const qs = useMemo(() => {
   162→    const p = new URLSearchParams();
   163→    if (search.trim()) p.set("q", search.trim());
   164→    if (statusFilter !== "ALL") p.set("status", statusFilter);
   165→    if (planFilter !== "ALL") p.set("planId", planFilter);
   166→    const s = p.toString();
   167→    return s ? `?${s}` : "";
   168→  }, [search, statusFilter, planFilter]);
   169→
   170→  const { data, loading, error, refetch } = useApiFetch<StudentsResponse>(
   171→    qs ? `/api/admin/students${qs}` : "/api/admin/students"
   172→  );
   173→
   174→  // Auto-open Add dialog when query.action=new — React 19 "store previous value" pattern
   175→  const [lastAction, setLastAction] = useState<string | undefined>(undefined);
   176→  if (query.action !== lastAction) {
   177→    setLastAction(query.action);
   178→    if (query.action === "new") setAddOpen(true);
   179→  }
   180→
   181→  const students = data?.students || [];
   182→  const totalPages = Math.max(1, Math.ceil(students.length / PAGE_SIZE));
   183→  const currentPage = Math.min(Math.max(1, page), totalPages);
   184→  const pageItems = students.slice(
   185→    (currentPage - 1) * PAGE_SIZE,
   186→    currentPage * PAGE_SIZE
   187→  );
   188→
   189→  return (
   190→    <div className="space-y-4">
   191→      {/* Toolbar */}
   192→      <Card className="border-border bg-card p-4 shadow-sm">
   193→        <div className="flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   194→          <div className="flex flex-1 flex-col gap-3 sm:flex-row sm:items-center">
   195→            <div className="relative flex-1">
   196→              <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   197→              <Input
   198→                value={search}
   199→                onChange={(e) => {
   200→                  setSearch(e.target.value);
   201→                  setPage(1);
   202→                }}
   203→                placeholder="Search by name, email or phone…"
   204→                className="h-9 rounded-lg pl-9 focus-visible:ring-2 focus-visible:ring-ring"
   205→                aria-label="Search students"
   206→              />
   207→            </div>
   208→            <Select
   209→              value={statusFilter}
   210→              onValueChange={(v) => {
   211→                setStatusFilter(v as "ALL" | "ACTIVE" | "EXPIRED");
   212→                setPage(1);
   213→              }}
   214→            >
   215→              <SelectTrigger className="h-9 w-full rounded-lg sm:w-[140px]" aria-label="Filter by status">
   216→                <SelectValue placeholder="All statuses" />
   217→              </SelectTrigger>
   218→              <SelectContent>
   219→                <SelectItem value="ALL">All statuses</SelectItem>
   220→                <SelectItem value="ACTIVE">Active</SelectItem>
   221→                <SelectItem value="EXPIRED">Expired</SelectItem>
   222→              </SelectContent>
   223→            </Select>
   224→            <Select
   225→              value={planFilter}
   226→              onValueChange={(v) => {
   227→                setPlanFilter(v);
   228→                setPage(1);
   229→              }}
   230→            >
   231→              <SelectTrigger className="h-9 w-full rounded-lg sm:w-[180px]" aria-label="Filter by plan">
   232→                <SelectValue placeholder="All plans" />
   233→              </SelectTrigger>
   234→              <SelectContent>
   235→                <SelectItem value="ALL">All plans</SelectItem>
   236→                {(plansData?.plans || []).map((p) => (
   237→                  <SelectItem key={p.id} value={p.id}>
   238→                    {p.name}
   239→                  </SelectItem>
   240→                ))}
   241→              </SelectContent>
   242→            </Select>
   243→          </div>
   244→          <Button className="h-9 gap-2" onClick={() => setAddOpen(true)}>
   245→            <Plus className="h-4 w-4" /> Add Student
   246→          </Button>
   247→        </div>
   248→      </Card>
   249→
   250→      {/* Content */}
   251→      <Card className="border-border bg-card p-0 shadow-sm overflow-hidden">
   252→        {loading ? (
   253→          <div className="space-y-2 p-4">
   254→            {Array.from({ length: 6 }).map((_, i) => (
   255→              <Skeleton key={i} className="h-12 w-full" />
   256→            ))}
   257→          </div>
   258→        ) : error ? (
   259→          <EmptyState
   260→            className="m-4"
   261→            icon={<Users className="h-5 w-5" />}
   262→            title="Couldn't load students"
   263→            description={error}
   264→          />
   265→        ) : students.length === 0 ? (
   266→          <EmptyState
   267→            className="m-4"
   268→            icon={<Users className="h-5 w-5" />}
   269→            title="No students yet"
   270→            description="When students register, they will appear here."
   271→            action={<Button onClick={() => setAddOpen(true)} className="gap-2"><Plus className="h-4 w-4" /> Add Student</Button>}
   272→          />
   273→        ) : (
   274→          <>
   275→            <Table>
   276→              <TableHeader>
   277→                <TableRow className="hover:bg-transparent">
   278→                  <TableHead className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Student</TableHead>
   279→                  <TableHead className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Plan</TableHead>
   280→                  <TableHead className="hidden text-xs font-semibold uppercase tracking-wider text-muted-foreground md:table-cell">Start</TableHead>
   281→                  <TableHead className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Expiry</TableHead>
   282→                  <TableHead className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Status</TableHead>
   283→                  <TableHead className="text-right text-xs font-semibold uppercase tracking-wider text-muted-foreground">Actions</TableHead>
   284→                </TableRow>
   285→              </TableHeader>
   286→              <TableBody>
   287→                {pageItems.map((s) => (
   288→                  <TableRow
   289→                    key={s.id}
   290→                    className="cursor-pointer hover:bg-muted/40"
   291→                    onClick={() => setProfileId(s.id)}
   292→                  >
   293→                    <TableCell>
   294→                      <div className="flex items-center gap-3">
   295→                        <div className="flex h-9 w-9 items-center justify-center rounded-full bg-primary/10 text-primary text-xs font-semibold ring-1 ring-border shrink-0">
   296→                          {s.name
   297→                            .split(" ")
   298→                            .map((w) => w[0])
   299→                            .slice(0, 2)
   300→                            .join("")
   301→                            .toUpperCase()}
   302→                        </div>
   303→                        <div className="min-w-0">
   304→                          <p className="font-medium text-foreground truncate">{s.name}</p>
   305→                          <p className="text-xs text-muted-foreground truncate">{s.email}</p>
   306→                        </div>
   307→                      </div>
   308→                    </TableCell>
   309→                    <TableCell>
   310→                      {s.subscription ? (
   311→                        <span className="text-sm font-medium text-foreground">{s.subscription.planName}</span>
   312→                      ) : (
   313→                        <Badge variant="outline" className="text-muted-foreground">No plan</Badge>
   314→                      )}
   315→                    </TableCell>
   316→                    <TableCell className="hidden text-sm text-muted-foreground md:table-cell">
   317→                      {s.subscription ? formatDate(s.subscription.startDate) : "—"}
   318→                    </TableCell>
   319→                    <TableCell className="text-sm text-muted-foreground">
   320→                      {s.subscription ? formatDate(s.subscription.expiryDate) : "—"}
   321→                    </TableCell>
   322→                    <TableCell>
   323→                      {!s.subscription ? (
   324→                        <Badge variant="outline" className="text-muted-foreground">No plan</Badge>
   325→                      ) : s.subscription.status === "ACTIVE" ? (
   326→                        <Badge className="bg-success/15 text-success hover:bg-success/20">Active</Badge>
   327→                      ) : (
   328→                        <Badge className="bg-destructive/10 text-destructive hover:bg-destructive/15">Expired</Badge>
   329→                      )}
   330→                    </TableCell>
   331→                    <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   332→                      <DropdownMenu>
   333→                        <DropdownMenuTrigger asChild>
   334→                          <Button variant="ghost" size="icon" aria-label={`Actions for ${s.name}`} className="text-muted-foreground hover:text-foreground hover:bg-muted">
   335→                            <MoreHorizontal className="h-4 w-4" />
   336→                          </Button>
   337→                        </DropdownMenuTrigger>
   338→                        <DropdownMenuContent align="end" className="w-48">
   339→                          <DropdownMenuLabel className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">Actions</DropdownMenuLabel>
   340→                          <DropdownMenuItem onClick={() => setProfileId(s.id)} className="gap-2">
   341→                            <Eye className="h-4 w-4 text-muted-foreground" /> View profile
   342→                          </DropdownMenuItem>
   343→                          <DropdownMenuItem
   344→                            onClick={() => {
   345→                              setEditTarget(s);
   346→                              setEditOpen(true);
   347→                            }}
   348→                            className="gap-2"
   349→                          >
   350→                            <Pencil className="h-4 w-4 text-muted-foreground" /> Edit
   351→                          </DropdownMenuItem>
   352→                          <DropdownMenuSeparator />
   353→                          <DropdownMenuItem
   354→                            variant="destructive"
   355→                            onClick={() => setDeleteTarget(s)}
   356→                            className="gap-2"
   357→                          >
   358→                            <Trash2 className="h-4 w-4" /> Delete
   359→                          </DropdownMenuItem>
   360→                        </DropdownMenuContent>
   361→                      </DropdownMenu>
   362→                    </TableCell>
   363→                  </TableRow>
   364→                ))}
   365→              </TableBody>
   366→            </Table>
   367→
   368→            {/* Pagination */}
   369→            <div className="flex items-center justify-between gap-2 border-t border-border bg-muted/20 px-4 py-3">
   370→              <p className="text-xs text-muted-foreground">
   371→                Showing <span className="font-medium text-foreground">{(currentPage - 1) * PAGE_SIZE + 1}</span>–
   372→                <span className="font-medium text-foreground">{Math.min(currentPage * PAGE_SIZE, students.length)}</span> of <span className="font-medium text-foreground">{students.length}</span>
   373→              </p>
   374→              <div className="flex gap-1.5">
   375→                <Button
   376→                  variant="outline"
   377→                  size="sm"
   378→                  className="h-8"
   379→                  disabled={currentPage === 1}
   380→                  onClick={() => setPage((p) => Math.max(1, p - 1))}
   381→                >
   382→                  Previous
   383→                </Button>
   384→                <span className="px-3 py-1 text-xs font-medium text-muted-foreground self-center">
   385→                  Page <span className="text-foreground">{currentPage}</span> / {totalPages}
   386→                </span>
   387→                <Button
   388→                  variant="outline"
   389→                  size="sm"
   390→                  className="h-8"
   391→                  disabled={currentPage >= totalPages}
   392→                  onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   393→                >
   394→                  Next
   395→                </Button>
   396→              </div>
   397→            </div>
   398→          </>
   399→        )}
   400→      </Card>
   401→
   402→      {/* Add Student Dialog */}
   403→      <AddStudentDialog
   404→        open={addOpen}
   405→        onOpenChange={setAddOpen}
   406→        plans={plansData?.plans || []}
   407→        onCreated={() => {
   408→          refetch();
   409→          toast({ title: "Student created", description: "The student account has been created." });
   410→        }}
   411→      />
   412→
   413→      {/* Edit Student Dialog */}
   414→      <EditStudentDialog
   415→        open={editOpen}
   416→        onOpenChange={setEditOpen}
   417→        target={editTarget}
   418→        plans={plansData?.plans || []}
   419→        onSaved={() => {
   420→          refetch();
   421→          toast({ title: "Student updated", description: "Changes have been saved." });
   422→        }}
   423→      />
   424→
   425→      {/* Student Profile Sheet */}
   426→      <StudentProfileSheet
   427→        studentId={profileId}
   428→        onOpenChange={(open) => {
   429→          if (!open) setProfileId(null);
   430→        }}
   431→        plans={plansData?.plans || []}
   432→        onAssigned={() => refetch()}
   433→      />
   434→
   435→      {/* Delete Confirmation */}
   436→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   437→        <AlertDialogContent>
   438→          <AlertDialogHeader>
   439→            <AlertDialogTitle>Delete student?</AlertDialogTitle>
   440→            <AlertDialogDescription>
   441→              This will permanently delete <strong>{deleteTarget?.name}</strong> and all related
   442→              subscriptions, submissions, certificates and progress records. This action cannot be undone.
   443→            </AlertDialogDescription>
   444→          </AlertDialogHeader>
   445→          <AlertDialogFooter>
   446→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   447→            <AlertDialogAction
   448→              className="bg-destructive text-white hover:bg-destructive/90"
   449→              onClick={async () => {
   450→                if (!deleteTarget) return;
   451→                try {
   452→                  await apiPost(`/api/admin/students/${deleteTarget.id}`, null, {
   453→                    method: "DELETE",
   454→                  });
   455→                  toast({ title: "Student deleted", description: `${deleteTarget.name}'s account has been removed.` });
   456→                  setDeleteTarget(null);
   457→                  refetch();
   458→                } catch {
   459→                  /* toast shown by apiPost */
   460→                }
   461→              }}
   462→            >
   463→              Delete
   464→            </AlertDialogAction>
   465→          </AlertDialogFooter>
   466→        </AlertDialogContent>
   467→      </AlertDialog>
   468→    </div>
   469→  );
   470→}

   472→// ---------- Add Student Dialog ----------
   473→function AddStudentDialog({
   474→  open,
   475→  onOpenChange,
   476→  plans,
   477→  onCreated,
   478→}: {
   479→  open: boolean;
   480→  onOpenChange: (v: boolean) => void;
   481→  plans: SubscriptionPlanPublic[];
   482→  onCreated: () => void;
   483→}) {
   484→  const [name, setName] = useState("");
   485→  const [email, setEmail] = useState("");
   486→  const [password, setPassword] = useState("");
   487→  const [planId, setPlanId] = useState<string>("NONE");
   488→  const [saving, setSaving] = useState(false);
   489→
   490→  useEffect(() => {
   491→    if (open) {
   492→      setName("");
   493→      setEmail("");
   494→      setPassword("");
   495→      setPlanId("NONE");
   496→    }
   497→  }, [open]);
   498→
   499→  async function submit() {
   500→    if (!name.trim() || !email.trim() || !password) {
   501→      toast({ title: "Missing fields", description: "Name, email and password are required.", variant: "destructive" });
   502→      return;
   503→    }
   504→    setSaving(true);
   505→    try {
   506→      await apiPost("/api/admin/students", {
   507→        name: name.trim(),
   508→        email: email.trim(),
   509→        password,
   510→        planId: planId === "NONE" ? undefined : planId,
   511→      });
   512→      onCreated();
   513→      onOpenChange(false);
   514→    } catch {
   515→      /* toast shown by apiPost */
   516→    } finally {
   517→      setSaving(false);
   518→    }
   519→  }
   520→
   521→  return (
   522→    <Dialog open={open} onOpenChange={onOpenChange}>
   523→      <DialogContent className="sm:max-w-md">
   524→        <DialogHeader>
   525→          <DialogTitle className="font-heading text-lg font-semibold">Add New Student</DialogTitle>
   526→          <DialogDescription>Create a student account. You can optionally assign a plan now.</DialogDescription>
   527→        </DialogHeader>
   528→        <div className="space-y-3">
   529→          <div className="space-y-1.5">
   530→            <Label htmlFor="as-name" className="text-xs font-medium text-muted-foreground">Full Name</Label>
   531→            <Input id="as-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Aisha Khalid" className="h-9 rounded-lg" />
   532→          </div>
   533→          <div className="space-y-1.5">
   534→            <Label htmlFor="as-email" className="text-xs font-medium text-muted-foreground">Email</Label>
   535→            <Input id="as-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="aisha@example.com" className="h-9 rounded-lg" />
   536→          </div>
   537→          <div className="space-y-1.5">
   538→            <Label htmlFor="as-pass" className="text-xs font-medium text-muted-foreground">Password</Label>
   539→            <Input id="as-pass" type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="Minimum 6 characters" className="h-9 rounded-lg" />
   540→          </div>
   541→          <div className="space-y-1.5">
   542→            <Label htmlFor="as-plan" className="text-xs font-medium text-muted-foreground">Subscription Plan (optional)</Label>
   543→            <Select value={planId} onValueChange={setPlanId}>
   544→              <SelectTrigger id="as-plan" className="w-full h-9 rounded-lg">
   545→                <SelectValue placeholder="No plan — student only" />
   546→              </SelectTrigger>
   547→              <SelectContent>
   548→                <SelectItem value="NONE">No plan — student only</SelectItem>
   549→                {plans.map((p) => (
   550→                  <SelectItem key={p.id} value={p.id}>
   551→                    {p.name}
   552→                  </SelectItem>
   553→                ))}
   554→              </SelectContent>
   555→            </Select>
   556→          </div>
   557→        </div>
   558→        <DialogFooter>
   559→          <Button variant="outline" className="h-9" onClick={() => onOpenChange(false)}>Cancel</Button>
   560→          <Button disabled={saving} className="h-9" onClick={submit}>
   561→            {saving ? "Creating…" : "Create Student"}
   562→          </Button>
   563→        </DialogFooter>
   564→      </DialogContent>
   565→    </Dialog>
   566→  );
   567→}

   569→// ---------- Edit Student Dialog ----------
   570→function EditStudentDialog({
   571→  open,
   572→  onOpenChange,
   573→  target,
   574→  onSaved,
   575→}: {
   576→  open: boolean;
   577→  onOpenChange: (v: boolean) => void;
   578→  target: StudentRow | null;
   579→  plans: SubscriptionPlanPublic[];
   580→  onSaved: () => void;
   581→}) {
   582→  const [name, setName] = useState("");
   583→  const [phone, setPhone] = useState("");
   584→  const [bio, setBio] = useState("");
   585→  const [saving, setSaving] = useState(false);
   586→
   587→  useEffect(() => {
   588→    if (target) {
   589→      setName(target.name);
   590→      setPhone(target.phone || "");
   591→      setBio("");
   592→    }
   593→  }, [target]);
   594→
   595→  async function submit() {
   596→    if (!target) return;
   597→    if (!name.trim()) {
   598→      toast({ title: "Name required", variant: "destructive" });
   599→      return;
   600→    }
   601→    setSaving(true);
   602→    try {
   603→      await apiPost(`/api/admin/students/${target.id}`, {
   604→        name: name.trim(),
   605→        phone: phone.trim() || null,
   606→        bio: bio.trim() || null,
   607→      }, { method: "PUT" });
   608→      onSaved();
   609→      onOpenChange(false);
   610→    } catch {
   611→      /* toast */
   612→    } finally {
   613→      setSaving(false);
   614→    }
   615→  }
   616→
   617→  return (
   618→    <Dialog open={open} onOpenChange={onOpenChange}>
   619→      <DialogContent className="sm:max-w-md">
   620→        <DialogHeader>
   621→          <DialogTitle className="font-heading text-lg font-semibold">Edit Student</DialogTitle>
   622→          <DialogDescription>Update profile details for {target?.email}</DialogDescription>
   623→        </DialogHeader>
   624→        <div className="space-y-3">
   625→          <div className="space-y-1.5">
   626→            <Label htmlFor="es-name" className="text-xs font-medium text-muted-foreground">Full Name</Label>
   627→            <Input id="es-name" value={name} onChange={(e) => setName(e.target.value)} className="h-9 rounded-lg" />
   628→          </div>
   629→          <div className="space-y-1.5">
   630→            <Label htmlFor="es-phone" className="text-xs font-medium text-muted-foreground">Phone</Label>
   631→            <Input id="es-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+968 9XXX XXXX" className="h-9 rounded-lg" />
   632→          </div>
   633→          <div className="space-y-1.5">
   634→            <Label htmlFor="es-bio" className="text-xs font-medium text-muted-foreground">Bio</Label>
   635→            <Textarea id="es-bio" value={bio} onChange={(e) => setBio(e.target.value)} placeholder="Short bio (optional)" rows={3} className="rounded-lg" />
   636→          </div>
   637→        </div>
   638→        <DialogFooter>
   639→          <Button variant="outline" className="h-9" onClick={() => onOpenChange(false)}>Cancel</Button>
   640→          <Button disabled={saving} className="h-9" onClick={submit}>
   641→            {saving ? "Saving…" : "Save Changes"}
   642→          </Button>
   643→        </DialogFooter>
   644→      </DialogContent>
   645→    </Dialog>
   646→  );
   647→}

   649→// ---------- Student Profile Sheet ----------

   651→  studentId,
   652→  onOpenChange,
   653→  plans,

   686→    } finally {
