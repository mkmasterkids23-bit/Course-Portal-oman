    79→export function UniversityCourses() {
    80→  const query = useAppStore((s) => s.query);
    81→  const navigate = useAppStore((s) => s.navigate);
    82→  const [search, setSearch] = useState("");
    83→  const [category, setCategory] = useState("");
    84→  const [published, setPublished] = useState<"ALL" | "true" | "false">("ALL");
    85→  const [addOpen, setAddOpen] = useState(false);
    86→  const [editTarget, setEditTarget] = useState<CourseRow | null>(null);
    87→  const [deleteTarget, setDeleteTarget] = useState<CourseRow | null>(null);
    88→
    89→  const { data: plansData } = useApiFetch<PlansResponse>("/api/public/plans");
    90→
    91→  const qs = useMemo(() => {
    92→    const p = new URLSearchParams();
    93→    if (search.trim()) p.set("q", search.trim());
    94→    if (category) p.set("category", category);
    95→    if (published !== "ALL") p.set("published", published);
    96→    const s = p.toString();
    97→    return s ? `?${s}` : "";
    98→  }, [search, category, published]);
    99→
   100→  const { data, loading, error, refetch } = useApiFetch<CoursesResponse>(
   101→    qs ? `/api/university/courses${qs}` : "/api/university/courses"
   102→  );
   103→
   104→  // Auto-open Add dialog when query.action=new — React 19 "store previous value" pattern
   105→  const [lastAction, setLastAction] = useState<string | undefined>(undefined);
   106→  if (query.action !== lastAction) {
   107→    setLastAction(query.action);
   108→    if (query.action === "new") setAddOpen(true);
   109→  }
   110→
   111→  const courses = data?.courses || [];
   112→
   113→  const categories = useMemo(() => {
   114→    const set = new Set<string>();
   115→    courses.forEach((c) => {
   116→      if (c.category) set.add(c.category);
   117→    });
   118→    return Array.from(set).sort();
   119→  }, [courses]);
   120→
   121→  async function togglePublish(course: CourseRow) {
   122→    try {
   123→      await apiPost(
   124→        `/api/university/courses/${course.id}/publish`,
   125→        { isPublished: !course.isPublished },
   126→        { method: "PATCH" }
   127→      );
   128→      toast({
   129→        title: course.isPublished ? "Course unpublished" : "Course published",
   130→        description: course.title,
   131→      });
   132→      refetch();
   133→    } catch {
   134→      /* toast */
   135→    }
   136→  }
   137→
   138→  return (
   139→    <div className="space-y-4">
   140→      <Card className="border-border bg-card p-4 shadow-sm">
   141→        <div className="flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   142→          <div className="flex flex-1 flex-col gap-3 sm:flex-row sm:items-center">
   143→            <div className="relative flex-1">
   144→              <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   145→              <Input
   146→                value={search}
   147→                onChange={(e) => setSearch(e.target.value)}
   148→                placeholder="Search your courses…"
   149→                className="h-9 rounded-lg pl-9 focus-visible:ring-2 focus-visible:ring-ring"
   150→                aria-label="Search courses"
   151→              />
   152→            </div>
   153→            <Select
   154→              value={category || "ALL"}
   155→              onValueChange={(v) => setCategory(v === "ALL" ? "" : v)}
   156→            >
   157→              <SelectTrigger
   158→                className="h-9 w-full rounded-lg sm:w-[160px]"
   159→                aria-label="Filter by category"
   160→              >
   161→                <SelectValue placeholder="All categories" />
   162→              </SelectTrigger>
   163→              <SelectContent>
   164→                <SelectItem value="ALL">All categories</SelectItem>
   165→                {categories.map((c) => (
   166→                  <SelectItem key={c} value={c}>
   167→                    {c}
   168→                  </SelectItem>
   169→                ))}
   170→              </SelectContent>
   171→            </Select>
   172→            <Select
   173→              value={published}
   174→              onValueChange={(v) => setPublished(v as "ALL" | "true" | "false")}
   175→            >
   176→              <SelectTrigger
   177→                className="h-9 w-full rounded-lg sm:w-[140px]"
   178→                aria-label="Filter by published state"
   179→              >
   180→                <SelectValue placeholder="All statuses" />
   181→              </SelectTrigger>
   182→              <SelectContent>
   183→                <SelectItem value="ALL">All statuses</SelectItem>
   184→                <SelectItem value="true">Published</SelectItem>
   185→                <SelectItem value="false">Drafts</SelectItem>
   186→              </SelectContent>
   187→            </Select>
   188→          </div>
   189→          <Button className="h-9 gap-2" onClick={() => setAddOpen(true)}>
   190→            <Plus className="h-4 w-4" /> Create Course
   191→          </Button>
   192→        </div>
   193→      </Card>
   194→
   195→      {/* Course count summary */}
   196→      {!loading && !error && courses.length > 0 && (
   197→        <div className="flex items-center justify-between">
   198→          <p className="text-sm text-muted-foreground">
   199→            Showing <span className="font-medium text-foreground">{courses.length}</span>{" "}
   200→            course{courses.length !== 1 ? "s" : ""}
   201→          </p>
   202→        </div>
   203→      )}
   204→
   205→      {loading ? (
   206→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   207→          {Array.from({ length: 6 }).map((_, i) => (
   208→            <Card key={i} className="overflow-hidden border-border p-0">
   209→              <Skeleton className="h-36 w-full rounded-none" />
   210→              <div className="p-5 space-y-3">
   211→                <Skeleton className="h-4 w-1/3" />
   212→                <Skeleton className="h-5 w-3/4" />
   213→                <Skeleton className="h-4 w-full" />
   214→                <div className="flex gap-2 pt-2">
   215→                  <Skeleton className="h-8 w-20" />
   216→                  <Skeleton className="h-8 w-20" />
   217→                </div>
   218→              </div>
   219→            </Card>
   220→          ))}
   221→        </div>
   222→      ) : error ? (
   223→        <EmptyState
   224→          className="m-4"
   225→          icon={<BookOpen className="h-5 w-5" />}
   226→          title="Couldn't load courses"
   227→          description={error}
   228→        />
   229→      ) : courses.length === 0 ? (
   230→        <EmptyState
   231→          className="m-4"
   232→          icon={<BookOpen className="h-5 w-5" />}
   233→          title="No courses yet"
   234→          description="Create your first course to start adding videos, materials and assignments for your university."
   235→          action={
   236→            <Button onClick={() => setAddOpen(true)} className="gap-2">
   237→              <Plus className="h-4 w-4" /> Create Course
   238→            </Button>
   239→          }
   240→        />
   241→      ) : (
   242→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   243→          {courses.map((c) => (
   244→            <CourseCard
   245→              key={c.id}
   246→              course={c}
   247→              onManage={() => navigate(["university", "courses", c.id])}
   248→              onEdit={() => setEditTarget(c)}
   249→              onTogglePublish={() => togglePublish(c)}
   250→              onDelete={() => setDeleteTarget(c)}
   251→            />
   252→          ))}
   253→        </div>
   254→      )}
   255→
   256→      <CourseFormDialog
   257→        open={addOpen}
   258→        onOpenChange={setAddOpen}
   259→        plans={plansData?.plans || []}
   260→        onSaved={(id) => {
   261→          if (id) {
   262→            toast({
   263→              title: "Course created",
   264→              description: "Taking you to the course editor.",
   265→            });
   266→            setAddOpen(false);
   267→            navigate(["university", "courses", id]);
   268→          } else {
   269→            refetch();
   270→          }
   271→        }}
   272→      />
   273→
   274→      <CourseFormDialog
   275→        open={!!editTarget}
   276→        onOpenChange={(v) => !v && setEditTarget(null)}
   277→        target={editTarget}
   278→        plans={plansData?.plans || []}
   279→        onSaved={() => {
   280→          refetch();
   281→          setEditTarget(null);
   282→          toast({ title: "Course updated" });
   283→        }}
   284→      />
   285→
   286→      <AlertDialog
   287→        open={!!deleteTarget}
   288→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   289→      >
   290→        <AlertDialogContent>
   291→          <AlertDialogHeader>
   292→            <AlertDialogTitle>Delete course?</AlertDialogTitle>
   293→            <AlertDialogDescription>
   294→              This will permanently delete <strong>{deleteTarget?.title}</strong> along with all its
   295→              topics, videos, materials, assignments and submissions. This cannot be undone.
   296→            </AlertDialogDescription>
   297→          </AlertDialogHeader>
   298→          <AlertDialogFooter>
   299→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   300→            <AlertDialogAction
   301→              className="bg-destructive text-white hover:bg-destructive/90"
   302→              onClick={async () => {
   303→                if (!deleteTarget) return;
   304→                try {
   305→                  await apiPost(
   306→                    `/api/university/courses/${deleteTarget.id}`,
   307→                    null,
   308→                    { method: "DELETE" }
   309→                  );
   310→                  toast({ title: "Course deleted", description: deleteTarget.title });
   311→                  setDeleteTarget(null);
   312→                  refetch();
   313→                } catch {
   314→                  /* toast */
   315→                }
   316→              }}
   317→            >
   318→              Delete
   319→            </AlertDialogAction>
   320→          </AlertDialogFooter>
   321→        </AlertDialogContent>
   322→      </AlertDialog>
   323→    </div>
   324→  );
   325→}

   327→interface CourseFormValues {
   328→  title: string;
   329→  description: string;
   330→  category: string;
   331→  accessType: "FREE" | "PAID";
   332→  isPublished: boolean;
   333→  thumbnail: string;
   334→  planIds: string[];
   335→}
   336→
   337→function CourseFormDialog({
   338→  open,
   339→  onOpenChange,
   340→  target,
   341→  plans,
   342→  onSaved,
   343→}: {
   344→  open: boolean;
   345→  onOpenChange: (v: boolean) => void;
   346→  target?: CourseRow | null;
   347→  plans: SubscriptionPlanPublic[];
   348→  onSaved: (createdId?: string) => void;
   349→}) {
   350→  const isEdit = !!target;
   351→  const [values, setValues] = useState<CourseFormValues>({
   352→    title: "",
   353→    description: "",
   354→    category: "",
   355→    accessType: "FREE",
   356→    isPublished: false,
   357→    thumbnail: "",
   358→    planIds: [],
   359→  });
   360→  const [saving, setSaving] = useState(false);
   361→  const [uploadingThumb, setUploadingThumb] = useState(false);
   362→
   363→  useEffect(() => {
   364→    if (target) {
   365→      setValues({
   366→        title: target.title,
   367→        description: target.description || "",
   368→        category: target.category || "",
   369→        accessType: target.accessType,
   370→        isPublished: target.isPublished,
   371→        thumbnail: target.thumbnail || "",
   372→        planIds: [],
   373→      });
   374→    } else if (open) {
   375→      setValues({
   376→        title: "",
   377→        description: "",
   378→        category: "",
   379→        accessType: "FREE",
   380→        isPublished: false,
   381→        thumbnail: "",
   382→        planIds: [],
   383→      });
   384→    }
   385→  }, [target, open]);
   386→
   387→  function togglePlan(id: string) {
   388→    setValues((v) => ({
   389→      ...v,
   390→      planIds: v.planIds.includes(id)
   391→        ? v.planIds.filter((p) => p !== id)
   392→        : [...v.planIds, id],
   393→    }));
   394→  }
   395→
   396→  async function uploadThumbnail(file: File) {
   397→    if (!target) {
   398→      toast({
   399→        title: "Save the course first",
   400→        description: "Thumbnail upload is available after the course is created.",
   401→        variant: "default",
   402→      });
   403→      return;
   404→    }
   405→    setUploadingThumb(true);
   406→    try {
   407→      const fd = new FormData();
   408→      fd.append("file", file);
   409→      const res = await apiUpload<{ thumbnail: string }>(
   410→        `/api/university/courses/${target.id}/thumbnail`,
   411→        fd
   412→      );
   413→      setValues((v) => ({ ...v, thumbnail: res.thumbnail }));
   414→      toast({ title: "Thumbnail uploaded" });
   415→    } catch {
   416→      /* toast */
   417→    } finally {
   418→      setUploadingThumb(false);
   419→    }
   420→  }
   421→
   422→  async function submit() {
   423→    if (!values.title.trim()) {
   424→      toast({ title: "Title required", variant: "destructive" });
   425→      return;
   426→    }
   427→    if (values.accessType === "PAID" && values.planIds.length === 0) {
   428→      toast({
   429→        title: "Select at least one plan",
   430→        description: "Paid courses must be assigned to one or more subscription plans.",
   431→        variant: "destructive",
   432→      });
   433→      return;
   434→    }
   435→
   436→    setSaving(true);
   437→    try {
   438→      if (isEdit && target) {
   439→        await apiPost(`/api/university/courses/${target.id}`, {
   440→          title: values.title.trim(),
   441→          description: values.description.trim() || null,
   442→          category: values.category.trim() || null,
   443→          accessType: values.accessType,
   444→          isPublished: values.isPublished,
   445→          thumbnail: values.thumbnail || null,
   446→          ...(values.accessType === "PAID" ? { planIds: values.planIds } : {}),
   447→        }, { method: "PUT" });
   448→        onSaved();
   449→      } else {
   450→        const res = await apiPost<{ course: { id: string } }>(
   451→          "/api/university/courses",
   452→          {
   453→            title: values.title.trim(),
   454→            description: values.description.trim() || null,
   455→            category: values.category.trim() || null,
   456→            accessType: values.accessType,
   457→            isPublished: values.isPublished,
   458→            thumbnail: values.thumbnail || null,
   459→            planIds: values.planIds,
   460→          }
   461→        );
   462→        onSaved(res.course.id);
   463→      }
   464→    } catch {
   465→      /* toast */
   466→    } finally {
   467→      setSaving(false);
   468→    }
   469→  }
   470→
   471→  return (
   472→    <Dialog open={open} onOpenChange={onOpenChange}>
   473→      <DialogContent className="sm:max-w-lg">
   474→        <DialogHeader>
   475→          <DialogTitle className="font-heading text-lg font-semibold">
   476→            {isEdit ? "Edit Course" : "Create New Course"}
   477→          </DialogTitle>
   478→          <DialogDescription>
   479→            {isEdit
   480→              ? "Update course details. Plan assignment and content can be edited in the course manager."
   481→              : "Set up the basics. The course will be automatically scoped to your university."}
   482→          </DialogDescription>
   483→        </DialogHeader>
   484→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1 scrollbar-thin">
   485→          <div className="space-y-1.5">
   486→            <Label htmlFor="cf-title" className="text-xs font-medium text-muted-foreground">
   487→              Title
   488→            </Label>
   489→            <Input
   490→              id="cf-title"
   491→              value={values.title}
   492→              onChange={(e) => setValues({ ...values, title: e.target.value })}
   493→              placeholder="e.g. Digital Marketing Fundamentals"
   494→              className="h-9 rounded-lg"
   495→            />
   496→          </div>
   497→          <div className="space-y-1.5">
   498→            <Label htmlFor="cf-desc" className="text-xs font-medium text-muted-foreground">
   499→              Description
   500→            </Label>
   501→            <Textarea
   502→              id="cf-desc"
   503→              value={values.description}
   504→              onChange={(e) => setValues({ ...values, description: e.target.value })}
   505→              placeholder="Short summary shown on course cards"
   506→              rows={3}
   507→              className="rounded-lg"
   508→            />
   509→          </div>
   510→          <div className="grid grid-cols-2 gap-3">
   511→            <div className="space-y-1.5">
   512→              <Label htmlFor="cf-cat" className="text-xs font-medium text-muted-foreground">
   513→                Category
   514→              </Label>
   515→              <Input
   516→                id="cf-cat"
   517→                value={values.category}
   518→                onChange={(e) => setValues({ ...values, category: e.target.value })}
   519→                placeholder="e.g. Marketing"
   520→                className="h-9 rounded-lg"
   521→              />
   522→            </div>
   523→            <div className="space-y-1.5">
   524→              <Label className="text-xs font-medium text-muted-foreground">Access Type</Label>
   525→              <Select
   526→                value={values.accessType}
   527→                onValueChange={(v) =>
   528→                  setValues({ ...values, accessType: v as "FREE" | "PAID" })
   529→                }
   530→              >
   531→                <SelectTrigger className="w-full h-9 rounded-lg">
   532→                  <SelectValue />
   533→                </SelectTrigger>
   534→                <SelectContent>
   535→                  <SelectItem value="FREE">Free — Open to all</SelectItem>
   536→                  <SelectItem value="PAID">Paid — Plan-gated</SelectItem>
   537→                </SelectContent>
   538→              </Select>
   539→            </div>
   540→          </div>
   541→
   542→          {values.accessType === "PAID" && (
   543→            <div className="space-y-2 rounded-lg border border-border p-3 bg-muted/30">
   544→              <Label className="text-[11px] uppercase tracking-wider text-muted-foreground">
   545→                Required Plans
   546→              </Label>
   547→              <p className="text-xs text-muted-foreground -mt-1">
   548→                Students with any of these plans can access this course.
   549→              </p>
   550→              {plans.length === 0 ? (
   551→                <p className="text-xs text-destructive">
   552→                  No subscription plans exist yet. Contact the platform admin.
   553→                </p>
   554→              ) : (
   555→                <div className="space-y-2 max-h-32 overflow-y-auto scrollbar-thin">
   556→                  {plans.map((p) => (
   557→                    <label
   558→                      key={p.id}
   559→                      className="flex items-center gap-2 cursor-pointer rounded-md p-1.5 transition-colors hover:bg-background"
   560→                    >
   561→                      <Checkbox
   562→                        checked={values.planIds.includes(p.id)}
   563→                        onCheckedChange={() => togglePlan(p.id)}
   564→                      />
   565→                      <span className="text-sm text-foreground">{p.name}</span>
   566→                      <Badge variant="outline" className="ml-auto text-xs">
   567→                        {p.price} {p.currency}
   568→                      </Badge>
   569→                    </label>
   570→                  ))}
   571→                </div>
   572→              )}
   573→            </div>
   574→          )}
   575→
   576→          {/* Thumbnail */}
   577→          <div className="space-y-1.5">
   578→            <Label className="text-xs font-medium text-muted-foreground">Thumbnail</Label>
   579→            <div className="flex items-center gap-3">
   580→              <div className="h-16 w-24 overflow-hidden rounded-lg bg-muted flex items-center justify-center shrink-0 ring-1 ring-border">
   581→                {values.thumbnail ? (
   582→                  <img src={values.thumbnail} alt="" className="h-full w-full object-cover" />
   583→                ) : (
   584→                  <PlayCircle className="h-5 w-5 text-muted-foreground" />
   585→                )}
   586→              </div>
   587→              <div className="flex-1 space-y-1.5">
   588→                <Input
   589→                  value={values.thumbnail}
   590→                  onChange={(e) => setValues({ ...values, thumbnail: e.target.value })}
   591→                  placeholder="Paste image URL or upload below"
   592→                  className="h-9 rounded-lg"
   593→                />
   594→                <label className="inline-flex items-center gap-1.5 cursor-pointer text-xs text-accent hover:underline">
   595→                  <Upload className="h-3.5 w-3.5" />
   596→                  {uploadingThumb ? "Uploading…" : "Upload image"}
   597→                  <input
   598→                    type="file"
   599→                    accept="image/*"
   600→                    className="hidden"
   601→                    disabled={!isEdit || uploadingThumb}
   602→                    onChange={(e) => {
   603→                      const f = e.target.files?.[0];
   604→                      if (f) uploadThumbnail(f);
   605→                    }}
   606→                  />
   607→                </label>
   608→                {!isEdit && (
   609→                  <p className="text-xs text-muted-foreground">
   610→                    Save course first, then upload thumbnail.
   611→                  </p>
   612→                )}
   613→              </div>
   614→            </div>
   615→          </div>
   616→
   617→          {/* Published switch */}
   618→          <div className="flex items-center justify-between rounded-lg border border-border p-3">
   619→            <div>
   620→              <Label htmlFor="cf-pub" className="text-sm font-medium">
   621→                Publish immediately
   622→              </Label>
   623→              <p className="text-xs text-muted-foreground mt-0.5">If off, course stays as a draft.</p>
   624→            </div>
   625→            <Switch
   626→              id="cf-pub"
   627→              checked={values.isPublished}
   628→              onCheckedChange={(v) => setValues({ ...values, isPublished: v })}
   629→            />
   630→          </div>
   631→        </div>
   632→        <DialogFooter>
   633→          <Button variant="outline" className="h-9" onClick={() => onOpenChange(false)}>
   634→            Cancel
   635→          </Button>
   636→          <Button disabled={saving} className="h-9" onClick={submit}>
   637→            {saving ? "Saving…" : isEdit ? "Save Changes" : "Create Course"}
   638→          </Button>
   639→        </DialogFooter>
   640→      </DialogContent>
   641→    </Dialog>
   642→  );
   643→}
