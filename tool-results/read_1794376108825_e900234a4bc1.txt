   112→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   113→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   114→    `/api/student/courses/${courseId}`
   115→  );
   116→
   117→  // Pick the initial video: last watched, or first available.
   118→  const initialVideoId = (() => {
   119→    if (!data?.course) return null;
   120→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   121→    //    most recently updated progress entry in the topics tree.
   122→    let candidate: VideoItem | null = null;
   123→    for (const t of data.course.topics) {
   124→      for (const v of t.videos) {
   125→        if (v.progress && !v.progress.isCompleted) {
   126→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   127→            candidate = v;
   128→          }
   129→        }
   130→      }
   131→    }
   132→    if (candidate) return candidate.id;
   133→    // 2) First video of the first topic
   134→    return data.course.topics[0]?.videos[0]?.id ?? null;
   135→  })();
   136→
   137→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   138→  // Reset user selection when courseId changes — "adjust state during render"
   139→  // pattern (avoids setState-in-effect warnings).
   140→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   141→  if (courseId !== prevCourseId) {
   142→    setPrevCourseId(courseId);
   143→    setUserSelectedVideoId(null);
   144→  }
   145→  // Effective active video: explicit user selection overrides the auto-pick
   146→  // (which is the last-watched or first video). When data loads and we have an
   147→  // initial pick, this falls back to it without needing an effect.
   148→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   149→
   150→  if (loading) {
   151→    return (
   152→      <div className="space-y-5">
   153→        <Skeleton className="h-10 w-32 rounded-lg" />
   154→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   155→          <div className="space-y-4">
   156→            <Skeleton className="h-[420px] rounded-xl" />
   157→            <Skeleton className="h-24 rounded-xl" />
   158→          </div>
   159→          <Skeleton className="h-[600px] rounded-xl" />
   160→        </div>
   161→      </div>
   162→    );
   163→  }
   164→
   165→  if (error) {
   166→    return (
   167→      <div className="space-y-4">
   168→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   169→          <ArrowLeft className="h-4 w-4" /> Back to courses
   170→        </Button>
   171→        <EmptyState
   172→          icon={<AlertTriangle className="h-5 w-5" />}
   173→          title="Couldn't open course"
   174→          description={error}
   175→          action={<Button onClick={onBack}>Back to courses</Button>}
   176→        />
   177→      </div>
   178→    );
   179→  }
   180→
   181→  if (!data?.course) {
   182→    return (
   183→      <EmptyState
   184→        icon={<PlayCircle className="h-5 w-5" />}
   185→        title="Course not found"
   186→        description="This course may have been removed or is not accessible."
   187→        action={<Button onClick={onBack}>Back to courses</Button>}
   188→      />
   189→    );
   190→  }
   191→
   192→  const course = data.course;
   193→  const allVideos = course.topics.flatMap((t) => t.videos);
   194→  const activeVideo =
   195→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   196→
   197→  // Find the topic containing the active video
   198→  const activeTopic = course.topics.find((t) =>
   199→    t.videos.some((v) => v.id === activeVideo?.id)
   200→  );
   201→
   202→  // Build a flat ordered list for prev/next navigation
   203→  const flatVideoIds = allVideos.map((v) => v.id);
   204→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   205→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   206→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   207→
   208→  function handleProgressUpdate(updated: {
   209→    videoId: string;
   210→    progress: VideoProgressInfo;
   211→    justCompleted: boolean;
   212→  }) {
   213→    if (!data) return;
   214→    const newCourse = {
   215→      ...data.course,
   216→      topics: data.course.topics.map((t) => ({
   217→        ...t,
   218→        videos: t.videos.map((v) =>
   219→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   220→        ),
   221→      })),
   222→      courseCompletion: (() => {
   223→        // Recompute local snapshot
   224→        const videos = data.course.topics.flatMap((t) => t.videos);
   225→        const completed = videos.filter(
   226→          (v) =>
   227→            v.id === updated.videoId
   228→              ? updated.progress.isCompleted
   229→              : v.progress?.isCompleted
   230→        ).length;
   231→        const total = videos.length;
   232→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   233→        return {
   234→          totalVideos: total,
   235→          completedVideos: completed,
   236→          percent,
   237→          allCompleted: total > 0 && completed === total,
   238→        };
   239→      })(),
   240→    };
   241→    setData({ course: newCourse });
   242→
   243→    if (updated.justCompleted) {
   244→      toast({
   245→        title: "Lesson completed",
   246→        description: "Great work! Your progress has been saved.",
   247→      });
   248→    }
   249→  }
   250→
   251→  async function handleSubmissionSuccess() {
   252→    await refetch();
   253→    toast({
   254→      title: "Assignment submitted",
   255→      description: "Your submission has been received.",
   256→    });
   257→  }
   258→
   259→  return (
   260→    <div className="space-y-5">
   261→      {/* ─── Premium header ─── */}

   263→        <div className="flex flex-wrap items-center justify-between gap-3">
   264→          <div className="flex items-center gap-3 min-w-0 flex-1">
   265→            <Button variant="outline" size="icon" onClick={onBack} className="shrink-0 h-9 w-9" aria-label="Back to courses">
   266→              <ArrowLeft className="h-4 w-4" />
   267→            </Button>
   268→            <div className="min-w-0 flex-1">
   269→              <div className="flex items-center gap-2 mb-0.5">
   270→                {course.category && (
   271→                  <Badge variant="outline" className="text-[11px] border-border text-muted-foreground">
   272→                    {course.category}
   273→                  </Badge>
   274→                )}
   275→                <Badge
   276→                  className={
   277→                    course.accessType === "FREE"
   278→                      ? "bg-success/15 text-success border-transparent"
   279→                      : "bg-primary text-primary-foreground border-transparent"
   280→                  }
   281→                >
   282→                  {course.accessType === "FREE" ? "Free" : "Paid"}
   283→                </Badge>
   284→              </div>
   285→              <h1 className="font-heading text-lg md:text-xl font-semibold text-foreground truncate">
   286→                {course.title}
   287→              </h1>
   288→            </div>
   289→          </div>
   290→          <div className="flex items-center gap-4 shrink-0">
   291→            <div className="hidden sm:flex flex-col items-end gap-0.5">
   292→              <span className="text-xs font-medium text-foreground">
   293→                {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} lessons
   294→              </span>
   295→              <span className="text-[11px] text-muted-foreground">completed</span>
   296→            </div>
   297→            <div className="flex items-center gap-2.5 min-w-[160px]">
   298→              <Progress
   299→                value={course.courseCompletion.percent}
   300→                className={`h-2 w-32 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   301→              />
   302→              <span className="text-sm font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   303→            </div>
   304→          </div>
   305→        </div>
   306→      </div>
   307→
   308→      {/* ─── Mobile: stacked. Desktop: split. ─── */}
   309→      <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   310→        {/* Left: player + materials + assignment */}
   311→        <div className="space-y-5 min-w-0">
   312→          {activeVideo ? (
   313→            <VideoPlayer
   314→              key={activeVideo.id}
   315→              video={activeVideo}
   316→              topicTitle={activeTopic?.title || null}
   317→              onProgress={handleProgressUpdate}
   318→              prevVideo={prevVideo}
   319→              nextVideo={nextVideo}
   320→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   321→            />
   322→          ) : (
   323→            <EmptyState
   324→              icon={<VideoIcon className="h-5 w-5" />}
   325→              title="No videos in this course"
   326→              description="This course doesn't have any published video lessons yet."
   327→            />
   328→          )}
   329→
   330→          {/* Topic materials */}
   331→          {activeTopic && activeTopic.materials.length > 0 && (
   332→            <Card className="shadow-sm">
   333→              <CardHeader>
   334→                <CardTitle className="text-base flex items-center gap-2">
   335→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   336→                    <FileText className="h-4 w-4" />
   337→                  </span>
   338→                  Materials · {activeTopic.title}
   339→                </CardTitle>
   340→                <CardDescription>Reference documents for this topic</CardDescription>
   341→              </CardHeader>
   342→              <CardContent>
   343→                <ul className="space-y-2">
   344→                  {activeTopic.materials.map((m) => (
   345→                    <MaterialRow key={m.id} material={m} />
   346→                  ))}
   347→                </ul>
   348→              </CardContent>
   349→            </Card>
   350→          )}
   351→
   352→          {/* Assignment panel — completely hidden until ALL videos are completed.
   353→              Before completion, show a locked placeholder that does NOT reveal
   354→              the assignment title, description, or instructions. */}
   355→          {course.assignment && course.courseCompletion.allCompleted ? (
   356→            <AssignmentPanel
   357→              course={course}
   358→              onSubmitted={handleSubmissionSuccess}
   359→            />
   360→          ) : course.assignment ? (
   361→            <LockedAssignmentCard
   362→              completedVideos={course.courseCompletion.completedVideos}
   363→              totalVideos={course.courseCompletion.totalVideos}
   364→              percent={course.courseCompletion.percent}
   365→            />
   366→          ) : null}
   367→
   368→          {/* All materials (course-level) */}
   369→          {course.materials.length > 0 && (
   370→            <Card className="shadow-sm">
   371→              <CardHeader>
   372→                <CardTitle className="text-base flex items-center gap-2">
   373→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   374→                    <FileText className="h-4 w-4" />
   375→                  </span>
   376→                  All Course Materials
   377→                </CardTitle>
   378→                <CardDescription>
   379→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   380→                </CardDescription>
   381→              </CardHeader>
   382→              <CardContent>
   383→                <ul className="space-y-2">
   384→                  {course.materials.map((m) => (
   385→                    <MaterialRow key={m.id} material={m} />
   386→                  ))}
   387→                </ul>
   388→              </CardContent>
   389→            </Card>
   390→          )}
   391→        </div>
   392→
   393→        {/* Right: course sidebar */}
   394→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col shadow-sm overflow-hidden">
   395→          <CardHeader className="border-b border-border bg-gradient-to-br from-light-teal/60 to-card">
   396→            <CardTitle className="text-base flex items-center gap-2">
   397→              <BookOpen className="h-4 w-4 text-accent" />
   398→              Course Lessons
   399→            </CardTitle>
   400→            <CardDescription className="mt-1">
   401→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   402→            </CardDescription>
   403→            <div className="mt-2">
   404→              <div className="flex justify-between text-xs mb-1">
   405→                <span className="text-muted-foreground">Progress</span>
   406→                <span className="font-semibold text-foreground">{course.courseCompletion.percent}%</span>
   407→              </div>
   408→              <Progress
   409→                value={course.courseCompletion.percent}
   410→                className={`h-1.5 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   411→              />
   412→            </div>
   413→          </CardHeader>
   414→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-15rem)] scrollbar-thin">
   415→            {course.topics.length === 0 ? (
   416→              <p className="text-sm text-muted-foreground p-6 text-center">
   417→                No topics published yet.
   418→              </p>
   419→            ) : (
   420→              <Accordion
   421→                type="multiple"
   422→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   423→                className="px-3 py-2"
   424→              >
   425→                {course.topics.map((topic, topicIdx) => {
   426→                  const completedInTopic = topic.videos.filter(
   427→                    (v) => v.progress?.isCompleted
   428→                  ).length;
   429→                  const isActiveTopic = topic.id === activeTopic?.id;
   430→                  const topicComplete = completedInTopic === topic.videos.length && topic.videos.length > 0;
   431→                  return (
   432→                    <AccordionItem key={topic.id} value={topic.id} className="border-b border-border/60 last:border-b-0">
   433→                      <AccordionTrigger className="hover:no-underline px-2 py-3 rounded-lg hover:bg-muted/40 transition-colors">
   434→                        <div className="flex items-center gap-2.5 flex-1 min-w-0 text-left">
   435→                          <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-[11px] font-semibold ${
   436→                            topicComplete
   437→                              ? "bg-success/15 text-success"
   438→                              : isActiveTopic
   439→                                ? "bg-accent/15 text-accent"
   440→                                : "bg-muted text-muted-foreground"
   441→                          }`}>
   442→                            {topicComplete ? <CheckCircle2 className="h-3.5 w-3.5" /> : topicIdx + 1}
   443→                          </span>
   444→                          <div className="flex-1 min-w-0">
   445→                            <p className={`text-sm font-semibold truncate ${isActiveTopic ? "text-accent" : "text-foreground"}`}>
   446→                              {topic.title}
   447→                            </p>
   448→                            <p className="text-[11px] text-muted-foreground mt-0.5">
   449→                              {completedInTopic}/{topic.videos.length} lessons
   450→                            </p>
   451→                          </div>
   452→                        </div>
   453→                      </AccordionTrigger>
   454→                      <AccordionContent className="pb-2 pt-1">
   455→                        <ul className="space-y-0.5 ml-9">
   456→                          {topic.videos.map((video, videoIdx) => {
   457→                            const isActive = video.id === activeVideo?.id;
   458→                            const completed = video.progress?.isCompleted;
   459→                            return (
   460→                              <li key={video.id} className="relative">
   461→                                {/* Vertical connecting line */}
   462→                                {videoIdx < topic.videos.length - 1 && (
   463→                                  <span className="absolute left-3 top-7 bottom-0 w-px bg-border" aria-hidden />
   464→                                )}
   465→                                <button
   466→                                  onClick={() => setUserSelectedVideoId(video.id)}
   467→                                  className={`group flex w-full items-start gap-2.5 rounded-lg px-2 py-2 text-left text-sm transition-all duration-200 ${
   468→                                    isActive
   469→                                      ? "bg-light-teal text-accent"
   470→                                      : "text-foreground hover:bg-muted/60"
   471→                                  }`}
   472→                                >
   473→                                  {/* Icon: status indicator */}
   474→                                  <span className="relative z-10 mt-0.5 shrink-0 flex h-5 w-5 items-center justify-center rounded-full bg-card ring-1 ring-border">
   475→                                    {completed ? (
   476→                                      <CheckCircle2 className="h-4 w-4 text-success" />
   477→                                    ) : isActive ? (
   478→                                      <PlayCircle className="h-4 w-4 text-accent" />
   479→                                    ) : (
   480→                                      <PlayCircle className="h-4 w-4 text-muted-foreground/70 group-hover:text-foreground" />
   481→                                    )}
   482→                                  </span>
   483→                                  <span className="min-w-0 flex-1">
   484→                                    <span className={`block truncate ${isActive ? "font-semibold text-accent" : "font-medium text-foreground group-hover:text-accent"}`}>
   485→                                      {video.title}
   486→                                    </span>
   487→                                    {video.duration > 0 && (
   488→                                      <span className="block text-[11px] text-muted-foreground mt-0.5">
   489→                                        {formatVideoDuration(video.duration)}
   490→                                      </span>
   491→                                    )}
   492→                                  </span>
   493→                                </button>
   494→                              </li>
   495→                            );
   496→                          })}
   497→                        </ul>
   498→                      </AccordionContent>
   499→                    </AccordionItem>
   500→                  );
   501→                })}
   502→              </Accordion>
   503→            )}
   504→          </CardContent>
   505→        </Card>
   506→      </div>
   507→    </div>
   508→  );
   509→}

   512→
   513→function VideoPlayer({
   514→  video,
   515→  topicTitle,
   516→  onProgress,
   517→  prevVideo,
   518→  nextVideo,
   519→  onNavigateLesson,
   520→}: {
   521→  video: VideoItem;
   522→  topicTitle: string | null;
   523→  onProgress: (update: {
   524→    videoId: string;
   525→    progress: VideoProgressInfo;
   526→    justCompleted: boolean;
   527→  }) => void;
   528→  prevVideo: VideoItem | null;
   529→  nextVideo: VideoItem | null;
   530→  onNavigateLesson: (id: string) => void;
   531→}) {
   532→  const videoRef = useRef<HTMLVideoElement>(null);
   533→  const lastTimeRef = useRef<number>(0);
   534→  const watchTimeRef = useRef<number>(0);
   535→  const durationRef = useRef<number>(0);
   536→  const lastPostRef = useRef<number>(0);
   537→  const resumeAppliedRef = useRef<boolean>(false);
   538→  const [expanded, setExpanded] = useState(false);
   539→
   540→  // YouTube completion timer
   541→  const ytStartRef = useRef<number | null>(null);
   542→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   543→  const ytWatchSecondsRef = useRef<number>(0);
   544→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   545→
   546→  // Local UI state
   547→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   548→    video.progress?.isCompleted || false
   549→  );
   550→  const [posting, setPosting] = useState(false);
   551→
   552→  // Reset on video change
   553→  useEffect(() => {
   554→    lastTimeRef.current = 0;
   555→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   556→    durationRef.current = video.duration || video.progress?.duration || 0;
   557→    lastPostRef.current = 0;
   558→    resumeAppliedRef.current = false;
   559→    setLocalCompleted(video.progress?.isCompleted || false);
   560→    setYtCanMarkComplete(false);
   561→    ytWatchSecondsRef.current = 0;
   562→    if (ytTimerRef.current) {
   563→      clearInterval(ytTimerRef.current);
   564→      ytTimerRef.current = null;
   565→    }
   566→  }, [video.id]);
   567→
   568→  // For HTML5 video: apply resume position once metadata is loaded.
   569→  const handleLoadedMetadata = () => {
   570→    const el = videoRef.current;
   571→    if (!el) return;
   572→    durationRef.current = el.duration || durationRef.current;
   573→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   574→      // resume within (duration - 2) so we don't immediately fire ended
   575→      const target = Math.min(
   576→        video.progress.lastPosition,
   577→        Math.max(0, (el.duration || 0) - 3)
   578→      );
   579→      try {
   580→        el.currentTime = target;
   581→      } catch {}
   582→      lastTimeRef.current = target;
   583→      resumeAppliedRef.current = true;
   584→    }
   585→  };
   586→
   587→  // Core: timeupdate handler — accumulate real watch time.
   588→  const handleTimeUpdate = () => {
   589→    const el = videoRef.current;
   590→    if (!el) return;
   591→    const cur = el.currentTime;
   592→    const last = lastTimeRef.current;
   593→    const delta = cur - last;
   594→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   595→    // (large positive deltas) and rewinds (negative deltas).
   596→    if (delta > 0 && delta <= 1.5) {
   597→      watchTimeRef.current += delta;
   598→    }
   599→    // Update lastTime to current — even on seek, so next delta is small.
   600→    lastTimeRef.current = cur;
   601→
   602→    // Throttled server POST every 5 seconds
   603→    const now = Date.now();
   604→    if (now - lastPostRef.current > 5000) {
   605→      lastPostRef.current = now;
   606→      void postProgress(false);
   607→    }
   608→  };
   609→
   610→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   611→  const handleSeeked = () => {
   612→    const el = videoRef.current;
   613→    if (!el) return;
   614→    lastTimeRef.current = el.currentTime;
   615→  };
   616→
   617→  // Ended event — submit final progress with completion hint.
   618→  const handleEnded = () => {
   619→    void postProgress(true);
   620→  };
   621→
   622→  const postProgress = useCallback(
   623→    async (isCompletionHint: boolean) => {
   624→      setPosting(true);
   625→      try {
   626→        const res = await apiPost<{
   627→          progress: VideoProgressInfo;
   628→          justCompleted: boolean;
   629→          rejectedCompletion: boolean;
   630→          certificateIssued: boolean;
   631→          certificateId: string | null;
   632→        }>("/api/student/progress", {
   633→          videoId: video.id,
   634→          lastPosition: videoRef.current?.currentTime || 0,
   635→          watchTimeSeconds: watchTimeRef.current,
   636→          duration: durationRef.current,
   637→          isCompleted: isCompletionHint,
   638→        });
   639→        setLocalCompleted(res.progress.isCompleted);
   640→        onProgress({
   641→          videoId: video.id,
   642→          progress: res.progress,
   643→          justCompleted: res.justCompleted,
   644→        });
   645→        if (isCompletionHint && res.rejectedCompletion) {
   646→          toast({
   647→            title: "Completion not recorded",
   648→            description:
   649→              "You need to watch at least 85% of the video before it's marked complete.",
   650→            variant: "destructive",
   651→          });
   652→        }
   653→        // ─── Certificate auto-issued upon course completion ───
   654→        if (res.certificateIssued && res.certificateId) {
   655→          toast({
   656→            title: "🎉 Certificate Issued!",
   657→            description:
   658→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   659→          });
   660→        }
   661→      } catch {
   662→        // apiPost already surfaces errors
   663→      } finally {
   664→        setPosting(false);
   665→      }
   666→    },
   667→    [video.id, onProgress]
   668→  );
   669→
   670→  // Manual "Mark complete" for YouTube videos
   671→  const markYtComplete = async () => {
   672→    setPosting(true);
   673→    try {
   674→      const res = await apiPost<{
   675→        progress: VideoProgressInfo;
   676→        justCompleted: boolean;
   677→        rejectedCompletion: boolean;
   678→        certificateIssued: boolean;
   679→        certificateId: string | null;
   680→      }>("/api/student/progress", {
   681→        videoId: video.id,
   682→        lastPosition: durationRef.current,
   683→        watchTimeSeconds: ytWatchSecondsRef.current,
   684→        duration: durationRef.current,
   685→        isCompleted: true,
   686→      });
   687→      setLocalCompleted(res.progress.isCompleted);
   688→      onProgress({
   689→        videoId: video.id,
   690→        progress: res.progress,
   691→        justCompleted: res.justCompleted,
   692→      });
   693→      if (res.justCompleted) {
   694→        toast({
   695→          title: "Lesson completed",
   696→          description: "Great work! Your progress has been saved.",
   697→        });
   698→      }
   699→      if (res.rejectedCompletion) {
   700→        toast({
   701→          title: "Completion not recorded",
   702→          description:
   703→            "You need to keep the video playing for at least 85% of its duration.",
   704→          variant: "destructive",
   705→        });
   706→      }
   707→      // Certificate auto-issued upon course completion
   708→      if (res.certificateIssued && res.certificateId) {
   709→        toast({
   710→          title: "🎉 Certificate Issued!",
   711→          description:
   712→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   713→        });
   714→      }
   715→    } catch {
   716→    } finally {
   717→      setPosting(false);
   718→    }
   719→  };
   720→
   721→  /* YouTube handling */
   722→  const ytEmbedUrl = useMemo(() => toYouTubeEmbed(video.videoUrl), [video.videoUrl]);
   723→  const isYouTube = video.source === "YOUTUBE" && ytEmbedUrl !== null;
   724→
   725→  // YouTube timer: accumulate watch time + enable mark-complete after 85%.
   726→  useEffect(() => {
   727→    if (!isYouTube) return;
   728→    if (ytTimerRef.current) clearInterval(ytTimerRef.current);
   729→    ytStartRef.current = Date.now();
   730→
   731→    ytTimerRef.current = setInterval(() => {
   732→      ytWatchSecondsRef.current = (Date.now() - (ytStartRef.current || Date.now())) / 1000;
   733→      const dur = durationRef.current;
   734→      if (dur > 0 && ytWatchSecondsRef.current >= dur * 0.85) {
   735→        setYtCanMarkComplete(true);
   736→      }
   737→    }, 1000);
   738→
   739→    return () => {
   740→      if (ytTimerRef.current) {
   741→        clearInterval(ytTimerRef.current);
   742→        ytTimerRef.current = null;
   743→      }
   744→    };
   745→  }, [isYouTube, video.id]);
   746→
   747→  // Render the right player
   748→  return (
   749→    <>
   750→      {/* Expanded fullscreen overlay — rendered as a portal-like fixed layer */}
   751→      {expanded && (

   754→          <div className="flex items-center justify-between px-4 py-3 bg-gradient-to-b from-black/80 to-transparent backdrop-blur">
   755→            <div className="min-w-0 flex-1">
   756→              {topicTitle && (
   757→                <p className="text-[11px] font-medium uppercase tracking-wider text-white/60 mb-0.5 truncate">
   758→                  {topicTitle}
   759→                </p>
   760→              )}
   761→              <p className="text-sm font-semibold text-white truncate">{video.title}</p>
   762→            </div>
   763→            <div className="flex items-center gap-2 shrink-0">
   764→              {localCompleted && (
   765→                <Badge className="bg-success/20 text-success border-transparent">
   766→                  <CheckCircle2 className="h-3 w-3" /> Completed
   767→                </Badge>
   768→              )}
   769→              <button
   770→                onClick={() => setExpanded(false)}
   771→                className="flex items-center gap-1.5 rounded-lg bg-white/10 px-3 py-1.5 text-sm font-medium text-white hover:bg-white/20 transition-colors"
   772→                aria-label="Exit fullscreen"
   773→              >
   774→                <Minimize2 className="h-4 w-4" />
   775→                <span className="hidden sm:inline">Exit fullscreen</span>
   776→              </button>
   777→            </div>
   778→          </div>
   779→          {/* Video fills the remaining space */}
   780→          <div className="flex-1 flex items-center justify-center p-2 sm:p-4">
   781→            <div className="relative w-full h-full flex items-center justify-center">
   782→              {video.source === "YOUTUBE" && ytEmbedUrl ? (
   783→                <iframe
   784→                  key={video.id}
   785→                  src={ytEmbedUrl}
   786→                  title={video.title}
   787→                  className="w-full h-full max-w-[1600px]"
   788→                  style={{ aspectRatio: "16 / 9" }}
   789→                  frameBorder={0}
   790→                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share"
   791→                  allowFullScreen
   792→                />
   793→              ) : (
   794→                <video
   795→                  key={video.id}
   796→                  ref={videoRef}
   797→                  src={video.videoUrl}
   798→                  poster={video.thumbnail || undefined}
   799→                  controls
   800→                  autoPlay
   801→                  className="w-full h-full max-h-full object-contain"
   802→                  onLoadedMetadata={handleLoadedMetadata}
   803→                  onTimeUpdate={handleTimeUpdate}
   804→                  onSeeked={handleSeeked}
   805→                  onEnded={handleEnded}
   806→                />
   807→              )}
   808→            </div>
   809→          </div>
   810→        </div>
   811→      )}
   812→
   813→      {/* ─── Default compact view ─── */}
   814→      <Card className="overflow-hidden p-0 shadow-sm">
   815→        {/* Compact video area — centered, max 640px, dark frame */}
   816→        <div className="bg-gradient-to-b from-foreground/[0.03] to-card p-4 md:p-6">
   817→          <div className="relative mx-auto overflow-hidden rounded-xl bg-black shadow-lg ring-1 ring-black/20" style={{ maxWidth: "640px" }}>
   818→            <div className="aspect-video w-full flex items-center justify-center relative">
   819→              {video.source === "YOUTUBE" && ytEmbedUrl ? (
   820→                <iframe
   821→                  key={video.id}
   822→                  src={ytEmbedUrl}
   823→                  title={video.title}
   824→                  className="w-full h-full"
   825→                  frameBorder={0}
   826→                  allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share"
   827→                  allowFullScreen
   828→                />
   829→              ) : (
   830→                <video
   831→                  key={video.id}
   832→                  ref={videoRef}
   833→                  src={video.videoUrl}
   834→                  poster={video.thumbnail || undefined}
   835→                  controls
   836→                  className="w-full h-full"
   837→                  onLoadedMetadata={handleLoadedMetadata}
   838→                  onTimeUpdate={handleTimeUpdate}
   839→                  onSeeked={handleSeeked}
   840→                  onEnded={handleEnded}
   841→                />

   843→              {/* Expand button overlay */}

   845→                onClick={() => setExpanded(true)}

   847→                aria-label="Expand video to fullscreen"

   849→                <Maximize2 className="h-3.5 w-3.5" />
   850→                <span className="hidden sm:inline">Expand</span>
   851→              </button>
   852→            </div>
   853→          </div>
   854→        </div>
   855→
   856→        <CardContent className="p-5 space-y-4">
   857→          {/* Video info */}
   858→          <div className="flex flex-wrap items-start justify-between gap-3">
   859→            <div className="min-w-0 flex-1">
   860→              {topicTitle && (
   861→                <p className="text-[11px] font-semibold uppercase tracking-[0.12em] text-accent mb-1.5">
   862→                  {topicTitle}
   863→                </p>
   864→              )}
   865→              <h2 className="font-heading text-lg font-semibold text-foreground">{video.title}</h2>
   866→              {video.description && (

   888→          {/* Prev / Next navigation */}

   930→              </div>

   932→                <Tooltip>

   947→                    {ytCanMarkComplete

   949→                      : "Available after watching at least 85% of the video duration."}
   950→                  </TooltipContent>
   951→                </Tooltip>
   952→              </div>
   953→            </div>
   954→          )}
   955→        </CardContent>
   956→      </Card>
   957→    </>
   958→  );
   959→}

  1216→function toYouTubeEmbed(url: string): string | null {
  1217→  try {
  1218→    const u = new URL(url);
  1219→    if (u.hostname === "youtu.be") {
  1220→      const id = u.pathname.slice(1);
  1221→      return `https://www.youtube.com/embed/${id}?rel=0`;
  1222→    }
  1223→    if (u.hostname.includes("youtube.com")) {
  1224→      const id = u.searchParams.get("v");
  1225→      if (id) return `https://www.youtube.com/embed/${id}?rel=0`;
  1226→      // /embed/ID or /shorts/ID
  1227→      const m = u.pathname.match(/\/(embed|shorts)\/([\w-]+)/);
  1228→      if (m) return `https://www.youtube.com/embed/${m[2]}?rel=0`;
  1229→    }
  1230→    return null;
  1231→  } catch {
  1232→    return null;
  1233→  }
  1234→}

  1236→function formatVideoDuration(seconds: number): string {
  1237→  if (seconds <= 0) return "";
  1238→  const m = Math.floor(seconds / 60);
  1239→  const s = Math.floor(seconds % 60);
  1240→  if (m === 0) return `${s}s`;
  1241→  return `${m}:${s.toString().padStart(2, "0")}`;
  1242→}
