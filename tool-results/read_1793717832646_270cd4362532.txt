   469→   469→function AddStudentDialog({
   470→   470→  open,
   471→   471→  onOpenChange,
   472→   472→  plans,
   473→   473→  onCreated,
   474→   474→}: {
   475→   475→  open: boolean;
   476→   476→  onOpenChange: (v: boolean) => void;
   477→   477→  plans: SubscriptionPlanPublic[];
   478→   478→  onCreated: () => void;
   479→   479→}) {
   480→   480→  const [name, setName] = useState("");
   481→   481→  const [email, setEmail] = useState("");
   482→   482→  const [password, setPassword] = useState("");
   483→   483→  const [planId, setPlanId] = useState<string>("NONE");
   484→   484→  const [saving, setSaving] = useState(false);
   485→   485→
   486→   486→  useEffect(() => {
   487→   487→    if (open) {
   488→   488→      setName("");
   489→   489→      setEmail("");
   490→   490→      setPassword("");
   491→   491→      setPlanId("NONE");
   492→   492→    }
   493→   493→  }, [open]);
   494→   494→
   495→   495→  async function submit() {
   496→   496→    if (!name.trim() || !email.trim() || !password) {
