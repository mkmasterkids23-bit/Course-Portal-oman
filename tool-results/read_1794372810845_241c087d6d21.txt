   224→function TopicsTab({ course, onRefresh }: { course: CourseDetail; onRefresh: () => void }) {
   225→  const [expandedId, setExpandedId] = useState<string | null>(course.topics[0]?.id || null);
   226→  const [addTopicOpen, setAddTopicOpen] = useState(false);
   227→  const [editTopic, setEditTopic] = useState<TopicRow | null>(null);
   228→  const [deleteTopic, setDeleteTopic] = useState<TopicRow | null>(null);
   229→  const [addVideoFor, setAddVideoFor] = useState<TopicRow | null>(null);
   230→  const [editVideo, setEditVideo] = useState<VideoRow | null>(null);
   231→  const [deleteVideo, setDeleteVideo] = useState<VideoRow | null>(null);
   232→
   233→  async function moveTopic(topic: TopicRow, dir: -1 | 1) {
   234→    const sorted = [...course.topics].sort((a, b) => a.displayOrder - b.displayOrder);
   235→    const idx = sorted.findIndex((t) => t.id === topic.id);
   236→    const target = idx + dir;
   237→    if (target < 0 || target >= sorted.length) return;
   238→    const reordered = [...sorted];
   239→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   240→    try {
   241→      await apiPost(`/api/admin/courses/${course.id}/topics/reorder`, {
   242→        orderedIds: reordered.map((t) => t.id),
   243→      });
   244→      onRefresh();
   245→    } catch { /* toast */ }
   246→  }
   247→
   248→  return (
   249→    <div className="space-y-4">
   250→      <div className="flex items-center justify-between">
   251→        <div>
   252→          <h3 className="font-heading text-base font-semibold text-foreground">Course Topics</h3>
   253→          <p className="text-xs text-muted-foreground mt-0.5">
   254→            {course.topics.length} topic{course.topics.length === 1 ? "" : "s"} · {course.topics.reduce((a, t) => a + t.videos.length, 0)} videos
   255→          </p>
   256→        </div>
   257→        <Button className="h-9 gap-2" onClick={() => setAddTopicOpen(true)}>
   258→          <Plus className="h-4 w-4" /> Add Topic
   259→        </Button>
   260→      </div>
   261→
   262→      {course.topics.length === 0 ? (
   263→        <EmptyState
   264→          icon={<VideoIcon className="h-5 w-5" />}
   265→          title="No topics yet"
   266→          description="Topics are containers for video lessons. Add your first topic to get started."
   267→          action={<Button onClick={() => setAddTopicOpen(true)} className="gap-2"><Plus className="h-4 w-4" /> Add Topic</Button>}
   268→        />
   269→      ) : (
   270→        <div className="space-y-2">
   271→          {course.topics.map((topic, idx) => (
   272→            <Card key={topic.id} className="overflow-hidden border-border bg-card p-0 shadow-sm transition-shadow hover:shadow-md">
   273→              <div className="flex items-center gap-2 p-3 bg-gradient-to-r from-muted/40 to-muted/10">
   274→                <div className="flex flex-col">
   275→                  <Button
   276→                    variant="ghost" size="icon" className="h-5 w-5"
   277→                    disabled={idx === 0}
   278→                    onClick={() => moveTopic(topic, -1)}
   279→                    aria-label="Move topic up"
   280→                  >
   281→                    <ChevronUp className="h-3 w-3" />
   282→                  </Button>
   283→                  <Button
   284→                    variant="ghost" size="icon" className="h-5 w-5"
   285→                    disabled={idx === course.topics.length - 1}
   286→                    onClick={() => moveTopic(topic, 1)}
   287→                    aria-label="Move topic down"
   288→                  >
   289→                    <ChevronDown className="h-3 w-3" />
   290→                  </Button>
   291→                </div>
   292→                <GripVertical className="h-4 w-4 text-muted-foreground/40 hidden sm:block" />
   293→                <button
   294→                  className="flex flex-1 items-center gap-2 text-left min-w-0"
   295→                  onClick={() => setExpandedId(expandedId === topic.id ? null : topic.id)}
   296→                >
   297→                  <span className="flex h-7 w-7 items-center justify-center rounded-full bg-primary/10 text-primary text-xs font-semibold shrink-0 ring-1 ring-border">
   298→                    {idx + 1}
   299→                  </span>
   300→                  <div className="min-w-0">
   301→                    <p className="font-medium text-foreground truncate">{topic.title}</p>
   302→                    <p className="text-xs text-muted-foreground truncate">
   303→                      {topic.videos.length} video{topic.videos.length === 1 ? "" : "s"}
   304→                      {topic.materials.length > 0 && ` · ${topic.materials.length} material${topic.materials.length === 1 ? "" : "s"}`}
   305→                    </p>
   306→                  </div>
   307→                </button>
   308→                <div className="flex items-center gap-1">
   309→                  <Button variant="ghost" size="sm" className="h-8 gap-1 text-muted-foreground hover:text-accent" onClick={() => setAddVideoFor(topic)}>
   310→                    <Plus className="h-3.5 w-3.5" /> Add Video
   311→                  </Button>
   312→                  <Button variant="ghost" size="icon" aria-label="Edit topic" className="h-8 w-8 text-muted-foreground hover:text-foreground" onClick={() => setEditTopic(topic)}>

   334→                    <ul className="space-y-1">
   335→                      {topic.videos.map((v, vi) => (

   338→                            {v.thumbnail ? (
   339→                              <img src={v.thumbnail} alt="" className="h-full w-full object-cover" />
   340→                            ) : (
   341→                              <VideoIcon className="h-3.5 w-3.5 text-muted-foreground" />
   342→                            )}
   343→                          </div>
   344→                          <div className="min-w-0 flex-1">
   345→                            <p className="text-sm font-medium text-foreground truncate">
   346→                              <span className="text-muted-foreground mr-1.5">{vi + 1}.</span>
   347→                              {v.title}
   348→                            </p>
   349→                            <div className="flex items-center gap-2 text-xs text-muted-foreground mt-0.5">
   350→                              <Badge variant="outline" className="text-[10px] py-0 h-4">{v.source}</Badge>
   351→                              {v.isPublished ? (
   352→                                <span className="text-success">Published</span>
   353→                              ) : (

   464→function TopicFormDialog({
   465→  open, onOpenChange, courseId, target, onSaved,
   466→}: {
   467→  open: boolean;
   468→  onOpenChange: (v: boolean) => void;
   469→  courseId: string;
   470→  target?: TopicRow | null;
   471→  onSaved: () => void;
   472→}) {
   473→  const [title, setTitle] = useState("");
   474→  const [description, setDescription] = useState("");
   475→  const [saving, setSaving] = useState(false);
   476→
   477→  useEffect(() => {
   478→    if (target) {
   479→      setTitle(target.title);
   480→      setDescription(target.description || "");
   481→    } else if (open) {
   482→      setTitle("");
   483→      setDescription("");
   484→    }
   485→  }, [target, open]);
   486→
   487→  async function submit() {
   488→    if (!title.trim()) {
   489→      toast({ title: "Title required", variant: "destructive" });
   490→      return;
   491→    }
   492→    setSaving(true);
   493→    try {
   494→      if (target) {
   495→        await apiPost(`/api/admin/topics/${target.id}`, {
   496→          title: title.trim(),
   497→          description: description.trim() || null,
   498→        }, { method: "PUT" });
   499→        toast({ title: "Topic updated" });
   500→      } else {

  1168→function AssignmentFormDialog({
  1169→  open, onOpenChange, courseId, target, onSaved,
  1170→}: {
  1171→  open: boolean;
  1172→  onOpenChange: (v: boolean) => void;
  1173→  courseId: string;
  1174→  target?: AssignmentRow | null;
  1175→  onSaved: () => void;
  1176→}) {
  1177→  const [title, setTitle] = useState("");
  1178→  const [description, setDescription] = useState("");
  1179→  const [instructions, setInstructions] = useState("");
  1180→  const [dueDate, setDueDate] = useState("");

  1182→  const [allowResubmission, setAllowResubmission] = useState(false);
  1183→  const [file, setFile] = useState<File | null>(null);
  1184→  const [clearFile, setClearFile] = useState(false);
  1185→  const [saving, setSaving] = useState(false);
  1186→
  1187→  useEffect(() => {
  1188→    if (target) {
  1189→      setTitle(target.title);
  1190→      setDescription(target.description || "");
  1191→      setInstructions(target.instructions || "");
  1192→      setDueDate(target.dueDate ? target.dueDate.slice(0, 10) : "");
  1193→      setIsPublished(target.isPublished);
  1194→      setAllowResubmission(target.allowResubmission);

  1196→      setTitle("");
  1197→      setDescription("");
  1198→      setInstructions("");
  1199→      setDueDate("");

  1201→      setAllowResubmission(false);
  1202→    }
  1203→    setFile(null);
  1204→    setClearFile(false);
  1205→  }, [target, open]);
  1206→
  1207→  async function submit() {
  1208→    if (!title.trim()) {
  1209→      toast({ title: "Title required", variant: "destructive" });
  1210→      return;

  1212→    const fd = new FormData();
  1213→    fd.append("title", title.trim());

  1224→      if (target) {
  1225→        const res = await fetch(`/api/admin/assignments/${target.id}`, {
  1226→          method: "PUT",
  1227→          body: fd,
  1228→          credentials: "same-origin",
  1229→        });

  1234→        await apiUpload(`/api/admin/courses/${courseId}/assignments`, fd);
  1235→        toast({ title: "Assignment added" });
  1236→      }
  1237→      onSaved();
  1238→    } catch (e) {
  1239→      if (e instanceof Error) {
  1240→        toast({ title: "Error", description: e.message, variant: "destructive" });
  1241→      }
  1242→    } finally {
  1243→      setSaving(false);
  1244→    }
  1245→  }
  1246→

  1248→    <Dialog open={open} onOpenChange={onOpenChange}>
  1249→      <DialogContent className="sm:max-w-lg">

  1522→function SettingsTab({
  1523→  course, plans, onRefresh, onDeleted,
  1524→}: {
  1525→  course: CourseDetail;
  1526→  plans: SubscriptionPlanPublic[];
  1527→  onRefresh: () => void;
  1528→  onDeleted: () => void;
  1529→}) {
  1530→  const navigate = useAppStore((s) => s.navigate);
  1531→  const [title, setTitle] = useState(course.title);
  1532→  const [description, setDescription] = useState(course.description || "");
  1533→  const [category, setCategory] = useState(course.category || "");
  1534→  const [accessType, setAccessType] = useState<"FREE" | "PAID">(course.accessType);
  1535→  const [isPublished, setIsPublished] = useState(course.isPublished);
  1536→  const [planIds, setPlanIds] = useState<string[]>(course.planAccess.map((p) => p.planId));
  1537→  const [saving, setSaving] = useState(false);
  1538→  const [deleteOpen, setDeleteOpen] = useState(false);
  1539→
  1540→  useEffect(() => {
  1541→    setTitle(course.title);
  1542→    setDescription(course.description || "");
  1543→    setCategory(course.category || "");
  1544→    setAccessType(course.accessType);
  1545→    setIsPublished(course.isPublished);
  1546→    setPlanIds(course.planAccess.map((p) => p.planId));
  1547→  }, [course]);
  1548→
  1549→  function togglePlan(id: string) {
  1550→    setPlanIds((cur) =>
  1551→      cur.includes(id) ? cur.filter((p) => p !== id) : [...cur, id]
  1552→    );
  1553→  }
  1554→
  1555→  async function save() {
  1556→    if (!title.trim()) {
  1557→      toast({ title: "Title required", variant: "destructive" });
  1558→      return;
  1559→    }
  1560→    if (accessType === "PAID" && planIds.length === 0) {
  1561→      toast({ title: "Select at least one plan", variant: "destructive" });
  1562→      return;
  1563→    }
  1564→    setSaving(true);
  1565→    try {
  1566→      await apiPost(`/api/admin/courses/${course.id}`, {
  1567→        title: title.trim(),
  1568→        description: description.trim() || null,
  1569→        category: category.trim() || null,
  1570→        accessType,
  1571→        isPublished,
  1572→        planIds,
  1573→      }, { method: "PUT" });
  1574→      toast({ title: "Course updated" });
  1575→      onRefresh();
  1576→    } catch { /* toast */ } finally {
  1577→      setSaving(false);
  1578→    }
  1579→  }
  1580→
  1581→  async function doDelete() {
  1582→    try {
  1583→      await apiPost(`/api/admin/courses/${course.id}`, null, { method: "DELETE" });
  1584→      toast({ title: "Course deleted", description: course.title });
  1585→      onDeleted();
  1586→    } catch { /* toast */ }
  1587→  }
  1588→
  1589→  return (
  1590→    <div className="space-y-4">
  1591→      <Card className="p-5 space-y-3 border-border bg-card shadow-sm">
  1592→        <CardHeader className="p-0">
  1593→          <CardTitle className="font-heading text-base font-semibold">Course Details</CardTitle>
  1594→          <CardDescription className="mt-1">Update the basic information about this course.</CardDescription>
  1595→        </CardHeader>
  1596→        <CardContent className="p-0 space-y-3">
  1597→          <div className="space-y-1.5">
  1598→            <Label htmlFor="s-title" className="text-xs font-medium text-muted-foreground">Title</Label>
  1599→            <Input id="s-title" value={title} onChange={(e) => setTitle(e.target.value)} className="h-9 rounded-lg" />
  1600→          </div>
  1601→          <div className="space-y-1.5">
  1602→            <Label htmlFor="s-desc" className="text-xs font-medium text-muted-foreground">Description</Label>
  1603→            <Textarea id="s-desc" value={description} onChange={(e) => setDescription(e.target.value)} rows={3} className="rounded-lg" />
  1604→          </div>
  1605→          <div className="grid grid-cols-2 gap-3">
  1606→            <div className="space-y-1.5">
  1607→              <Label htmlFor="s-cat" className="text-xs font-medium text-muted-foreground">Category</Label>
  1608→              <Input id="s-cat" value={category} onChange={(e) => setCategory(e.target.value)} className="h-9 rounded-lg" />
  1609→            </div>
  1610→            <div className="space-y-1.5">
  1611→              <Label className="text-xs font-medium text-muted-foreground">Access Type</Label>
  1612→              <Select value={accessType} onValueChange={(v) => setAccessType(v as "FREE" | "PAID")}>
  1613→                <SelectTrigger className="w-full h-9 rounded-lg">
  1614→                  <SelectValue />
  1615→                </SelectTrigger>
  1616→                <SelectContent>
  1617→                  <SelectItem value="FREE">Free</SelectItem>
  1618→                  <SelectItem value="PAID">Paid</SelectItem>
  1619→                </SelectContent>
  1620→              </Select>
  1621→            </div>
  1622→          </div>
  1623→
  1624→          {accessType === "PAID" && (
  1625→            <div className="space-y-2 rounded-lg border border-border p-3 bg-muted/30">
  1626→              <Label className="text-[11px] uppercase tracking-wider text-muted-foreground">Required Plans</Label>
  1627→              {plans.length === 0 ? (
  1628→                <p className="text-xs text-destructive">No subscription plans exist. Create one in the Plans section first.</p>
  1629→              ) : (
  1630→                <div className="space-y-2 max-h-32 overflow-y-auto scrollbar-thin">
  1631→                  {plans.map((p) => (
  1632→                    <label key={p.id} className="flex items-center gap-2 cursor-pointer rounded-md p-1.5 transition-colors hover:bg-background">
  1633→                      <Checkbox
  1634→                        checked={planIds.includes(p.id)}
  1635→                        onCheckedChange={() => togglePlan(p.id)}
  1636→                      />
  1637→                      <span className="text-sm text-foreground">{p.name}</span>

  1642→            </div>

  1644→

  1661→      {/* Danger zone */}
