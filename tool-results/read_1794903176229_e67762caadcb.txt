   419→// ─── Form Dialog (Add / Edit) ───────────────────────────────────────────
   420→interface FormValues {
   421→  name: string;
   422→  email: string;
   423→  password: string;
   424→  description: string;
   425→  logoFile: File | null;
   426→  logoPreview: string | null;
   427→}
   428→
   429→function UniversityFormDialog({
   430→  open,
   431→  onOpenChange,
   432→  target,
   433→  onSaved,
   434→}: {
   435→  open: boolean;
   436→  onOpenChange: (v: boolean) => void;
   437→  target?: UniversityRow | null;
   438→  onSaved: () => void;
   439→}) {
   440→  const { t } = useI18n();
   441→  const isEdit = !!target;
   442→  const [values, setValues] = useState<FormValues>({
   443→    name: "",
   444→    email: "",
   445→    password: "",
   446→    description: "",
   447→    logoFile: null,
   448→    logoPreview: null,
   449→  });
   450→  const [saving, setSaving] = useState(false);
   451→
   452→  useEffect(() => {
   453→    if (target) {
   454→      setValues({
   455→        name: target.name,
   456→        email: target.email,
   457→        password: "",
   458→        description: target.description || "",
   459→        logoFile: null,
   460→        logoPreview: target.logo,
   461→      });
   462→    } else if (open) {
   463→      setValues({
   464→        name: "",
   465→        email: "",
   466→        password: "",
   467→        description: "",
   468→        logoFile: null,
   469→        logoPreview: null,
   470→      });
   471→    }
   472→  }, [target, open]);
   473→
   474→  function onLogoSelected(file: File | null) {
   475→    if (!file) {
   476→      setValues((v) => ({ ...v, logoFile: null, logoPreview: target?.logo || null }));
   477→      return;
   478→    }
   479→    const previewUrl = URL.createObjectURL(file);
   480→    setValues((v) => ({ ...v, logoFile: file, logoPreview: previewUrl }));
   481→  }
   482→
   483→  async function submit() {
   484→    if (!values.name.trim()) {
   485→      toast({ title: t("common.name") + " " + t("common.required"), variant: "destructive" });
   486→      return;
   487→    }
   488→    if (!values.email.trim()) {
   489→      toast({ title: t("common.email") + " " + t("common.required"), variant: "destructive" });
   490→      return;
   491→    }
   492→    if (!isEdit && !values.password) {
   493→      toast({ title: t("common.password") + " " + t("common.required"), variant: "destructive" });
   494→      return;
   495→    }
   496→    if (values.password && values.password.length < 6) {
   497→      toast({
   498→        title: t("common.password"),
   499→        description: t("common.required"),
   500→        variant: "destructive",
   501→      });
   502→      return;
   503→    }
   504→
   505→    setSaving(true);
   506→    try {
   507→      const fd = new FormData();
   508→      fd.append("name", values.name.trim());
   509→      fd.append("email", values.email.trim());
   510→      if (values.password) fd.append("password", values.password);
   511→      if (values.description.trim()) fd.append("description", values.description.trim());
   512→      if (values.logoFile) fd.append("logo", values.logoFile);
   513→
   514→      const url = isEdit
   515→        ? `/api/admin/universities/${target!.id}`
   516→        : "/api/admin/universities";
   517→
   518→      // apiUpload uses POST; for PUT we fetch directly with FormData.
   519→      if (isEdit) {
   520→        const res = await fetch(url, {
   521→          method: "PUT",
   522→          body: fd,
   523→          credentials: "include",
   524→        });
   525→        const json = await res.json();
   526→        if (!res.ok) {
   527→          const msg = json.error || `Request failed (${res.status})`;
   528→          toast({ title: t("common.error"), description: msg, variant: "destructive" });
   529→          throw new Error(msg);
   530→        }
   531→      } else {
   532→        await apiUpload(url, fd);
   533→      }
   534→      onSaved();
   535→    } catch {
   536→      /* toast handled in helpers */
   537→    } finally {
   538→      setSaving(false);
   539→    }
   540→  }
   541→
   542→  const previewInitials = values.name
   543→    .split(/\s+/)
   544→    .map((w) => w[0])
   545→    .filter(Boolean)
   546→    .slice(0, 2)
   547→    .join("")
   548→    .toUpperCase();
   549→
   550→  return (
   551→    <Dialog open={open} onOpenChange={onOpenChange}>
   552→      <DialogContent className="animate-scale-in w-[95vw] max-w-lg max-h-[90vh] overflow-y-auto">
   553→        <DialogHeader>
   554→          <DialogTitle className="font-heading text-lg font-semibold">
   555→            {isEdit ? t("courses.editCourse") : t("admin.addUniversity")}
   556→          </DialogTitle>
   557→          <DialogDescription>
   558→            {isEdit
   559→              ? t("admin.universityDesc")
   560→              : t("admin.universityDesc")}
   561→          </DialogDescription>
   562→        </DialogHeader>
   563→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1">
   564→          {/* Logo preview + upload */}
   565→          <div className="space-y-1.5">
   566→            <Label className="text-xs font-medium text-muted-foreground">{t("admin.universityLogo")}</Label>
   567→            <div className="flex items-center gap-3">
   568→              <div className="h-16 w-16 overflow-hidden rounded-lg bg-muted flex items-center justify-center shrink-0 ring-1 ring-border">
   569→                {values.logoPreview ? (
   570→                  <img
   571→                    src={values.logoPreview}
   572→                    alt=""
   573→                    className="h-full w-full object-contain p-1"
   574→                  />
   575→                ) : previewInitials ? (
   576→                  <span className="font-heading text-lg font-bold text-primary">
   577→                    {previewInitials}
   578→                  </span>
   579→                ) : (
   580→                  <Building2 className="h-5 w-5 text-muted-foreground" />
   581→                )}
   582→              </div>
   583→              <div className="flex-1 space-y-1.5">
   584→                <label className="inline-flex items-center gap-1.5 cursor-pointer text-xs text-accent hover:underline">
   585→                  <Upload className="h-3.5 w-3.5" />
   586→                  {values.logoFile ? t("common.edit") : t("common.upload")}
   587→                  <input
   588→                    type="file"
   589→                    accept="image/*"
   590→                    className="hidden"
   591→                    onChange={(e) => {

   673→function UniversityDetailDrawer({
   674→  target,
   675→  onOpenChange,
   676→}: {
   677→  target: UniversityRow | null;
   678→  onOpenChange: (v: boolean) => void;
   679→}) {
   680→  const { t } = useI18n();
   681→  const open = !!target;
   682→  const { data, loading, error } = useApiFetch<UniversityDetailResponse>(
   683→    target ? `/api/admin/universities/${target.id}` : null,
   684→  );
   685→
   686→  // Reset fetch when drawer closes — pass null url above handles that
   687→  const university = data?.university || target;
   688→  const initials = useMemo(() => {
   689→    if (!university) return "";
   690→    return university.name
   691→      .split(/\s+/)
   692→      .map((w) => w[0])
   693→      .filter(Boolean)
   694→      .slice(0, 2)
   695→      .join("")
   696→      .toUpperCase();
   697→  }, [university]);
   698→
   699→  return (
   700→    <Drawer open={open} onOpenChange={onOpenChange}>
   701→      <DrawerContent className="max-h-[88vh]">
   702→        <div className="mx-auto w-full max-w-3xl">

   706→            </DrawerTitle>
   707→            <DrawerDescription>
   708→              {university?.description || t("admin.universityDesc")}
   709→            </DrawerDescription>
   710→          </DrawerHeader>
   711→
   712→          <div className="space-y-4 p-4 md:p-6 overflow-y-auto max-h-[calc(88vh-6rem)] scrollbar-thin">
   713→            {loading ? (
   714→              <div className="space-y-3">
   715→                <Skeleton className="h-24 w-full rounded-lg" />
   716→                <Skeleton className="h-6 w-40" />
   717→                <Skeleton className="h-32 w-full rounded-lg" />
   718→              </div>
   719→            ) : error ? (
   720→              <EmptyState
   721→                icon={<Building2 className="h-5 w-5" />}
   722→                title={t("common.error")}
   723→                description={error}
   724→              />
   725→            ) : university ? (
   726→              <>
   727→                {/* Header card */}

   729→                  <div className="relative h-24 bg-gradient-to-br from-primary via-primary to-[#2A3F35]">
   730→                    <div
   731→                      className="absolute inset-0 opacity-30"
   732→                      style={{
   733→                        backgroundImage:
   734→                          "radial-gradient(circle at 20% 50%, rgba(120,151,134,0.55), transparent 55%), radial-gradient(circle at 80% 20%, rgba(252,252,250,0.12), transparent 45%)",
   735→                      }}

   739→                    <div className="flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
   740→                      <div className="flex items-end gap-3">

   742→                          <img
   743→                            src={university.logo}
   744→                            alt={`${university.name} logo`}
   745→                            className="h-20 w-20 rounded-xl bg-background object-contain p-1.5 ring-2 ring-background shadow-sm"
   746→                          />
   747→                        ) : (
   748→                          <div className="flex h-20 w-20 items-center justify-center rounded-xl bg-background text-primary font-heading text-2xl font-bold ring-2 ring-background shadow-sm">

   751→                        )}

   777→                    label={t("common.date")}
   778→                    value={formatDate(university.createdAt)}
   779→                    isText
   780→                  />
   781→                  <DetailStat
   782→                    icon={<GraduationCap className="h-4 w-4" />}
   783→                    label={t("common.status")}
   784→                    value={t("common.active")}
   785→                    isText
   786→                  />

   835→function DetailStat({
   836→  icon,
   837→  label,
   838→  value,
   839→  isText,
   840→}: {
   841→  icon: React.ReactNode;
   842→  label: string;
   843→  value: number | string;
   844→  isText?: boolean;
   845→}) {
   846→  return (
   847→    <Card className="border-border p-3">
   848→      <div className="flex items-center gap-2 text-accent">
   849→        {icon}
   850→        <span className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground">
   851→          {label}
   852→        </span>
   853→      </div>
   854→      <p
   855→        className={`mt-1.5 font-heading font-semibold text-foreground ${
   856→          isText ? "text-sm truncate" : "text-xl"
   857→        }`}
   858→        title={isText ? String(value) : undefined}
   859→      >
   860→        {isText && typeof value === "string"
   861→          ? value.length > 22
   862→            ? value.slice(0, 20) + "…"
   863→            : value
   864→          : typeof value === "number"
   865→            ? value.toLocaleString()
   866→            : value}
   867→      </p>
   868→    </Card>
   869→  );
