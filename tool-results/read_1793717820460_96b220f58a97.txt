   112→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   113→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   114→    `/api/student/courses/${courseId}`
   115→  );
   116→
   117→  // Pick the initial video: last watched, or first available.
   118→  const initialVideoId = (() => {
   119→    if (!data?.course) return null;
   120→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   121→    //    most recently updated progress entry in the topics tree.
   122→    let candidate: VideoItem | null = null;
   123→    for (const t of data.course.topics) {
   124→      for (const v of t.videos) {
   125→        if (v.progress && !v.progress.isCompleted) {
   126→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   127→            candidate = v;
   128→          }
   129→        }
   130→      }
   131→    }
   132→    if (candidate) return candidate.id;
   133→    // 2) First video of the first topic
   134→    return data.course.topics[0]?.videos[0]?.id ?? null;
   135→  })();
   136→
   137→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   138→  // Reset user selection when courseId changes — "adjust state during render"
   139→  // pattern (avoids setState-in-effect warnings).
   140→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   141→  if (courseId !== prevCourseId) {
   142→    setPrevCourseId(courseId);
   143→    setUserSelectedVideoId(null);
   144→  }
   145→  // Effective active video: explicit user selection overrides the auto-pick
   146→  // (which is the last-watched or first video). When data loads and we have an
   147→  // initial pick, this falls back to it without needing an effect.
   148→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   149→
   150→  if (loading) {
   151→    return (
   152→      <div className="space-y-4">
   153→        <Skeleton className="h-12 w-32 rounded-md" />
   154→        <div className="grid gap-4 lg:grid-cols-[1fr_360px]">
   155→          <Skeleton className="h-[480px] rounded-xl" />
   156→          <Skeleton className="h-[480px] rounded-xl" />
   157→        </div>
   158→      </div>
   159→    );
   160→  }
   161→
   162→  if (error) {
   163→    return (
   164→      <div className="space-y-4">
   165→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   166→          <ArrowLeft className="h-4 w-4" /> Back to courses
   167→        </Button>
   168→        <EmptyState
   169→          icon={<AlertTriangle className="h-5 w-5" />}
   170→          title="Couldn't open course"
   171→          description={error}
   172→          action={<Button onClick={onBack}>Back to courses</Button>}
   173→        />
   174→      </div>
   175→    );
   176→  }
   177→
   178→  if (!data?.course) {
   179→    return (
   180→      <EmptyState
   181→        icon={<PlayCircle className="h-5 w-5" />}
   182→        title="Course not found"
   183→        description="This course may have been removed or is not accessible."
   184→        action={<Button onClick={onBack}>Back to courses</Button>}
   185→      />
   186→    );
   187→  }
   188→
   189→  const course = data.course;
   190→  const allVideos = course.topics.flatMap((t) => t.videos);
   191→  const activeVideo =
   192→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   193→
   194→  // Find the topic containing the active video
   195→  const activeTopic = course.topics.find((t) =>
   196→    t.videos.some((v) => v.id === activeVideo?.id)
   197→  );
   198→
   199→  function handleProgressUpdate(updated: {
   200→    videoId: string;
   201→    progress: VideoProgressInfo;
   202→    justCompleted: boolean;
   203→  }) {
   204→    if (!data) return;
   205→    const newCourse = {
   206→      ...data.course,
   207→      topics: data.course.topics.map((t) => ({
   208→        ...t,
   209→        videos: t.videos.map((v) =>
   210→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   211→        ),
   212→      })),
   213→      courseCompletion: (() => {
   214→        // Recompute local snapshot
   215→        const videos = data.course.topics.flatMap((t) => t.videos);
   216→        const completed = videos.filter(
   217→          (v) =>
   218→            v.id === updated.videoId
   219→              ? updated.progress.isCompleted
   220→              : v.progress?.isCompleted
   221→        ).length;
   222→        const total = videos.length;
   223→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   224→        return {
   225→          totalVideos: total,
   226→          completedVideos: completed,
   227→          percent,
   228→          allCompleted: total > 0 && completed === total,
   229→        };
   230→      })(),
   231→    };
   232→    setData({ course: newCourse });
   233→
   234→    if (updated.justCompleted) {
   235→      toast({
   236→        title: "Lesson completed",
   237→        description: "Great work! Your progress has been saved.",
   238→      });
   239→    }
   240→  }
   241→
   242→  async function handleSubmissionSuccess() {
   243→    await refetch();
   244→    toast({
   245→      title: "Assignment submitted",
   246→      description: "Your submission has been received.",
   247→    });
   248→  }
   249→
   250→  return (
   251→    <div className="space-y-4">
   252→      {/* Top bar */}
   253→      <div className="flex flex-wrap items-center justify-between gap-3">
   254→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   255→          <ArrowLeft className="h-4 w-4" /> Back to courses
   256→        </Button>
   257→        <div className="flex items-center gap-3 min-w-0 flex-1">
   258→          <h1 className="font-heading text-lg md:text-xl font-semibold text-foreground truncate">
   259→            {course.title}
   260→          </h1>
   261→        </div>
   262→        <div className="flex items-center gap-3 shrink-0">
   263→          <div className="flex items-center gap-2 text-sm text-muted-foreground">
   264→            <span>{course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} lessons</span>
   265→          </div>
   266→          <div className="flex items-center gap-2 min-w-[120px]">
   267→            <Progress value={course.courseCompletion.percent} className="h-2 w-24" />
   268→            <span className="text-sm font-medium text-foreground">{course.courseCompletion.percent}%</span>
   269→          </div>
   270→        </div>
   271→      </div>
   272→
   273→      {/* Mobile: stacked. Desktop: split. */}
   274→      <div className="grid gap-4 lg:grid-cols-[1fr_360px]">
   275→        {/* Left: player + materials + assignment */}
   276→        <div className="space-y-4 min-w-0">
   277→          {activeVideo ? (
   278→            <VideoPlayer
   279→              key={activeVideo.id}
   280→              video={activeVideo}
   281→              onProgress={handleProgressUpdate}
   282→            />
   283→          ) : (
   284→            <EmptyState
   285→              icon={<VideoIcon className="h-5 w-5" />}
   286→              title="No videos in this course"
   287→              description="This course doesn't have any published video lessons yet."
   288→            />
   289→          )}
   290→
   291→          {/* Materials */}
   292→          {activeTopic && activeTopic.materials.length > 0 && (
   293→            <Card>
   294→              <CardHeader>
   295→                <CardTitle className="text-base flex items-center gap-2">
   296→                  <FileText className="h-4 w-4 text-accent" />
   297→                  Materials for {activeTopic.title}
   298→                </CardTitle>
   299→                <CardDescription>Reference documents for this topic</CardDescription>
   300→              </CardHeader>
   301→              <CardContent>
   302→                <ul className="space-y-2">
   303→                  {activeTopic.materials.map((m) => (
   304→                    <MaterialRow key={m.id} material={m} />
   305→                  ))}
   306→                </ul>
   307→              </CardContent>
   308→            </Card>
   309→          )}
   310→
   311→          {/* Assignment panel — completely hidden until ALL videos are completed.
   312→              Before completion, show a locked placeholder that does NOT reveal
   313→              the assignment title, description, or instructions. */}
   314→          {course.assignment && course.courseCompletion.allCompleted ? (
   315→            <AssignmentPanel
   316→              course={course}
   317→              onSubmitted={handleSubmissionSuccess}
   318→            />
   319→          ) : course.assignment ? (
   320→            <LockedAssignmentCard
   321→              completedVideos={course.courseCompletion.completedVideos}
   322→              totalVideos={course.courseCompletion.totalVideos}
   323→              percent={course.courseCompletion.percent}
   324→            />
   325→          ) : null}
   326→
   327→          {/* All materials (course-level) */}
   328→          {course.materials.length > 0 && (
   329→            <Card>
   330→              <CardHeader>
   331→                <CardTitle className="text-base flex items-center gap-2">
   332→                  <FileText className="h-4 w-4 text-accent" />
   333→                  All Course Materials
   334→                </CardTitle>
   335→                <CardDescription>
   336→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   337→                </CardDescription>
   338→              </CardHeader>
   339→              <CardContent>
   340→                <ul className="space-y-2">
   341→                  {course.materials.map((m) => (
   342→                    <MaterialRow key={m.id} material={m} />
   343→                  ))}
   344→                </ul>
   345→              </CardContent>
   346→            </Card>
   347→          )}
   348→        </div>
   349→
   350→        {/* Right: topic sidebar */}
   351→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col">
   352→          <CardHeader className="border-b border-border pb-3">
   353→            <CardTitle className="text-base">Course Lessons</CardTitle>
   354→            <CardDescription>
   355→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   356→            </CardDescription>
   357→          </CardHeader>
   358→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-12rem)] scrollbar-thin">
   359→            {course.topics.length === 0 ? (
   360→              <p className="text-sm text-muted-foreground p-6 text-center">
   361→                No topics published yet.
   362→              </p>
   363→            ) : (
   364→              <Accordion
   365→                type="multiple"
   366→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   367→                className="px-3"
   368→              >
   369→                {course.topics.map((topic) => {
   370→                  const completedInTopic = topic.videos.filter(
   371→                    (v) => v.progress?.isCompleted
   372→                  ).length;
   373→                  const isActiveTopic = topic.id === activeTopic?.id;
   374→                  return (
   375→                    <AccordionItem key={topic.id} value={topic.id} className="border-b border-border last:border-b-0">
   376→                      <AccordionTrigger className="hover:no-underline px-2">
   377→                        <div className="flex-1 min-w-0 text-left">
   378→                          <p className={`text-sm font-semibold truncate ${isActiveTopic ? "text-accent" : "text-foreground"}`}>
   379→                            {topic.title}
   380→                          </p>
   381→                          <p className="text-xs text-muted-foreground mt-0.5">
   382→                            {completedInTopic}/{topic.videos.length} videos
   383→                          </p>
   384→                        </div>
   385→                      </AccordionTrigger>
   386→                      <AccordionContent className="pb-2">
   387→                        <ul className="space-y-0.5">
   388→                          {topic.videos.map((video) => {
   389→                            const isActive = video.id === activeVideo?.id;
   390→                            const completed = video.progress?.isCompleted;
   391→                            return (
   392→                              <li key={video.id}>
   393→                                <button
   394→                                  onClick={() => setUserSelectedVideoId(video.id)}
   395→                                  className={`flex w-full items-start gap-2 rounded-md px-2 py-2 text-left text-sm transition-colors ${
   396→                                    isActive
   397→                                      ? "bg-accent/10 text-accent"
   398→                                      : "text-foreground hover:bg-muted"
   399→                                  }`}
   400→                                >
   401→                                  <span className="mt-0.5 shrink-0">
   402→                                    {completed ? (
   403→                                      <CheckCircle2 className="h-4 w-4 text-success" />
   404→                                    ) : isActive ? (
   405→                                      <PlayCircle className="h-4 w-4 text-accent" />
   406→                                    ) : (
   407→                                      <PlayCircle className="h-4 w-4 text-muted-foreground" />
   408→                                    )}
   409→                                  </span>
   410→                                  <span className="min-w-0 flex-1">
   411→                                    <span className="block truncate font-medium">{video.title}</span>
   412→                                    {video.duration > 0 && (
   413→                                      <span className="block text-xs text-muted-foreground mt-0.5">
   414→                                        {formatVideoDuration(video.duration)}
   415→                                      </span>
   416→                                    )}
   417→                                  </span>
   418→                                </button>
   419→                              </li>
   420→                            );
   421→                          })}
   422→                        </ul>
   423→                      </AccordionContent>
   424→                    </AccordionItem>
   425→                  );
   426→                })}
   427→              </Accordion>
   428→            )}
   429→          </CardContent>
   430→        </Card>
   431→      </div>
   432→    </div>
   433→  );
   434→}

   437→
   438→function VideoPlayer({
   439→  video,
   440→  onProgress,
   441→}: {
   442→  video: VideoItem;
   443→  onProgress: (update: {
   444→    videoId: string;
   445→    progress: VideoProgressInfo;
   446→    justCompleted: boolean;
   447→  }) => void;
   448→}) {
   449→  const videoRef = useRef<HTMLVideoElement>(null);
   450→  const lastTimeRef = useRef<number>(0);
   451→  const watchTimeRef = useRef<number>(0);
   452→  const durationRef = useRef<number>(0);
   453→  const lastPostRef = useRef<number>(0);
   454→  const resumeAppliedRef = useRef<boolean>(false);
   455→  const [expanded, setExpanded] = useState(false);
   456→
   457→  // YouTube completion timer
   458→  const ytStartRef = useRef<number | null>(null);
   459→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   460→  const ytWatchSecondsRef = useRef<number>(0);
   461→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   462→
   463→  // Local UI state
   464→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   465→    video.progress?.isCompleted || false
   466→  );
   467→  const [posting, setPosting] = useState(false);
   468→
   469→  // Reset on video change
   470→  useEffect(() => {
   471→    lastTimeRef.current = 0;
   472→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   473→    durationRef.current = video.duration || video.progress?.duration || 0;
   474→    lastPostRef.current = 0;
   475→    resumeAppliedRef.current = false;
   476→    setLocalCompleted(video.progress?.isCompleted || false);
   477→    setYtCanMarkComplete(false);
   478→    ytWatchSecondsRef.current = 0;
   479→    if (ytTimerRef.current) {
   480→      clearInterval(ytTimerRef.current);
   481→      ytTimerRef.current = null;
   482→    }
   483→  }, [video.id]);
   484→
   485→  // For HTML5 video: apply resume position once metadata is loaded.
   486→  const handleLoadedMetadata = () => {
   487→    const el = videoRef.current;
   488→    if (!el) return;
   489→    durationRef.current = el.duration || durationRef.current;
   490→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   491→      // resume within (duration - 2) so we don't immediately fire ended
   492→      const target = Math.min(
   493→        video.progress.lastPosition,
   494→        Math.max(0, (el.duration || 0) - 3)
   495→      );
   496→      try {
   497→        el.currentTime = target;
   498→      } catch {}
   499→      lastTimeRef.current = target;
   500→      resumeAppliedRef.current = true;
   501→    }
   502→  };
   503→
   504→  // Core: timeupdate handler — accumulate real watch time.
   505→  const handleTimeUpdate = () => {
   506→    const el = videoRef.current;
   507→    if (!el) return;
   508→    const cur = el.currentTime;
   509→    const last = lastTimeRef.current;
   510→    const delta = cur - last;
   511→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   512→    // (large positive deltas) and rewinds (negative deltas).
   513→    if (delta > 0 && delta <= 1.5) {
   514→      watchTimeRef.current += delta;
   515→    }
   516→    // Update lastTime to current — even on seek, so next delta is small.
   517→    lastTimeRef.current = cur;
   518→
   519→    // Throttled server POST every 5 seconds
   520→    const now = Date.now();
   521→    if (now - lastPostRef.current > 5000) {
   522→      lastPostRef.current = now;
   523→      void postProgress(false);
   524→    }
   525→  };
   526→
   527→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   528→  const handleSeeked = () => {
   529→    const el = videoRef.current;
   530→    if (!el) return;
   531→    lastTimeRef.current = el.currentTime;
   532→  };
   533→
   534→  // Ended event — submit final progress with completion hint.
   535→  const handleEnded = () => {
   536→    void postProgress(true);
   537→  };
   538→
   539→  const postProgress = useCallback(
   540→    async (isCompletionHint: boolean) => {
   541→      setPosting(true);
   542→      try {
   543→        const res = await apiPost<{
   544→          progress: VideoProgressInfo;
   545→          justCompleted: boolean;
   546→          rejectedCompletion: boolean;
   547→          certificateIssued: boolean;
   548→          certificateId: string | null;
   549→        }>("/api/student/progress", {
   550→          videoId: video.id,
   551→          lastPosition: videoRef.current?.currentTime || 0,
   552→          watchTimeSeconds: watchTimeRef.current,
   553→          duration: durationRef.current,
   554→          isCompleted: isCompletionHint,
   555→        });
   556→        setLocalCompleted(res.progress.isCompleted);
   557→        onProgress({
   558→          videoId: video.id,
   559→          progress: res.progress,
   560→          justCompleted: res.justCompleted,
   561→        });
   562→        if (isCompletionHint && res.rejectedCompletion) {
   563→          toast({
   564→            title: "Completion not recorded",
   565→            description:
   566→              "You need to watch at least 85% of the video before it's marked complete.",
   567→            variant: "destructive",
   568→          });
   569→        }
   570→        // ─── Certificate auto-issued upon course completion ───
   571→        if (res.certificateIssued && res.certificateId) {
   572→          toast({
   573→            title: "🎉 Certificate Issued!",
   574→            description:
   575→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   576→          });
   577→        }
   578→      } catch {
   579→        // apiPost already surfaces errors
   580→      } finally {
   581→        setPosting(false);
   582→      }
   583→    },
   584→    [video.id, onProgress]
   585→  );
   586→
   587→  // Manual "Mark complete" for YouTube videos
   588→  const markYtComplete = async () => {
   589→    setPosting(true);
   590→    try {
   591→      const res = await apiPost<{
   592→        progress: VideoProgressInfo;
   593→        justCompleted: boolean;
   594→        rejectedCompletion: boolean;
   595→        certificateIssued: boolean;

  1079→// Local helper for YouTube URL parsing
  1080→function toYouTubeEmbed(url: string): string | null {
  1081→  try {
  1082→    const u = new URL(url);
  1083→    if (u.hostname === "youtu.be") {
  1084→      const id = u.pathname.slice(1);
  1085→      return `https://www.youtube.com/embed/${id}?rel=0`;
  1086→    }
  1087→    if (u.hostname.includes("youtube.com")) {
  1088→      const id = u.searchParams.get("v");
  1089→      if (id) return `https://www.youtube.com/embed/${id}?rel=0`;
  1090→      // /embed/ID or /shorts/ID
  1091→      const m = u.pathname.match(/\/(embed|shorts)\/([\w-]+)/);
  1092→      if (m) return `https://www.youtube.com/embed/${m[2]}?rel=0`;
  1093→    }
  1094→    return null;
  1095→  } catch {
  1096→    return null;
  1097→  }
  1098→}

  1100→function formatVideoDuration(seconds: number): string {
  1101→  if (seconds <= 0) return "";
  1102→  const m = Math.floor(seconds / 60);
  1103→  const s = Math.floor(seconds % 60);
  1104→  if (m === 0) return `${s}s`;
  1105→  return `${m}:${s.toString().padStart(2, "0")}`;
  1106→}
