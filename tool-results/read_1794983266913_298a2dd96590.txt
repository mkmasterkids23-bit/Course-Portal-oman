    37→let ytApiPromise: Promise<any> | null = null;
    38→function loadYouTubeAPI(): Promise<any> {
    39→  if (typeof window === "undefined") return Promise.reject(new Error("no window"));
    40→  if (window.YT && window.YT.Player) return Promise.resolve(window.YT);
    41→  if (ytApiPromise) return ytApiPromise;
    42→  ytApiPromise = new Promise((resolve) => {
    43→    const prev = window.onYouTubeIframeAPIReady;
    44→    window.onYouTubeIframeAPIReady = () => {
    45→      if (typeof prev === "function") { try { prev(); } catch {} }
    46→      resolve(window.YT);
    47→    };
    48→    const existing = document.querySelector('script[src="https://www.youtube.com/iframe_api"]');
    49→    if (!existing) {
    50→      const tag = document.createElement("script");
    51→      tag.src = "https://www.youtube.com/iframe_api";
    52→      document.head.appendChild(tag);
    53→    }
    54→  });
    55→  return ytApiPromise;
    56→}

   197→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   198→  const { t } = useI18n();
   199→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   200→    `/api/student/courses/${courseId}`
   201→  );
   202→
   203→  // Read the hash query for deep-link support. The global Discussions
   204→  // view navigates here with `?tab=discussions&discussion=<id>`; we honour
   205→  // that by defaulting the Materials/Discussions tab to "discussions" and
   206→  // asking the panel to highlight (and scroll to) the matching thread.
   207→  const routeQuery = useAppStore((s) => s.query);
   208→  const highlightDiscussionId = routeQuery.discussion || null;
   209→  const discussionsTabDefault = routeQuery.tab === "discussions";
   210→
   211→  // Pick the initial video: last watched, or first available.
   212→  const initialVideoId = (() => {
   213→    if (!data?.course) return null;
   214→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   215→    //    most recently updated progress entry in the topics tree.
   216→    let candidate: VideoItem | null = null;
   217→    for (const t of data.course.topics) {
   218→      for (const v of t.videos) {
   219→        if (v.progress && !v.progress.isCompleted) {
   220→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   221→            candidate = v;
   222→          }
   223→        }
   224→      }
   225→    }
   226→    if (candidate) return candidate.id;
   227→    // 2) First video of the first topic
   228→    return data.course.topics[0]?.videos[0]?.id ?? null;
   229→  })();
   230→
   231→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   232→  // Reset user selection when courseId changes — "adjust state during render"
   233→  // pattern (avoids setState-in-effect warnings).
   234→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   235→  if (courseId !== prevCourseId) {
   236→    setPrevCourseId(courseId);
   237→    setUserSelectedVideoId(null);
   238→  }
   239→  // Effective active video: explicit user selection overrides the auto-pick
   240→  // (which is the last-watched or first video). When data loads and we have an
   241→  // initial pick, this falls back to it without needing an effect.
   242→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   243→
   244→  if (loading) {
   245→    return (
   246→      <div className="space-y-5">
   247→        <Skeleton className="h-10 w-32 rounded-lg" />
   248→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   249→          <div className="space-y-4">
   250→            <Skeleton className="h-[420px] rounded-xl" />
   251→            <Skeleton className="h-24 rounded-xl" />
   252→          </div>
   253→          <Skeleton className="h-[600px] rounded-xl" />
   254→        </div>
   255→      </div>
   256→    );
   257→  }
   258→
   259→  if (error) {
   260→    return (
   261→      <div className="space-y-4">
   262→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   263→          <ArrowLeft className="h-4 w-4" /> {t("learning.backToCourses")}
   264→        </Button>
   265→        <EmptyState
   266→          icon={<AlertTriangle className="h-5 w-5" />}
   267→          title={t("learning.couldntOpenCourse")}
   268→          description={error}
   269→          action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   270→        />
   271→      </div>
   272→    );
   273→  }
   274→
   275→  if (!data?.course) {
   276→    return (
   277→      <EmptyState
   278→        icon={<PlayCircle className="h-5 w-5" />}
   279→        title={t("learning.courseNotFound")}
   280→        description={t("learning.courseMayRemoved")}

   282→      />
   283→    );
   284→  }
   285→
   286→  const course = data.course;
   287→  const allVideos = course.topics.flatMap((t) => t.videos);
   288→  const activeVideo =
   289→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   290→
   291→  // Find the topic containing the active video
   292→  const activeTopic = course.topics.find((t) =>
   293→    t.videos.some((v) => v.id === activeVideo?.id)
   294→  );
   295→
   296→  // Build a flat ordered list for prev/next navigation
   297→  const flatVideoIds = allVideos.map((v) => v.id);
   298→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   299→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   300→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   301→
   302→  function handleProgressUpdate(updated: {
   303→    videoId: string;
   304→    progress: VideoProgressInfo;
   305→    justCompleted: boolean;
   306→  }) {
   307→    if (!data) return;
   308→    const newCourse = {
   309→      ...data.course,
   310→      topics: data.course.topics.map((t) => ({
   311→        ...t,
   312→        videos: t.videos.map((v) =>
   313→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   314→        ),
   315→      })),
   316→      courseCompletion: (() => {
   317→        // Recompute local snapshot
   318→        const videos = data.course.topics.flatMap((t) => t.videos);
   319→        const completed = videos.filter(
   320→          (v) =>
   321→            v.id === updated.videoId
   322→              ? updated.progress.isCompleted
   323→              : v.progress?.isCompleted
   324→        ).length;
   325→        const total = videos.length;
   326→        // No videos → course is considered fully watched (assignment unlocked).
   327→        const percent = total > 0 ? Math.round((completed / total) * 100) : 100;
   328→        const totalTopics = data.course.topics.length;
   329→        const completedTopics = data.course.topics.filter((topic) =>
   330→          topic.videos.length > 0 &&
   331→          topic.videos.every(
   332→            (v) =>
   333→              (v.id === updated.videoId ? updated.progress.isCompleted : v.progress?.isCompleted)
   334→          )
   335→        ).length;
   336→        return {
   337→          totalVideos: total,
   338→          completedVideos: completed,
   339→          completedTopics,
   340→          totalTopics,
   341→          percent,
   342→          allCompleted: total === 0 || completed === total,
   343→        };
   344→      })(),
   345→    };
   346→    setData({ course: newCourse });
   347→
   348→    if (updated.justCompleted) {
   349→      toast({
   350→        title: t("learning.lessonCompletedToast"),
   351→        description: t("learning.greatWorkSaved"),

   367→      <div className="shadow-premium flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-4 py-3">
   368→        <div className="flex items-center gap-2 min-w-0 flex-1">
   369→          <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0 h-8 w-8" aria-label={t("learning.backToCourses")}>
   370→            <ArrowLeft className="h-4 w-4" />
   371→          </Button>
   372→          <div className="min-w-0 flex-1">
   373→            <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   374→              <button onClick={onBack} className="hover:text-foreground transition-colors">{t("learning.myCourses")}</button>
   375→              <span>/</span>
   376→              <span className="truncate">{course.title}</span>
   377→              {activeVideo && (
   378→                <>
   379→                  <span>/</span>
   380→                  <span className="truncate text-foreground font-medium">{activeVideo.title}</span>
   381→                </>
   382→              )}
   383→            </div>
   384→            <div className="flex items-center gap-2 mt-0.5">
   385→              {course.category && (
   386→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   387→                  {course.category}
   388→                </Badge>

   390→              <Badge
   391→                className={
   392→                  course.accessType === "FREE"
   393→                    ? "bg-success/15 text-success border-transparent text-[10px] px-1.5 py-0 h-4"
   394→                    : "bg-primary text-primary-foreground border-transparent text-[10px] px-1.5 py-0 h-4"
   395→                }
   396→              >
   397→                {course.accessType === "FREE" ? t("common.free") : t("common.paid")}
   398→              </Badge>
   399→            </div>
   400→          </div>
   401→        </div>
   402→        <div className="flex items-center gap-3 shrink-0">
   403→          <div className="hidden sm:flex flex-col items-end">
   404→            <span className="text-xs font-medium text-foreground">
   405→              {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} {t("courses.lessons")}
   406→            </span>
   407→            <span className="text-[10px] text-muted-foreground">{t("learning.completed")}</span>
   408→          </div>
   409→          <div className="flex items-center gap-2 min-w-[120px]">
   410→            <Progress
   411→              value={course.courseCompletion.percent}
   412→              className={`h-1.5 w-24 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   413→            />

   416→        </div>

   494→            </div>

   496→            {/* Tab 1: Overview */}
   497→            <TabsContent value="overview" className="mt-0 p-4">

   755→/* ===================== Video player ===================== */
   756→
   757→function VideoPlayer({
   758→  video,
   759→  topicTitle,
   760→  courseId,
   761→  topicId,
   762→  onProgress,
   763→  prevVideo,
   764→  nextVideo,
   765→  onNavigateLesson,
   766→  refetchCourse,
   767→}: {
   768→  video: VideoItem;
   769→  topicTitle: string | null;
   770→  courseId: string;
   771→  topicId: string | null;
   772→  onProgress: (update: {
   773→    videoId: string;
   774→    progress: VideoProgressInfo;
   775→    justCompleted: boolean;
   776→  }) => void;
   777→  prevVideo: VideoItem | null;
   778→  nextVideo: VideoItem | null;
   779→  onNavigateLesson: (id: string) => void;
   780→  refetchCourse?: () => void;
   781→}) {
   782→  const videoRef = useRef<HTMLVideoElement>(null);
   783→  const lastTimeRef = useRef<number>(0);
   784→  const watchTimeRef = useRef<number>(0);
   785→  const durationRef = useRef<number>(0);
   786→  const lastPostRef = useRef<number>(0);
   787→  const resumeAppliedRef = useRef<boolean>(false);
   788→  const [expanded, setExpanded] = useState(false);
   789→
   790→  // ─── Video Assessment state ───
   791→  // Fetches the questions for THIS video (without correctAnswer — security).
   792→  // The hook is unconditional (rules of hooks); the URL just changes per video.
   793→  const questionsUrl = `/api/student/videos/${video.id}/questions`;
   794→  const {
   795→    data: questionsData,
   796→    loading: questionsLoading,
   797→    refetch: refetchQuestions,
   798→  } = useApiFetch<{
   799→    questions: VideoQuestionData[];
   800→    assessment: VideoAssessmentSummary | null;
   801→  }>(questionsUrl);
   802→
   803→  // Currently-displayed question popup (null = no popup shown).
   804→  const [activeQuestion, setActiveQuestion] = useState<VideoQuestionData | null>(null);
   805→  // Map of questionId → selected option index (answers given this session).
   806→  const [answeredMap, setAnsweredMap] = useState<Map<string, number>>(new Map());
   807→  // Currently-selected option in the active popup (null = nothing selected yet).
   808→  const [selectedAnswer, setSelectedAnswer] = useState<number | null>(null);
   809→  // Brief "Answer saved" confirmation shown after submitting a popup answer.
   810→  const [showFeedback, setShowFeedback] = useState(false);
   811→  // Result returned from POST /assessment — when set, the result overlay shows.
   812→  const [assessmentResult, setAssessmentResult] = useState<AssessmentResult | null>(null);
   813→  // True while the assessment POST is in flight.
   814→  const [submittingAssessment, setSubmittingAssessment] = useState(false);
   815→  // Timer ref for the post-submit feedback delay (cleared on unmount/video change).
   816→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   817→
   818→  // ─── Player abstraction + YouTube IFrame player ───
   819→  const playerApiRef = useRef<PlayerApi | null>(null);
   820→  const ytPlayerRef = useRef<any>(null);
   821→  const ytPollRef = useRef<ReturnType<typeof setInterval> | null>(null);
   822→  const ytContainerRef = useRef<HTMLDivElement | null>(null);
   823→
   824→  // Local UI state
   825→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   826→    video.progress?.isCompleted || false
   827→  );
   828→  const [posting, setPosting] = useState(false);
   829→
   830→  // Are question popups active for this viewing session?
   831→  // They are NOT active if the student has already passed the assessment —
   832→  // in that case the video plays freely and the "Passed" badge is shown.
   833→  const questionsActive = Boolean(
   834→    questionsData &&
   835→      questionsData.questions.length > 0 &&
   836→      questionsData.assessment?.passed !== true
   837→  );
   838→
   839→  // Reset on video change
   840→  useEffect(() => {
   841→    lastTimeRef.current = 0;
   842→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   843→    durationRef.current = video.duration || video.progress?.duration || 0;
   844→    lastPostRef.current = 0;
   845→    resumeAppliedRef.current = false;
   846→    setLocalCompleted(video.progress?.isCompleted || false);
   847→    // Tear down any YouTube player + polling from the previous video.
   848→    if (ytPollRef.current) {
   849→      clearInterval(ytPollRef.current);
   850→      ytPollRef.current = null;
   851→    }
   852→    try { ytPlayerRef.current?.destroy?.(); } catch {}
   853→    ytPlayerRef.current = null;
   854→    playerApiRef.current = null;
   855→    // ─── Reset assessment state ───
   856→    setActiveQuestion(null);
   857→    setSelectedAnswer(null);
   858→    setAnsweredMap(new Map());
   859→    setShowFeedback(false);
   860→    setAssessmentResult(null);
   861→    setSubmittingAssessment(false);
   862→    if (feedbackTimerRef.current) {
   863→      clearTimeout(feedbackTimerRef.current);
   864→      feedbackTimerRef.current = null;
   865→    }
   866→  }, [video.id]);
   867→
   868→  // ─── Missed-question catch-up ───
   869→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   870→  // the student answers a question. If the playhead has already moved past an
   871→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   872→  // the questions fetch landed late), pause and surface that question now.
   873→  useEffect(() => {
   874→    if (!questionsActive || !questionsData) return;
   875→    if (activeQuestion || assessmentResult || showFeedback || submittingAssessment) return;
   876→    const api = playerApiRef.current;
   877→    if (!api) return;
   878→    const cur = api.currentTime();
   879→    const missed = questionsData.questions.find(
   880→      (q) => !answeredMap.has(q.id) && cur >= q.timestamp - 0.5
   881→    );
   882→    if (missed && !api.paused()) {
   883→      try {
   884→        api.pause();
   885→        lastTimeRef.current = api.currentTime();
   886→      } catch {}
   887→      setActiveQuestion(missed);
   888→      setSelectedAnswer(null);
   889→    }
   890→  }, [
   891→    questionsActive,
   892→    questionsData,
   893→    activeQuestion,
   894→    assessmentResult,
   895→    showFeedback,
   896→    submittingAssessment,
   897→    answeredMap,
   898→  ]);
   899→
   900→  // For HTML5 video: apply resume position once metadata is loaded.
   901→  const handleLoadedMetadata = () => {
   902→    const el = videoRef.current;
   903→    if (!el) return;
   904→    // Wire up the shared player abstraction for HTML5 video.
   905→    playerApiRef.current = {
   906→      currentTime: () => el.currentTime,
   907→      duration: () => el.duration,
   908→      pause: () => { el.pause(); },
   909→      play: () => { void el.play().catch(() => {}); },
   910→      seekTo: (t) => { el.currentTime = t; },
   911→      paused: () => el.paused,
   912→    };
   913→    durationRef.current = el.duration || durationRef.current;
   914→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   915→      // resume within (duration - 2) so we don't immediately fire ended
   916→      const target = Math.min(
   917→        video.progress.lastPosition,
   918→        Math.max(0, (el.duration || 0) - 3)
   919→      );
   920→      try {
   921→        el.currentTime = target;
   922→      } catch {}
   923→      lastTimeRef.current = target;
   924→      resumeAppliedRef.current = true;
   925→    }
   926→  };
   927→
   928→  // Core: timeupdate handler — accumulate real watch time.
   929→  const handleTimeUpdate = () => {
   930→    const api = playerApiRef.current;
   931→    if (!api) return;
   932→    const cur = api.currentTime();
   933→    const last = lastTimeRef.current;
   934→
   935→    // ─── Question trigger check ───
   936→    // Pause and surface a popup the moment the playhead crosses an unanswered
   937→    // question's timestamp going forward. We require BOTH:
   938→    //   - cur >= q.timestamp    (we're at or past it)
   939→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   940→    //                            this is a forward crossing, not a re-tick while
   941→    //                            paused, and not a backward seek landing past it)
   942→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   943→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   944→    if (
   945→      questionsActive &&
   946→      !activeQuestion &&
   947→      !assessmentResult &&
   948→      !showFeedback &&
   949→      !submittingAssessment &&
   950→      questionsData
   951→    ) {
   952→      for (const q of questionsData.questions) {
   953→        if (answeredMap.has(q.id)) continue;
   954→        if (cur >= q.timestamp && last < q.timestamp) {
   955→          try {
   956→            api.pause();
   957→            lastTimeRef.current = cur;
   958→          } catch {}
   959→          setActiveQuestion(q);
   960→          setSelectedAnswer(null);
   961→          return; // do NOT accumulate watch time for this tick
   962→        }
   963→      }
   964→    }
   965→
   966→    // Watch time tracking — only counts normal forward playback deltas.
   967→    const delta = cur - last;
   968→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   969→    // (large positive deltas) and rewinds (negative deltas).
   970→    if (delta > 0 && delta <= 1.5) {
   971→      watchTimeRef.current += delta;
   972→    }
   973→    // Update lastTime to current — even on seek, so next delta is small.
   974→    lastTimeRef.current = cur;
   975→
   976→    // Forward-seek past an unanswered question (YouTube doesn't fire seeked).
   977→    if (delta > 1.5 && questionsActive && !activeQuestion && !assessmentResult && !showFeedback && !submittingAssessment && questionsData) {
   978→      const skipped = questionsData.questions.find((q) => !answeredMap.has(q.id) && cur > q.timestamp + 0.5);
   979→      if (skipped) { try { api.pause(); } catch {} setActiveQuestion(skipped); setSelectedAnswer(null); return; }
   980→    }
   981→
   982→    // Throttled server POST every 5 seconds
   983→    const now = Date.now();
   984→    if (now - lastPostRef.current > 5000) {
   985→      lastPostRef.current = now;
   986→      void postProgress(false);
   987→    }
   988→  };
   989→
   990→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   991→  // Also catches forward-seek-past-question: if the student scrubs the
   992→  // progress bar past an unanswered question, pause and surface that question
   993→  // so they can't skip it.
   994→  const handleSeeked = () => {
   995→    const api = playerApiRef.current;
   996→    if (!api) return;
   997→    const newTime = api.currentTime();
   998→    lastTimeRef.current = newTime;
   999→
  1000→    if (
  1001→      questionsActive &&
  1002→      !activeQuestion &&
  1003→      !assessmentResult &&
  1004→      !showFeedback &&
  1005→      !submittingAssessment &&
  1006→      questionsData
  1007→    ) {
  1008→      // Find the earliest unanswered question we've now skipped past.
  1009→      const skipped = questionsData.questions.find(
  1010→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
  1011→      );
  1012→      if (skipped) {
  1013→        try {
  1014→          api.pause();
  1015→        } catch {}
  1016→        setActiveQuestion(skipped);
  1017→        setSelectedAnswer(null);
  1018→      }
  1019→    }
  1020→  };
  1021→
  1022→  // Ended event — submit final progress with completion hint.
  1023→  const handleEnded = () => {
  1024→    void postProgress(true);
  1025→  };
  1026→
  1027→  const postProgress = useCallback(
  1028→    async (isCompletionHint: boolean) => {
  1029→      setPosting(true);
  1030→      try {
  1031→        const res = await apiPost<{
  1032→          progress: VideoProgressInfo;
  1033→          justCompleted: boolean;
  1034→          rejectedCompletion: boolean;
  1035→          certificateIssued: boolean;
  1036→          certificateId: string | null;
  1037→        }>("/api/student/progress", {
  1038→          videoId: video.id,
  1039→          lastPosition: isCompletionHint && Number.isFinite(durationRef.current) && durationRef.current > 0
  1040→            ? durationRef.current : (videoRef.current?.currentTime || 0),
  1041→          watchTimeSeconds: watchTimeRef.current,
  1042→          duration: Number.isFinite(durationRef.current) ? durationRef.current : 0,
  1043→          isCompleted: isCompletionHint,
  1044→          ended: isCompletionHint,
  1045→        });
  1046→        setLocalCompleted(res.progress.isCompleted);
  1047→        onProgress({
  1048→          videoId: video.id,
  1049→          progress: res.progress,
  1050→          justCompleted: res.justCompleted,
  1051→        });
  1052→        if (isCompletionHint && res.rejectedCompletion) {
  1053→          toast({
  1054→            title: "Completion not recorded",
