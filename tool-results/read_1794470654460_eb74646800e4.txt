   158→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   159→  const { t } = useI18n();
   160→  const { data, loading, error, refetch } = useApiFetch<CourseResponse>(
   161→    `/api/student/courses/${courseId}`
   162→  );
   163→
   164→  // Read the hash query for deep-link support. The global Discussions
   165→  // view navigates here with `?tab=discussions&discussion=<id>`; we honour
   166→  // that by defaulting the Materials/Discussions tab to "discussions" and
   167→  // asking the panel to highlight (and scroll to) the matching thread.
   168→  const routeQuery = useAppStore((s) => s.query);
   169→  const highlightDiscussionId = routeQuery.discussion || null;
   170→  const discussionsTabDefault = routeQuery.tab === "discussions";
   171→
   172→  // Pick the initial video: last watched, or first available.
   173→  const initialVideoId = (() => {
   174→    if (!data?.course) return null;
   175→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   176→    //    most recently updated progress entry in the topics tree.
   177→    let candidate: VideoItem | null = null;
   178→    for (const t of data.course.topics) {
   179→      for (const v of t.videos) {
   180→        if (v.progress && !v.progress.isCompleted) {
   181→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   182→            candidate = v;
   183→          }
   184→        }
   185→      }
   186→    }
   187→    if (candidate) return candidate.id;
   188→    // 2) First video of the first topic
   189→    return data.course.topics[0]?.videos[0]?.id ?? null;
   190→  })();
   191→
   192→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   193→  // Reset user selection when courseId changes — "adjust state during render"
   194→  // pattern (avoids setState-in-effect warnings).
   195→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   196→  if (courseId !== prevCourseId) {
   197→    setPrevCourseId(courseId);
   198→    setUserSelectedVideoId(null);
   199→  }
   200→  // Effective active video: explicit user selection overrides the auto-pick
   201→  // (which is the last-watched or first video). When data loads and we have an
   202→  // initial pick, this falls back to it without needing an effect.
   203→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   204→
   205→  if (loading) {
   206→    return (
   207→      <div className="space-y-5">
   208→        <Skeleton className="h-10 w-32 rounded-lg" />
   209→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   210→          <div className="space-y-4">
   211→            <Skeleton className="h-[420px] rounded-xl" />
   212→            <Skeleton className="h-24 rounded-xl" />
   213→          </div>
   214→          <Skeleton className="h-[600px] rounded-xl" />
   215→        </div>
   216→      </div>
   217→    );
   218→  }
   219→
   220→  if (error) {
   221→    return (
   222→      <div className="space-y-4">
   223→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   224→          <ArrowLeft className="h-4 w-4" /> {t("learning.backToCourses")}
   225→        </Button>
   226→        <EmptyState
   227→          icon={<AlertTriangle className="h-5 w-5" />}
   228→          title={t("learning.couldntOpenCourse")}
   229→          description={error}
   230→          action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   231→        />
   232→      </div>
   233→    );
   234→  }
   235→
   236→  if (!data?.course) {
   237→    return (
   238→      <EmptyState
   239→        icon={<PlayCircle className="h-5 w-5" />}
   240→        title={t("learning.courseNotFound")}
   241→        description={t("learning.courseMayRemoved")}
   242→        action={<Button onClick={onBack}>{t("learning.backToCourses")}</Button>}
   243→      />
   244→    );
   245→  }
   246→
   247→  const course = data.course;
   248→  const allVideos = course.topics.flatMap((t) => t.videos);
   249→  const activeVideo =
   250→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   251→
   252→  // Find the topic containing the active video
   253→  const activeTopic = course.topics.find((t) =>
   254→    t.videos.some((v) => v.id === activeVideo?.id)
   255→  );
   256→
   257→  // Build a flat ordered list for prev/next navigation
   258→  const flatVideoIds = allVideos.map((v) => v.id);
   259→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   260→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   261→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   262→
   263→  function handleProgressUpdate(updated: {
   264→    videoId: string;
   265→    progress: VideoProgressInfo;
   266→    justCompleted: boolean;
   267→  }) {
   268→    if (!data) return;
   269→    const newCourse = {
   270→      ...data.course,
   271→      topics: data.course.topics.map((t) => ({
   272→        ...t,
   273→        videos: t.videos.map((v) =>
   274→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   275→        ),
   276→      })),
   277→      courseCompletion: (() => {
   278→        // Recompute local snapshot
   279→        const videos = data.course.topics.flatMap((t) => t.videos);
   280→        const completed = videos.filter(
   281→          (v) =>
   282→            v.id === updated.videoId
   283→              ? updated.progress.isCompleted
   284→              : v.progress?.isCompleted
   285→        ).length;
   286→        const total = videos.length;
   287→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   288→        return {
   289→          totalVideos: total,
   290→          completedVideos: completed,
   291→          percent,
   292→          allCompleted: total > 0 && completed === total,
   293→        };
   294→      })(),
   295→    };
   296→    setData({ course: newCourse });
   297→
   298→    if (updated.justCompleted) {
   299→      toast({
   300→        title: t("learning.lessonCompletedToast"),
   301→        description: t("learning.greatWorkSaved"),
   302→      });
   303→    }
   304→  }
   305→
   306→  async function handleSubmissionSuccess() {
   307→    await refetch();
   308→    toast({
   309→      title: t("learning.assignmentSubmittedToast"),
   310→      description: t("learning.assignmentSubmittedDesc"),
   311→    });
   312→  }
   313→
   314→  return (
   315→    <div className="space-y-4 animate-page-in">
   316→      {/* ─── Compact breadcrumb header ─── */}
   317→      <div className="flex items-center justify-between gap-3 rounded-lg border border-border bg-card px-4 py-3 shadow-sm">
   318→        <div className="flex items-center gap-2 min-w-0 flex-1">
   319→          <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0 h-8 w-8" aria-label={t("learning.backToCourses")}>
   320→            <ArrowLeft className="h-4 w-4" />
   321→          </Button>
   322→          <div className="min-w-0 flex-1">
   323→            <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   324→              <button onClick={onBack} className="hover:text-foreground transition-colors">{t("learning.myCourses")}</button>
   325→              <span>/</span>
   326→              <span className="truncate">{course.title}</span>
   327→              {activeVideo && (
   328→                <>
   329→                  <span>/</span>
   330→                  <span className="truncate text-foreground font-medium">{activeVideo.title}</span>
   331→                </>
   332→              )}
   333→            </div>
   334→            <div className="flex items-center gap-2 mt-0.5">
   335→              {course.category && (
   336→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   337→                  {course.category}
   338→                </Badge>
   339→              )}
   340→              <Badge
   341→                className={
   342→                  course.accessType === "FREE"
   343→                    ? "bg-success/15 text-success border-transparent text-[10px] px-1.5 py-0 h-4"
   344→                    : "bg-primary text-primary-foreground border-transparent text-[10px] px-1.5 py-0 h-4"
   345→                }
   346→              >
   347→                {course.accessType === "FREE" ? t("common.free") : t("common.paid")}
   348→              </Badge>
   349→            </div>
   350→          </div>
   351→        </div>
   352→        <div className="flex items-center gap-3 shrink-0">
   353→          <div className="hidden sm:flex flex-col items-end">
   354→            <span className="text-xs font-medium text-foreground">
   355→              {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} {t("courses.lessons")}
   356→            </span>
   357→            <span className="text-[10px] text-muted-foreground">{t("learning.completed")}</span>
   358→          </div>
   359→          <div className="flex items-center gap-2 min-w-[120px]">
   360→            <Progress
   361→              value={course.courseCompletion.percent}
   362→              className={`h-1.5 w-24 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   363→            />
   364→            <span className="text-xs font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   365→          </div>
   366→        </div>
   367→      </div>
   368→
   369→      {/* ─── Main Layout: LEFT content + RIGHT curriculum ─── */}
   370→      <div className="grid gap-4 lg:grid-cols-[1fr_340px]">
   371→        {/* ─── LEFT: Video + Tabs ─── */}
   372→        <div className="space-y-4 min-w-0">
   373→          {/* Video player */}
   374→          {activeVideo ? (
   375→            <VideoPlayer
   376→              key={activeVideo.id}
   377→              video={activeVideo}
   378→              topicTitle={activeTopic?.title || null}
   379→              courseId={course.id}
   380→              topicId={activeTopic?.id || null}
   381→              onProgress={handleProgressUpdate}
   382→              prevVideo={prevVideo}
   383→              nextVideo={nextVideo}
   384→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   385→              refetchCourse={refetch}
   386→            />
   387→          ) : (
   388→            <EmptyState
   389→              icon={<VideoIcon className="h-5 w-5" />}
   390→              title={t("courses.noVideosYet")}
   391→              description={t("courses.noVideosDesc")}
   392→            />
   393→          )}
   394→
   395→          {/* Assignment panel — hidden until all videos completed */}
   396→          {course.assignment && course.courseCompletion.allCompleted ? (
   397→            <AssignmentPanel course={course} onSubmitted={handleSubmissionSuccess} />
   398→          ) : course.assignment ? (
   399→            <LockedAssignmentCard
   400→              completedVideos={course.courseCompletion.completedVideos}
   401→              totalVideos={course.courseCompletion.totalVideos}
   402→              percent={course.courseCompletion.percent}
   403→            />
   404→          ) : null}
   405→
   406→          {/* ─── 4-Tab Interface below video ─── */}
   407→          <Tabs defaultValue="overview" className="gap-0">
   408→            <div className="border-b border-border">
   409→              <TabsList className="bg-transparent border-0 p-0 h-auto gap-0">
   410→                <TabsTrigger
   411→                  value="overview"
   412→                  className="rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   413→                >
   414→                  <Info className="h-3.5 w-3.5" />
   415→                  {t("learning.tab.overview")}
   416→                </TabsTrigger>
   417→                <TabsTrigger
   418→                  value="notes"
   419→                  className="rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   420→                >
   421→                  <StickyNote className="h-3.5 w-3.5" />
   422→                  {t("learning.tab.notes")}
   423→                </TabsTrigger>
   424→                <TabsTrigger
   425→                  value="discussions"
   426→                  className="rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   427→                >
   428→                  <MessageSquare className="h-3.5 w-3.5" />
   429→                  {t("learning.tab.discussion")}
   430→                </TabsTrigger>
   431→                <TabsTrigger
   432→                  value="resources"
   433→                  className="rounded-none border-b-2 border-transparent data-[state=active]:border-accent data-[state=active]:bg-transparent data-[state=active]:text-accent gap-1.5 px-4 py-2.5 text-sm font-medium transition-all"
   434→                >
   435→                  <FileText className="h-3.5 w-3.5" />
   436→                  {t("learning.tab.resources")}
   437→                  {(course.materials.length > 0 || (activeTopic && activeTopic.materials.length > 0)) && (
   438→                    <Badge variant="outline" className="ml-0.5 text-[10px] px-1.5 py-0 h-4 border-border text-muted-foreground">
   439→                      {course.materials.length + (activeTopic?.materials.length || 0)}
   440→                    </Badge>
   441→                  )}
   442→                </TabsTrigger>
   443→              </TabsList>
   444→            </div>
   445→
   446→            {/* Tab 1: Overview */}
   447→            <TabsContent value="overview" className="mt-0 p-4">
   448→              <div className="space-y-3">
   449→                {activeVideo && (
   450→                  <>
   451→                    <div>
   452→                      <h3 className="font-heading text-base font-semibold text-foreground">
   453→                        {activeVideo.title}
   454→                      </h3>
   455→                      {activeTopic && (
   456→                        <p className="text-xs text-muted-foreground mt-0.5">
   457→                          {t("learning.topic")} {activeTopic.title}
   458→                        </p>
   459→                      )}
   460→                    </div>
   461→                    {activeVideo.description && (
   462→                      <p className="text-sm text-muted-foreground leading-relaxed">
   463→                        {activeVideo.description}
   464→                      </p>
   465→                    )}
   466→                    <div className="flex items-center gap-3 pt-2 border-t border-border">
   467→                      {activeVideo.progress?.isCompleted ? (
   468→                        <Badge className="bg-success/15 text-success border-transparent gap-1">
   469→                          <CheckCircle2 className="h-3 w-3" /> {t("common.completed")}
   470→                        </Badge>
   471→                      ) : activeVideo.progress?.lastPosition ? (
   472→                        <Badge className="bg-warning/15 text-warning border-transparent gap-1">
   473→                          <Clock className="h-3 w-3" /> {t("common.inProgress")}
   474→                        </Badge>
   475→                      ) : (
   476→                        <Badge variant="outline" className="text-muted-foreground gap-1">
   477→                          <PlayCircle className="h-3 w-3" /> {t("common.notStarted")}
   478→                        </Badge>
   479→                      )}
   480→                      {activeVideo.duration > 0 && (
   481→                        <span className="text-xs text-muted-foreground">
   482→                          {t("common.duration")}: {formatVideoDuration(activeVideo.duration)}
   483→                        </span>
   484→                      )}
   485→                    </div>
   486→                  </>
   487→                )}
   488→              </div>
   489→            </TabsContent>
   490→
   491→            {/* Tab 2: My Notes */}
   492→            <TabsContent value="notes" className="mt-0 p-4">
   493→              {activeVideo && (
   494→                <VideoNotesPanel
   495→                  videoId={activeVideo.id}
   496→                  courseId={course.id}
   497→                  topicId={activeTopic?.id || null}
   498→                  videoTitle={activeVideo.title}
   499→                  canCaptureTimestamp={false}
   500→                  getCurrentTime={() => 0}
   501→                  seekTo={() => {}}
   502→                />
   503→              )}
   504→            </TabsContent>
   505→
   506→            {/* Tab 3: Discussion */}
   507→            <TabsContent value="discussions" className="mt-0">
   508→              <CourseDiscussionsPanel
   509→                courseId={course.id}
   510→                highlightDiscussionId={highlightDiscussionId}
   511→              />
   512→            </TabsContent>
   513→
   514→            {/* Tab 4: Resources */}
   515→            <TabsContent value="resources" className="mt-0 p-4">
   516→              <div className="space-y-3">
   517→                {activeTopic && activeTopic.materials.length > 0 && (
   518→                  <div>
   519→                    <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   520→                      {t("learning.topicMaterialsLabel")} · {activeTopic.title}
   521→                    </p>
   522→                    <ul className="space-y-2">
   523→                      {activeTopic.materials.map((m) => (
   524→                        <MaterialRow key={m.id} material={m} />
   525→                      ))}
   526→                    </ul>
   527→                  </div>
   528→                )}
   529→                {course.materials.length > 0 && (
   530→                  <div>
   531→                    <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   532→                      {t("learning.courseMaterials")}
   533→                    </p>
   534→                    <ul className="space-y-2">
   535→                      {course.materials.map((m) => (
   536→                        <MaterialRow key={m.id} material={m} />
   537→                      ))}
   538→                    </ul>
   539→                  </div>
   540→                )}
   541→                {course.materials.length === 0 && (!activeTopic || activeTopic.materials.length === 0) && (
   542→                  <div className="flex flex-col items-center justify-center py-8 text-center">
   543→                    <div className="flex h-10 w-10 items-center justify-center rounded-full bg-muted text-muted-foreground mb-2">

   705→/* ===================== Video player ===================== */
   706→
   707→function VideoPlayer({
   708→  video,
   709→  topicTitle,
   710→  courseId,
   711→  topicId,
   712→  onProgress,
   713→  prevVideo,
   714→  nextVideo,
   715→  onNavigateLesson,
   716→  refetchCourse,
   717→}: {
   718→  video: VideoItem;
   719→  topicTitle: string | null;
   720→  courseId: string;
   721→  topicId: string | null;
   722→  onProgress: (update: {
   723→    videoId: string;
   724→    progress: VideoProgressInfo;
   725→    justCompleted: boolean;
   726→  }) => void;
   727→  prevVideo: VideoItem | null;
   728→  nextVideo: VideoItem | null;
   729→  onNavigateLesson: (id: string) => void;
   730→  refetchCourse?: () => void;
   731→}) {
   732→  const videoRef = useRef<HTMLVideoElement>(null);
   733→  const lastTimeRef = useRef<number>(0);
   734→  const watchTimeRef = useRef<number>(0);
   735→  const durationRef = useRef<number>(0);
   736→  const lastPostRef = useRef<number>(0);
   737→  const resumeAppliedRef = useRef<boolean>(false);
   738→  const [expanded, setExpanded] = useState(false);
   739→
   740→  // ─── Video Assessment state ───
   741→  // Fetches the questions for THIS video (without correctAnswer — security).
   742→  // The hook is unconditional (rules of hooks); the URL just changes per video.
   743→  const questionsUrl = `/api/student/videos/${video.id}/questions`;
   744→  const {
   745→    data: questionsData,
   746→    loading: questionsLoading,
   747→    refetch: refetchQuestions,
   748→  } = useApiFetch<{
   749→    questions: VideoQuestionData[];
   750→    assessment: VideoAssessmentSummary | null;
   751→  }>(questionsUrl);
   752→
   753→  // Currently-displayed question popup (null = no popup shown).
   754→  const [activeQuestion, setActiveQuestion] = useState<VideoQuestionData | null>(null);
   755→  // Map of questionId → selected option index (answers given this session).
   756→  const [answeredMap, setAnsweredMap] = useState<Map<string, number>>(new Map());
   757→  // Currently-selected option in the active popup (null = nothing selected yet).
   758→  const [selectedAnswer, setSelectedAnswer] = useState<number | null>(null);
   759→  // Brief "Answer saved" confirmation shown after submitting a popup answer.
   760→  const [showFeedback, setShowFeedback] = useState(false);
   761→  // Result returned from POST /assessment — when set, the result overlay shows.
   762→  const [assessmentResult, setAssessmentResult] = useState<AssessmentResult | null>(null);
   763→  // True while the assessment POST is in flight.
   764→  const [submittingAssessment, setSubmittingAssessment] = useState(false);
   765→  // Timer ref for the post-submit feedback delay (cleared on unmount/video change).
   766→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   767→
   768→  // YouTube completion timer
   769→  const ytStartRef = useRef<number | null>(null);
   770→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   771→  const ytWatchSecondsRef = useRef<number>(0);
   772→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   773→
   774→  // Local UI state
   775→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   776→    video.progress?.isCompleted || false
   777→  );
   778→  const [posting, setPosting] = useState(false);
   779→
   780→  // Are question popups active for this viewing session?
   781→  // They are NOT active if the student has already passed the assessment —
   782→  // in that case the video plays freely and the "Passed" badge is shown.
   783→  const questionsActive = Boolean(
   784→    questionsData &&
   785→      questionsData.questions.length > 0 &&
   786→      questionsData.assessment?.passed !== true
   787→  );
   788→
   789→  // Reset on video change
   790→  useEffect(() => {
   791→    lastTimeRef.current = 0;
   792→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   793→    durationRef.current = video.duration || video.progress?.duration || 0;
   794→    lastPostRef.current = 0;
   795→    resumeAppliedRef.current = false;
   796→    setLocalCompleted(video.progress?.isCompleted || false);
   797→    setYtCanMarkComplete(false);
   798→    ytWatchSecondsRef.current = 0;
   799→    if (ytTimerRef.current) {
   800→      clearInterval(ytTimerRef.current);
   801→      ytTimerRef.current = null;
   802→    }
   803→    // ─── Reset assessment state ───
   804→    setActiveQuestion(null);
   805→    setSelectedAnswer(null);
   806→    setAnsweredMap(new Map());
   807→    setShowFeedback(false);
   808→    setAssessmentResult(null);
   809→    setSubmittingAssessment(false);
   810→    if (feedbackTimerRef.current) {
   811→      clearTimeout(feedbackTimerRef.current);
   812→      feedbackTimerRef.current = null;
   813→    }
   814→  }, [video.id]);
   815→
   816→  // ─── Missed-question catch-up ───
   817→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   818→  // the student answers a question. If the playhead has already moved past an
   819→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   820→  // the questions fetch landed late), pause and surface that question now.
   821→  useEffect(() => {
   822→    if (!questionsActive || !questionsData) return;
   823→    if (activeQuestion || assessmentResult || showFeedback || submittingAssessment) return;
   824→    const el = videoRef.current;
   825→    if (!el) return;
   826→    const cur = el.currentTime;
   827→    const missed = questionsData.questions.find(
   828→      (q) => !answeredMap.has(q.id) && cur >= q.timestamp - 0.5
   829→    );
   830→    if (missed && !el.paused) {
   831→      try {
   832→        el.pause();
   833→        lastTimeRef.current = el.currentTime;
   834→      } catch {}
   835→      setActiveQuestion(missed);
   836→      setSelectedAnswer(null);
   837→    }
   838→  }, [
   839→    questionsActive,
   840→    questionsData,
   841→    activeQuestion,
   842→    assessmentResult,
   843→    showFeedback,
   844→    submittingAssessment,
   845→    answeredMap,
   846→  ]);
   847→
   848→  // For HTML5 video: apply resume position once metadata is loaded.
   849→  const handleLoadedMetadata = () => {
   850→    const el = videoRef.current;
   851→    if (!el) return;
   852→    durationRef.current = el.duration || durationRef.current;
   853→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   854→      // resume within (duration - 2) so we don't immediately fire ended
   855→      const target = Math.min(
   856→        video.progress.lastPosition,
   857→        Math.max(0, (el.duration || 0) - 3)
   858→      );
   859→      try {
   860→        el.currentTime = target;
   861→      } catch {}
   862→      lastTimeRef.current = target;
   863→      resumeAppliedRef.current = true;
   864→    }
   865→  };
   866→
   867→  // Core: timeupdate handler — accumulate real watch time.
   868→  const handleTimeUpdate = () => {
   869→    const el = videoRef.current;
   870→    if (!el) return;
   871→    const cur = el.currentTime;
   872→    const last = lastTimeRef.current;
   873→
   874→    // ─── Question trigger check ───
   875→    // Pause and surface a popup the moment the playhead crosses an unanswered
   876→    // question's timestamp going forward. We require BOTH:
   877→    //   - cur >= q.timestamp    (we're at or past it)
   878→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   879→    //                            this is a forward crossing, not a re-tick while
   880→    //                            paused, and not a backward seek landing past it)
   881→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   882→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   883→    if (
   884→      questionsActive &&
   885→      !activeQuestion &&
   886→      !assessmentResult &&
   887→      !showFeedback &&
   888→      !submittingAssessment &&
   889→      questionsData
   890→    ) {
   891→      for (const q of questionsData.questions) {
   892→        if (answeredMap.has(q.id)) continue;
   893→        if (cur >= q.timestamp && last < q.timestamp) {
   894→          try {
   895→            el.pause();
   896→            lastTimeRef.current = cur;
   897→          } catch {}
   898→          setActiveQuestion(q);
   899→          setSelectedAnswer(null);
   900→          return; // do NOT accumulate watch time for this tick
   901→        }
   902→      }
   903→    }
   904→
   905→    // Watch time tracking — only counts normal forward playback deltas.
   906→    const delta = cur - last;
   907→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   908→    // (large positive deltas) and rewinds (negative deltas).
   909→    if (delta > 0 && delta <= 1.5) {
   910→      watchTimeRef.current += delta;
   911→    }
   912→    // Update lastTime to current — even on seek, so next delta is small.
   913→    lastTimeRef.current = cur;
   914→
   915→    // Throttled server POST every 5 seconds
   916→    const now = Date.now();
   917→    if (now - lastPostRef.current > 5000) {
   918→      lastPostRef.current = now;
   919→      void postProgress(false);
   920→    }
   921→  };
   922→
   923→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   924→  // Also catches forward-seek-past-question: if the student scrubs the
   925→  // progress bar past an unanswered question, pause and surface that question
   926→  // so they can't skip it.
   927→  const handleSeeked = () => {
   928→    const el = videoRef.current;
   929→    if (!el) return;
   930→    const newTime = el.currentTime;
   931→    lastTimeRef.current = newTime;
   932→
   933→    if (
   934→      questionsActive &&
   935→      !activeQuestion &&
   936→      !assessmentResult &&
   937→      !showFeedback &&
   938→      !submittingAssessment &&
   939→      questionsData
   940→    ) {
   941→      // Find the earliest unanswered question we've now skipped past.
   942→      const skipped = questionsData.questions.find(
   943→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
   944→      );
   945→      if (skipped) {
   946→        try {
   947→          el.pause();
   948→        } catch {}
   949→        setActiveQuestion(skipped);
   950→        setSelectedAnswer(null);
   951→      }
   952→    }
   953→  };
   954→
   955→  // Ended event — submit final progress with completion hint.
   956→  const handleEnded = () => {
   957→    void postProgress(true);
   958→  };
   959→
   960→  const postProgress = useCallback(
   961→    async (isCompletionHint: boolean) => {
   962→      setPosting(true);
   963→      try {
   964→        const res = await apiPost<{
   965→          progress: VideoProgressInfo;
   966→          justCompleted: boolean;
   967→          rejectedCompletion: boolean;
   968→          certificateIssued: boolean;
   969→          certificateId: string | null;
   970→        }>("/api/student/progress", {
   971→          videoId: video.id,
   972→          lastPosition: videoRef.current?.currentTime || 0,
   973→          watchTimeSeconds: watchTimeRef.current,
   974→          duration: durationRef.current,
   975→          isCompleted: isCompletionHint,
   976→        });
   977→        setLocalCompleted(res.progress.isCompleted);
   978→        onProgress({
   979→          videoId: video.id,
   980→          progress: res.progress,
   981→          justCompleted: res.justCompleted,
   982→        });
   983→        if (isCompletionHint && res.rejectedCompletion) {
   984→          toast({
   985→            title: "Completion not recorded",
   986→            description:
   987→              "You need to watch at least 85% of the video before it's marked complete.",
   988→            variant: "destructive",
   989→          });
   990→        }
   991→        // ─── Certificate auto-issued upon course completion ───
   992→        if (res.certificateIssued && res.certificateId) {
   993→          toast({
   994→            title: "🎉 Certificate Issued!",
   995→            description:
   996→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   997→          });
   998→        }
   999→      } catch {
  1000→        // apiPost already surfaces errors
  1001→      } finally {
  1002→        setPosting(false);
  1003→      }
  1004→    },
  1005→    [video.id, onProgress]
  1006→  );
  1007→
  1008→  // Manual "Mark complete" for YouTube videos
  1009→  const markYtComplete = async () => {
  1010→    setPosting(true);
  1011→    try {
  1012→      const res = await apiPost<{
  1013→        progress: VideoProgressInfo;
  1014→        justCompleted: boolean;
  1015→        rejectedCompletion: boolean;
  1016→        certificateIssued: boolean;
  1017→        certificateId: string | null;
  1018→      }>("/api/student/progress", {
  1019→        videoId: video.id,
  1020→        lastPosition: durationRef.current,
  1021→        watchTimeSeconds: ytWatchSecondsRef.current,
  1022→        duration: durationRef.current,
  1023→        isCompleted: true,
  1024→      });
  1025→      setLocalCompleted(res.progress.isCompleted);
  1026→      onProgress({
  1027→        videoId: video.id,
  1028→        progress: res.progress,
  1029→        justCompleted: res.justCompleted,
  1030→      });
  1031→      if (res.justCompleted) {
  1032→        toast({
  1033→          title: "Lesson completed",
  1034→          description: "Great work! Your progress has been saved.",
  1035→        });
  1036→      }
  1037→      if (res.rejectedCompletion) {
  1038→        toast({
  1039→          title: "Completion not recorded",
  1040→          description:
  1041→            "You need to keep the video playing for at least 85% of its duration.",
  1042→          variant: "destructive",
  1043→        });
  1044→      }
  1045→      // Certificate auto-issued upon course completion
  1046→      if (res.certificateIssued && res.certificateId) {
  1047→        toast({
  1048→          title: "🎉 Certificate Issued!",
  1049→          description:
  1050→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
  1051→        });
  1052→      }
  1053→    } catch {
  1054→    } finally {
  1055→      setPosting(false);
  1056→    }
  1057→  };
  1058→
  1059→  /* ──────────────────────────────────────────────────────────────
  1060→     Video Assessment — submit, retake, dismiss handlers
  1061→     ────────────────────────────────────────────────────────────── */
  1062→
  1063→  // Submit the collected answers to the server. The server evaluates
  1064→  // correctness (correctAnswer never leaves the server) and returns the
  1065→  // score, pass/fail, and per-question correctness for review.
  1066→  const submitAssessment = useCallback(
  1067→    async (answersMap: Map<string, number>) => {
  1068→      if (!questionsData || submittingAssessment) return;
  1069→      setSubmittingAssessment(true);
  1070→      try {
  1071→        const answers = questionsData.questions.map((q) => ({
  1072→          questionId: q.id,
  1073→          selectedAnswer: answersMap.get(q.id) ?? -1,
  1074→        }));
