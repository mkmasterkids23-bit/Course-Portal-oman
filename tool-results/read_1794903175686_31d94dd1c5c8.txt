   361→                            <Trash2 className="h-4 w-4" /> {t("common.delete")}
   362→                          </DropdownMenuItem>

   376→                <span className="font-medium text-foreground">{Math.min(currentPage * PAGE_SIZE, students.length)}</span> {t("common.of")} <span className="font-medium text-foreground">{students.length}</span>
   377→              </p>
   378→              <div className="flex gap-1.5">
   379→                <Button
   380→                  variant="outline"
   381→                  size="sm"
   382→                  className="h-8"
   383→                  disabled={currentPage === 1}
   384→                  onClick={() => setPage((p) => Math.max(1, p - 1))}
   385→                >
   386→                  {t("common.previous")}
   387→                </Button>
   388→                <span className="px-3 py-1 text-xs font-medium text-muted-foreground self-center">
   389→                  {t("common.page")} <span className="text-foreground">{currentPage}</span> / {totalPages}
   390→                </span>
   391→                <Button
   392→                  variant="outline"
   393→                  size="sm"
   394→                  className="h-8"
   395→                  disabled={currentPage >= totalPages}
   396→                  onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   397→                >
   398→                  {t("common.next")}
   399→                </Button>
   400→              </div>
   401→            </div>
   402→          </>
   403→        )}
   404→      </Card>
   405→
   406→      {/* Add Student Dialog */}
   407→      <AddStudentDialog
   408→        open={addOpen}
   409→        onOpenChange={setAddOpen}
   410→        plans={plansData?.plans || []}
   411→        onCreated={() => {
   412→          refetch();
   413→          toast({ title: t("common.success") });
   414→        }}
   415→      />
   416→
   417→      {/* Edit Student Dialog */}
   418→      <EditStudentDialog
   419→        open={editOpen}
   420→        onOpenChange={setEditOpen}
   421→        target={editTarget}
   422→        plans={plansData?.plans || []}
   423→        onSaved={() => {
   424→          refetch();
   425→          toast({ title: t("common.success") });
   426→        }}
   427→      />
   428→
   429→      {/* Student Profile Sheet */}
   430→      <StudentProfileSheet
   431→        studentId={profileId}
   432→        onOpenChange={(open) => {
   433→          if (!open) setProfileId(null);
   434→        }}
   435→        plans={plansData?.plans || []}
   436→        onAssigned={() => refetch()}
   437→      />
   438→
   439→      {/* Delete Confirmation */}
   440→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   441→        <AlertDialogContent>
   442→          <AlertDialogHeader>
   443→            <AlertDialogTitle>{t("common.delete")}</AlertDialogTitle>
   444→            <AlertDialogDescription>
   445→              {t("courses.deleteCourseConfirm")}
   446→            </AlertDialogDescription>
   447→          </AlertDialogHeader>
   448→          <AlertDialogFooter>
   449→            <AlertDialogCancel>{t("common.cancel")}</AlertDialogCancel>
   450→            <AlertDialogAction
   451→              className="bg-destructive text-white hover:bg-destructive/90"
   452→              onClick={async () => {
   453→                if (!deleteTarget) return;
   454→                try {
   455→                  await apiPost(`/api/admin/students/${deleteTarget.id}`, null, {
   456→                    method: "DELETE",
   457→                  });
   458→                  toast({ title: t("common.success") });
   459→                  setDeleteTarget(null);
   460→                  refetch();
   461→                } catch {
   462→                  /* toast shown by apiPost */
   463→                }
   464→              }}
   465→            >
   466→              {t("common.delete")}
   467→            </AlertDialogAction>
   468→          </AlertDialogFooter>
   469→        </AlertDialogContent>
   470→      </AlertDialog>
   471→    </div>
   472→  );
   473→}

   475→// ---------- Add Student Dialog ----------
   476→function AddStudentDialog({
   477→  open,
   478→  onOpenChange,
   479→  plans,
   480→  onCreated,
   481→}: {
   482→  open: boolean;
   483→  onOpenChange: (v: boolean) => void;
   484→  plans: SubscriptionPlanPublic[];
   485→  onCreated: () => void;
   486→}) {
   487→  const { t } = useI18n();
   488→  const [name, setName] = useState("");
   489→  const [email, setEmail] = useState("");
   490→  const [password, setPassword] = useState("");
   491→  const [planId, setPlanId] = useState<string>("NONE");
   492→  const [saving, setSaving] = useState(false);
   493→
   494→  useEffect(() => {
   495→    if (open) {
   496→      setName("");
   497→      setEmail("");
   498→      setPassword("");
   499→      setPlanId("NONE");
   500→    }
   501→  }, [open]);
   502→
   503→  async function submit() {
   504→    if (!name.trim() || !email.trim() || !password) {
   505→      toast({ title: t("common.required"), description: t("common.required"), variant: "destructive" });
   506→      return;
   507→    }
   508→    setSaving(true);
   509→    try {
   510→      await apiPost("/api/admin/students", {
   511→        name: name.trim(),
   512→        email: email.trim(),
   513→        password,
   514→        planId: planId === "NONE" ? undefined : planId,
   515→      });
   516→      onCreated();
   517→      onOpenChange(false);
   518→    } catch {
   519→      /* toast shown by apiPost */
   520→    } finally {
   521→      setSaving(false);
   522→    }
   523→  }
   524→
   525→  return (
   526→    <Dialog open={open} onOpenChange={onOpenChange}>
   527→      <DialogContent className="animate-scale-in w-[95vw] max-w-md max-h-[90vh] overflow-y-auto">
   528→        <DialogHeader>
   529→          <DialogTitle className="font-heading text-lg font-semibold">{t("dashboard.addStudent")}</DialogTitle>
   530→          <DialogDescription>{t("dashboard.addStudent")}</DialogDescription>
   531→        </DialogHeader>
   532→        <div className="space-y-3">
   533→          <div className="space-y-1.5">
   534→            <Label htmlFor="as-name" className="text-xs font-medium text-muted-foreground">{t("auth.fullName")}</Label>
   535→            <Input id="as-name" value={name} onChange={(e) => setName(e.target.value)} placeholder={t("auth.fullName")} className="h-9 rounded-lg" />
   536→          </div>
   537→          <div className="space-y-1.5">
   538→            <Label htmlFor="as-email" className="text-xs font-medium text-muted-foreground">{t("common.email")}</Label>
   539→            <Input id="as-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder={t("common.email")} className="h-9 rounded-lg" />
   540→          </div>
   541→          <div className="space-y-1.5">
   542→            <Label htmlFor="as-pass" className="text-xs font-medium text-muted-foreground">{t("common.password")}</Label>
   543→            <Input id="as-pass" type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder={t("common.password")} className="h-9 rounded-lg" />
   544→          </div>
   545→          <div className="space-y-1.5">
   546→            <Label htmlFor="as-plan" className="text-xs font-medium text-muted-foreground">{t("common.subscriptionPlans")} ({t("common.optional")})</Label>
   547→            <Select value={planId} onValueChange={setPlanId}>
   548→              <SelectTrigger id="as-plan" className="w-full h-9 rounded-lg">
   549→                <SelectValue placeholder={t("common.optional")} />

   573→// ---------- Edit Student Dialog ----------
   574→function EditStudentDialog({
   575→  open,
   576→  onOpenChange,
   577→  target,
   578→  onSaved,
   579→}: {
   580→  open: boolean;
   581→  onOpenChange: (v: boolean) => void;
   582→  target: StudentRow | null;
   583→  plans: SubscriptionPlanPublic[];
   584→  onSaved: () => void;
   585→}) {
   586→  const { t } = useI18n();
   587→  const [name, setName] = useState("");
   588→  const [phone, setPhone] = useState("");
   589→  const [bio, setBio] = useState("");
   590→  const [saving, setSaving] = useState(false);
   591→
   592→  useEffect(() => {
   593→    if (target) {
   594→      setName(target.name);
   595→      setPhone(target.phone || "");
   596→      setBio("");
   597→    }
   598→  }, [target]);
   599→
   600→  async function submit() {
   601→    if (!target) return;
   602→    if (!name.trim()) {
   603→      toast({ title: t("common.name") + " " + t("common.required"), variant: "destructive" });
   604→      return;
   605→    }
   606→    setSaving(true);
   607→    try {
   608→      await apiPost(`/api/admin/students/${target.id}`, {
   609→        name: name.trim(),
   610→        phone: phone.trim() || null,
   611→        bio: bio.trim() || null,
   612→      }, { method: "PUT" });
   613→      onSaved();
   614→      onOpenChange(false);
   615→    } catch {
   616→      /* toast */
   617→    } finally {
   618→      setSaving(false);
   619→    }
   620→  }
   621→
   622→  return (
   623→    <Dialog open={open} onOpenChange={onOpenChange}>
   624→      <DialogContent className="animate-scale-in w-[95vw] max-w-md max-h-[90vh] overflow-y-auto">
   625→        <DialogHeader>
   626→          <DialogTitle className="font-heading text-lg font-semibold">{t("profile.editProfile")}</DialogTitle>
   627→          <DialogDescription>{target?.email}</DialogDescription>
   628→        </DialogHeader>
   629→        <div className="space-y-3">
   630→          <div className="space-y-1.5">
   631→            <Label htmlFor="es-name" className="text-xs font-medium text-muted-foreground">{t("auth.fullName")}</Label>
   632→            <Input id="es-name" value={name} onChange={(e) => setName(e.target.value)} className="h-9 rounded-lg" />
   633→          </div>
   634→          <div className="space-y-1.5">
   635→            <Label htmlFor="es-phone" className="text-xs font-medium text-muted-foreground">{t("profile.phone")}</Label>
   636→            <Input id="es-phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder={t("profile.phone")} className="h-9 rounded-lg" />
   637→          </div>
   638→          <div className="space-y-1.5">
   639→            <Label htmlFor="es-bio" className="text-xs font-medium text-muted-foreground">{t("profile.bio")}</Label>
   640→            <Textarea id="es-bio" value={bio} onChange={(e) => setBio(e.target.value)} placeholder={t("profile.bio")} rows={3} className="rounded-lg" />
   641→          </div>
   642→        </div>
   643→        <DialogFooter>
   644→          <Button variant="outline" className="h-9" onClick={() => onOpenChange(false)}>{t("common.cancel")}</Button>
   645→          <Button disabled={saving} className="h-9" onClick={submit}>
   646→            {saving ? t("common.saving") : t("profile.saveChanges")}
   647→          </Button>
   648→        </DialogFooter>
   649→      </DialogContent>
   650→    </Dialog>
   651→  );
   652→}

   654→// ---------- Student Profile Sheet ----------
   655→function StudentProfileSheet({
   656→  studentId,

   658→  plans,

   668→    studentId ? `/api/admin/students/${studentId}` : null,
   669→    { skip: !studentId }

   684→        title: t("common.success"),

   688→      refetch();

   691→      /* toast */
   692→    } finally {
