   420→   420→interface FormValues {
   421→   421→  name: string;
   422→   422→  email: string;
   423→   423→  password: string;
   424→   424→  description: string;

   429→   429→function UniversityFormDialog({
   430→   430→  open,
   431→   431→  onOpenChange,
   432→   432→  target,
   433→   433→  onSaved,
   434→   434→}: {
   435→   435→  open: boolean;
   436→   436→  onOpenChange: (v: boolean) => void;
   437→   437→  target?: UniversityRow | null;
   438→   438→  onSaved: () => void;
   439→   439→}) {
   440→   440→  const isEdit = !!target;
   441→   441→  const [values, setValues] = useState<FormValues>({
   442→   442→    name: "",
   443→   443→    email: "",
   444→   444→    password: "",
   445→   445→    description: "",
   446→   446→    logoFile: null,
   447→   447→    logoPreview: null,
   448→   448→  });
   449→   449→  const [saving, setSaving] = useState(false);
   450→   450→
   451→   451→  useEffect(() => {
   452→   452→    if (target) {
   453→   453→      setValues({
   454→   454→        name: target.name,
   455→   455→        email: target.email,
   456→   456→        password: "",
   457→   457→        description: target.description || "",
   458→   458→        logoFile: null,
   459→   459→        logoPreview: target.logo,
   460→   460→      });
   461→   461→    } else if (open) {
   462→   462→      setValues({
   463→   463→        name: "",
   464→   464→        email: "",
   465→   465→        password: "",
   466→   466→        description: "",
   467→   467→        logoFile: null,
   468→   468→        logoPreview: null,
   469→   469→      });
   470→   470→    }
   471→   471→  }, [target, open]);
   472→   472→
   473→   473→  function onLogoSelected(file: File | null) {
   474→   474→    if (!file) {
   475→   475→      setValues((v) => ({ ...v, logoFile: null, logoPreview: target?.logo || null }));
   476→   476→      return;
   477→   477→    }
   478→   478→    const previewUrl = URL.createObjectURL(file);
   479→   479→    setValues((v) => ({ ...v, logoFile: file, logoPreview: previewUrl }));
   480→   480→  }
   481→   481→
   482→   482→  async function submit() {
   483→   483→    if (!values.name.trim()) {
   484→   484→      toast({ title: "Name required", variant: "destructive" });
   485→   485→      return;
   486→   486→    }
   487→   487→    if (!values.email.trim()) {
   488→   488→      toast({ title: "Email required", variant: "destructive" });
   489→   489→      return;
   490→   490→    }
   491→   491→    if (!isEdit && !values.password) {
   492→   492→      toast({ title: "Password required", variant: "destructive" });
   493→   493→      return;
   494→   494→    }
   495→   495→    if (values.password && values.password.length < 6) {
   496→   496→      toast({
   497→   497→        title: "Password too short",
