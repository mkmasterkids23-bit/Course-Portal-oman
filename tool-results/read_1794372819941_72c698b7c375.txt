   464→function TopicFormDialog({
   465→  open, onOpenChange, courseId, target, onSaved,
   466→}: {
   467→  open: boolean;
   468→  onOpenChange: (v: boolean) => void;
   469→  courseId: string;
   470→  target?: TopicRow | null;
   471→  onSaved: () => void;
   472→}) {

   482→      setTitle("");
   483→      setDescription("");

   539→function VideoFormDialog({
   540→  open, onOpenChange, course, fixedTopicId, target, onSaved,
   541→}: {
   542→  open: boolean;
   543→  onOpenChange: (v: boolean) => void;
   544→  course: CourseDetail;
   545→  fixedTopicId?: string;
   546→  target?: VideoRow | null;
   547→  onSaved: () => void;
   548→}) {
   549→  const [title, setTitle] = useState("");
   550→  const [description, setDescription] = useState("");
   551→  const [topicId, setTopicId] = useState<string>(fixedTopicId || "");
   552→  const [source, setSource] = useState<"UPLOAD" | "YOUTUBE" | "URL">("UPLOAD");
   553→  const [videoUrl, setVideoUrl] = useState("");
   554→  const [displayOrder, setDisplayOrder] = useState("");
   555→  const [isPublished, setIsPublished] = useState(true);
   556→  const [saving, setSaving] = useState(false);
   557→  const [file, setFile] = useState<File | null>(null);
   558→  const [thumbFile, setThumbFile] = useState<File | null>(null);
   559→
   560→  useEffect(() => {
   561→    if (target) {
   562→      setTitle(target.title);
   563→      setDescription(target.description || "");
   564→      setTopicId(fixedTopicId || "");
   565→      setSource(target.source);
   566→      setVideoUrl(target.source !== "UPLOAD" ? target.videoUrl : "");
   567→      setDisplayOrder(String(target.displayOrder));
   568→      setIsPublished(target.isPublished);
   569→    } else if (open) {
   570→      setTitle("");
   571→      setDescription("");
   572→      setTopicId(fixedTopicId || course.topics[0]?.id || "");
   573→      setSource("UPLOAD");
   574→      setVideoUrl("");
   575→      setDisplayOrder("");
   576→      setIsPublished(true);
   577→    }
   578→    setFile(null);
   579→    setThumbFile(null);
   580→  }, [target, open, fixedTopicId, course.topics]);
   581→
   582→  async function submit() {
   583→    if (!title.trim()) {
   584→      toast({ title: "Title required", variant: "destructive" });
   585→      return;
   586→    }
   587→    if (!topicId) {
   588→      toast({ title: "Select a topic", variant: "destructive" });
   589→      return;
   590→    }
   591→    if (source !== "UPLOAD" && !videoUrl.trim() && !target) {
   592→      toast({ title: "Video URL required", variant: "destructive" });
   593→      return;
   594→    }
   595→    if (source === "UPLOAD" && !file && !target) {
   596→      toast({ title: "Video file required", variant: "destructive" });
   597→      return;
   598→    }
   599→
   600→    const fd = new FormData();
   601→    fd.append("title", title.trim());
   602→    fd.append("description", description.trim());
   603→    fd.append("topicId", topicId);
   604→    fd.append("source", source);
   605→    if (source !== "UPLOAD") fd.append("videoUrl", videoUrl.trim());
   606→    if (file) fd.append("file", file);
   607→    if (thumbFile) fd.append("thumbnail", thumbFile);
   608→    if (displayOrder) fd.append("displayOrder", displayOrder);
   609→    fd.append("isPublished", String(isPublished));
   610→
   611→    setSaving(true);
   612→    try {
   613→      if (target) {
   614→        // PUT with FormData — use a custom fetch since apiUpload is POST-only
   615→        const res = await fetch(`/api/admin/videos/${target.id}`, {
   616→          method: "PUT",
   617→          body: fd,
   618→          credentials: "same-origin",
   619→        });
   620→        const json = await res.json();
   621→        if (!res.ok) throw new Error(json.error || "Failed to update video");
   622→        toast({ title: "Video updated" });
   623→      } else {
   624→        await apiUpload(`/api/admin/courses/${course.id}/videos`, fd);
   625→        toast({ title: "Video added" });
   626→      }
   627→      onSaved();
   628→    } catch (e) {
   629→      if (target && e instanceof Error) {
   630→        toast({ title: "Error", description: e.message, variant: "destructive" });
   631→      }
   632→    } finally {
   633→      setSaving(false);
   634→    }
   635→  }
   636→
   637→  return (
   638→    <Dialog open={open} onOpenChange={onOpenChange}>
   639→      <DialogContent className="sm:max-w-lg">
   640→        <DialogHeader>
   641→          <DialogTitle className="font-heading text-lg font-semibold">{target ? "Edit Video" : "Add Video Lesson"}</DialogTitle>
   642→          <DialogDescription>Upload a video file, embed a YouTube link, or stream from a direct URL.</DialogDescription>
   643→        </DialogHeader>
   644→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1 scrollbar-thin">
   645→          <div className="space-y-1.5">
   646→            <Label htmlFor="v-title" className="text-xs font-medium text-muted-foreground">Title</Label>
   647→            <Input id="v-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Lesson 1: Design fundamentals" className="h-9 rounded-lg" />
   648→          </div>
   649→          <div className="space-y-1.5">
   650→            <Label htmlFor="v-desc" className="text-xs font-medium text-muted-foreground">Description (optional)</Label>
   651→            <Textarea id="v-desc" value={description} onChange={(e) => setDescription(e.target.value)} rows={2} className="rounded-lg" />
   652→          </div>
   653→          <div className="grid grid-cols-2 gap-3">
   654→            <div className="space-y-1.5">
   655→              <Label className="text-xs font-medium text-muted-foreground">Topic</Label>
   656→              <Select value={topicId} onValueChange={setTopicId} disabled={!!fixedTopicId}>
   657→                <SelectTrigger className="w-full h-9 rounded-lg">
   658→                  <SelectValue placeholder="Select topic" />
   659→                </SelectTrigger>
   660→                <SelectContent>
   661→                  {course.topics.map((t) => (
   662→                    <SelectItem key={t.id} value={t.id}>{t.title}</SelectItem>
   663→                  ))}
   664→                </SelectContent>
   665→              </Select>
   666→            </div>
   667→            <div className="space-y-1.5">
   668→              <Label className="text-xs font-medium text-muted-foreground">Source</Label>
   669→              <Select value={source} onValueChange={(v) => setSource(v as "UPLOAD" | "YOUTUBE" | "URL")}>
   670→                <SelectTrigger className="w-full h-9 rounded-lg">
   671→                  <SelectValue />
   672→                </SelectTrigger>
   673→                <SelectContent>
   674→                  <SelectItem value="UPLOAD">Upload File</SelectItem>
   675→                  <SelectItem value="YOUTUBE">YouTube</SelectItem>
   676→                  <SelectItem value="URL">Direct URL</SelectItem>
   677→                </SelectContent>
   678→              </Select>
   679→            </div>
   680→          </div>
   681→
   682→          {source === "UPLOAD" ? (
   683→            <div className="space-y-1.5">
   684→              <Label htmlFor="v-file" className="text-xs font-medium text-muted-foreground">Video File (.mp4, .webm, .mov, .mkv)</Label>

   687→                type="file"
   688→                accept="video/*"
   689→                onChange={(e) => setFile(e.target.files?.[0] || null)}
   690→              />
   691→              {target && !file && (
   692→                <p className="text-xs text-muted-foreground">Leave empty to keep current video file.</p>
   693→              )}
   694→            </div>
   695→          ) : (
   696→            <div className="space-y-1.5">
   697→              <Label htmlFor="v-url" className="text-xs font-medium text-muted-foreground">

   700→              <Input
   701→                id="v-url"
   702→                value={videoUrl}
   703→                onChange={(e) => setVideoUrl(e.target.value)}
   704→                placeholder={source === "YOUTUBE" ? "https://www.youtube.com/watch?v=…" : "https://example.com/video.mp4"}
   705→                className="h-9 rounded-lg"
