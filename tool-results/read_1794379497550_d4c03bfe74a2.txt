   154→export function StudentLearning({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   155→  const { data, loading, error, refetch, setData } = useApiFetch<CourseResponse>(
   156→    `/api/student/courses/${courseId}`
   157→  );
   158→
   159→  // Pick the initial video: last watched, or first available.
   160→  const initialVideoId = (() => {
   161→    if (!data?.course) return null;
   162→    // 1) lastWatchedVideoId is implicitly captured via progress; find the
   163→    //    most recently updated progress entry in the topics tree.
   164→    let candidate: VideoItem | null = null;
   165→    for (const t of data.course.topics) {
   166→      for (const v of t.videos) {
   167→        if (v.progress && !v.progress.isCompleted) {
   168→          if (!candidate || (v.progress.watchTimeSeconds > (candidate.progress?.watchTimeSeconds || 0))) {
   169→            candidate = v;
   170→          }
   171→        }
   172→      }
   173→    }
   174→    if (candidate) return candidate.id;
   175→    // 2) First video of the first topic
   176→    return data.course.topics[0]?.videos[0]?.id ?? null;
   177→  })();
   178→
   179→  const [userSelectedVideoId, setUserSelectedVideoId] = useState<string | null>(null);
   180→  // Reset user selection when courseId changes — "adjust state during render"
   181→  // pattern (avoids setState-in-effect warnings).
   182→  const [prevCourseId, setPrevCourseId] = useState(courseId);
   183→  if (courseId !== prevCourseId) {
   184→    setPrevCourseId(courseId);
   185→    setUserSelectedVideoId(null);
   186→  }
   187→  // Effective active video: explicit user selection overrides the auto-pick
   188→  // (which is the last-watched or first video). When data loads and we have an
   189→  // initial pick, this falls back to it without needing an effect.
   190→  const activeVideoId = userSelectedVideoId ?? initialVideoId;
   191→
   192→  if (loading) {
   193→    return (
   194→      <div className="space-y-5">
   195→        <Skeleton className="h-10 w-32 rounded-lg" />
   196→        <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   197→          <div className="space-y-4">
   198→            <Skeleton className="h-[420px] rounded-xl" />
   199→            <Skeleton className="h-24 rounded-xl" />
   200→          </div>
   201→          <Skeleton className="h-[600px] rounded-xl" />
   202→        </div>
   203→      </div>
   204→    );
   205→  }
   206→
   207→  if (error) {
   208→    return (
   209→      <div className="space-y-4">
   210→        <Button variant="outline" size="sm" onClick={onBack} className="gap-2">
   211→          <ArrowLeft className="h-4 w-4" /> Back to courses
   212→        </Button>
   213→        <EmptyState
   214→          icon={<AlertTriangle className="h-5 w-5" />}
   215→          title="Couldn't open course"
   216→          description={error}
   217→          action={<Button onClick={onBack}>Back to courses</Button>}
   218→        />
   219→      </div>
   220→    );
   221→  }
   222→
   223→  if (!data?.course) {
   224→    return (
   225→      <EmptyState
   226→        icon={<PlayCircle className="h-5 w-5" />}
   227→        title="Course not found"
   228→        description="This course may have been removed or is not accessible."
   229→        action={<Button onClick={onBack}>Back to courses</Button>}
   230→      />
   231→    );
   232→  }
   233→
   234→  const course = data.course;
   235→  const allVideos = course.topics.flatMap((t) => t.videos);
   236→  const activeVideo =
   237→    allVideos.find((v) => v.id === activeVideoId) || allVideos[0] || null;
   238→
   239→  // Find the topic containing the active video
   240→  const activeTopic = course.topics.find((t) =>
   241→    t.videos.some((v) => v.id === activeVideo?.id)
   242→  );
   243→
   244→  // Build a flat ordered list for prev/next navigation
   245→  const flatVideoIds = allVideos.map((v) => v.id);
   246→  const activeIndex = activeVideo ? flatVideoIds.indexOf(activeVideo.id) : -1;
   247→  const prevVideo = activeIndex > 0 ? allVideos[activeIndex - 1] : null;
   248→  const nextVideo = activeIndex >= 0 && activeIndex < allVideos.length - 1 ? allVideos[activeIndex + 1] : null;
   249→
   250→  function handleProgressUpdate(updated: {
   251→    videoId: string;
   252→    progress: VideoProgressInfo;
   253→    justCompleted: boolean;
   254→  }) {
   255→    if (!data) return;
   256→    const newCourse = {
   257→      ...data.course,
   258→      topics: data.course.topics.map((t) => ({
   259→        ...t,
   260→        videos: t.videos.map((v) =>
   261→          v.id === updated.videoId ? { ...v, progress: updated.progress } : v
   262→        ),
   263→      })),
   264→      courseCompletion: (() => {
   265→        // Recompute local snapshot
   266→        const videos = data.course.topics.flatMap((t) => t.videos);
   267→        const completed = videos.filter(
   268→          (v) =>
   269→            v.id === updated.videoId
   270→              ? updated.progress.isCompleted
   271→              : v.progress?.isCompleted
   272→        ).length;
   273→        const total = videos.length;
   274→        const percent = total > 0 ? Math.round((completed / total) * 100) : 0;
   275→        return {
   276→          totalVideos: total,
   277→          completedVideos: completed,
   278→          percent,
   279→          allCompleted: total > 0 && completed === total,
   280→        };
   281→      })(),
   282→    };
   283→    setData({ course: newCourse });
   284→
   285→    if (updated.justCompleted) {
   286→      toast({
   287→        title: "Lesson completed",
   288→        description: "Great work! Your progress has been saved.",
   289→      });
   290→    }
   291→  }
   292→
   293→  async function handleSubmissionSuccess() {
   294→    await refetch();
   295→    toast({
   296→      title: "Assignment submitted",
   297→      description: "Your submission has been received.",
   298→    });
   299→  }
   300→
   301→  return (
   302→    <div className="space-y-5">
   303→      {/* ─── Premium header ─── */}
   304→      <div className="rounded-xl border border-border bg-card p-4 md:p-5 shadow-sm">
   305→        <div className="flex flex-wrap items-center justify-between gap-3">
   306→          <div className="flex items-center gap-3 min-w-0 flex-1">
   307→            <Button variant="outline" size="icon" onClick={onBack} className="shrink-0 h-9 w-9" aria-label="Back to courses">
   308→              <ArrowLeft className="h-4 w-4" />
   309→            </Button>
   310→            <div className="min-w-0 flex-1">
   311→              <div className="flex items-center gap-2 mb-0.5">
   312→                {course.category && (
   313→                  <Badge variant="outline" className="text-[11px] border-border text-muted-foreground">
   314→                    {course.category}
   315→                  </Badge>
   316→                )}
   317→                <Badge
   318→                  className={
   319→                    course.accessType === "FREE"
   320→                      ? "bg-success/15 text-success border-transparent"
   321→                      : "bg-primary text-primary-foreground border-transparent"
   322→                  }
   323→                >
   324→                  {course.accessType === "FREE" ? "Free" : "Paid"}
   325→                </Badge>
   326→              </div>
   327→              <h1 className="font-heading text-lg md:text-xl font-semibold text-foreground truncate">
   328→                {course.title}
   329→              </h1>
   330→            </div>
   331→          </div>
   332→          <div className="flex items-center gap-4 shrink-0">
   333→            <div className="hidden sm:flex flex-col items-end gap-0.5">
   334→              <span className="text-xs font-medium text-foreground">
   335→                {course.courseCompletion.completedVideos}/{course.courseCompletion.totalVideos} lessons
   336→              </span>
   337→              <span className="text-[11px] text-muted-foreground">completed</span>
   338→            </div>
   339→            <div className="flex items-center gap-2.5 min-w-[160px]">
   340→              <Progress
   341→                value={course.courseCompletion.percent}
   342→                className={`h-2 w-32 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   343→              />
   344→              <span className="text-sm font-semibold text-foreground tabular-nums">{course.courseCompletion.percent}%</span>
   345→            </div>
   346→          </div>
   347→        </div>
   348→      </div>
   349→
   350→      {/* ─── Mobile: stacked. Desktop: split. ─── */}
   351→      <div className="grid gap-5 lg:grid-cols-[1fr_360px]">
   352→        {/* Left: player + materials + assignment */}
   353→        <div className="space-y-5 min-w-0">
   354→          {activeVideo ? (
   355→            <VideoPlayer
   356→              key={activeVideo.id}
   357→              video={activeVideo}
   358→              topicTitle={activeTopic?.title || null}
   359→              onProgress={handleProgressUpdate}
   360→              prevVideo={prevVideo}
   361→              nextVideo={nextVideo}
   362→              onNavigateLesson={(id) => setUserSelectedVideoId(id)}
   363→              refetchCourse={refetch}
   364→            />
   365→          ) : (
   366→            <EmptyState
   367→              icon={<VideoIcon className="h-5 w-5" />}
   368→              title="No videos in this course"
   369→              description="This course doesn't have any published video lessons yet."
   370→            />
   371→          )}
   372→
   373→          {/* Topic materials */}
   374→          {activeTopic && activeTopic.materials.length > 0 && (
   375→            <Card className="shadow-sm">
   376→              <CardHeader>
   377→                <CardTitle className="text-base flex items-center gap-2">
   378→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   379→                    <FileText className="h-4 w-4" />
   380→                  </span>
   381→                  Materials · {activeTopic.title}
   382→                </CardTitle>
   383→                <CardDescription>Reference documents for this topic</CardDescription>
   384→              </CardHeader>
   385→              <CardContent>
   386→                <ul className="space-y-2">
   387→                  {activeTopic.materials.map((m) => (
   388→                    <MaterialRow key={m.id} material={m} />
   389→                  ))}
   390→                </ul>
   391→              </CardContent>
   392→            </Card>
   393→          )}
   394→
   395→          {/* Assignment panel — completely hidden until ALL videos are completed.
   396→              Before completion, show a locked placeholder that does NOT reveal
   397→              the assignment title, description, or instructions. */}
   398→          {course.assignment && course.courseCompletion.allCompleted ? (
   399→            <AssignmentPanel
   400→              course={course}
   401→              onSubmitted={handleSubmissionSuccess}
   402→            />
   403→          ) : course.assignment ? (
   404→            <LockedAssignmentCard
   405→              completedVideos={course.courseCompletion.completedVideos}
   406→              totalVideos={course.courseCompletion.totalVideos}
   407→              percent={course.courseCompletion.percent}
   408→            />
   409→          ) : null}
   410→
   411→          {/* All materials (course-level) */}
   412→          {course.materials.length > 0 && (
   413→            <Card className="shadow-sm">
   414→              <CardHeader>
   415→                <CardTitle className="text-base flex items-center gap-2">
   416→                  <span className="flex h-7 w-7 items-center justify-center rounded-lg bg-accent/10 text-accent">
   417→                    <FileText className="h-4 w-4" />
   418→                  </span>
   419→                  All Course Materials
   420→                </CardTitle>
   421→                <CardDescription>
   422→                  {course.materials.length} document{course.materials.length > 1 ? "s" : ""} attached to this course
   423→                </CardDescription>
   424→              </CardHeader>
   425→              <CardContent>
   426→                <ul className="space-y-2">
   427→                  {course.materials.map((m) => (
   428→                    <MaterialRow key={m.id} material={m} />
   429→                  ))}
   430→                </ul>
   431→              </CardContent>
   432→            </Card>
   433→          )}
   434→        </div>
   435→
   436→        {/* Right: course sidebar */}
   437→        <Card className="lg:sticky lg:top-20 lg:self-start lg:max-h-[calc(100vh-6rem)] flex flex-col shadow-sm overflow-hidden">
   438→          <CardHeader className="border-b border-border bg-gradient-to-br from-soft-sage/60 to-card">
   439→            <CardTitle className="text-base flex items-center gap-2">
   440→              <BookOpen className="h-4 w-4 text-accent" />
   441→              Course Lessons
   442→            </CardTitle>
   443→            <CardDescription className="mt-1">
   444→              {course.topics.length} topic{course.topics.length > 1 ? "s" : ""} · {course.courseCompletion.totalVideos} lessons
   445→            </CardDescription>
   446→            <div className="mt-2">
   447→              <div className="flex justify-between text-xs mb-1">
   448→                <span className="text-muted-foreground">Progress</span>
   449→                <span className="font-semibold text-foreground">{course.courseCompletion.percent}%</span>
   450→              </div>
   451→              <Progress
   452→                value={course.courseCompletion.percent}
   453→                className={`h-1.5 bg-muted ${course.courseCompletion.allCompleted ? "[&_[data-slot=progress-indicator]]:bg-success" : "[&_[data-slot=progress-indicator]]:bg-accent"}`}
   454→              />
   455→            </div>
   456→          </CardHeader>
   457→          <CardContent className="p-0 overflow-y-auto max-h-[70vh] lg:max-h-[calc(100vh-15rem)] scrollbar-thin">
   458→            {course.topics.length === 0 ? (
   459→              <p className="text-sm text-muted-foreground p-6 text-center">
   460→                No topics published yet.
   461→              </p>
   462→            ) : (
   463→              <Accordion
   464→                type="multiple"
   465→                defaultValue={[course.topics[0]?.id].filter(Boolean) as string[]}
   466→                className="px-3 py-2"
   467→              >
   468→                {course.topics.map((topic, topicIdx) => {
   469→                  const completedInTopic = topic.videos.filter(
   470→                    (v) => v.progress?.isCompleted
   471→                  ).length;
   472→                  const isActiveTopic = topic.id === activeTopic?.id;
   473→                  const topicComplete = completedInTopic === topic.videos.length && topic.videos.length > 0;
   474→                  return (
   475→                    <AccordionItem key={topic.id} value={topic.id} className="border-b border-border/60 last:border-b-0">
   476→                      <AccordionTrigger className="hover:no-underline px-2 py-3 rounded-lg hover:bg-muted/40 transition-colors">
   477→                        <div className="flex items-center gap-2.5 flex-1 min-w-0 text-left">
   478→                          <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-[11px] font-semibold ${
   479→                            topicComplete
   480→                              ? "bg-success/15 text-success"
   481→                              : isActiveTopic
   482→                                ? "bg-accent/15 text-accent"
   483→                                : "bg-muted text-muted-foreground"
   484→                          }`}>
   485→                            {topicComplete ? <CheckCircle2 className="h-3.5 w-3.5" /> : topicIdx + 1}
   486→                          </span>
   487→                          <div className="flex-1 min-w-0">
   488→                            <p className={`text-sm font-semibold truncate ${isActiveTopic ? "text-accent" : "text-foreground"}`}>
   489→                              {topic.title}
   490→                            </p>
   491→                            <p className="text-[11px] text-muted-foreground mt-0.5">
   492→                              {completedInTopic}/{topic.videos.length} lessons
   493→                            </p>
   494→                          </div>
   495→                        </div>
   496→                      </AccordionTrigger>
   497→                      <AccordionContent className="pb-2 pt-1">
   498→                        <ul className="space-y-0.5 ml-9">
   499→                          {topic.videos.map((video, videoIdx) => {
   500→                            const isActive = video.id === activeVideo?.id;
   501→                            const completed = video.progress?.isCompleted;
   502→                            return (
   503→                              <li key={video.id} className="relative">
   504→                                {/* Vertical connecting line */}
   505→                                {videoIdx < topic.videos.length - 1 && (
   506→                                  <span className="absolute left-3 top-7 bottom-0 w-px bg-border" aria-hidden />
   507→                                )}
   508→                                <button
   509→                                  onClick={() => setUserSelectedVideoId(video.id)}
   510→                                  className={`group flex w-full items-start gap-2.5 rounded-lg px-2 py-2 text-left text-sm transition-all duration-200 ${
   511→                                    isActive
   512→                                      ? "bg-soft-sage text-primary"
   513→                                      : "text-foreground hover:bg-muted/60"
   514→                                  }`}
   515→                                >
   516→                                  {/* Icon: status indicator */}
   517→                                  <span className="relative z-10 mt-0.5 shrink-0 flex h-5 w-5 items-center justify-center rounded-full bg-card ring-1 ring-border">
   518→                                    {completed ? (
   519→                                      <CheckCircle2 className="h-4 w-4 text-success" />
   520→                                    ) : isActive ? (
   521→                                      <PlayCircle className="h-4 w-4 text-primary" />
   522→                                    ) : (
   523→                                      <PlayCircle className="h-4 w-4 text-muted-foreground/70 group-hover:text-foreground" />
   524→                                    )}
   525→                                  </span>
   526→                                  <span className="min-w-0 flex-1">
   527→                                    <span className={`flex items-center gap-1.5 truncate ${isActive ? "font-semibold text-primary" : "font-medium text-foreground group-hover:text-accent"}`}>
   528→                                      <span className="truncate">{video.title}</span>
   529→                                      {video.hasQuestions && (
   530→                                        <Tooltip>
   531→                                          <TooltipTrigger asChild>
   532→                                            <span className={cn(
   533→                                              "inline-flex items-center gap-0.5 rounded px-1 py-0.5 text-[10px] font-semibold shrink-0",
   534→                                              video.assessmentPassed === true
   535→                                                ? "bg-success/15 text-success"
   536→                                                : video.assessmentPassed === false
   537→                                                  ? "bg-destructive/10 text-destructive"
   538→                                                  : "bg-accent/10 text-accent"
   539→                                            )}>
   540→                                              <HelpCircle className="h-3 w-3" />
   541→                                              {video.questionCount}
   542→                                            </span>
   543→                                          </TooltipTrigger>
   544→                                          <TooltipContent side="right">
   545→                                            {video.assessmentPassed === true
   546→                                              ? "Assessment passed"
   547→                                              : video.assessmentPassed === false
   548→                                                ? "Assessment not passed — retake needed"
   549→                                                : `${video.questionCount} question${video.questionCount > 1 ? "s" : ""} in this video`}
   550→                                          </TooltipContent>
   551→                                        </Tooltip>
   552→                                      )}
   553→                                    </span>
   554→                                    {video.duration > 0 && (
   555→                                      <span className="block text-[11px] text-muted-foreground mt-0.5">
   556→                                        {formatVideoDuration(video.duration)}
   557→                                      </span>
   558→                                    )}
   559→                                  </span>
   560→                                </button>
   561→                              </li>
   562→                            );
   563→                          })}
   564→                        </ul>
   565→                      </AccordionContent>
   566→                    </AccordionItem>
   567→                  );
   568→                })}
   569→              </Accordion>
   570→            )}
   571→          </CardContent>
   572→        </Card>
   573→      </div>
   574→    </div>
   575→  );
   576→}
   577→
   578→/* ===================== Video player ===================== */
   579→
   580→function VideoPlayer({
   581→  video,
   582→  topicTitle,
   583→  onProgress,
   584→  prevVideo,
   585→  nextVideo,
   586→  onNavigateLesson,
   587→  refetchCourse,
   588→}: {
   589→  video: VideoItem;
   590→  topicTitle: string | null;
   591→  onProgress: (update: {
   592→    videoId: string;
   593→    progress: VideoProgressInfo;
   594→    justCompleted: boolean;
   595→  }) => void;
   596→  prevVideo: VideoItem | null;
   597→  nextVideo: VideoItem | null;
   598→  onNavigateLesson: (id: string) => void;
   599→  refetchCourse?: () => void;
   600→}) {
   601→  const videoRef = useRef<HTMLVideoElement>(null);
   602→  const lastTimeRef = useRef<number>(0);
   603→  const watchTimeRef = useRef<number>(0);
   604→  const durationRef = useRef<number>(0);
   605→  const lastPostRef = useRef<number>(0);
   606→  const resumeAppliedRef = useRef<boolean>(false);
   607→  const [expanded, setExpanded] = useState(false);
   608→
   609→  // ─── Video Assessment state ───
   610→  // Fetches the questions for THIS video (without correctAnswer — security).
   611→  // The hook is unconditional (rules of hooks); the URL just changes per video.
   612→  const questionsUrl = `/api/student/videos/${video.id}/questions`;
   613→  const {
   614→    data: questionsData,
   615→    loading: questionsLoading,
   616→    refetch: refetchQuestions,
   617→  } = useApiFetch<{
   618→    questions: VideoQuestionData[];
   619→    assessment: VideoAssessmentSummary | null;
   620→  }>(questionsUrl);
   621→
   622→  // Currently-displayed question popup (null = no popup shown).
   623→  const [activeQuestion, setActiveQuestion] = useState<VideoQuestionData | null>(null);
   624→  // Map of questionId → selected option index (answers given this session).
   625→  const [answeredMap, setAnsweredMap] = useState<Map<string, number>>(new Map());
   626→  // Currently-selected option in the active popup (null = nothing selected yet).
   627→  const [selectedAnswer, setSelectedAnswer] = useState<number | null>(null);
   628→  // Brief "Answer saved" confirmation shown after submitting a popup answer.
   629→  const [showFeedback, setShowFeedback] = useState(false);
   630→  // Result returned from POST /assessment — when set, the result overlay shows.
   631→  const [assessmentResult, setAssessmentResult] = useState<AssessmentResult | null>(null);
   632→  // True while the assessment POST is in flight.
   633→  const [submittingAssessment, setSubmittingAssessment] = useState(false);
   634→  // Timer ref for the post-submit feedback delay (cleared on unmount/video change).
   635→  const feedbackTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   636→
   637→  // YouTube completion timer
   638→  const ytStartRef = useRef<number | null>(null);
   639→  const ytTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   640→  const ytWatchSecondsRef = useRef<number>(0);
   641→  const [ytCanMarkComplete, setYtCanMarkComplete] = useState(false);
   642→
   643→  // Local UI state
   644→  const [localCompleted, setLocalCompleted] = useState<boolean>(
   645→    video.progress?.isCompleted || false
   646→  );
   647→  const [posting, setPosting] = useState(false);
   648→
   649→  // Are question popups active for this viewing session?
   650→  // They are NOT active if the student has already passed the assessment —
   651→  // in that case the video plays freely and the "Passed" badge is shown.
   652→  const questionsActive = Boolean(
   653→    questionsData &&
   654→      questionsData.questions.length > 0 &&
   655→      questionsData.assessment?.passed !== true
   656→  );
   657→
   658→  // Reset on video change
   659→  useEffect(() => {
   660→    lastTimeRef.current = 0;
   661→    watchTimeRef.current = video.progress?.watchTimeSeconds || 0;
   662→    durationRef.current = video.duration || video.progress?.duration || 0;
   663→    lastPostRef.current = 0;
   664→    resumeAppliedRef.current = false;
   665→    setLocalCompleted(video.progress?.isCompleted || false);
   666→    setYtCanMarkComplete(false);
   667→    ytWatchSecondsRef.current = 0;
   668→    if (ytTimerRef.current) {
   669→      clearInterval(ytTimerRef.current);
   670→      ytTimerRef.current = null;
   671→    }
   672→    // ─── Reset assessment state ───
   673→    setActiveQuestion(null);
   674→    setSelectedAnswer(null);
   675→    setAnsweredMap(new Map());
   676→    setShowFeedback(false);
   677→    setAssessmentResult(null);
   678→    setSubmittingAssessment(false);
   679→    if (feedbackTimerRef.current) {
   680→      clearTimeout(feedbackTimerRef.current);
   681→      feedbackTimerRef.current = null;
   682→    }
   683→  }, [video.id]);
   684→
   685→  // ─── Missed-question catch-up ───
   686→  // Fires when questions data arrives, when a popup/feedback dismisses, or when
   687→  // the student answers a question. If the playhead has already moved past an
   688→  // unanswered question's timestamp (e.g. they resumed a half-watched video, or
   689→  // the questions fetch landed late), pause and surface that question now.
   690→  useEffect(() => {
   691→    if (!questionsActive || !questionsData) return;
   692→    if (activeQuestion || assessmentResult || showFeedback || submittingAssessment) return;
   693→    const el = videoRef.current;
   694→    if (!el) return;
   695→    const cur = el.currentTime;
   696→    const missed = questionsData.questions.find(
   697→      (q) => !answeredMap.has(q.id) && cur >= q.timestamp - 0.5
   698→    );
   699→    if (missed && !el.paused) {
   700→      try {
   701→        el.pause();
   702→        lastTimeRef.current = el.currentTime;
   703→      } catch {}
   704→      setActiveQuestion(missed);
   705→      setSelectedAnswer(null);
   706→    }
   707→  }, [
   708→    questionsActive,
   709→    questionsData,
   710→    activeQuestion,
   711→    assessmentResult,
   712→    showFeedback,
   713→    submittingAssessment,
   714→    answeredMap,
   715→  ]);
   716→
   717→  // For HTML5 video: apply resume position once metadata is loaded.
   718→  const handleLoadedMetadata = () => {
   719→    const el = videoRef.current;
   720→    if (!el) return;
   721→    durationRef.current = el.duration || durationRef.current;
   722→    if (!resumeAppliedRef.current && video.progress?.lastPosition) {
   723→      // resume within (duration - 2) so we don't immediately fire ended
   724→      const target = Math.min(
   725→        video.progress.lastPosition,
   726→        Math.max(0, (el.duration || 0) - 3)
   727→      );
   728→      try {
   729→        el.currentTime = target;
   730→      } catch {}
   731→      lastTimeRef.current = target;
   732→      resumeAppliedRef.current = true;
   733→    }
   734→  };
   735→
   736→  // Core: timeupdate handler — accumulate real watch time.
   737→  const handleTimeUpdate = () => {
   738→    const el = videoRef.current;
   739→    if (!el) return;
   740→    const cur = el.currentTime;
   741→    const last = lastTimeRef.current;
   742→
   743→    // ─── Question trigger check ───
   744→    // Pause and surface a popup the moment the playhead crosses an unanswered
   745→    // question's timestamp going forward. We require BOTH:
   746→    //   - cur >= q.timestamp    (we're at or past it)
   747→    //   - last < q.timestamp    (we were before it on the previous tick — i.e.
   748→    //                            this is a forward crossing, not a re-tick while
   749→    //                            paused, and not a backward seek landing past it)
   750→    // The 0.5s tolerance from the spec is absorbed by `last < q.timestamp` —
   751→    // timeupdate fires ~4×/s so `last` is at most ~0.25s behind `cur`.
   752→    if (
   753→      questionsActive &&
   754→      !activeQuestion &&
   755→      !assessmentResult &&
   756→      !showFeedback &&
   757→      !submittingAssessment &&
   758→      questionsData
   759→    ) {
   760→      for (const q of questionsData.questions) {
   761→        if (answeredMap.has(q.id)) continue;
   762→        if (cur >= q.timestamp && last < q.timestamp) {
   763→          try {
   764→            el.pause();
   765→            lastTimeRef.current = cur;
   766→          } catch {}
   767→          setActiveQuestion(q);
   768→          setSelectedAnswer(null);
   769→          return; // do NOT accumulate watch time for this tick
   770→        }
   771→      }
   772→    }
   773→
   774→    // Watch time tracking — only counts normal forward playback deltas.
   775→    const delta = cur - last;
   776→    // Only count normal playback deltas (0 to 1.5s). This rejects both seeks
   777→    // (large positive deltas) and rewinds (negative deltas).
   778→    if (delta > 0 && delta <= 1.5) {
   779→      watchTimeRef.current += delta;
   780→    }
   781→    // Update lastTime to current — even on seek, so next delta is small.
   782→    lastTimeRef.current = cur;
   783→
   784→    // Throttled server POST every 5 seconds
   785→    const now = Date.now();
   786→    if (now - lastPostRef.current > 5000) {
   787→      lastPostRef.current = now;
   788→      void postProgress(false);
   789→    }
   790→  };
   791→
   792→  // Seeked event — log; backend rejects completion if watchTime < 85%.
   793→  // Also catches forward-seek-past-question: if the student scrubs the
   794→  // progress bar past an unanswered question, pause and surface that question
   795→  // so they can't skip it.
   796→  const handleSeeked = () => {
   797→    const el = videoRef.current;
   798→    if (!el) return;
   799→    const newTime = el.currentTime;
   800→    lastTimeRef.current = newTime;
   801→
   802→    if (
   803→      questionsActive &&
   804→      !activeQuestion &&
   805→      !assessmentResult &&
   806→      !showFeedback &&
   807→      !submittingAssessment &&
   808→      questionsData
   809→    ) {
   810→      // Find the earliest unanswered question we've now skipped past.
   811→      const skipped = questionsData.questions.find(
   812→        (q) => !answeredMap.has(q.id) && newTime > q.timestamp + 0.5
   813→      );
   814→      if (skipped) {
   815→        try {
   816→          el.pause();
   817→        } catch {}
   818→        setActiveQuestion(skipped);
   819→        setSelectedAnswer(null);
   820→      }
   821→    }
   822→  };
   823→
   824→  // Ended event — submit final progress with completion hint.
   825→  const handleEnded = () => {
   826→    void postProgress(true);
   827→  };
   828→
   829→  const postProgress = useCallback(
   830→    async (isCompletionHint: boolean) => {
   831→      setPosting(true);
   832→      try {
   833→        const res = await apiPost<{
   834→          progress: VideoProgressInfo;
   835→          justCompleted: boolean;
   836→          rejectedCompletion: boolean;
   837→          certificateIssued: boolean;
   838→          certificateId: string | null;
   839→        }>("/api/student/progress", {
   840→          videoId: video.id,
   841→          lastPosition: videoRef.current?.currentTime || 0,
   842→          watchTimeSeconds: watchTimeRef.current,
   843→          duration: durationRef.current,
   844→          isCompleted: isCompletionHint,
   845→        });
   846→        setLocalCompleted(res.progress.isCompleted);
   847→        onProgress({
   848→          videoId: video.id,
   849→          progress: res.progress,
   850→          justCompleted: res.justCompleted,
   851→        });
   852→        if (isCompletionHint && res.rejectedCompletion) {
   853→          toast({
   854→            title: "Completion not recorded",
   855→            description:
   856→              "You need to watch at least 85% of the video before it's marked complete.",
   857→            variant: "destructive",
   858→          });
   859→        }
   860→        // ─── Certificate auto-issued upon course completion ───
   861→        if (res.certificateIssued && res.certificateId) {
   862→          toast({
   863→            title: "🎉 Certificate Issued!",
   864→            description:
   865→              "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   866→          });
   867→        }
   868→      } catch {
   869→        // apiPost already surfaces errors
   870→      } finally {
   871→        setPosting(false);
   872→      }
   873→    },
   874→    [video.id, onProgress]
   875→  );
   876→
   877→  // Manual "Mark complete" for YouTube videos
   878→  const markYtComplete = async () => {
   879→    setPosting(true);
   880→    try {
   881→      const res = await apiPost<{
   882→        progress: VideoProgressInfo;
   883→        justCompleted: boolean;
   884→        rejectedCompletion: boolean;
   885→        certificateIssued: boolean;
   886→        certificateId: string | null;
   887→      }>("/api/student/progress", {
   888→        videoId: video.id,
   889→        lastPosition: durationRef.current,
   890→        watchTimeSeconds: ytWatchSecondsRef.current,
   891→        duration: durationRef.current,
   892→        isCompleted: true,
   893→      });
   894→      setLocalCompleted(res.progress.isCompleted);
   895→      onProgress({
   896→        videoId: video.id,
   897→        progress: res.progress,
   898→        justCompleted: res.justCompleted,
   899→      });
   900→      if (res.justCompleted) {
   901→        toast({
   902→          title: "Lesson completed",
   903→          description: "Great work! Your progress has been saved.",
   904→        });
   905→      }
   906→      if (res.rejectedCompletion) {
   907→        toast({
   908→          title: "Completion not recorded",
   909→          description:
   910→            "You need to keep the video playing for at least 85% of its duration.",
   911→          variant: "destructive",
   912→        });
   913→      }
   914→      // Certificate auto-issued upon course completion
   915→      if (res.certificateIssued && res.certificateId) {
   916→        toast({
   917→          title: "🎉 Certificate Issued!",
   918→          description:
   919→            "Congratulations! You've completed the course and your certificate has been automatically generated. View it in My Certificates.",
   920→        });
   921→      }
   922→    } catch {
   923→    } finally {
   924→      setPosting(false);
   925→    }
   926→  };
   927→
   928→  /* ──────────────────────────────────────────────────────────────
   929→     Video Assessment — submit, retake, dismiss handlers
   930→     ────────────────────────────────────────────────────────────── */
   931→
   932→  // Submit the collected answers to the server. The server evaluates
   933→  // correctness (correctAnswer never leaves the server) and returns the
   934→  // score, pass/fail, and per-question correctness for review.
   935→  const submitAssessment = useCallback(
   936→    async (answersMap: Map<string, number>) => {
   937→      if (!questionsData || submittingAssessment) return;
   938→      setSubmittingAssessment(true);
   939→      try {
   940→        const answers = questionsData.questions.map((q) => ({
   941→          questionId: q.id,
   942→          selectedAnswer: answersMap.get(q.id) ?? -1,
   943→        }));
   944→        const res = await apiPost<{ assessment: AssessmentResult }>(
   945→          `/api/student/videos/${video.id}/assessment`,
   946→          { answers }
   947→        );
   948→        setAssessmentResult(res.assessment);
   949→        // Refresh the questions endpoint so the assessment badge reflects
   950→        // the latest passed/attempts state, and refresh the parent course
   951→        // data so the sidebar + course completion update.
   952→        refetchQuestions();
   953→        refetchCourse?.();
   954→        if (res.assessment.passed) {
   955→          toast({
   956→            title: "Assessment Passed!",
   957→            description: `You scored ${res.assessment.score}% (${res.assessment.correctCount}/${res.assessment.totalQuestions} correct).`,
   958→          });
   959→        }
   960→      } catch {
   961→        // apiPost surfaces the error toast
   962→      } finally {
   963→        setSubmittingAssessment(false);
   964→      }
   965→    },
   966→    [questionsData, submittingAssessment, video.id, refetchQuestions, refetchCourse]
   967→  );
   968→
   969→  // After a popup answer is submitted + the brief confirmation shows, decide
   970→  // what to do next: either surface the next missed question (e.g. if the
   971→  // student had seeked forward past several), or auto-submit if all answered,
   972→  // or simply resume playback.
   973→  const afterAnswerSubmitted = useCallback(
   974→    (newMap: Map<string, number>) => {
   975→      setShowFeedback(false);
   976→      setActiveQuestion(null);
   977→      setSelectedAnswer(null);
   978→
   979→      if (questionsData && newMap.size >= questionsData.questions.length) {
   980→        // All answered → auto-submit the assessment
   981→        void submitAssessment(newMap);
   982→        return;
   983→      }
   984→
   985→      // If the playhead is already past another unanswered question (e.g.
   986→      // after a forward seek that skipped several), surface the next one
   987→      // instead of resuming playback.
   988→      const cur = videoRef.current?.currentTime || 0;
   989→      if (questionsData) {
   990→        const nextQ = questionsData.questions.find(
   991→          (q) => !newMap.has(q.id) && cur >= q.timestamp - 0.5
   992→        );
   993→        if (nextQ) {
   994→          setActiveQuestion(nextQ);
   995→          setSelectedAnswer(null);
   996→          return;
   997→        }
   998→      }
   999→
  1000→      // Otherwise resume normal playback
  1001→      try {
  1002→        const el = videoRef.current;
  1003→        if (el && el.paused) {
  1004→          void el.play().catch(() => {});
  1005→        }
  1006→      } catch {}
  1007→    },
  1008→    [questionsData, submitAssessment]
  1009→  );
  1010→
  1011→  // Submit the currently-selected answer for the active popup question.
  1012→  // Shows a brief "Answer saved" confirmation, then advances via
  1013→  // afterAnswerSubmitted.
  1014→  const handleSubmitAnswer = useCallback(() => {
  1015→    if (!activeQuestion || selectedAnswer === null) return;
  1016→    const qid = activeQuestion.id;
  1017→    const newMap = new Map(answeredMap);
  1018→    newMap.set(qid, selectedAnswer);
  1019→    setAnsweredMap(newMap);
  1020→    setShowFeedback(true);
  1021→
  1022→    if (feedbackTimerRef.current) clearTimeout(feedbackTimerRef.current);
  1023→    feedbackTimerRef.current = setTimeout(() => {
  1024→      feedbackTimerRef.current = null;
  1025→      afterAnswerSubmitted(newMap);
  1026→    }, 1200);
  1027→  }, [activeQuestion, selectedAnswer, answeredMap, afterAnswerSubmitted]);
  1028→
  1029→  // Reset all in-session assessment state and let the student try again.
  1030→  // Seeks back to ~1s before the first question so popups re-trigger as the
  1031→  // video plays forward.
  1032→  const handleRetake = useCallback(() => {
  1033→    setAssessmentResult(null);
  1034→    setActiveQuestion(null);
  1035→    setSelectedAnswer(null);
  1036→    setAnsweredMap(new Map());
  1037→    setShowFeedback(false);
  1038→    setSubmittingAssessment(false);
  1039→    if (feedbackTimerRef.current) {
  1040→      clearTimeout(feedbackTimerRef.current);
  1041→      feedbackTimerRef.current = null;
  1042→    }
  1043→    const el = videoRef.current;
  1044→    if (el && questionsData && questionsData.questions.length > 0) {
  1045→      const firstTs = questionsData.questions[0].timestamp;
  1046→      const target = Math.max(0, firstTs - 1);
  1047→      try {
  1048→        el.currentTime = target;
  1049→        lastTimeRef.current = target;
  1050→        void el.play().catch(() => {});
  1051→      } catch {}
  1052→    }
  1053→  }, [questionsData]);
  1054→
  1055→  // Dismiss the result overlay after a passing attempt — just resume the video.
  1056→  const handleDismissResult = useCallback(() => {
  1057→    setAssessmentResult(null);
  1058→    try {
  1059→      const el = videoRef.current;
  1060→      if (el && el.paused) {
  1061→        void el.play().catch(() => {});
  1062→      }
  1063→    } catch {}
  1064→  }, []);
  1065→
  1066→  // Cleanup the feedback timer on unmount so it can't fire on stale state.
  1067→  useEffect(() => {
  1068→    return () => {
  1069→      if (feedbackTimerRef.current) {
  1070→        clearTimeout(feedbackTimerRef.current);
  1071→        feedbackTimerRef.current = null;
  1072→      }
  1073→    };
  1074→  }, []);
  1075→
  1076→  /* YouTube handling */
  1077→  const ytEmbedUrl = useMemo(() => toYouTubeEmbed(video.videoUrl), [video.videoUrl]);
  1078→  const isYouTube = video.source === "YOUTUBE" && ytEmbedUrl !== null;
  1079→
  1080→  // YouTube timer: accumulate watch time + enable mark-complete after 85%.
  1081→  useEffect(() => {
  1082→    if (!isYouTube) return;
  1083→    if (ytTimerRef.current) clearInterval(ytTimerRef.current);
  1084→    ytStartRef.current = Date.now();
  1085→
  1086→    ytTimerRef.current = setInterval(() => {
  1087→      ytWatchSecondsRef.current = (Date.now() - (ytStartRef.current || Date.now())) / 1000;
  1088→      const dur = durationRef.current;
  1089→      if (dur > 0 && ytWatchSecondsRef.current >= dur * 0.85) {
  1090→        setYtCanMarkComplete(true);
  1091→      }
  1092→    }, 1000);
  1093→
  1094→    return () => {
  1095→      if (ytTimerRef.current) {
  1096→        clearInterval(ytTimerRef.current);
  1097→        ytTimerRef.current = null;
  1098→      }
  1099→    };
  1100→  }, [isYouTube, video.id]);
  1101→
  1102→  // Pre-compute the assessment status badge + overlay JSX so both the compact
  1103→  // and expanded views can render the same overlays without duplication.
  1104→  const currentAssessment: VideoAssessmentSummary | null =
  1105→    assessmentResult
  1106→      ? {
  1107→          score: assessmentResult.score,
  1108→          passed: assessmentResult.passed,
  1109→          attempts: assessmentResult.attempts,
  1110→        }
  1111→      : questionsData?.assessment ?? null;
  1112→
  1113→  const hasQuestions = Boolean(questionsData && questionsData.questions.length > 0);
  1114→
  1115→  const assessmentStatusBadge = hasQuestions ? (
  1116→    currentAssessment?.passed === true ? (
  1117→      <Badge className="bg-success/15 text-success border-transparent">
  1118→        <CheckCircle2 className="h-3 w-3" /> Assessment Passed ✓
  1119→      </Badge>
  1120→    ) : currentAssessment && currentAssessment.attempts > 0 ? (
  1121→      <Badge className="bg-destructive/10 text-destructive border-transparent">
  1122→        <AlertTriangle className="h-3 w-3" /> Assessment Failed — Retake Needed
  1123→      </Badge>
  1124→    ) : (
  1125→      <Badge className="bg-warning/15 text-warning border-transparent">
  1126→        <ClipboardList className="h-3 w-3" /> Assessment Required
  1127→      </Badge>
  1128→    )

  1920→// Local helper for YouTube URL parsing
  1921→function toYouTubeEmbed(url: string): string | null {
  1922→  try {
  1923→    const u = new URL(url);
  1924→    if (u.hostname === "youtu.be") {
  1925→      const id = u.pathname.slice(1);
  1926→      return `https://www.youtube.com/embed/${id}?rel=0`;
  1927→    }
  1928→    if (u.hostname.includes("youtube.com")) {
  1929→      const id = u.searchParams.get("v");
  1930→      if (id) return `https://www.youtube.com/embed/${id}?rel=0`;
  1931→      // /embed/ID or /shorts/ID
  1932→      const m = u.pathname.match(/\/(embed|shorts)\/([\w-]+)/);
  1933→      if (m) return `https://www.youtube.com/embed/${m[2]}?rel=0`;
  1934→    }
  1935→    return null;
  1936→  } catch {
  1937→    return null;
  1938→  }
  1939→}

  1941→function formatVideoDuration(seconds: number): string {
  1942→  if (seconds <= 0) return "";
  1943→  const m = Math.floor(seconds / 60);
  1944→  const s = Math.floor(seconds % 60);
  1945→  if (m === 0) return `${s}s`;
  1946→  return `${m}:${s.toString().padStart(2, "0")}`;
  1947→}
