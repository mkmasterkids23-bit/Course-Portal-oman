    74→export function AdminCourses() {
    75→  const query = useAppStore((s) => s.query);
    76→  const navigate = useAppStore((s) => s.navigate);
    77→  const [search, setSearch] = useState("");
    78→  const [category, setCategory] = useState("");
    79→  const [published, setPublished] = useState<"ALL" | "true" | "false">("ALL");
    80→  const [addOpen, setAddOpen] = useState(false);
    81→  const [editTarget, setEditTarget] = useState<CourseRow | null>(null);
    82→  const [deleteTarget, setDeleteTarget] = useState<CourseRow | null>(null);
    83→
    84→  const { data: plansData } = useApiFetch<PlansResponse>("/api/public/plans");
    85→
    86→  const qs = useMemo(() => {
    87→    const p = new URLSearchParams();
    88→    if (search.trim()) p.set("q", search.trim());
    89→    if (category) p.set("category", category);
    90→    if (published !== "ALL") p.set("published", published);
    91→    const s = p.toString();
    92→    return s ? `?${s}` : "";
    93→  }, [search, category, published]);
    94→
    95→  const { data, loading, error, refetch } = useApiFetch<CoursesResponse>(
    96→    qs ? `/api/admin/courses${qs}` : "/api/admin/courses"
    97→  );
    98→
    99→  // Auto-open Add dialog when query.action=new — React 19 "store previous value" pattern
   100→  const [lastAction, setLastAction] = useState<string | undefined>(undefined);
   101→  if (query.action !== lastAction) {
   102→    setLastAction(query.action);
   103→    if (query.action === "new") setAddOpen(true);
   104→  }
   105→
   106→  const courses = data?.courses || [];
   107→
   108→  // Derive unique categories for filter dropdown
   109→  const categories = useMemo(() => {
   110→    const set = new Set<string>();
   111→    courses.forEach((c) => { if (c.category) set.add(c.category); });
   112→    return Array.from(set).sort();
   113→  }, [courses]);
   114→
   115→  async function togglePublish(course: CourseRow) {
   116→    try {
   117→      await apiPost(`/api/admin/courses/${course.id}/publish`, {
   118→        isPublished: !course.isPublished,
   119→      }, { method: "PATCH" });
   120→      toast({
   121→        title: course.isPublished ? "Course unpublished" : "Course published",
   122→        description: course.title,
   123→      });
   124→      refetch();
   125→    } catch {
   126→      /* toast */
   127→    }
   128→  }
   129→
   130→  return (
   131→    <div className="space-y-4">
   132→      <Card className="border-border bg-card p-4 shadow-sm">
   133→        <div className="flex flex-col gap-3 md:flex-row md:items-center md:justify-between">
   134→          <div className="flex flex-1 flex-col gap-3 sm:flex-row sm:items-center">
   135→            <div className="relative flex-1">
   136→              <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   137→              <Input
   138→                value={search}
   139→                onChange={(e) => setSearch(e.target.value)}
   140→                placeholder="Search courses…"
   141→                className="h-9 rounded-lg pl-9 focus-visible:ring-2 focus-visible:ring-ring"
   142→                aria-label="Search courses"
   143→              />
   144→            </div>
   145→            <Select value={category || "ALL"} onValueChange={(v) => setCategory(v === "ALL" ? "" : v)}>
   146→              <SelectTrigger className="h-9 w-full rounded-lg sm:w-[160px]" aria-label="Filter by category">
   147→                <SelectValue placeholder="All categories" />
   148→              </SelectTrigger>
   149→              <SelectContent>
   150→                <SelectItem value="ALL">All categories</SelectItem>
   151→                {categories.map((c) => (
   152→                  <SelectItem key={c} value={c}>{c}</SelectItem>
   153→                ))}
   154→              </SelectContent>
   155→            </Select>
   156→            <Select
   157→              value={published}
   158→              onValueChange={(v) => setPublished(v as "ALL" | "true" | "false")}
   159→            >
   160→              <SelectTrigger className="h-9 w-full rounded-lg sm:w-[140px]" aria-label="Filter by published state">
   161→                <SelectValue placeholder="All statuses" />
   162→              </SelectTrigger>
   163→              <SelectContent>
   164→                <SelectItem value="ALL">All statuses</SelectItem>
   165→                <SelectItem value="true">Published</SelectItem>
   166→                <SelectItem value="false">Drafts</SelectItem>
   167→              </SelectContent>
   168→            </Select>
   169→          </div>
   170→          <Button className="h-9 gap-2" onClick={() => setAddOpen(true)}>
   171→            <Plus className="h-4 w-4" /> Create Course
   172→          </Button>
   173→        </div>
   174→      </Card>
   175→
   176→      {/* Course count summary */}
   177→      {!loading && !error && courses.length > 0 && (
   178→        <div className="flex items-center justify-between">
   179→          <p className="text-sm text-muted-foreground">
   180→            Showing <span className="font-medium text-foreground">{courses.length}</span> course{courses.length !== 1 ? "s" : ""}
   181→          </p>
   182→        </div>
   183→      )}
   184→
   185→      {loading ? (
   186→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   187→          {Array.from({ length: 6 }).map((_, i) => (
   188→            <Card key={i} className="overflow-hidden border-border p-0">
   189→              <Skeleton className="h-36 w-full rounded-none" />
   190→              <div className="p-5 space-y-3">
   191→                <Skeleton className="h-4 w-1/3" />
   192→                <Skeleton className="h-5 w-3/4" />
   193→                <Skeleton className="h-4 w-full" />
   194→                <div className="flex gap-2 pt-2">
   195→                  <Skeleton className="h-8 w-20" />
   196→                  <Skeleton className="h-8 w-20" />
   197→                </div>
   198→              </div>
   199→            </Card>
   200→          ))}
   201→        </div>
   202→      ) : error ? (
   203→        <EmptyState
   204→          className="m-4"
   205→          icon={<BookOpen className="h-5 w-5" />}
   206→          title="Couldn't load courses"
   207→          description={error}
   208→        />
   209→      ) : courses.length === 0 ? (
   210→        <EmptyState
   211→          className="m-4"
   212→          icon={<BookOpen className="h-5 w-5" />}
   213→          title="No courses yet"
   214→          description="Create your first course to start adding videos, materials and assignments."
   215→          action={<Button onClick={() => setAddOpen(true)} className="gap-2"><Plus className="h-4 w-4" /> Create Course</Button>}
   216→        />
   217→      ) : (
   218→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   219→          {courses.map((c) => (
   220→            <CourseCard
   221→              key={c.id}
   222→              course={c}
   223→              onManage={() => navigate(["admin", "courses", c.id])}
   224→              onEdit={() => setEditTarget(c)}
   225→              onTogglePublish={() => togglePublish(c)}
   226→              onDelete={() => setDeleteTarget(c)}
   227→            />
   228→          ))}
   229→        </div>
   230→      )}
   231→
   232→      <CourseFormDialog
   233→        open={addOpen}
   234→        onOpenChange={setAddOpen}
   235→        plans={plansData?.plans || []}
   236→        onSaved={(id) => {
   237→          if (id) {
   238→            toast({ title: "Course created", description: "Taking you to the course editor." });
   239→            setAddOpen(false);
   240→            navigate(["admin", "courses", id]);
   241→          } else {
   242→            refetch();
   243→          }
   244→        }}
   245→      />
   246→
   247→      <CourseFormDialog
   248→        open={!!editTarget}
   249→        onOpenChange={(v) => !v && setEditTarget(null)}
   250→        target={editTarget}
   251→        plans={plansData?.plans || []}
   252→        onSaved={() => {
   253→          refetch();
   254→          setEditTarget(null);
   255→          toast({ title: "Course updated" });
   256→        }}
   257→      />
   258→
   259→      <AlertDialog open={!!deleteTarget} onOpenChange={(v) => !v && setDeleteTarget(null)}>
   260→        <AlertDialogContent>
   261→          <AlertDialogHeader>
   262→            <AlertDialogTitle>Delete course?</AlertDialogTitle>
   263→            <AlertDialogDescription>
   264→              This will permanently delete <strong>{deleteTarget?.title}</strong> along with all its
   265→              topics, videos, materials, assignments and submissions. This cannot be undone.
   266→            </AlertDialogDescription>
   267→          </AlertDialogHeader>
   268→          <AlertDialogFooter>
   269→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   270→            <AlertDialogAction
   271→              className="bg-destructive text-white hover:bg-destructive/90"
   272→              onClick={async () => {
   273→                if (!deleteTarget) return;
   274→                try {
   275→                  await apiPost(`/api/admin/courses/${deleteTarget.id}`, null, { method: "DELETE" });
   276→                  toast({ title: "Course deleted", description: deleteTarget.title });
   277→                  setDeleteTarget(null);
   278→                  refetch();
   279→                } catch { /* toast */ }
   280→              }}
   281→            >
   282→              Delete
   283→            </AlertDialogAction>
   284→          </AlertDialogFooter>
   285→        </AlertDialogContent>
   286→      </AlertDialog>
   287→    </div>
   288→  );
   289→}

   291→interface CourseFormValues {
   292→  title: string;
   293→  description: string;
   294→  category: string;
   295→  accessType: "FREE" | "PAID";
   296→  isPublished: boolean;
   297→  thumbnail: string;
   298→  planIds: string[];
   299→}
   300→
   301→function CourseFormDialog({
   302→  open,
   303→  onOpenChange,
   304→  target,
   305→  plans,
   306→  onSaved,
   307→}: {
   308→  open: boolean;
   309→  onOpenChange: (v: boolean) => void;
   310→  target?: CourseRow | null;
   311→  plans: SubscriptionPlanPublic[];
   312→  onSaved: (createdId?: string) => void;
   313→}) {
   314→  const isEdit = !!target;
   315→  const [values, setValues] = useState<CourseFormValues>({
   316→    title: "",
   317→    description: "",
   318→    category: "",
   319→    accessType: "FREE",
   320→    isPublished: false,
   321→    thumbnail: "",
   322→    planIds: [],
   323→  });
   324→  const [saving, setSaving] = useState(false);
   325→  const [uploadingThumb, setUploadingThumb] = useState(false);
   326→
   327→  useEffect(() => {
   328→    if (target) {
   329→      setValues({
   330→        title: target.title,
   331→        description: target.description || "",
   332→        category: target.category || "",
   333→        accessType: target.accessType,
   334→        isPublished: target.isPublished,
   335→        thumbnail: target.thumbnail || "",
   336→        planIds: [],
   337→      });
   338→    } else if (open) {
   339→      setValues({
   340→        title: "",
   341→        description: "",
   342→        category: "",
   343→        accessType: "FREE",
   344→        isPublished: false,
   345→        thumbnail: "",
   346→        planIds: [],
   347→      });
   348→    }
   349→  }, [target, open]);
   350→
   351→  function togglePlan(id: string) {
   352→    setValues((v) => ({
   353→      ...v,
   354→      planIds: v.planIds.includes(id)
   355→        ? v.planIds.filter((p) => p !== id)
   356→        : [...v.planIds, id],
   357→    }));
   358→  }
   359→
   360→  async function uploadThumbnail(file: File) {
   361→    if (!target) {
   362→      // No course yet — defer; just keep file URL empty and warn user
   363→      toast({
   364→        title: "Save the course first",
   365→        description: "Thumbnail upload is available after the course is created.",
   366→        variant: "default",
   367→      });
   368→      return;
   369→    }
   370→    setUploadingThumb(true);
   371→    try {
   372→      const fd = new FormData();
   373→      fd.append("file", file);
   374→      const res = await apiUpload<{ thumbnail: string }>(`/api/admin/courses/${target.id}/thumbnail`, fd);
   375→      setValues((v) => ({ ...v, thumbnail: res.thumbnail }));
   376→      toast({ title: "Thumbnail uploaded" });
   377→    } catch {
   378→      /* toast */
   379→    } finally {
   380→      setUploadingThumb(false);
   381→    }
   382→  }
   383→
   384→  async function submit() {
   385→    if (!values.title.trim()) {
   386→      toast({ title: "Title required", variant: "destructive" });
   387→      return;
   388→    }
   389→    if (values.accessType === "PAID" && values.planIds.length === 0) {
   390→      toast({
   391→        title: "Select at least one plan",
   392→        description: "Paid courses must be assigned to one or more subscription plans.",
   393→        variant: "destructive",
   394→      });
   395→      return;
   396→    }
   397→
   398→    setSaving(true);
   399→    try {
   400→      if (isEdit && target) {
   401→        await apiPost(`/api/admin/courses/${target.id}`, {
   402→          title: values.title.trim(),
   403→          description: values.description.trim() || null,
   404→          category: values.category.trim() || null,
   405→          accessType: values.accessType,
   406→          isPublished: values.isPublished,
   407→          thumbnail: values.thumbnail || null,
   408→          ...(values.accessType === "PAID" ? { planIds: values.planIds } : {}),
   409→        }, { method: "PUT" });
   410→        onSaved();
   411→      } else {
   412→        const res = await apiPost<{ course: { id: string } }>("/api/admin/courses", {
   413→          title: values.title.trim(),
   414→          description: values.description.trim() || null,
   415→          category: values.category.trim() || null,
   416→          accessType: values.accessType,
   417→          isPublished: values.isPublished,
   418→          thumbnail: values.thumbnail || null,
   419→          planIds: values.planIds,
   420→        });
   421→        onSaved(res.course.id);
   422→      }
   423→    } catch {
   424→      /* toast */
   425→    } finally {
   426→      setSaving(false);
   427→    }
   428→  }
   429→
   430→  return (
   431→    <Dialog open={open} onOpenChange={onOpenChange}>
   432→      <DialogContent className="sm:max-w-lg">
   433→        <DialogHeader>
   434→          <DialogTitle className="font-heading text-lg font-semibold">{isEdit ? "Edit Course" : "Create New Course"}</DialogTitle>
   435→          <DialogDescription>
   436→            {isEdit
   437→              ? "Update course details. Plan assignment and content can be edited in the course manager."
   438→              : "Set up the basics. You'll be able to add topics, videos and assignments next."}
   439→          </DialogDescription>
   440→        </DialogHeader>
   441→        <div className="space-y-3 max-h-[60vh] overflow-y-auto pr-1">
   442→          <div className="space-y-1.5">
   443→            <Label htmlFor="cf-title" className="text-xs font-medium text-muted-foreground">Title</Label>
   444→            <Input id="cf-title" value={values.title} onChange={(e) => setValues({ ...values, title: e.target.value })} placeholder="e.g. Digital Marketing Fundamentals" className="h-9 rounded-lg" />
   445→          </div>
   446→          <div className="space-y-1.5">
   447→            <Label htmlFor="cf-desc" className="text-xs font-medium text-muted-foreground">Description</Label>
   448→            <Textarea id="cf-desc" value={values.description} onChange={(e) => setValues({ ...values, description: e.target.value })} placeholder="Short summary shown on course cards" rows={3} className="rounded-lg" />
   449→          </div>
   450→          <div className="grid grid-cols-2 gap-3">
   451→            <div className="space-y-1.5">
   452→              <Label htmlFor="cf-cat" className="text-xs font-medium text-muted-foreground">Category</Label>
   453→              <Input id="cf-cat" value={values.category} onChange={(e) => setValues({ ...values, category: e.target.value })} placeholder="e.g. Marketing" className="h-9 rounded-lg" />
   454→            </div>
   455→            <div className="space-y-1.5">
   456→              <Label className="text-xs font-medium text-muted-foreground">Access Type</Label>
   457→              <Select value={values.accessType} onValueChange={(v) => setValues({ ...values, accessType: v as "FREE" | "PAID" })}>
   458→                <SelectTrigger className="w-full h-9 rounded-lg">
   459→                  <SelectValue />
   460→                </SelectTrigger>
   461→                <SelectContent>
   462→                  <SelectItem value="FREE">Free — Open to all</SelectItem>
   463→                  <SelectItem value="PAID">Paid — Plan-gated</SelectItem>
   464→                </SelectContent>
   465→              </Select>
   466→            </div>
   467→          </div>
   468→
   469→          {values.accessType === "PAID" && (
   470→            <div className="space-y-2 rounded-lg border border-border p-3 bg-muted/30">
   471→              <Label className="text-[11px] uppercase tracking-wider text-muted-foreground">Required Plans</Label>
   472→              <p className="text-xs text-muted-foreground -mt-1">
   473→                Students with any of these plans can access this course.
   474→              </p>
   475→              {plans.length === 0 ? (
   476→                <p className="text-xs text-destructive">No subscription plans exist yet. Create one first.</p>
   477→              ) : (
   478→                <div className="space-y-2 max-h-32 overflow-y-auto scrollbar-thin">
   479→                  {plans.map((p) => (
   480→                    <label key={p.id} className="flex items-center gap-2 cursor-pointer rounded-md p-1.5 transition-colors hover:bg-background">
   481→                      <Checkbox
   482→                        checked={values.planIds.includes(p.id)}
   483→                        onCheckedChange={() => togglePlan(p.id)}
   484→                      />
   485→                      <span className="text-sm text-foreground">{p.name}</span>
   486→                      <Badge variant="outline" className="ml-auto text-xs">{p.price} {p.currency}</Badge>
   487→                    </label>
   488→                  ))}
   489→                </div>
   490→              )}
   491→            </div>
   492→          )}
   493→
   494→          {/* Thumbnail */}
   495→          <div className="space-y-1.5">
   496→            <Label className="text-xs font-medium text-muted-foreground">Thumbnail</Label>
   497→            <div className="flex items-center gap-3">
   498→              <div className="h-16 w-24 overflow-hidden rounded-lg bg-muted flex items-center justify-center shrink-0 ring-1 ring-border">
   499→                {values.thumbnail ? (
   500→                  <img src={values.thumbnail} alt="" className="h-full w-full object-cover" />
   501→                ) : (
   502→                  <PlayCircle className="h-5 w-5 text-muted-foreground" />
   503→                )}
   504→              </div>
   505→              <div className="flex-1 space-y-1.5">
   506→                <Input
   507→                  value={values.thumbnail}
   508→                  onChange={(e) => setValues({ ...values, thumbnail: e.target.value })}
   509→                  placeholder="Paste image URL or upload below"
   510→                  className="h-9 rounded-lg"
   511→                />
   512→                <label className="inline-flex items-center gap-1.5 cursor-pointer text-xs text-accent hover:underline">
   513→                  <Upload className="h-3.5 w-3.5" />
   514→                  {uploadingThumb ? "Uploading…" : "Upload image"}
   515→                  <input
   516→                    type="file"
   517→                    accept="image/*"
   518→                    className="hidden"
   519→                    disabled={!isEdit || uploadingThumb}
   520→                    onChange={(e) => {
   521→                      const f = e.target.files?.[0];
   522→                      if (f) uploadThumbnail(f);
   523→                    }}
   524→                  />
   525→                </label>
   526→                {!isEdit && (
   527→                  <p className="text-xs text-muted-foreground">Save course first, then upload thumbnail.</p>
   528→                )}
   529→              </div>
   530→            </div>
   531→          </div>
   532→
   533→          {/* Published switch */}
   534→          <div className="flex items-center justify-between rounded-lg border border-border p-3">
   535→            <div>
   536→              <Label htmlFor="cf-pub" className="text-sm font-medium">Publish immediately</Label>
   537→              <p className="text-xs text-muted-foreground mt-0.5">If off, course stays as a draft.</p>
   538→            </div>
   539→            <Switch
   540→              id="cf-pub"
   541→              checked={values.isPublished}
   542→              onCheckedChange={(v) => setValues({ ...values, isPublished: v })}
   543→            />
   544→          </div>
   545→        </div>
   546→        <DialogFooter>
   547→          <Button variant="outline" className="h-9" onClick={() => onOpenChange(false)}>Cancel</Button>
   548→          <Button disabled={saving} className="h-9" onClick={submit}>
   549→            {saving ? "Saving…" : isEdit ? "Save Changes" : "Create Course"}
   550→          </Button>
   551→        </DialogFooter>
   552→      </DialogContent>
   553→    </Dialog>
   554→  );
   555→}

   557→// ─── Premium Course Card ────────────────────────────────────────────────
   558→function CourseCard({
